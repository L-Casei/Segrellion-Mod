# =========================================================
# THE UNICORN RETURNS
# =========================================================

segrellion_unicorn_returns_cb = {

	icon = invasion
	group = invasion

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = {
		always = yes
	}

	allow_hostages = no


	# =====================================================
	# ONLY DAEMON, AND ONLY AFTER THE DECISION EVENT
	# =====================================================

	allowed_for_character = {

		this = character:Segrellion_60

		has_character_flag = segrellion_unicorn_returns_cb_authorized
	}


	allowed_against_character = {

		scope:defender = title:e_flamelands.holder
	}


	# e_flamelands is mechanically empire-tier even though
	# AGOT presents it as the Lord Paramountcy.
	target_titles = all
	target_title_tier = empire

	target_de_jure_regions_above = no

	# Title transfer is handled explicitly in on_victory.
	# This prevents the normal CB transfer from stripping
	# unrelated Flamelands lords of their domains.
	ignore_effect = change_title_holder


	valid_to_start = {

		scope:target = {
			this = title:e_flamelands
		}
	}


	should_invalidate = {

		# =====================================================
		# THE WAR MUST SURVIVE DAEMON BECOMING LANDED
		#
		# Vanilla adventurer invasions invalidate immediately
		# when the attacker gains land.
		#
		# This is NOT appropriate for The Unicorn Returns.
		#
		# Daemon may inherit Vhal, Blackmark or another title
		# during the war without abandoning his restoration.
		# =====================================================

		NOT = {

			any_in_list = {

				list = target_titles

				any_in_de_jure_hierarchy = {

					tier = tier_county

					is_landless_type_title = no

					holder = {

						target_is_same_character_or_above = scope:defender
					}
				}
			}
		}
	}


	cost = {
		prestige = 250
	}


	# =====================================================
	# WAR BEGINS
	# =====================================================

	on_declaration = {

		on_declared_war = yes


		scope:attacker = {

			remove_character_flag = segrellion_unicorn_returns_cb_authorized

			add_character_flag = segrellion_unicorn_returns_war_active

			set_variable = {
				name = segrellion_unicorn_returns_war
				value = scope:war
			}

			# =================================================
			# THE STORMFLAMES GATHER AT THE BEACON
			# =================================================

			hidden_effect = {
				segrellion_gather_stormflames_at_beacon_effect = yes
			}

			# =================================================
			# THE STORMFLAME HOST
			#
			# Leo remains at the Beacon.
			# Clifford commands the field army.
			# =================================================

			if = {
				limit = {

					scope:defender = character:Stormflame_3

					character:Stormflame_3 ?= {
						is_alive = yes
					}

					character:Stormflame_13 ?= {
						is_alive = yes
					}
				}

				# Leo is far too old to ride into battle himself.
				character:Stormflame_3 = {

					hidden_effect = {

						set_knight_status = forbid

						character:Stormflame_12 ?= {
							set_employer = character:Stormflame_3
						}

						character:Stormflame_13 ?= {
							set_employer = character:Stormflame_3
						}
					}

					spawn_army = {

						name = iaf_leo_stormflame_host

						men_at_arms = {
							type = armored_footmen
							men = 800
						}

						men_at_arms = {
							type = pikemen_unit
							men = 600
						}

						men_at_arms = {
							type = bowmen
							men = 600
						}

						levies = 3000

						inheritable = yes

						location = title:c_farring_cross.title_province
						origin = title:c_farring_cross.title_province

						war = scope:war

						save_scope_as = segrellion_stormflame_war_host
					}
				}

				scope:segrellion_stormflame_war_host ?= {

					assign_commander = character:Stormflame_13

					character:Stormflame_3 = {
						set_variable = {
							name = segrellion_stormflame_war_host
							value = scope:segrellion_stormflame_war_host
						}
					}
				}
			}


			# ---------------------------------------------
			# THE UNICORN UNBOUND
			# ---------------------------------------------

			add_character_modifier = {
				modifier = segrellion_unicorn_unbound_modifier
				years = 2
			}


			# ---------------------------------------------
			# 250 UNICORN RIDERS RALLY TO DAEMON
			# ---------------------------------------------

			spawn_army = {

				name = segrellion_unicorn_returns_riders

				men_at_arms = {
					type = segrellion_unicorn_riders
					men = 250
				}

				inheritable = yes

				location = scope:attacker.capital_province
				origin = scope:attacker.capital_province

				war = scope:war
				war_keep_on_attacker_victory = yes
			}
		}


		# =================================================
		# JOFFREY VLADWYN AND THE TRUE SENTINELS
		# =================================================

		if = {
			limit = {
				character:Vladwyn_5 ?= {
					is_alive = yes
				}
			}

			scope:war ?= {
				add_attacker = character:Vladwyn_5
			}

			character:Vladwyn_5 = {

				trigger_event = {
					id = segrellion_winds.0002
					days = 1
				}
			}
		}


		# =================================================
		# HUGH SEGRELLION
		#
		# Hugh only joins if he personally promised military
		# support when Daemon demanded the Plate of Segrell.
		# =================================================

		character:Segrellion_43 ?= {

			if = {
				limit = {
					is_alive = yes

					has_character_flag = segrellion_hugh_will_support_daemon

					this != scope:defender
				}

				save_scope_as = unicorn_returns_hugh
			}
		}

		scope:unicorn_returns_hugh ?= {


			# Persistent memory:
			# Hugh actually answered Daemon's call and fought
			# in The Unicorn Returns.

			add_character_flag = segrellion_hugh_supported_unicorn_returns


			scope:war ?= {

				add_attacker = scope:unicorn_returns_hugh
			}

			scope:unicorn_returns_hugh = {

				trigger_event = {
					id = segrellion_winds.0003
					days = 1
				}
			}


			remove_character_flag = segrellion_hugh_will_support_daemon
		}


		# =================================================
		# LORD MASSEY
		# =================================================

		title:d_masseys_hook.holder ?= {

			if = {

				limit = {

					is_alive = yes

					this != scope:defender


					character:Segrellion_60 = {
						has_character_flag = segrellion_massey_secret_support_received
					}
				}


				save_scope_as = unicorn_returns_massey
			}
		}

		scope:unicorn_returns_massey ?= {

			scope:war ?= {
				add_attacker = scope:unicorn_returns_massey
			}

			scope:unicorn_returns_massey = {

				trigger_event = {
					id = segrellion_winds.0004
					days = 1
				}
			}
		}


		# =========================================================
		# YOUNG GRIFF
		#
		# If Aegon still honours his promise to Daemon, he joins
		# automatically and is immediately informed that the
		# Unicorn Returns has begun.
		# =========================================================

		character:IAF_Young_Griff_1 ?= {

			if = {

				limit = {

					is_alive = yes

					has_character_flag = segrellion_young_griff_supports_daemon

					this != scope:defender
				}


				scope:war ?= {

					if = {

						limit = {

							NOT = {
								is_attacker = character:IAF_Young_Griff_1
							}
						}


						add_attacker = character:IAF_Young_Griff_1
					}
				}


				add_character_flag = segrellion_young_griff_unicorn_returns_notice_pending


				if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_young_griff_unicorn_returns_notice_seen
						}
					}


					trigger_event = segrellion_winds.0005
				}
			}
		}

		# =========================================================
		# FLAMELANDS HOUSES
		#
		# The realm turns against House Stormflame.
		# Gaunt and Pyle remain loyal to Leo.
		# Landed Stormflames remain loyal to their own house.
		# =========================================================

		scope:defender = {

			# =====================================================
			# MOST FLAMELANDS HOUSES JOIN DAEMON
			# =====================================================

			every_vassal_or_below = {

				limit = {

					is_alive = yes
					is_landed = yes

					highest_held_title_tier > tier_barony

					this != scope:attacker
					this != scope:defender

					NOT = {
						government_has_flag = government_is_nw
					}

					# House Stormflame remains with Leo.
					NOT = {
						house = house:house_Stormflame
					}

					# House Gaunt remains with Leo.
					NOT = {
						has_title = title:c_the_gauntlet
					}

					# House Pyle remains with Leo.
					NOT = {
						has_title = title:c_greyhelm
					}
				}


				save_temporary_scope_as = segrellion_unicorn_returns_daemon_house


				scope:war ?= {

					if = {

						limit = {

							NOR = {

								any_war_attacker = {
									this = scope:segrellion_unicorn_returns_daemon_house
								}

								any_war_defender = {
									this = scope:segrellion_unicorn_returns_daemon_house
								}
							}
						}

						add_attacker = scope:segrellion_unicorn_returns_daemon_house
					}
				}
			}


			# =====================================================
			# LANDED STORMFLAMES JOIN LEO
			# =====================================================

			every_vassal_or_below = {

				limit = {

					is_alive = yes
					is_landed = yes

					highest_held_title_tier > tier_barony

					this != scope:attacker
					this != scope:defender

					house = house:house_Stormflame
				}


				save_temporary_scope_as = segrellion_unicorn_returns_stormflame_house


				scope:war ?= {

					if = {

						limit = {

							NOR = {

								any_war_attacker = {
									this = scope:segrellion_unicorn_returns_stormflame_house
								}

								any_war_defender = {
									this = scope:segrellion_unicorn_returns_stormflame_house
								}
							}
						}

						add_defender = scope:segrellion_unicorn_returns_stormflame_house
					}
				}
			}
		}


		# =========================================================
		# HOUSE GAUNT SUPPORTS LEO STORMFLAME
		# =========================================================

		title:c_the_gauntlet.holder ?= {

			if = {

				limit = {

					is_alive = yes

					this != scope:attacker
					this != scope:defender
				}

				save_scope_as = segrellion_unicorn_returns_gaunt
			}
		}


		scope:segrellion_unicorn_returns_gaunt ?= {

			scope:war ?= {

				if = {

					limit = {

						NOR = {

							any_war_attacker = {
								this = scope:segrellion_unicorn_returns_gaunt
							}

							any_war_defender = {
								this = scope:segrellion_unicorn_returns_gaunt
							}
						}
					}

					add_defender = scope:segrellion_unicorn_returns_gaunt
				}
			}
		}


		# =========================================================
		# HOUSE PYLE SUPPORTS LEO STORMFLAME
		# =========================================================

		title:c_greyhelm.holder ?= {

			if = {

				limit = {

					is_alive = yes

					this != scope:attacker
					this != scope:defender
				}

				save_scope_as = segrellion_unicorn_returns_pyle
			}
		}


		scope:segrellion_unicorn_returns_pyle ?= {

			scope:war ?= {

				if = {

					limit = {

						NOR = {

							any_war_attacker = {
								this = scope:segrellion_unicorn_returns_pyle
							}

							any_war_defender = {
								this = scope:segrellion_unicorn_returns_pyle
							}
						}
					}

					add_defender = scope:segrellion_unicorn_returns_pyle
				}
			}
		}

		# =================================================
		# STORMFLAME ALLIES
		# =================================================

		# Walder Frey answers Leo's call.
		if = {
			limit = {
				character:Frey_1 ?= {
					is_alive = yes
				}
			}

			scope:war ?= {
				add_defender = character:Frey_1
			}
		}


		# Driftmark answers Leo's call.
		title:d_driftmark.holder ?= {

			if = {
				limit = {
					is_alive = yes
					this != scope:attacker
					this != scope:defender
				}

				save_scope_as = unicorn_returns_driftmark
			}
		}

		scope:unicorn_returns_driftmark ?= {

			scope:war ?= {
				add_defender = scope:unicorn_returns_driftmark
			}
		}

		scope:unicorn_returns_young_griff ?= {

			scope:war ?= {
				add_attacker = scope:unicorn_returns_young_griff
			}
		}
	}


	# =====================================================
	# VICTORY
	# =====================================================

	on_victory_desc = {
		desc = segrellion_unicorn_returns_victory_desc
	}


	on_victory = {

		scope:attacker = {

			remove_character_flag = segrellion_unicorn_returns_war_active

			add_character_flag = segrellion_unicorn_returns_completed

			segrellion_sansa_story_progression_effect = yes

			title:d_masseys_hook.holder ?= {
				if = {
					limit = {
						is_alive = yes
					}

					save_scope_as = massey_lord

					if = {
						limit = {
							has_character_modifier = segrellion_defiant_against_stormflames_modifier
						}
						remove_character_modifier = segrellion_defiant_against_stormflames_modifier
					}

					if = {
						limit = {
							NOT = {
								has_character_modifier = segrellion_loyal_to_the_unicorn_modifier
							}
						}

						add_character_modifier = {
							modifier = segrellion_loyal_to_the_unicorn_modifier
						}
					}

					house:house_Segrellion = {
						every_house_member = {
							limit = {
								is_alive = yes
							}

							save_scope_as = segrellion_house_member

							scope:massey_lord = {
								add_opinion = {
									target = scope:segrellion_house_member
									modifier = segrellion_massey_loyal_to_house_segrellion_opinion
								}
							}
						}
					}
				}
			}

			hidden_effect = {
				segrellion_imprison_unicorn_returns_traitors_effect = yes
			}

			# =========================================================
			# VISENYA SEGRELLION
			# THE BLACK UNICORN
			# =========================================================

			character:Black_Unicorn_1 ?= {

				if = {

					limit = {

						is_alive = yes

						has_character_flag = segrellion_black_unicorn_visenya_revealed
						has_character_flag = segrellion_black_unicorn_visenya_chose_daemon
					}


					give_nickname = nick_segrellion_the_black_unicorn


					add_character_flag = segrellion_black_unicorn_named_after_restoration
				}
			}


			# =========================================================
			# FUTURE BLACK UNICORN RETURN
			#
			# Daemon abandoned or completely lost the trail.
			# We only prepare the future branch here.
			# No event is fired yet.
			# =========================================================

			if = {

				limit = {

					has_character_flag = segrellion_black_unicorn_trail_gone_cold

					has_character_flag = segrellion_black_unicorn_future_post_war_return
				}


				add_character_flag = segrellion_black_unicorn_post_war_return_ready
			}


			# =================================================
			# THE TRUE UNICORN
			#
			# Daemon has finally achieved the purpose that
			# defined his exile.
			# =================================================

			add_stress = -100

			add_character_modifier = {
				modifier = segrellion_true_unicorn_modifier
				years = 10
			}


			remove_variable = segrellion_unicorn_returns_war


			ep3_become_landed_warning_effect = {

				TITLE_RECEIVER = scope:attacker
				TITLE = title:e_flamelands
			}

			# =========================================================
			# ABSOLUTE BEACON JUDGMENT FAILSAFE
			#
			# The siege/arrival events provide flavor, but victory
			# itself guarantees that Leo and his family are captured
			# and judged.
			# =========================================================

			if = {

				limit = {

					NOT = {
						has_character_flag = segrellion_beacon_final_judgment_resolved
					}

					NOT = {
						has_character_flag = segrellion_beacon_final_judgment_pending
					}
				}


				segrellion_capture_stormflame_court_effect = yes


				add_character_flag = segrellion_beacon_final_judgment_pending


				trigger_event = {

					id = segrellion_winds.0404
				}
			}


			# =========================================================
			# RESTORE THE FLAMELANDS WITHOUT DESTROYING ITS LORDS
			#
			# Daemon takes:
			#
			# - Paramountcy of the Flamelands
			# - Kingdom/Lordship of the Flamelands
			# - The Beacon
			# - White Point / White Hall
			# - Lion's Tail
			#
			# Existing landed lords otherwise remain landed and simply
			# pass beneath the restored ruler.
			# =========================================================

			create_title_and_vassal_change = {

				type = conquest

				save_scope_as = segrellion_flamelands_restoration_change

				add_claim_on_loss = no
			}


			# =====================================================
			# PARAMOUNTCY + EXISTING VASSALS
			# =====================================================

			title:e_flamelands = {

				change_title_holder_include_vassals = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change

					government_base = scope:defender
				}
			}


			# =====================================================
			# PRINCIPAL FLAMELANDS TITLE
			# =====================================================

			title:k_flamelands = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			# =====================================================
			# THE BEACON
			# =====================================================

			title:d_wendwater_forest = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			title:c_farring_cross = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			title:b_castle_farring = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			# =====================================================
			# WHITE POINT / WHITE HALL
			# =====================================================

			title:c_winded_glade = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			title:b_benthill_dale = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			# =====================================================
			# LION'S TAIL
			#
			# Current AGOT history in the mod identifies the Manning
			# county here as c_stripeharbor.
			# =====================================================

			title:c_stripeharbor = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			resolve_title_and_vassal_change = scope:segrellion_flamelands_restoration_change

			# =====================================================
			# LEO'S SENTENCE
			#
			# Only now is Leo allowed to die.
			#
			# e_flamelands and the restored holdings already belong
			# to Daemon, so Leo's death can no longer change the
			# primary defender or create a Stormflame successor.
			# =====================================================

			if = {

				limit = {

					has_character_flag = segrellion_leo_execution_pending_after_unicorn_returns_victory
				}


				segrellion_queue_leo_execution_after_unicorn_returns_effect = yes
			}


			# =====================================================
			# DAEMON'S CAPITAL
			# =====================================================

			set_realm_capital = title:c_farring_cross

			ep3_become_landed_cleanup_effect = yes

			# =====================================================
			# VISENYA - PART III
			#
			# The Unicorn Returns is the canonical moment from
			# which the one-year Part III clock begins.
			#
			# Do NOT rely on on_title_gain to detect this.
			# =====================================================

			segrellion_black_unicorn_start_post_beacon_clock_effect = yes



			# =====================================================
			# THE TRUE SEGRELLION
			#
			# Daemon has physically restored the Flamelands.
			# One day later he formally reclaims leadership of
			# the main House Segrellion line.
			# =====================================================

			trigger_event = {

				id = segrellion_support.4930

				days = 1
			}


			# =====================================================
			# POSTWAR POLITICAL SETTLEMENT
			# =====================================================

			trigger_event = {

				id = segrellion_support.4940

				days = 2
			}

			segrellion_young_griff_after_unicorn_returns_effect = yes
			
		}


		add_truce_attacker_victory_effect = yes
	}


	# =====================================================
	# WHITE PEACE
	# =====================================================

	on_white_peace_desc = {
		desc = segrellion_unicorn_returns_white_peace_desc
	}

	on_white_peace = {

		scope:attacker = {
			remove_character_flag = segrellion_unicorn_returns_war_active
			remove_variable = segrellion_unicorn_returns_war
			segrellion_young_griff_after_unicorn_returns_effect = yes
		}

		add_truce_white_peace_effect = yes
	}


	# =====================================================
	# DEFEAT
	# =====================================================

	on_defeat_desc = {
		desc = segrellion_unicorn_returns_defeat_desc
	}

	on_defeat = {

		scope:attacker = {
			remove_character_flag = segrellion_unicorn_returns_war_active
			remove_variable = segrellion_unicorn_returns_war
			segrellion_young_griff_after_unicorn_returns_effect = yes
		}

		add_truce_attacker_defeat_effect = yes
	}


	on_invalidated = {

		scope:attacker = {

			remove_character_flag = segrellion_unicorn_returns_cb_authorized
			remove_character_flag = segrellion_unicorn_returns_war_active
			remove_variable = segrellion_unicorn_returns_war
			segrellion_young_griff_after_unicorn_returns_effect = yes
		}
	}


	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes


	war_name = "segrellion_unicorn_returns_war_name"
	war_name_base = "segrellion_unicorn_returns_war_name_base"
	cb_name = "segrellion_unicorn_returns_cb_name"


	interface_priority = 100

	use_de_jure_wargoal_only = yes

	attacker_wargoal_percentage = 0.8
