# =========================================================
# THE UNICORN RETURNS
# =========================================================

segrellion_unicorn_returns_cb = {

	icon = invasion
	group = invasion

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = {
		always = yes
	}

	allow_hostages = no


	# =====================================================
	# ONLY DAEMON, AND ONLY AFTER THE DECISION EVENT
	# =====================================================

	allowed_for_character = {

		this = character:Segrellion_60

		has_character_flag = segrellion_unicorn_returns_cb_authorized
	}


	allowed_against_character = {

		scope:defender = title:e_flamelands.holder
	}


	# e_flamelands is mechanically empire-tier even though
	# AGOT presents it as the Lord Paramountcy.
	target_titles = all
	target_title_tier = empire

	target_de_jure_regions_above = no

	# Title transfer is handled explicitly in on_victory.
	# This prevents the normal CB transfer from stripping
	# unrelated Flamelands lords of their domains.
	ignore_effect = change_title_holder


	valid_to_start = {

		scope:target = {
			this = title:e_flamelands
		}
	}


	should_invalidate = {

		ep3_adventurer_landless_should_invalidate_trigger = yes
	}


	cost = {
	}


	# =====================================================
	# WAR BEGINS
	# =====================================================

	on_declaration = {

		on_declared_war = yes


		scope:attacker = {

			remove_character_flag = segrellion_unicorn_returns_cb_authorized

			add_character_flag = segrellion_unicorn_returns_war_active

			set_variable = {
				name = segrellion_unicorn_returns_war
				value = scope:war
			}

			# =================================================
			# THE STORMFLAMES GATHER AT THE BEACON
			# =================================================

			hidden_effect = {
				segrellion_gather_stormflames_at_beacon_effect = yes
			}

			# =================================================
			# THE STORMFLAME HOST
			#
			# Leo remains at the Beacon.
			# Clifford commands the field army.
			# =================================================

			if = {
				limit = {

					scope:defender = character:Stormflame_3

					character:Stormflame_3 ?= {
						is_alive = yes
					}

					character:Stormflame_13 ?= {
						is_alive = yes
					}
				}

				# Leo is far too old to ride into battle himself.
				character:Stormflame_3 = {

					hidden_effect = {

						set_knight_status = forbid

						character:Stormflame_12 ?= {
							set_employer = character:Stormflame_3
						}

						character:Stormflame_13 ?= {
							set_employer = character:Stormflame_3
						}
					}

					spawn_army = {

						name = iaf_leo_stormflame_host

						men_at_arms = {
							type = armored_footmen
							men = 800
						}

						men_at_arms = {
							type = pikemen_unit
							men = 600
						}

						men_at_arms = {
							type = bowmen
							men = 600
						}

						levies = 3000

						inheritable = yes

						location = title:c_farring_cross.title_province
						origin = title:c_farring_cross.title_province

						war = scope:war

						save_scope_as = segrellion_stormflame_war_host
					}
				}

				scope:segrellion_stormflame_war_host ?= {

					assign_commander = character:Stormflame_13

					character:Stormflame_3 = {
						set_variable = {
							name = segrellion_stormflame_war_host
							value = scope:segrellion_stormflame_war_host
						}
					}
				}
			}


			# ---------------------------------------------
			# THE UNICORN UNBOUND
			# ---------------------------------------------

			add_character_modifier = {
				modifier = segrellion_unicorn_unbound_modifier
				years = 2
			}


			# ---------------------------------------------
			# 250 UNICORN RIDERS RALLY TO DAEMON
			# ---------------------------------------------

			spawn_army = {

				name = segrellion_unicorn_returns_riders

				men_at_arms = {
					type = segrellion_unicorn_riders
					men = 250
				}

				inheritable = yes

				location = scope:attacker.capital_province
				origin = scope:attacker.capital_province

				war = scope:war
				war_keep_on_attacker_victory = yes
			}
		}


		# =================================================
		# JOFFREY VLADWYN AND THE TRUE SENTINELS
		# =================================================

		if = {
			limit = {
				character:Vladwyn_5 ?= {
					is_alive = yes
				}
			}

			scope:war ?= {
				add_attacker = character:Vladwyn_5
			}
		}

		# =================================================
		# HUGH SEGRELLION
		#
		# Hugh only joins if he personally promised military
		# support when Daemon demanded the Plate of Segrell.
		# =================================================

		character:Segrellion_43 ?= {

			if = {
				limit = {
					is_alive = yes

					has_character_flag = segrellion_hugh_will_support_daemon

					this != scope:defender
				}

				save_scope_as = unicorn_returns_hugh
			}
		}

		scope:unicorn_returns_hugh ?= {


			# Persistent memory:
			# Hugh actually answered Daemon's call and fought
			# in The Unicorn Returns.

			add_character_flag = segrellion_hugh_supported_unicorn_returns


			scope:war ?= {

				add_attacker = scope:unicorn_returns_hugh
			}


			remove_character_flag = segrellion_hugh_will_support_daemon
		}


		# =================================================
		# LORD MASSEY
		# =================================================

		title:d_masseys_hook.holder ?= {

			if = {
				limit = {
					is_alive = yes
					this != scope:defender
				}

				save_scope_as = unicorn_returns_massey
			}
		}

		scope:unicorn_returns_massey ?= {

			scope:war ?= {
				add_attacker = scope:unicorn_returns_massey
			}
		}


		# =================================================
		# YOUNG GRIFF
		#
		# We deliberately require the flag that his 300 AC
		# setup will give him. This prevents Stannis from
		# accidentally joining the war before we replace him
		# as holder of the Stormlands.
		# =================================================

		# =================================================
		# YOUNG GRIFF
		#
		# If Aegon still honors the Pentos Compact, he joins
		# Daemon directly. His participation does not depend
		# on already possessing the Stormlands.
		# =================================================

		character:IAF_Young_Griff_1 ?= {

			if = {

				limit = {

					is_alive = yes

					has_character_flag = segrellion_young_griff_supports_daemon

					this != scope:defender
				}


				scope:war ?= {

					add_attacker = character:IAF_Young_Griff_1
				}
			}
		}

		# =================================================
		# STORMFLAME ALLIES
		# =================================================

		# Walder Frey answers Leo's call.
		if = {
			limit = {
				character:Frey_1 ?= {
					is_alive = yes
				}
			}

			scope:war ?= {
				add_defender = character:Frey_1
			}
		}


		# Driftmark answers Leo's call.
		title:d_driftmark.holder ?= {

			if = {
				limit = {
					is_alive = yes
					this != scope:attacker
					this != scope:defender
				}

				save_scope_as = unicorn_returns_driftmark
			}
		}

		scope:unicorn_returns_driftmark ?= {

			scope:war ?= {
				add_defender = scope:unicorn_returns_driftmark
			}
		}

		scope:unicorn_returns_young_griff ?= {

			scope:war ?= {
				add_attacker = scope:unicorn_returns_young_griff
			}
		}
	}


	# =====================================================
	# VICTORY
	# =====================================================

	on_victory_desc = {
		desc = segrellion_unicorn_returns_victory_desc
	}


	on_victory = {

		scope:attacker = {

			remove_character_flag = segrellion_unicorn_returns_war_active

			add_character_flag = segrellion_unicorn_returns_completed

			# =========================================================
			# VISENYA SEGRELLION
			# THE BLACK UNICORN
			# =========================================================

			character:Black_Unicorn_1 ?= {

				if = {

					limit = {

						is_alive = yes

						has_character_flag = segrellion_black_unicorn_visenya_revealed
						has_character_flag = segrellion_black_unicorn_visenya_chose_daemon
					}


					give_nickname = nick_segrellion_the_black_unicorn


					add_character_flag = segrellion_black_unicorn_named_after_restoration
				}
			}


			# =========================================================
			# FUTURE BLACK UNICORN RETURN
			#
			# Daemon abandoned or completely lost the trail.
			# We only prepare the future branch here.
			# No event is fired yet.
			# =========================================================

			if = {

				limit = {

					has_character_flag = segrellion_black_unicorn_trail_gone_cold

					has_character_flag = segrellion_black_unicorn_future_post_war_return
				}


				add_character_flag = segrellion_black_unicorn_post_war_return_ready
			}


			# =================================================
			# THE TRUE UNICORN
			#
			# Daemon has finally achieved the purpose that
			# defined his exile.
			# =================================================

			add_stress = -100

			add_character_modifier = {
				modifier = segrellion_true_unicorn_modifier
				years = 10
			}


			remove_variable = segrellion_unicorn_returns_war


			ep3_become_landed_warning_effect = {

				TITLE_RECEIVER = scope:attacker
				TITLE = title:e_flamelands
			}

			# =========================================================
			# ABSOLUTE BEACON JUDGMENT FAILSAFE
			#
			# The siege/arrival events provide flavor, but victory
			# itself guarantees that Leo and his family are captured
			# and judged.
			# =========================================================

			if = {

				limit = {

					NOT = {
						has_character_flag = segrellion_beacon_final_judgment_resolved
					}

					NOT = {
						has_character_flag = segrellion_beacon_final_judgment_pending
					}
				}


				segrellion_capture_stormflame_court_effect = yes


				add_character_flag = segrellion_beacon_final_judgment_pending


				trigger_event = {

					id = segrellion_winds.0404

					days = 1
				}
			}


			# =========================================================
			# RESTORE THE FLAMELANDS WITHOUT DESTROYING ITS LORDS
			#
			# Daemon takes:
			#
			# - Paramountcy of the Flamelands
			# - Kingdom/Lordship of the Flamelands
			# - The Beacon
			# - White Point / White Hall
			# - Lion's Tail
			#
			# Existing landed lords otherwise remain landed and simply
			# pass beneath the restored ruler.
			# =========================================================

			create_title_and_vassal_change = {

				type = conquest

				save_scope_as = segrellion_flamelands_restoration_change

				add_claim_on_loss = no
			}


			# =====================================================
			# PARAMOUNTCY + EXISTING VASSALS
			# =====================================================

			title:e_flamelands = {

				change_title_holder_include_vassals = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change

					government_base = scope:defender
				}
			}


			# =====================================================
			# PRINCIPAL FLAMELANDS TITLE
			# =====================================================

			title:k_flamelands = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			# =====================================================
			# THE BEACON
			# =====================================================

			title:d_wendwater_forest = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			title:c_farring_cross = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			title:b_castle_farring = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			# =====================================================
			# WHITE POINT / WHITE HALL
			# =====================================================

			title:c_winded_glade = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			title:b_benthill_dale = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			# =====================================================
			# LION'S TAIL
			#
			# Current AGOT history in the mod identifies the Manning
			# county here as c_stripeharbor.
			# =====================================================

			title:c_stripeharbor = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:segrellion_flamelands_restoration_change
				}
			}


			resolve_title_and_vassal_change = scope:segrellion_flamelands_restoration_change


			# =====================================================
			# DAEMON'S CAPITAL
			# =====================================================

			set_realm_capital = title:c_farring_cross

			ep3_become_landed_cleanup_effect = yes

			# =====================================================
			# THE TRUE SEGRELLION
			#
			# Daemon has physically restored the Flamelands.
			# One day later he formally reclaims leadership of
			# the main House Segrellion line.
			# =====================================================

			trigger_event = {

				id = segrellion_support.4930

				days = 1
			}


			# =====================================================
			# POSTWAR POLITICAL SETTLEMENT
			# =====================================================

			trigger_event = {

				id = segrellion_support.4940

				days = 2
			}

			segrellion_young_griff_after_unicorn_returns_effect = yes
			
		}


		add_truce_attacker_victory_effect = yes
	}


	# =====================================================
	# WHITE PEACE
	# =====================================================

	on_white_peace_desc = {
		desc = segrellion_unicorn_returns_white_peace_desc
	}

	on_white_peace = {

		scope:attacker = {
			remove_character_flag = segrellion_unicorn_returns_war_active
			remove_variable = segrellion_unicorn_returns_war
			segrellion_young_griff_after_unicorn_returns_effect = yes
		}

		add_truce_white_peace_effect = yes
	}


	# =====================================================
	# DEFEAT
	# =====================================================

	on_defeat_desc = {
		desc = segrellion_unicorn_returns_defeat_desc
	}

	on_defeat = {

		scope:attacker = {
			remove_character_flag = segrellion_unicorn_returns_war_active
			remove_variable = segrellion_unicorn_returns_war
			segrellion_young_griff_after_unicorn_returns_effect = yes
		}

		add_truce_attacker_defeat_effect = yes
	}


	on_invalidated = {

		scope:attacker = {

			remove_character_flag = segrellion_unicorn_returns_cb_authorized
			remove_character_flag = segrellion_unicorn_returns_war_active
			remove_variable = segrellion_unicorn_returns_war
			segrellion_young_griff_after_unicorn_returns_effect = yes
		}
	}


	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes


	war_name = "segrellion_unicorn_returns_war_name"
	war_name_base = "segrellion_unicorn_returns_war_name_base"
	cb_name = "segrellion_unicorn_returns_cb_name"


	interface_priority = 100

	use_de_jure_wargoal_only = yes

	attacker_wargoal_percentage = 0.8

	max_attacker_score_from_battles = 75

	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}

# =========================================================
# STANNIS BARATHEON
# THE NORTHERN CAMPAIGN
#
# Stannis is NOT claiming the North for himself.
#
# The purpose of this war is to destroy Bolton rule.
# The political settlement happens through events after
# victory.
# =========================================================

segrellion_stannis_northern_campaign_cb = {

	icon = invasion

	group = invasion


	combine_into_one = yes

	should_show_war_goal_subview = yes


	mutually_exclusive_titles = {
		always = yes
	}


	allow_hostages = no


	# =====================================================
	# STANNIS ONLY
	# =====================================================

	allowed_for_character = {

		this = character:Baratheon_3

		has_character_flag = segrellion_stannis_northern_cb_authorized
	}


	# =====================================================
	# CURRENT LORD PARAMOUNT OF THE NORTH
	# =====================================================

	allowed_against_character = {

		scope:defender = title:e_the_north.holder
	}


	target_titles = all

	target_title_tier = empire

	target_de_jure_regions_above = no


	# Do not automatically give the North to Stannis.
	ignore_effect = change_title_holder


	valid_to_start = {

		scope:target = {
			this = title:e_the_north
		}
	}


	should_invalidate = {

		OR = {

			scope:attacker = {
				is_alive = no
			}

			scope:defender = {

				NOT = {
					has_title = title:e_the_north
				}
			}
		}
	}


	cost = {
	}


	# =====================================================
	# WAR DECLARATION
	# =====================================================

	on_declaration = {

		on_declared_war = yes


		scope:attacker = {

			remove_character_flag = segrellion_stannis_northern_cb_authorized
			remove_character_flag = segrellion_stannis_preparing_northern_campaign

			add_character_flag = segrellion_stannis_northern_campaign_started
			add_character_flag = segrellion_stannis_northern_campaign_active


			set_variable = {

				name = segrellion_stannis_northern_war

				value = scope:war
			}


			if = {

				limit = {
					has_character_modifier = segrellion_stannis_northern_story_ai_modifier
				}

				remove_character_modifier = segrellion_stannis_northern_story_ai_modifier
			}


			# =================================================
			# STANNIS' NORTHERN HOST
			#
			# 500 Storm Knights  = 10 x 50
			# 1000 Antlermen     = 10 x 100
			# 150 Mangonels      = 15 x 10
			# 2000 Levies
			#
			# Stannis personally commands the host.
			# =================================================

			spawn_army = {

				name = segrellion_stannis_northern_host


				men_at_arms = {

					type = storm_knights

					men = 1000
				}

				men_at_arms = {

					type = storm_knights

					men = 1000
				}


				men_at_arms = {

					type = stormlands_heavy_infantry

					men = 1000
				}

				men_at_arms = {

					type = stormlands_heavy_infantry

					men = 1000
				}


				men_at_arms = {

					type = mangonel

					men = 150
				}


				levies = 5000


				inheritable = no


				location = title:c_deepwood.title_province
				origin = title:c_deepwood.title_province


				war = scope:war

				war_keep_on_attacker_victory = yes

				save_scope_as = segrellion_stannis_northern_host_army
			}
		}


		# Stannis PERSONALLY leads the army.

		scope:segrellion_stannis_northern_host_army ?= {

			assign_commander = character:Baratheon_3
		}


		# =====================================================
		# HOUSE MANDERLY
		# =====================================================

		title:c_white_harbor.holder ?= {

			if = {

				limit = {

					is_alive = yes

					this != scope:attacker
					this != scope:defender
				}

				save_scope_as = stannis_manderly_supporter
			}
		}


		scope:stannis_manderly_supporter ?= {

			scope:war ?= {
				add_attacker = scope:stannis_manderly_supporter
			}
		}


		# =====================================================
		# HOUSE GLOVER
		# =====================================================

		title:c_deepwood.holder ?= {

			if = {

				limit = {

					is_alive = yes

					this != scope:attacker
					this != scope:defender
				}

				save_scope_as = stannis_glover_supporter
			}
		}

		# =====================================================
		# HOUSE UMBER
		# =====================================================

		title:c_last_hearth.holder ?= {

			if = {

				limit = {

					is_alive = yes

					this != scope:attacker
					this != scope:defender
				}

				save_scope_as = stannis_umber_supporter
			}
		}


		scope:stannis_umber_supporter ?= {

			scope:war ?= {

				add_attacker = scope:stannis_umber_supporter
			}
		}


		# =====================================================
		# HOUSE REED
		# =====================================================

		title:c_greywater_watch.holder ?= {

			if = {

				limit = {

					is_alive = yes

					this != scope:attacker
					this != scope:defender
				}

				save_scope_as = stannis_reed_supporter
			}
		}


		scope:stannis_reed_supporter ?= {

			scope:war ?= {

				add_attacker = scope:stannis_reed_supporter
			}
		}


		scope:stannis_glover_supporter ?= {

			scope:war ?= {
				add_attacker = scope:stannis_glover_supporter
			}
		}
	}


	# =====================================================
	# VICTORY
	# =====================================================

	on_victory_desc = {
		desc = segrellion_stannis_northern_campaign_victory_desc
	}


	on_victory = {

		scope:attacker = {

			remove_character_flag = segrellion_stannis_northern_campaign_active

			segrellion_remove_stannis_pre_winterfell_modifiers_effect = yes

			add_character_flag = segrellion_stannis_northern_campaign_victory

			remove_variable = segrellion_stannis_northern_war


						# =================================================
			# RAMSAY / CURRENT NORTHERN RULER
			# FINAL CAPTURE FAILSAFE
			# =================================================

			if = {

				limit = {

					scope:defender = {

						is_alive = yes

						is_imprisoned = no
					}
				}


				imprison = {

					target = scope:defender

					type = dungeon
				}
			}


			# =================================================
			# REMOVE RAMSAY FROM UNRELATED WARS
			#
			# Example:
			#
			# Young Griff attacks the Iron Throne.
			# Ramsay joins the Iron Throne's side.
			#
			# Without this cleanup CK3 can transfer that
			# participation to the new Northern ruler when
			# e_the_north changes hands.
			#
			# We do NOT touch this Northern Campaign itself.
			# =================================================

			scope:defender = {

				every_character_war = {

					limit = {

						this != scope:war


						# Do not attempt to remove him from
						# a war where he is the actual main
						# attacker or defender.

						NOT = {
							primary_attacker = scope:defender
						}

						NOT = {
							primary_defender = scope:defender
						}
					}


					remove_participant = scope:defender
				}
			}


			# =================================================
			# STANNIS TAKES TEMPORARY CUSTODY OF THE NORTH
			#
			# THIS WAS MISSING FROM THE ACTUAL CB.
			# =================================================

			segrellion_stannis_secure_north_after_victory_effect = yes


			# =================================================

# =========================================================
# SECOND WAR OF THE FIVE KINGS
# RIVAL CLAIMANT WAR
#
# This war does NOT transfer the Iron Throne.
#
# It exists because two rival claimants cannot both survive
# their struggle for the crown.
#
# Victory = defender dies.
# Defeat = attacker dies.
# No white peace.
# =========================================================

segrellion_second_war_claimant_elimination_cb = {

	icon = invasion
	group = invasion

	combine_into_one = no
	should_show_war_goal_subview = yes

	allow_hostages = no

	white_peace_possible = no


	war_name = "segrellion_second_war_of_five_kings_war_name"

	war_name_base = "segrellion_second_war_of_five_kings_war_name_base"

	cb_name = "segrellion_second_war_claimant_elimination_cb_name"


	# =====================================================
	# ONLY ACTIVE IRON THRONE CONTENDERS
	# =====================================================

	allowed_for_character = {

		has_character_flag = segrellion_second_war_active_contender
	}


	allowed_against_character = {

		scope:defender = {

			has_character_flag = segrellion_second_war_active_contender
		}
	}


	# =====================================================
	# VALIDITY
	# =====================================================

	valid_to_start = {

		scope:attacker = {

			is_alive = yes

			has_character_flag = segrellion_second_war_active_contender
		}


		scope:defender = {

			is_alive = yes

			has_character_flag = segrellion_second_war_active_contender
		}
	}


	should_invalidate = {

		OR = {

			scope:attacker = {
				is_alive = no
			}

			scope:defender = {
				is_alive = no
			}
		}
	}


	cost = {
	}


	# =====================================================
	# DECLARATION
	# =====================================================

	on_declaration = {

		on_declared_war = yes
	}


	# =====================================================
	# ATTACKER WINS
	# =====================================================

	on_victory_desc = {

		desc = segrellion_second_war_claimant_elimination_victory_desc
	}


	on_victory = {

		scope:defender = {

			if = {

				limit = {
					is_alive = yes
				}

				death = {

					death_reason = death_execution

					killer = scope:attacker
				}
			}
		}


		add_truce_attacker_victory_effect = yes
	}


	# =====================================================
	# ATTACKER LOSES
	# =====================================================

	on_defeat_desc = {

		desc = segrellion_second_war_claimant_elimination_defeat_desc
	}


	on_defeat = {

		scope:attacker = {

			if = {

				limit = {
					is_alive = yes
				}

				death = {

					death_reason = death_execution

					killer = scope:defender
				}
			}
		}


		add_truce_attacker_defeat_effect = yes
	}


	on_invalidated = {
	}


	transfer_behavior = transfer
}

# =========================================================
# DAENERYS - THE IRON THRONE
#
# Daemon promised to win the Iron Throne for Daenerys
# in exchange for her accepting his proposed Valyrian union.
# =========================================================

segrellion_daenerys_iron_throne_cb = {


	icon = invasion

	group = invasion


	combine_into_one = yes

	should_show_war_goal_subview = yes


	allow_hostages = no


	# =====================================================
	# ONLY DAEMON, ONLY WHILE THE TWO-YEAR PACT IS ACTIVE
	# =====================================================

	allowed_for_character = {


		this = character:Segrellion_60


		has_character_flag = segrellion_daenerys_throne_pact_active


		NOT = {
			has_character_flag = segrellion_daenerys_throne_war_declared
		}


		character:Targaryen_9 ?= {
			is_alive = yes
		}
	}


	allowed_against_character = {

		scope:defender = title:e_the_iron_throne.holder
	}


	target_titles = all

	target_title_tier = empire

	target_de_jure_regions_above = no


	ignore_effect = change_title_holder


	valid_to_start = {

		scope:target = {
			this = title:e_the_iron_throne
		}
	}


	cost = {
	}


	# =====================================================
	# WAR BEGINS
	# =====================================================

	on_declaration = {


		on_declared_war = yes


		scope:attacker = {


			remove_character_flag = segrellion_daenerys_throne_pact_active


			add_character_flag = segrellion_daenerys_throne_war_declared


			set_variable = {

				name = segrellion_daenerys_throne_war

				value = scope:war
			}
		}


		character:Targaryen_9 = {


			remove_character_flag = segrellion_daenerys_waiting_for_throne_war


			add_character_flag = segrellion_daenerys_fighting_for_iron_throne
		}


		scope:war ?= {

			add_attacker = character:Targaryen_9
		}
	}


	# =====================================================
	# VICTORY
	# DAENERYS RECEIVES THE IRON THRONE
	# =====================================================

	on_victory = {


		scope:attacker = {


			remove_variable = segrellion_daenerys_throne_war


			add_character_flag = segrellion_daenerys_throne_war_won
		}


		character:Targaryen_9 = {

			remove_character_flag = segrellion_daenerys_fighting_for_iron_throne

			add_character_flag = segrellion_daenerys_iron_throne_won_by_daemon
		}


		create_title_and_vassal_change = {

			type = conquest

			save_scope_as = segrellion_daenerys_throne_change

			add_claim_on_loss = no
		}


		title:e_the_iron_throne = {

			change_title_holder_include_vassals = {

				holder = character:Targaryen_9

				change = scope:segrellion_daenerys_throne_change

				government_base = scope:defender
			}
		}


		resolve_title_and_vassal_change = scope:segrellion_daenerys_throne_change


		# =================================================
		# DAENERYS FULFILLS HER SIDE OF THE PACT
		#
		# If Visenya remained, Daemon now has two wives.
		# =================================================

		scope:attacker = {


			if = {

				limit = {

					NOT = {
						is_spouse_of = character:Targaryen_9
					}
				}


				marry = character:Targaryen_9
			}


			add_character_flag = segrellion_daemon_married_daenerys_after_throne_war


			if = {

				limit = {

					character:Black_Unicorn_1 ?= {

						is_alive = yes

						is_spouse_of = scope:attacker
					}
				}


				add_character_flag = segrellion_daemon_two_wives
			}
		}


		character:Targaryen_9 = {

			add_character_flag = segrellion_daenerys_married_daemon_after_throne_war
		}
	}


	# =====================================================
	# DEFEAT
	# =====================================================

	on_defeat = {


		scope:attacker = {

			remove_variable = segrellion_daenerys_throne_war

			add_character_flag = segrellion_daenerys_throne_war_lost
		}


		character:Targaryen_9 = {


			remove_character_flag = segrellion_daenerys_fighting_for_iron_throne


			add_opinion = {

				target = scope:attacker

				modifier = segrellion_daenerys_throne_pact_abandoned_opinion
			}
		}
	}


	on_white_peace = {


		scope:attacker = {

			remove_variable = segrellion_daenerys_throne_war

			add_character_flag = segrellion_daenerys_throne_war_white_peace
		}


		character:Targaryen_9 = {

			remove_character_flag = segrellion_daenerys_fighting_for_iron_throne
		}
	}
}
