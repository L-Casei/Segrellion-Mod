# =========================================================
# THE UNICORN RETURNS
# =========================================================

segrellion_unicorn_returns_cb = {

	icon = invasion
	group = invasion

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = {
		always = yes
	}

	allow_hostages = no


	# =====================================================
	# ONLY DAEMON, AND ONLY AFTER THE DECISION EVENT
	# =====================================================

	allowed_for_character = {

		this = character:Segrellion_60

		has_character_flag = segrellion_unicorn_returns_cb_authorized
	}


	allowed_against_character = {

		scope:defender = title:e_flamelands.holder
	}


	# e_flamelands is mechanically empire-tier even though
	# AGOT presents it as the Lord Paramountcy.
	target_titles = all
	target_title_tier = empire

	target_de_jure_regions_above = no

	# We handle title transfer using the landless invasion
	# effects below.
	ignore_effect = change_title_holder


	valid_to_start = {

		scope:target = {
			this = title:e_flamelands
		}
	}


	should_invalidate = {

		ep3_adventurer_landless_should_invalidate_trigger = yes
	}


	cost = {
	}


	# =====================================================
	# WAR BEGINS
	# =====================================================

	on_declaration = {

		on_declared_war = yes


		scope:attacker = {

			remove_character_flag = segrellion_unicorn_returns_cb_authorized

			add_character_flag = segrellion_unicorn_returns_war_active

			set_variable = {
				name = segrellion_unicorn_returns_war
				value = scope:war
			}

			# =================================================
			# THE STORMFLAMES GATHER AT THE BEACON
			# =================================================

			hidden_effect = {
				segrellion_gather_stormflames_at_beacon_effect = yes
			}

			# =================================================
			# THE STORMFLAME HOST
			#
			# Leo remains at the Beacon.
			# Clifford commands the field army.
			# =================================================

			if = {
				limit = {

					scope:defender = character:Stormflame_3

					character:Stormflame_3 ?= {
						is_alive = yes
					}

					character:Stormflame_13 ?= {
						is_alive = yes
					}
				}

				# Leo is far too old to ride into battle himself.
				character:Stormflame_3 = {

					hidden_effect = {

						set_knight_status = forbid

						character:Stormflame_12 ?= {
							set_employer = character:Stormflame_3
						}

						character:Stormflame_13 ?= {
							set_employer = character:Stormflame_3
						}
					}

					spawn_army = {

						name = iaf_leo_stormflame_host

						men_at_arms = {
							type = armored_footmen
							men = 800
						}

						men_at_arms = {
							type = pikemen_unit
							men = 600
						}

						men_at_arms = {
							type = bowmen
							men = 600
						}

						levies = 3000

						inheritable = yes

						location = title:c_farring_cross.title_province
						origin = title:c_farring_cross.title_province

						war = scope:war

						save_scope_as = segrellion_stormflame_war_host
					}
				}

				scope:segrellion_stormflame_war_host ?= {

					assign_commander = character:Stormflame_13

					character:Stormflame_3 = {
						set_variable = {
							name = segrellion_stormflame_war_host
							value = scope:segrellion_stormflame_war_host
						}
					}
				}
			}


			# ---------------------------------------------
			# THE UNICORN UNBOUND
			# ---------------------------------------------

			add_character_modifier = {
				modifier = segrellion_unicorn_unbound_modifier
				years = 2
			}


			# ---------------------------------------------
			# 250 UNICORN RIDERS RALLY TO DAEMON
			# ---------------------------------------------

			spawn_army = {

				name = segrellion_unicorn_returns_riders

				men_at_arms = {
					type = segrellion_unicorn_riders
					men = 250
				}

				inheritable = yes

				location = scope:attacker.capital_province
				origin = scope:attacker.capital_province

				war = scope:war
				war_keep_on_attacker_victory = yes
			}
		}


		# =================================================
		# JOFFREY VLADWYN AND THE TRUE SENTINELS
		# =================================================

		if = {
			limit = {
				character:Vladwyn_5 ?= {
					is_alive = yes
				}
			}

			scope:war ?= {
				add_attacker = character:Vladwyn_5
			}
		}

		# =================================================
		# HUGH SEGRELLION
		#
		# Hugh only joins if he personally promised military
		# support when Daemon demanded the Plate of Segrell.
		# =================================================

		character:Segrellion_43 ?= {

			if = {
				limit = {
					is_alive = yes

					has_character_flag = segrellion_hugh_will_support_daemon

					this != scope:defender
				}

				save_scope_as = unicorn_returns_hugh
			}
		}

		scope:unicorn_returns_hugh ?= {

			scope:war ?= {
				add_attacker = scope:unicorn_returns_hugh
			}

			remove_character_flag = segrellion_hugh_will_support_daemon
		}


		# =================================================
		# LORD MASSEY
		# =================================================

		title:d_masseys_hook.holder ?= {

			if = {
				limit = {
					is_alive = yes
					this != scope:defender
				}

				save_scope_as = unicorn_returns_massey
			}
		}

		scope:unicorn_returns_massey ?= {

			scope:war ?= {
				add_attacker = scope:unicorn_returns_massey
			}
		}


		# =================================================
		# YOUNG GRIFF
		#
		# We deliberately require the flag that his 300 AC
		# setup will give him. This prevents Stannis from
		# accidentally joining the war before we replace him
		# as holder of the Stormlands.
		# =================================================

		title:e_the_stormlands.holder ?= {

			if = {
				limit = {
					is_alive = yes

					has_character_flag = segrellion_young_griff_supports_daemon

					this != scope:defender
				}

				save_scope_as = unicorn_returns_young_griff
			}
		}

				# =================================================
		# STORMFLAME ALLIES
		# =================================================

		# Walder Frey answers Leo's call.
		if = {
			limit = {
				character:Frey_1 ?= {
					is_alive = yes
				}
			}

			scope:war ?= {
				add_defender = character:Frey_1
			}
		}


		# Driftmark answers Leo's call.
		title:d_driftmark.holder ?= {

			if = {
				limit = {
					is_alive = yes
					this != scope:attacker
					this != scope:defender
				}

				save_scope_as = unicorn_returns_driftmark
			}
		}

		scope:unicorn_returns_driftmark ?= {

			scope:war ?= {
				add_defender = scope:unicorn_returns_driftmark
			}
		}

		scope:unicorn_returns_young_griff ?= {

			scope:war ?= {
				add_attacker = scope:unicorn_returns_young_griff
			}
		}
	}


	# =====================================================
	# VICTORY
	# =====================================================

	on_victory_desc = {
		desc = segrellion_unicorn_returns_victory_desc
	}


	on_victory = {

		scope:attacker = {

			remove_character_flag = segrellion_unicorn_returns_war_active

			add_character_flag = segrellion_unicorn_returns_completed


			# =================================================
			# THE TRUE UNICORN
			#
			# Daemon has finally achieved the purpose that
			# defined his exile.
			# =================================================

			add_stress = -100

			add_character_modifier = {
				modifier = segrellion_true_unicorn_modifier
				years = 10
			}


			remove_variable = segrellion_unicorn_returns_war


			# Identify e_flamelands from the war target list.
			ordered_in_list = {
				list = target_titles
				order_by = tier
				save_scope_as = target
			}


			# Standard Roads to Power warning / preparation
			# for a landless character becoming landed.
			ep3_become_landed_warning_effect = {
				TITLE_RECEIVER = scope:attacker
				TITLE = scope:target
			}


			# Collect the titles belonging to the conquered
			# Flamelands realm.
			scope:defender = {

				every_held_title = {

					limit = {
						empire ?= scope:target
					}

					add_to_list = target_titles
				}


				# Make sure at least the capital county and
				# directly-held counties are included.
				if = {

					limit = {

						NOT = {
							any_in_list = {
								list = target_titles
								tier = tier_county
							}
						}
					}


					every_sub_realm_county = {

						limit = {

							OR = {

								AND = {
									empire = scope:target
									holder = scope:target.holder
								}

								scope:target.title_capital_county = this
							}
						}

						add_to_list = target_titles
					}
				}
			}


			# Transfer the conquered realm to Daemon using
			# the vanilla adventurer-invasion machinery.
			ep3_landless_invasion_titles_taken_effect = {

				TITLE_GIVER = scope:defender
				TITLE_RECEIVER = scope:attacker
				TITLE = scope:target
				TITLE_LIST = target_titles
			}


			# The Beacon becomes Daemon's capital.
			title:c_farring_cross = {
				save_temporary_scope_as = unicorn_returns_new_capital
			}

			set_realm_capital = scope:unicorn_returns_new_capital


			# Daemon ceases to be a landless adventurer.
			ep3_become_landed_cleanup_effect = yes
			segrellion_young_griff_after_unicorn_returns_effect = yes
			
		}


		add_truce_attacker_victory_effect = yes
	}


	# =====================================================
	# WHITE PEACE
	# =====================================================

	on_white_peace_desc = {
		desc = segrellion_unicorn_returns_white_peace_desc
	}

	on_white_peace = {

		scope:attacker = {
			remove_character_flag = segrellion_unicorn_returns_war_active
			remove_variable = segrellion_unicorn_returns_war
			segrellion_young_griff_after_unicorn_returns_effect = yes
		}

		add_truce_white_peace_effect = yes
	}


	# =====================================================
	# DEFEAT
	# =====================================================

	on_defeat_desc = {
		desc = segrellion_unicorn_returns_defeat_desc
	}

	on_defeat = {

		scope:attacker = {
			remove_character_flag = segrellion_unicorn_returns_war_active
			remove_variable = segrellion_unicorn_returns_war
			segrellion_young_griff_after_unicorn_returns_effect = yes
		}

		add_truce_attacker_defeat_effect = yes
	}


	on_invalidated = {

		scope:attacker = {

			remove_character_flag = segrellion_unicorn_returns_cb_authorized
			remove_character_flag = segrellion_unicorn_returns_war_active
			remove_variable = segrellion_unicorn_returns_war
			segrellion_young_griff_after_unicorn_returns_effect = yes
		}
	}


	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes


	war_name = "segrellion_unicorn_returns_war_name"
	war_name_base = "segrellion_unicorn_returns_war_name_base"
	cb_name = "segrellion_unicorn_returns_cb_name"


	interface_priority = 100

	use_de_jure_wargoal_only = yes

	attacker_wargoal_percentage = 0.8

	max_attacker_score_from_battles = 75

	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}

# =========================================================
# STANNIS BARATHEON
# THE NORTHERN CAMPAIGN
#
# Stannis is NOT claiming the North for himself.
#
# The purpose of this war is to destroy Bolton rule.
# The political settlement happens through events after
# victory.
# =========================================================

segrellion_stannis_northern_campaign_cb = {

	icon = invasion

	group = invasion


	combine_into_one = yes

	should_show_war_goal_subview = yes


	mutually_exclusive_titles = {
		always = yes
	}


	allow_hostages = no


	# =====================================================
	# STANNIS ONLY
	# =====================================================

	allowed_for_character = {

		this = character:Baratheon_3

		has_character_flag = segrellion_stannis_northern_cb_authorized
	}


	# =====================================================
	# CURRENT LORD PARAMOUNT OF THE NORTH
	# =====================================================

	allowed_against_character = {

		scope:defender = title:e_the_north.holder
	}


	target_titles = all

	target_title_tier = empire

	target_de_jure_regions_above = no


	# Do not automatically give the North to Stannis.
	ignore_effect = change_title_holder


	valid_to_start = {

		scope:target = {
			this = title:e_the_north
		}
	}


	should_invalidate = {

		OR = {

			scope:attacker = {
				is_alive = no
			}

			scope:defender = {

				NOT = {
					has_title = title:e_the_north
				}
			}
		}
	}


	cost = {
	}


	# =====================================================
	# WAR DECLARATION
	# =====================================================

	on_declaration = {

		on_declared_war = yes


		scope:attacker = {

			remove_character_flag = segrellion_stannis_northern_cb_authorized
			remove_character_flag = segrellion_stannis_preparing_northern_campaign

			add_character_flag = segrellion_stannis_northern_campaign_started
			add_character_flag = segrellion_stannis_northern_campaign_active


			set_variable = {

				name = segrellion_stannis_northern_war

				value = scope:war
			}


			if = {

				limit = {
					has_character_modifier = segrellion_stannis_northern_story_ai_modifier
				}

				remove_character_modifier = segrellion_stannis_northern_story_ai_modifier
			}


			# =================================================
			# STANNIS' NORTHERN HOST
			#
			# 500 Storm Knights  = 10 x 50
			# 1000 Antlermen     = 10 x 100
			# 150 Mangonels      = 15 x 10
			# 2000 Levies
			#
			# Stannis personally commands the host.
			# =================================================

			spawn_army = {

				name = segrellion_stannis_northern_host


				men_at_arms = {

					type = storm_knights

					men = 1000
				}

				men_at_arms = {

					type = storm_knights

					men = 1000
				}


				men_at_arms = {

					type = stormlands_heavy_infantry

					men = 1000
				}

				men_at_arms = {

					type = stormlands_heavy_infantry

					men = 1000
				}


				men_at_arms = {

					type = mangonel

					men = 150
				}


				levies = 5000


				inheritable = no


				location = title:c_castle_black.title_province

				origin = title:c_castle_black.title_province


				war = scope:war

				war_keep_on_attacker_victory = yes

				save_scope_as = segrellion_stannis_northern_host_army
			}
		}


		# Stannis PERSONALLY leads the army.

		scope:segrellion_stannis_northern_host_army ?= {

			assign_commander = character:Baratheon_3
		}


		# =====================================================
		# HOUSE MANDERLY
		# =====================================================

		title:c_white_harbor.holder ?= {

			if = {

				limit = {

					is_alive = yes

					this != scope:attacker
					this != scope:defender
				}

				save_scope_as = stannis_manderly_supporter
			}
		}


		scope:stannis_manderly_supporter ?= {

			scope:war ?= {
				add_attacker = scope:stannis_manderly_supporter
			}
		}


		# =====================================================
		# HOUSE GLOVER
		# =====================================================

		title:c_deepwood.holder ?= {

			if = {

				limit = {

					is_alive = yes

					this != scope:attacker
					this != scope:defender
				}

				save_scope_as = stannis_glover_supporter
			}
		}

		# =====================================================
		# HOUSE UMBER
		# =====================================================

		title:c_last_hearth.holder ?= {

			if = {

				limit = {

					is_alive = yes

					this != scope:attacker
					this != scope:defender
				}

				save_scope_as = stannis_umber_supporter
			}
		}


		scope:stannis_umber_supporter ?= {

			scope:war ?= {

				add_attacker = scope:stannis_umber_supporter
			}
		}


		# =====================================================
		# HOUSE REED
		# =====================================================

		title:c_greywater_watch.holder ?= {

			if = {

				limit = {

					is_alive = yes

					this != scope:attacker
					this != scope:defender
				}

				save_scope_as = stannis_reed_supporter
			}
		}


		scope:stannis_reed_supporter ?= {

			scope:war ?= {

				add_attacker = scope:stannis_reed_supporter
			}
		}


		scope:stannis_glover_supporter ?= {

			scope:war ?= {
				add_attacker = scope:stannis_glover_supporter
			}
		}
	}


	# =====================================================
	# VICTORY
	# =====================================================

	on_victory_desc = {
		desc = segrellion_stannis_northern_campaign_victory_desc
	}


	on_victory = {

		scope:attacker = {

			remove_character_flag = segrellion_stannis_northern_campaign_active

			add_character_flag = segrellion_stannis_northern_campaign_victory

			remove_variable = segrellion_stannis_northern_war


						# =================================================
			# RAMSAY / CURRENT NORTHERN RULER
			# FINAL CAPTURE FAILSAFE
			# =================================================

			if = {

				limit = {

					scope:defender = {

						is_alive = yes

						is_imprisoned = no
					}
				}


				imprison = {

					target = scope:defender

					type = dungeon
				}
			}


			# =================================================
			# REMOVE RAMSAY FROM UNRELATED WARS
			#
			# Example:
			#
			# Young Griff attacks the Iron Throne.
			# Ramsay joins the Iron Throne's side.
			#
			# Without this cleanup CK3 can transfer that
			# participation to the new Northern ruler when
			# e_the_north changes hands.
			#
			# We do NOT touch this Northern Campaign itself.
			# =================================================

			scope:defender = {

				every_character_war = {

					limit = {

						this != scope:war


						# Do not attempt to remove him from
						# a war where he is the actual main
						# attacker or defender.

						NOT = {
							primary_attacker = scope:defender
						}

						NOT = {
							primary_defender = scope:defender
						}
					}


					remove_participant = scope:defender
				}
			}


			# =================================================
			# STANNIS TAKES TEMPORARY CUSTODY OF THE NORTH
			#
			# THIS WAS MISSING FROM THE ACTUAL CB.
			# =================================================

			segrellion_stannis_secure_north_after_victory_effect = yes


			# =================================================
			# CONTINUE TO THE POLITICAL SETTLEMENT
			# =================================================

			trigger_event = {

				id = segrellion_stannis.1000

				days = { 2 4 }
			}
		}


		add_truce_attacker_victory_effect = yes
	}


	# =====================================================
	# WHITE PEACE
	# =====================================================

	on_white_peace_desc = {
		desc = segrellion_stannis_northern_campaign_white_peace_desc
	}


	on_white_peace = {

		scope:attacker = {

			remove_character_flag = segrellion_stannis_northern_campaign_active

			add_character_flag = segrellion_stannis_northern_campaign_failed

			remove_variable = segrellion_stannis_northern_war


			trigger_event = {

				id = segrellion_stannis.0900

				days = { 2 5 }
			}
		}


		add_truce_white_peace_effect = yes
	}


	# =====================================================
	# DEFEAT
	# =====================================================

	on_defeat_desc = {
		desc = segrellion_stannis_northern_campaign_defeat_desc
	}


	on_defeat = {

		scope:attacker = {

			remove_character_flag = segrellion_stannis_northern_campaign_active

			add_character_flag = segrellion_stannis_northern_campaign_failed

			remove_variable = segrellion_stannis_northern_war


			trigger_event = {

				id = segrellion_stannis.0900

				days = { 2 5 }
			}
		}


		add_truce_attacker_defeat_effect = yes
	}


	on_invalidated = {

		scope:attacker = {

			remove_character_flag = segrellion_stannis_northern_cb_authorized
			remove_character_flag = segrellion_stannis_northern_campaign_active

			remove_variable = segrellion_stannis_northern_war
		}
	}


	transfer_behavior = transfer


	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit


	attacker_allies_inherit = yes
	defender_allies_inherit = yes


	war_name = "segrellion_stannis_northern_campaign_war_name"
	war_name_base = "segrellion_stannis_northern_campaign_war_name_base"

	cb_name = "segrellion_stannis_northern_campaign_cb_name"


	interface_priority = 100


	use_de_jure_wargoal_only = yes


	attacker_wargoal_percentage = 0.8


	max_attacker_score_from_battles = 75

	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}
