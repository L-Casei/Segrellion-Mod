# =========================================================
# HOUSE SERVICE INTERACTIONS
#
# Voluntarily ask a close family member to:
#   - Join the Kingsguard
#   - Take the Black
#
# Both use normal interaction acceptance logic.
# =========================================================


# =========================================================
# ASK A CLOSE RELATIVE TO TRAIN FOR THE KINGSGUARD
#
# This does NOT appoint the character to the Kingsguard.
#
# If accepted:
#   - training_for_kingsguard modifier
#   - refusing_marriage trait
#
# AGOT's normal Kingsguard system can later recruit them.
# =========================================================

segrellion_ask_join_kingsguard_interaction = {

	category = interaction_category_friendly

	icon = segrellion_kingsguard_interaction

	common_interaction = yes
	interface_priority = 85

	desc = segrellion_ask_join_kingsguard_interaction_desc


	# =====================================================
	# RESPONSE
	# =====================================================

	ai_maybe = yes

	ai_min_reply_days = 4
	ai_max_reply_days = 10

	can_send_despite_rejection = yes

	popup_on_receive = yes
	pause_on_receive = yes

	use_diplomatic_range = yes

	greeting = negative

	notification_text = REQUEST_VOWS_NOTIFICATION_TEXT

	answer_accept_key = segrellion_kingsguard_answer_accept
	answer_reject_key = segrellion_kingsguard_answer_reject

	on_decline_summary = segrellion_kingsguard_family_decline_summary


	# =====================================================
	# COST
	# =====================================================

	cost = {
		prestige = 500
	}


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {

		scope:actor = {

			is_alive = yes
			is_adult = yes
			is_imprisoned = no
		}


		scope:recipient = {

			NOT = {
				this = scope:actor
			}

			is_close_family_of = scope:actor

			is_alive = yes


			NOR = {

				has_trait = kingsguard

				has_trait = lc_kingsguard

				has_trait = nightswatch

				has_trait = maester

				has_trait = order_member

				has_trait = devoted

				has_character_modifier = training_for_kingsguard
			}
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid_showing_failures_only = {

		scope:actor = {

			prestige >= 500
		}


		scope:recipient = {

			# Can begin preparing while still young.
			age >= 10

			is_landed = no

			valid_kingsguard_gender_trigger = yes


			custom_description = {

				text = segrellion_kingsguard_not_married_tt

				is_married = no
			}


			custom_description = {

				text = segrellion_kingsguard_not_already_training_tt

				NOT = {
					has_character_modifier = training_for_kingsguard
				}
			}


			custom_description = {

				text = segrellion_kingsguard_not_refusing_marriage_tt

				NOT = {
					has_trait = refusing_marriage
				}
			}
		}
	}


	# =====================================================
	# HOOK
	# =====================================================

	auto_accept = {

		AND = {

			scope:hook = yes

			scope:actor = {
				has_strong_hook = scope:recipient
			}
		}
	}


	send_option = {

		is_valid = {

			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}

		flag = hook

		localization = SCHEME_HOOK
	}


	should_use_extra_icon = {

		scope:actor = {
			has_usable_hook = scope:recipient
		}
	}


	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_options_exclusive = no


	# =====================================================
	# ACCEPTANCE
	#
	# A prestigious path, but one that means giving up
	# marriage and potentially inheritance.
	# =====================================================

	ai_accept = {

		base = 0


		# -------------------------------------------------
		# HOOK
		# -------------------------------------------------

		modifier = {

			add = 50

			scope:hook = yes

			desc = SCHEME_WEAK_HOOK_USED
		}


		# -------------------------------------------------
		# OPINION OF ACTOR
		# -------------------------------------------------

		opinion_modifier = {

			who = scope:recipient

			opinion_target = scope:actor

			multiplier = 0.25

			desc = AI_SIMPLE_OPINION_REASON
		}


		# -------------------------------------------------
		# HONOUR
		# -------------------------------------------------

		modifier = {

			add = {
				value = ai_honor
				divide = 2
			}

			scope:recipient = {
				ai_honor > 0
			}

			desc = ORDER_VOWS_HONOR
		}


		# -------------------------------------------------
		# BOLDNESS
		# -------------------------------------------------

		modifier = {

			add = {
				value = ai_boldness
				divide = 2
			}

			scope:recipient = {
				ai_boldness > 0
			}

			desc = ORDER_VOWS_BOLD
		}


		modifier = {

			add = {
				value = ai_boldness
			}

			scope:recipient = {
				ai_boldness < 0
			}

			desc = ORDER_VOWS_CRAVEN
		}


		# -------------------------------------------------
		# BRAVE
		# -------------------------------------------------

		modifier = {

			add = 25

			scope:recipient = {
				has_trait = brave
			}

			desc = segrellion_service_reason_brave
		}


		# -------------------------------------------------
		# CRAVEN
		# -------------------------------------------------

		modifier = {

			add = -25

			scope:recipient = {
				has_trait = craven
			}

			desc = segrellion_service_reason_craven
		}


		# -------------------------------------------------
		# AMBITIOUS
		#
		# Ambitious characters may like the prestige,
		# but dislike surrendering their own inheritance.
		# -------------------------------------------------

		modifier = {

			add = -20

			scope:recipient = {
				has_trait = ambitious
			}

			desc = segrellion_service_reason_ambitious
		}


		# -------------------------------------------------
		# CONTENT
		# -------------------------------------------------

		modifier = {

			add = 20

			scope:recipient = {
				has_trait = content
			}

			desc = TAKE_THE_VOWS_CONTENT
		}


		# -------------------------------------------------
		# CHASTE
		# -------------------------------------------------

		modifier = {

			add = 25

			scope:recipient = {
				has_trait = chaste
			}

			desc = TAKE_THE_VOWS_CHASTE
		}


		# -------------------------------------------------
		# LUSTFUL
		# -------------------------------------------------

		modifier = {

			add = -50

			scope:recipient = {

				OR = {
					has_trait = lustful
					has_trait = lifestyle_reveler
					has_trait = seducer
				}
			}

			desc = ORDER_VOWS_HORNY
		}


		# -------------------------------------------------
		# PRIMARY HEIR
		# -------------------------------------------------

		modifier = {

			add = -100

			scope:recipient = {

				any_heir_title = {

					exists = this

					place_in_line_of_succession = {
						target = prev
						value == 1
					}
				}
			}

			desc = ORDER_VOWS_HEIR_1
		}


		# -------------------------------------------------
		# OTHER HEIR
		# -------------------------------------------------

		modifier = {

			add = -35

			scope:recipient = {
				any_heir_title = {
					exists = this
				}
			}

			desc = ORDER_VOWS_HEIR_2
		}


		# -------------------------------------------------
		# MARTIAL EDUCATION
		# -------------------------------------------------

		modifier = {

			add = 25

			scope:recipient = {

				OR = {

					has_trait = education_martial_prowess_1
					has_trait = education_martial_prowess_2
					has_trait = education_martial_prowess_3
					has_trait = education_martial_prowess_4
					has_trait = education_martial_prowess_5
				}
			}

			desc = segrellion_kingsguard_reason_martial_training
		}


		# -------------------------------------------------
		# GOOD PROWESS
		# -------------------------------------------------

		modifier = {

			add = 20

			scope:recipient = {
				prowess >= 15
			}

			desc = segrellion_kingsguard_reason_good_prowess
		}


		modifier = {

			add = 20

			scope:recipient = {
				prowess >= 25
			}

			desc = segrellion_kingsguard_reason_great_prowess
		}
	}


	# =====================================================
	# ACCEPT
	# =====================================================

	on_accept = {

		# =================================================
		# HOOK
		# =================================================

		if = {
			limit = {
				scope:hook = yes
			}

			scope:actor = {
				use_hook = scope:recipient
			}
		}


		# =================================================
		# BEGIN TRAINING FOR THE KINGSGUARD
		#
		# Mirrors AGOT's native Kingsguard ambition setup.
		# =================================================

		scope:recipient = {

			if = {
				limit = {
					NOT = {
						has_trait = refusing_marriage
					}
				}

				add_trait = refusing_marriage
			}


			if = {
				limit = {
					NOT = {
						has_character_modifier = training_for_kingsguard
					}
				}

				add_character_modifier = {
					modifier = training_for_kingsguard
				}
			}


			if = {
				limit = {
					NOT = {
						has_character_flag = kingsguard_ambition
					}
				}

				add_character_flag = kingsguard_ambition
			}


			add_prowess_skill = 1
		}


		# =================================================
		# RESPONSE LETTER
		# =================================================

		scope:actor = {

			trigger_event = {
				id = segrellion_family_service.0001
				days = 1
			}
		}
	}


	# =====================================================
	# DECLINE
	# =====================================================

	on_decline = {

		scope:recipient = {

			add_opinion = {
				target = scope:actor
				modifier = segrellion_asked_me_to_join_kingsguard_opinion
			}
		}


		scope:actor = {

			trigger_event = {
				id = segrellion_family_service.0002
				days = 1
			}
		}
	}


	# =====================================================
	# AI
	# =====================================================

	ai_frequency = 0

	ai_will_do = {
		base = 0
	}
}

		# CHASTE

				has_trait = chaste
			}

			desc = TAKE_THE_VOWS_CHASTE
		}

			add = {
				value = ai_greed

				divide = 2

			scope:recipient = {
				ai_honor > 0
			}

			desc = segrellion_service_reason_honorable
		}


		# -------------------------------------------------
		# BRAVE
		# -------------------------------------------------

		modifier = {

			add = 20

			scope:recipient = {
				has_trait = brave
			}

			desc = segrellion_service_reason_brave
		}


		# -------------------------------------------------

	# =====================================================

		# =================================================

		# =================================================

			limit = {
				scope:hook = yes

				use_hook = scope:recipient
			}
		}


		# =================================================
		# JOIN THE NIGHT'S WATCH
		# =================================================

		scope:recipient = {

			agot_add_to_nightswatch_effect = yes


			title:k_the_wall.holder = {

				trigger_event = agot_nights_watch.0100

				add_courtier = scope:recipient
			}
		}


		# =================================================
		# RESPONSE LETTER
		# =================================================

		scope:actor = {

			trigger_event = {
				id = segrellion_family_service.0101

		}
	}


	# =====================================================
	# DECLINE
	# =====================================================

	on_decline = {

		scope:recipient = {

			add_character_flag = {
				flag = segrellion_refused_family_nightswatch_request
				years = 5
			}


			add_opinion = {
				target = scope:actor
				modifier = segrellion_asked_me_to_take_black_opinion
			}
		}


		scope:actor = {

			trigger_event = {
				id = segrellion_family_service.0102
				days = 1
			}
		}
	}


	# =====================================================
	# AI
	# =====================================================

	ai_frequency = 0

	ai_will_do = {
		base = 0
	}
}
