# =========================================================
# PETITION THE HIGH SEPTON
#
# Only an emperor/hegemon may make this demand.
# The interaction always resolves immediately on click.
# =========================================================

segrellion_petition_high_septon_second_wife_interaction = {

	category = interaction_category_religion
	icon = icon_marriage

	common_interaction = yes
	interface_priority = 100
	use_diplomatic_range = no

	desc = segrellion_petition_high_septon_second_wife_interaction_desc
	greeting = neutral
	notification_text = segrellion_petition_high_septon_second_wife_notification

	localization_values = {
		HIGH_SEPTON_ACCEPTANCE_CHANCE = segrellion_second_wife_high_septon_acceptance_chance_value
	}

	auto_accept = yes
	send_options_exclusive = no
	# Actor-wide: replacing the High Septon must not allow another immediate roll.
	cooldown = { years = 1 }


	should_use_extra_icon = {

		scope:actor = {
			has_usable_hook = scope:recipient
		}
	}

	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"


	is_shown = {

		scope:actor != scope:recipient

		scope:recipient = {
			has_title = title:k_the_most_devout
		}

		scope:actor = {

			is_alive = yes
			is_male = yes
			is_adult = yes
			is_married = yes

			number_of_spouses = 1

			segrellion_can_seek_second_wife_trigger = yes

			NOT = {
				has_character_flag = segrellion_two_wives_authorized
			}

			NOT = {
				has_character_flag = segrellion_second_wife_petition_pending
			}
		}
	}


	is_valid_showing_failures_only = {

		scope:actor = {

			is_imprisoned = no
			is_incapable = no

			custom_description = {

				text = segrellion_second_wife_empire_requirement

				highest_held_title_tier >= tier_empire
			}
		}

		scope:recipient = {
			is_alive = yes
			is_imprisoned = no
		}
	}


	send_option = {

		is_shown = {
			always = yes
		}

		is_valid = {

			scope:actor = {
				gold >= 1000
			}
		}

		flag = bribe

		localization = segrellion_second_wife_offer_1000_gold
		current_description = segrellion_second_wife_offer_1000_gold_desc

		starts_enabled = {
			always = no
		}
	}


	send_option = {

		is_shown = {
			always = yes
		}

		is_valid = {

			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}

		flag = hook

		localization = segrellion_second_wife_use_hook
		current_description = segrellion_second_wife_use_hook_desc

		starts_enabled = {
			always = no
		}
	}


	cost = {

		gold = {

			value = 0

			if = {

				limit = {
					scope:bribe = yes
				}

				add = 1000
			}
		}
	}


	on_send = {

		scope:actor = {

			add_character_flag = {
				flag = segrellion_second_wife_petition_pending
				days = 5
			}

			remove_character_flag = segrellion_second_wife_petition_roll_success
		}


		if = {

			limit = {
				scope:bribe = yes
			}

			scope:recipient = {
				add_gold = 1000
			}
		}


		# Resolve the High Septon's answer immediately.
		random = {

			chance = segrellion_second_wife_high_septon_acceptance_chance_value

			scope:actor = {
				add_character_flag = segrellion_second_wife_petition_roll_success
			}
		}


		if = {

			limit = {

				scope:actor = {
					has_character_flag = segrellion_second_wife_petition_roll_success
				}
			}


			scope:actor = {

				remove_character_flag = segrellion_second_wife_petition_roll_success
				remove_character_flag = segrellion_second_wife_petition_pending

				add_character_flag = segrellion_two_wives_authorized


				send_interface_toast = {

					type = event_toast_effect_good

					title = segrellion_second_wife_high_septon_accept_toast

					left_icon = scope:recipient
				}
			}


			scope:recipient = {

				add_opinion = {

					target = scope:actor

					modifier = pleased_opinion

					opinion = 10
				}
			}


			scope:actor = {

				trigger_event = {
					id = segrellion_second_wife.0001
					days = 1
				}
			}
		}


		else = {

			scope:actor = {

				remove_character_flag = segrellion_second_wife_petition_pending

				segrellion_second_wife_rejected_petition_censure_effect = yes


				send_interface_toast = {

					type = event_toast_effect_bad

					title = segrellion_second_wife_high_septon_reject_toast

					left_icon = scope:recipient
				}


				trigger_event = {
					id = segrellion_second_wife.0002
					days = 1
				}
			}
		}


		if = {

			limit = {
				scope:hook = yes
			}

			scope:actor = {

				if = {

					limit = {
						has_usable_hook = scope:recipient
					}

					use_hook = scope:recipient
				}
			}
		}
	}


	# UI preview of the same factors used by the immediate roll.

	ai_accept = {

		base = -150


		modifier = {

			add = 100
			desc = segrellion_second_wife_dragonrider_reason

			scope:actor = {
				is_current_dragonrider = yes
			}
		}


		modifier = {

			add = 100
			desc = segrellion_second_wife_bribe_reason

			scope:bribe = yes
		}


		modifier = {

			add = 100
			desc = segrellion_second_wife_hook_reason

			scope:hook = yes
		}


		modifier = {

			add = {

				value = scope:actor.piety_level
				multiply = 15
			}

			desc = segrellion_second_wife_piety_reason
		}


		opinion_modifier = {

			opinion_target = scope:actor
			who = scope:recipient
			multiplier = 0.50
		}


		modifier = {
			add = 15
			desc = segrellion_second_wife_military_reason
			scope:actor.current_military_strength >= 10000
			scope:actor.current_military_strength < 25000
		}

		modifier = {
			add = 30
			desc = segrellion_second_wife_military_reason
			scope:actor.current_military_strength >= 25000
			scope:actor.current_military_strength < 50000
		}

		modifier = {
			add = 50
			desc = segrellion_second_wife_military_reason
			scope:actor.current_military_strength >= 50000
			scope:actor.current_military_strength < 100000
		}

		modifier = {
			add = 75
			desc = segrellion_second_wife_military_reason
			scope:actor.current_military_strength >= 100000
		}
	}


	ai_frequency = 0
}


# =========================================================
# TAKE HER AS A SECOND WIFE
#
# Direct character interaction. This MUST stay separate from
# the portrait-slot finder so right-clicking a woman opens the
# normal proposal window instead of the concubine picker.
# =========================================================

segrellion_take_as_second_wife_interaction = {

	category = interaction_category_diplomacy
	icon = icon_marriage

	common_interaction = yes
	interface_priority = 90
	use_diplomatic_range = yes

	desc = segrellion_take_as_second_wife_interaction_desc
	send_name = segrellion_take_as_second_wife_interaction
	greeting = positive
	notification_text = segrellion_take_as_second_wife_notification

	ai_maybe = yes
	can_send_despite_rejection = yes

	ai_min_reply_days = 1
	ai_max_reply_days = 3


	is_shown = {

		scope:actor != scope:recipient

		scope:actor = {
			is_alive = yes
			is_male = yes
			is_adult = yes
			is_married = yes
			number_of_spouses = 1
			segrellion_has_second_wife_slot_trigger = yes
		}

		scope:recipient = {
			is_alive = yes
			is_female = yes
			is_adult = yes
			is_married = no
			is_betrothed = no
		}
	}


	is_valid_showing_failures_only = {

		scope:actor = {
			is_imprisoned = no
			is_incapable = no
		}

		scope:recipient = {
			segrellion_second_wife_candidate_trigger = yes
		}
	}


	on_accept = {

		scope:recipient = {

			if = {

				limit = {
					this = character:Black_Unicorn_1
					scope:actor = character:Segrellion_60
					has_trait = refusing_marriage
				}

				remove_trait = refusing_marriage
			}
		}


		scope:actor = {

			segrellion_create_second_wife_marriage_effect = {
				SECOND_WIFE = scope:recipient
			}
		}
	}

	on_decline = {

		scope:actor = {

			send_interface_toast = {

				type = event_toast_effect_bad

				title = segrellion_second_wife_candidate_declined_toast

				left_icon = scope:recipient
			}
		}
	}


	ai_accept = {

		base = -35


		# Visenya and Daemon receive a strong story-specific exception.
		modifier = {

			add = 300
			desc = segrellion_second_wife_visenya_daemon_reason

			scope:actor = character:Segrellion_60
			scope:recipient = character:Black_Unicorn_1
		}


		# A woman already living in the petitioner's court is much
		# more likely to consider the arrangement.
		modifier = {

			add = 60
			desc = segrellion_second_wife_courtier_reason

			scope:recipient = {
				is_courtier_of = scope:actor
			}
		}


		opinion_modifier = {

			opinion_target = scope:actor
			who = scope:recipient
			multiplier = 0.50
		}


		modifier = {

			add = 80
			desc = segrellion_second_wife_lover_reason

			scope:recipient = {
				has_relation_lover = scope:actor
			}
		}


		modifier = {

			add = 160
			desc = segrellion_second_wife_soulmate_reason

			scope:recipient = {
				has_relation_soulmate = scope:actor
			}
		}


		modifier = {
			add = 20
			desc = segrellion_second_wife_lustful_reason
			scope:recipient = { has_trait = lustful }
		}

		modifier = {
			add = 15
			desc = segrellion_second_wife_ambitious_reason
			scope:recipient = { has_trait = ambitious }
		}

		modifier = {
			add = 15
			desc = segrellion_second_wife_cynical_reason
			scope:recipient = { has_trait = cynical }
		}

		modifier = {
			add = 10
			desc = segrellion_second_wife_trusting_reason
			scope:recipient = { has_trait = trusting }
		}

		modifier = {
			add = 10
			desc = segrellion_second_wife_content_reason
			scope:recipient = { has_trait = content }
		}

		modifier = {
			add = -25
			desc = segrellion_second_wife_chaste_reason
			scope:recipient = { has_trait = chaste }
		}

		modifier = {
			add = -35
			desc = segrellion_second_wife_zealous_reason
			scope:recipient = { has_trait = zealous }
		}

		modifier = {
			add = -20
			desc = segrellion_second_wife_arrogant_reason
			scope:recipient = { has_trait = arrogant }
		}

		modifier = {
			add = -15
			desc = segrellion_second_wife_stubborn_reason
			scope:recipient = { has_trait = stubborn }
		}


		# Same faith helps. A different faith makes a second-wife
		# arrangement substantially harder.
		modifier = {

			add = 15
			desc = segrellion_second_wife_same_faith_reason

			scope:recipient = {
				faith = scope:actor.faith
			}
		}

		modifier = {

			add = -30
			desc = segrellion_second_wife_different_faith_reason

			scope:recipient = {
				faith != scope:actor.faith
			}
		}


		# Noblewomen are more reluctant, especially landed ladies.
		# A famous dynasty also makes the proposal politically harder.
		modifier = {

			add = -10
			desc = segrellion_second_wife_notable_dynasty_reason

			scope:recipient = {
				exists = dynasty

				dynasty = {
					dynasty_prestige_level >= 2
					dynasty_prestige_level < 4
				}
			}
		}


		modifier = {

			add = -25
			desc = segrellion_second_wife_powerful_dynasty_reason

			scope:recipient = {
				exists = dynasty

				dynasty = {
					dynasty_prestige_level >= 4
					dynasty_prestige_level < 7
				}
			}
		}


		modifier = {

			add = -45
			desc = segrellion_second_wife_great_dynasty_reason

			scope:recipient = {
				exists = dynasty

				dynasty = {
					dynasty_prestige_level >= 7
				}
			}
		}


		modifier = {

			add = -15
			desc = segrellion_second_wife_noblewoman_reason

			scope:recipient = {
				is_lowborn = no
			}
		}


		modifier = {

			add = -20
			desc = segrellion_second_wife_countess_reason

			scope:recipient = {
				highest_held_title_tier = tier_county
			}
		}

		modifier = {

			add = -40
			desc = segrellion_second_wife_duchess_reason

			scope:recipient = {
				highest_held_title_tier = tier_duchy
			}
		}

		modifier = {

			add = -70
			desc = segrellion_second_wife_queen_reason

			scope:recipient = {
				highest_held_title_tier = tier_kingdom
			}
		}

		modifier = {

			add = -100
			desc = segrellion_second_wife_empress_reason

			scope:recipient = {
				highest_held_title_tier >= tier_empire
			}
		}


	}


	ai_frequency = 0
}


# =========================================================
# FIND A SECOND WIFE
#
# Used ONLY by the custom empty portrait in the character
# window. This follows the native find_concubine redirect
# contract so actor / recipient scopes do not get scrambled.
# =========================================================

segrellion_find_second_wife_interaction = {

	category = interaction_category_diplomacy
	icon = icon_marriage

	interface_priority = 89
	use_diplomatic_range = yes

	desc = segrellion_find_second_wife_interaction_desc
	send_name = segrellion_take_as_second_wife_interaction
	greeting = positive
	notification_text = segrellion_take_as_second_wife_notification

	# Marriage UI, not concubine UI. This gives the picker the
	# real marriage-match Prestige Gain information and sorting.
	interface = marriage
	needs_recipient_to_open = no


	redirect = {

		scope:actor = {
			save_scope_as = secondary_actor
		}


		if = {

			limit = {

				exists = scope:recipient
				scope:actor != scope:recipient
			}


			scope:recipient = {
				save_scope_as = secondary_recipient
			}


			scope:actor = {
				save_scope_as = recipient
			}
		}
	}


	# The plus-slot only contains women whose REAL second-wife
	# proposal passes the interaction acceptance test.

	populate_recipient_list = {

		scope:actor = {

			ordered_living_character = {

				limit = {

					segrellion_second_wife_candidate_trigger = yes
					in_diplomatic_range = scope:actor

					save_temporary_scope_as = segrellion_second_wife_candidate


					scope:actor = {

						is_character_interaction_potentially_accepted = {

							interaction = segrellion_take_as_second_wife_interaction

							recipient = scope:segrellion_second_wife_candidate
						}
					}
				}


				# Highest actual marriage-prestige gain first.
				# This is based on the candidate dynasty's Splendor,
				# matching CK3's marriage-prestige scale.
				order_by = segrellion_second_wife_marriage_prestige_gain_value
				check_range_bounds = no


				add_to_list = characters
			}
		}
	}


	can_be_picked = {

		segrellion_second_wife_candidate_trigger = yes
		in_diplomatic_range = scope:actor

		save_temporary_scope_as = segrellion_second_wife_candidate


		scope:actor = {

			is_character_interaction_potentially_accepted = {

				interaction = segrellion_take_as_second_wife_interaction

				recipient = scope:segrellion_second_wife_candidate
			}
		}
	}


	# If she is in this list, there is no second rejection roll.
	auto_accept = yes
	ai_instant_response = yes


	is_shown = {

		scope:actor = {

			is_alive = yes
			is_male = yes
			is_adult = yes
			is_married = yes
			number_of_spouses = 1
			segrellion_has_second_wife_slot_trigger = yes


			trigger_if = {

				limit = {
					NOT = { exists = scope:secondary_recipient }
				}

				scope:actor = scope:recipient
			}
		}


		trigger_if = {

			limit = {
				exists = scope:secondary_recipient
			}


			scope:secondary_recipient = {

				segrellion_second_wife_candidate_trigger = yes
			}
		}
	}


	is_valid_showing_failures_only = {

		scope:actor = {
			is_imprisoned = no
			is_incapable = no
		}


		trigger_if = {

			limit = {
				exists = scope:secondary_recipient
			}


			scope:secondary_recipient = {

				segrellion_second_wife_candidate_trigger = yes
			}
		}
	}


	on_auto_accept = {

		scope:secondary_recipient = {

			if = {

				limit = {
					this = character:Black_Unicorn_1
					scope:actor = character:Segrellion_60
					has_trait = refusing_marriage
				}

				remove_trait = refusing_marriage
			}
		}


		scope:actor = {

			segrellion_create_second_wife_marriage_effect = {
				SECOND_WIFE = scope:secondary_recipient
			}
		}
	}


	ai_accept = {

		base = -35


		# Visenya and Daemon receive a strong story-specific exception.
		modifier = {

			add = 300
			desc = segrellion_second_wife_visenya_daemon_reason

			scope:actor = character:Segrellion_60
			scope:recipient = character:Black_Unicorn_1
		}


		# A woman already living in the petitioner's court is much
		# more likely to consider the arrangement.
		modifier = {

			add = 60
			desc = segrellion_second_wife_courtier_reason

			scope:recipient = {
				is_courtier_of = scope:actor
			}
		}


		opinion_modifier = {

			opinion_target = scope:actor
			who = scope:recipient
			multiplier = 0.50
		}

		}

			add = -25
			desc = segrellion_second_wife_chaste_reason
