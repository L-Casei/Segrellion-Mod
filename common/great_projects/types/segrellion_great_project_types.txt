	# VISIBILITY
	# =====================================================

		dynasty = dynasty:dynn_Segrellion

		var:segrellion_lore >= 10

		NOT = {
			has_global_variable = segrellion_lost_legacy_completed

			has_global_variable = segrellion_lost_legacy_project_active

		NOT = {

	# =====================================================
	# LOCATION
	#
	# The project is intellectually dynastic, but the Great
	# Project framework requires a physical province.
	# It is anchored to the founder's capital.
	# =====================================================

	province_filter = capital

	is_location_valid = {
		always = yes
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	can_start_planning = {

		dynasty = dynasty:dynn_Segrellion

			text = segrellion_lost_legacy_dragonlore_requirement_tt

			var:segrellion_lore >= 10
		}

		NOT = {
			has_global_variable = segrellion_lost_legacy_completed
		}

		NOT = {
			has_global_variable = segrellion_lost_legacy_project_active
		}
	}

		always = yes

	# OWNERSHIP
	#
	# Allows the project to remain connected to the founder's
	# principal political position if succession occurs.
	# =====================================================

	owner = founder_primary_title_owner


	is_valid = {

		scope:owner ?= {
			dynasty = dynasty:dynn_Segrellion

	# INITIAL COST
	# =====================================================

		prestige = 250

	# BIND THE CHRONICLE
	#
	# Once all six mandatory chapters are funded, the actual
	# construction phase represents editing, copying,
	# illumination and binding.
	# =====================================================

	# No new historical sources may be added once binding begins.
	contribution_threshold = 0

	contributor_cooldown = 0


	# =====================================================
	# PROJECT PLANNED
	# =====================================================

		set_global_variable = {
			name = segrellion_lost_legacy_project_active
			value = yes
		}

		scope:great_project = {

			set_variable = {
				name = segrellion_chronicle_quality
				value = 0
			}
		}

		scope:owner = {

			trigger_event = {
				id = segrellion_lost_legacy.1000
				days = 1
			}
		}
	}


	# =====================================================
	# BINDING BEGINS
	# =====================================================

			trigger_event = {
