segrellion_icebringer_affinity_value = {
	value = 0

	# =========================================================
	# BLOOD
	# Being a Segrellion helps, but it is far from enough.
	# =========================================================

	if = {
		limit = {
			dynasty = dynasty:dynn_Segrellion
		}
		add = 3
	}


	# =========================================================
	# CHARACTER
	# Icebringer responds to truth of purpose, resolve,
	# courage and consistency rather than simple morality.
	# =========================================================

	if = {
		limit = { has_trait = brave }
		add = 1
	}

	if = {
		limit = { has_trait = honest }
		add = 1
	}

	if = {
		limit = { has_trait = just }
		add = 1
	}

	if = {
		limit = { has_trait = diligent }
		add = 1
	}

	if = {
		limit = { has_trait = stubborn }
		add = 1
	}

	if = {
		limit = { has_trait = loyal }
		add = 1
	}


	# =========================================================
	# NEGATIVE CHARACTER TRAITS
	# =========================================================

	if = {
		limit = { has_trait = craven }
		add = -1
	}

	if = {
		limit = { has_trait = deceitful }
		add = -1
	}

	if = {
		limit = { has_trait = arbitrary }
		add = -1
	}

	if = {
		limit = { has_trait = lazy }
		add = -1
	}

	if = {
		limit = { has_trait = fickle }
		add = -2
	}

	if = {
		limit = { has_trait = disloyal }
		add = -1
	}


	# =========================================================
	# ABILITY TO WIELD THE BLADE
	# Prowess helps, but cannot make Icebringer accept someone.
	# =========================================================

	if = {
		limit = {
			prowess_no_portrait >= 25
		}
		add = 1
	}

	if = {
		limit = {
			prowess_no_portrait >= 35
		}
		add = 2
	}

	if = {
		limit = {
			has_trait = lifestyle_blademaster
		}
		add = 1
	}


	# =========================================================
	# DEEDS
	# These flags are awarded by narrative events.
	# They will become increasingly important as we add
	# Icebringer's moral / purpose events.
	# =========================================================

	if = {
		limit = {
			has_character_flag = segrellion_icebringer_kept_oath
		}
		add = 2
	}

	if = {
		limit = {
			has_character_flag = segrellion_icebringer_refused_reward
		}
		add = 1
	}

	if = {
		limit = {
			has_character_flag = segrellion_icebringer_saved_charge
		}
		add = 1
	}

	if = {
		limit = {
			has_character_flag = segrellion_icebringer_spared_enemy
		}
		add = 1
	}

	if = {
		limit = {
			has_character_flag = segrellion_icebringer_survived_magic
		}
		add = 1
	}

	if = {
		limit = {
			has_character_flag = segrellion_icebringer_guardian_touched
		}
		add = 1
	}


	# =========================================================
	# DEEDS AGAINST ONE'S DECLARED NATURE / PURPOSE
	# =========================================================

	if = {
		limit = {
			has_character_flag = segrellion_icebringer_betrayed_house
		}
		add = -3
	}

	if = {
		limit = {
			has_character_flag = segrellion_icebringer_broke_oath
		}
		add = -3
	}

	if = {
		limit = {
			has_character_flag = segrellion_icebringer_betrayed_declared_purpose
		}
		add = -4
	}

	if = {
		limit = {
			has_character_flag = segrellion_icebringer_lied_for_gain
		}
		add = -1
	}


	# =========================================================
	# FUTURE:
	#
	# Awakened Icebringer will NOT be determined by this score.
	# It will be a unique narrative condition for a single
	# chosen bearer.
	# =========================================================
}
