# =========================================================
# SEGRELLION - SECOND WIFE VALUES
# =========================================================


# =========================================================
# HIGH SEPTON ACCEPTANCE
# =========================================================

segrellion_second_wife_high_septon_acceptance_score_value = {

	value = -150


	if = {

		limit = {
			scope:actor = {
				is_current_dragonrider = yes
			}
		}

		add = 100
	}


	if = {

		limit = {
			scope:bribe = yes
		}

		add = 100
	}


	if = {

		limit = {
			scope:hook = yes
		}

		add = 100
	}


	# -50 to +50 from the High Septon's opinion.

	add = {

		value = "scope:recipient.opinion(scope:actor)"

		multiply = 0.50
	}


	# +15 for each Level of Devotion.

	add = {

		value = scope:actor.piety_level

		multiply = 15
	}


	if = {

		limit = {
			scope:actor.current_military_strength >= 10000
			scope:actor.current_military_strength < 25000
		}

		add = 15
	}


	else_if = {

		limit = {
			scope:actor.current_military_strength >= 25000
			scope:actor.current_military_strength < 50000
		}

		add = 30
	}


	else_if = {

		limit = {
			scope:actor.current_military_strength >= 50000
			scope:actor.current_military_strength < 100000
		}

		add = 50
	}


	else_if = {

		limit = {
			scope:actor.current_military_strength >= 100000
		}

		add = 75
	}
}


segrellion_second_wife_high_septon_acceptance_chance_value = {

	value = segrellion_second_wife_high_septon_acceptance_score_value

	min = 5
	max = 95
	round = yes
}
