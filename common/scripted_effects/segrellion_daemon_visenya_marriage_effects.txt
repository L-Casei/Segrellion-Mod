# Character scope: only Daemon, after a successful marriage to Visenya.
# Safe to call from both scripted weddings and the general marriage hook.

segrellion_daemon_visenya_marriage_traits_effect = {

	if = {

		limit = {

			this = character:Segrellion_60

			is_spouse_of = character:Black_Unicorn_1
		}


		if = {

			limit = {
				has_trait = lustful
			}

			remove_trait = lustful
		}


		if = {

			limit = {

				NOT = {
					has_trait = temperate
				}
			}

			add_trait = temperate
		}


		# =================================================
		# DAEMON ABANDONED A WIFE / BETROTHAL FOR VISENYA
		# =================================================

		if = {

			limit = {

				has_character_flag = segrellion_daemon_broke_previous_union_for_visenya


				NOT = {
					has_character_modifier = segrellion_daemon_visenya_true_love_modifier
				}
			}


			add_character_modifier = {

				modifier = segrellion_daemon_visenya_true_love_modifier
			}
		}


		if = {

			limit = {

				has_character_flag = segrellion_daemon_broke_previous_union_for_visenya


				NOT = {
					has_character_modifier = segrellion_publicly_disloyal_husband_modifier
				}
			}


			add_character_modifier = {

				modifier = segrellion_publicly_disloyal_husband_modifier

				years = 10
			}
		}
	}
}

# =========================================================
# VISENYA - PRE-DAEMON MARRIAGE GUARD
#
# THIS = VISENYA; ROOT may be another spouse or the startup scope.
#
# Before her Black Unicorn storyline resolves, Visenya may
# only be married or betrothed to Daemon. Any AI / vanilla
# union with another character is immediately annulled.
# =========================================================

segrellion_visenya_pre_daemon_marriage_guard_effect = {

	if = {

		limit = {

			this = character:Black_Unicorn_1

			segrellion_visenya_marriage_story_locked_trigger = yes
		}


		if = {

			limit = {

				exists = betrothed

				betrothed != character:Segrellion_60
			}


			break_betrothal = betrothed
		}


		every_spouse = {

			limit = {

				this != character:Segrellion_60
			}


			divorce = character:Black_Unicorn_1
		}
	}
}


# =========================================================
# VISENYA - MARRIAGE LOCK WATCHDOG
#
# Rechecks her betrothal state while the story is unresolved.
# The on_marriage guard below handles direct marriages.
# =========================================================

segrellion_visenya_schedule_marriage_guard_watch_effect = {

	if = {

		limit = {

			this = character:Black_Unicorn_1

			is_alive = yes

			segrellion_visenya_marriage_story_locked_trigger = yes


			NOT = {
				has_character_flag = segrellion_visenya_marriage_guard_watch_running
			}
		}


		add_character_flag = segrellion_visenya_marriage_guard_watch_running


		trigger_event = {

			id = segrellion_black_unicorn.0998

			days = 1
		}
	}
}
