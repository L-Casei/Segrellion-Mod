# =========================================================
# DAENERYS TARGARYEN
# POST-THRONE PACIFICATION
# =========================================================

# =========================================================
# CANCEL AGOT'S GENERIC WAR'S AFTERMATH
#
# ROOT = Daenerys
#
# Mirrors Young Griff's custom Iron Throne aftermath guard.
# Daenerys handles the defeated realm through her own
# pacification storyline, so AGOT must not open its generic
# punishment court.
# =========================================================

segrellion_daenerys_cancel_agot_wars_aftermath_effect = {

	if = {

		limit = {

			this = character:Targaryen_9
		}


		# =================================================
		# HARD-CLEAR AGOT'S PUNISHMENT STATE ON DAENERYS
		#
		# AGOT's "The War's Aftermath" (0600/0601) is fed
		# by these character-side lists / variables.
		# =================================================

		clear_variable_list = mw_traitors_list
		clear_variable_list = mw_punishers_list
		clear_variable_list = mw_post_war_rebels

		remove_variable = mw_punishment_window
		remove_variable = lords_to_be_punished


		# =================================================
		# INVALIDATE ONLY DAENERYS' THRONE-WAR MW STORIES
		#
		# IMPORTANT:
		# The direct Daenerys war uses
		# segrellion_daenerys_fire_and_blood_cb, NOT claim_cb.
		# Keep claim_cb for old saves created before the
		# dedicated CB existed.
		# =================================================

		every_owned_story = {

			limit = {

				story_type = story_agot_mw_rebel

				var:mw_source = flag:mw_external

				OR = {
					var:mw_war_cb = flag:segrellion_daenerys_fire_and_blood_cb
					var:mw_war_cb = flag:segrellion_daenerys_iron_throne_cb
					var:mw_war_cb = flag:claim_cb
				}
			}


			remove_variable = lords_to_be_punished

			clear_variable_list = mw_punishers_list


			set_variable = {

				name = mw_outcome

				value = flag:invalidated
			}
		}


		# =================================================
		# DAEMON-BACKED THRONE WAR
		#
		# In this route Daemon is the primary attacker and
		# can own AGOT's rebel-side Mega War story even though
		# Daenerys receives the Iron Throne.
		# =================================================

		character:Segrellion_60 ?= {

			clear_variable_list = mw_traitors_list
			clear_variable_list = mw_punishers_list
			clear_variable_list = mw_post_war_rebels

			remove_variable = mw_punishment_window
			remove_variable = lords_to_be_punished


			every_owned_story = {

				limit = {

					story_type = story_agot_mw_rebel

					var:mw_source = flag:mw_external

					var:mw_war_cb = flag:segrellion_daenerys_iron_throne_cb
				}


				remove_variable = lords_to_be_punished

				clear_variable_list = mw_punishers_list


				set_variable = {

					name = mw_outcome

					value = flag:invalidated
				}
			}
		}


		add_character_flag = segrellion_daenerys_custom_aftermath_active
	}
}

# =========================================================
# START POST-THRONE STORY
#
# IMPORTANT:
#
# This effect is intentionally idempotent.
#
# It may safely be called from:
#
# - on_title_gain
# - monthly recovery
# - war victory recovery
# - King's Landing surrender
#
# without restarting the story.
# =========================================================

segrellion_start_daenerys_post_throne_effect = {

	if = {

		limit = {

			this = character:Targaryen_9

			is_alive = yes

			has_title = title:h_the_iron_throne
		}


		# =================================================
		# ALWAYS KILL AGOT'S GENERIC AFTERMATH FIRST
		#
		# This deliberately sits OUTSIDE the
		# post_throne_started check.
		#
		# Every recovery pass repeats the cleanup in case
		# AGOT created / repaired its story after our first
		# initialization.
		# =================================================

		segrellion_daenerys_cancel_agot_wars_aftermath_effect = yes


		# =================================================
		# RESTORE DAENERYS' FERTILITY
		#
		# Once Daenerys has taken the Iron Throne, the
		# no-more-children state is permanently removed.
		#
		# This also works as a recovery safeguard for old
		# saves because this initializer is idempotent.
		# =================================================

		segrellion_daenerys_restore_fertility_effect = yes

		# =================================================
		# ROYAL TYRANNY UI / BACKEND
		#
		# Ensure the 0-100 value exists before any visible
		# post-throne event is opened.
		# =================================================

		segrellion_daenerys_ensure_pacification_story_effect = yes

		# =================================================
		# BROKEN-SAVE RECOVERY
		#
		# Older versions could leave fate_pending behind
		# without preserving the actual defeated ruler.
		#
		# That state blocks BOTH The Fallen Crown and the
		# normal post-throne opening forever.
		#
		# Fresh wars now preserve the defeated ruler before
		# war resolution, so this is strictly a recovery
		# path for already-corrupted saves.
		# =================================================

		if = {

			limit = {

				has_character_flag = segrellion_daenerys_defeated_throne_holder_fate_pending


				NOT = {
					has_variable = segrellion_daenerys_defeated_throne_holder
				}
			}


			remove_character_flag = segrellion_daenerys_defeated_throne_holder_fate_pending

			remove_character_flag = segrellion_daenerys_defeated_throne_holder_event_scheduled

			add_character_flag = segrellion_daenerys_defeated_throne_holder_fate_resolved
		}

		# =================================================
		# DEFEATED IRON THRONE HOLDER
		#
		# The fallen king is dealt with BEFORE the normal
		# post-throne government and regional settlements.
		# =================================================

		if = {

			limit = {

				has_character_flag = segrellion_daenerys_defeated_throne_holder_fate_pending

				has_variable = segrellion_daenerys_defeated_throne_holder


				NOT = {
					has_character_flag = segrellion_daenerys_defeated_throne_holder_fate_resolved
				}


				NOT = {
					has_character_flag = segrellion_daenerys_defeated_throne_holder_event_scheduled
				}
			}


			add_character_flag = {

				flag = segrellion_daenerys_defeated_throne_holder_event_scheduled

				days = 30
			}


			add_character_flag = {
				flag = segrellion_daenerys_post_throne_retry_guard
				days = 3
			}


			trigger_event = {

				id = segrellion_daenerys_post_throne.0000
				days = 1
			}
		}

		# =================================================
		# INITIALIZE THE STORY
		# =================================================

		if = {

			limit = {

				NOT = {
					has_character_flag = segrellion_daenerys_post_throne_started
				}
			}


			add_character_flag = segrellion_daenerys_post_throne_started

			add_character_flag = segrellion_daenerys_pacification_active
		}


		# =================================================
		# REPAIR OLD SAVES
		# =================================================

		if = {

			limit = {

				NOT = {
					has_variable = segrellion_daenerys_pacification_tyranny
				}
			}


			set_variable = {

				name = segrellion_daenerys_pacification_tyranny

				value = 0
			}
		}

		# =================================================
		# INVASION STATE IS OVER
		# =================================================

		remove_character_flag = sworn_peace

		remove_character_flag = segrellion_daenerys_invasion_abandoned

		remove_character_flag = segrellion_daenerys_special_story_abandoned

		remove_character_flag = segrellion_daenerys_free_cities_path


		if = {

			limit = {

				has_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
			}


			remove_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
		}


		# =================================================
		# KING'S LANDING WAS BURNED
		# =================================================

		if = {

			limit = {

				has_character_flag = segrellion_daenerys_burned_kings_landing


				NOT = {
					has_character_flag = segrellion_daenerys_kings_landing_burn_consequence_applied
				}
			}


			add_character_flag = segrellion_daenerys_kings_landing_burn_consequence_applied


			segrellion_change_daenerys_pacification_tyranny_effect = {
				VALUE = 40
			}
		}


		# =================================================
		# OPENING EVENT
		#
		# NO DAY DELAY.
		# =================================================

		if = {

			limit = {

				NOT = {
					has_character_flag = segrellion_daenerys_post_throne_opening_seen
				}


				NOT = {
					has_character_flag = segrellion_daenerys_post_throne_opening_scheduled
				}

				NOT = {
					has_character_flag = segrellion_daenerys_defeated_throne_holder_fate_pending
				}
			}


			add_character_flag = {

				flag = segrellion_daenerys_post_throne_opening_scheduled

				days = 30
			}


			add_character_flag = {
				flag = segrellion_daenerys_post_throne_retry_guard
				days = 3
			}


			trigger_event = {
				id = segrellion_daenerys_post_throne.0001
				days = 1
			}
		}
	}
}

# =========================================================
# POST-THRONE SAVE COMPATIBILITY / FAILSAFE
# =========================================================

segrellion_daenerys_post_throne_recovery_effect = {

	character:Targaryen_9 ?= {

		if = {

			limit = {

				is_alive = yes

				has_title = title:h_the_iron_throne
			}


			# =============================================
			# ALWAYS RUN THE IDEMPOTENT INITIALIZER.
			#
			# This repairs saves where post_throne_started
			# exists but the opening event never appeared.
			# =============================================

			segrellion_start_daenerys_post_throne_effect = yes

			# =============================================
			# THE QUEEN'S PEACE SITUATION RECOVERY
			#
			# Only attempt this AFTER the opening event has
			# successfully appeared.
			# =============================================

			if = {

				limit = {

					has_character_flag = segrellion_daenerys_post_throne_opening_seen

					NOT = {
						exists = situation:segrellion_daenerys_pacification
					}
				}


				segrellion_daenerys_ensure_pacification_situation_effect = yes
			}


			# =============================================
			# TYRION'S HAND EVENT FAILED / WAS LOST
			# =============================================

			if = {

				limit = {

					has_character_flag = segrellion_daenerys_post_throne_opening_seen

					NOT = {
						has_character_flag = segrellion_daenerys_has_post_throne_hand_promise
					}

					NOT = {
						has_character_flag = segrellion_tyrion_named_hand
					}

					NOT = {
						has_character_flag = segrellion_daenerys_hand_event_scheduled
					}

					character:Lannister_8 ?= {

						is_alive = yes
					}
				}


				add_character_flag = {

					flag = segrellion_daenerys_hand_event_scheduled

					days = 30
				}


				trigger_event = {

					id = segrellion_daenerys_post_throne.0002
				}
			}


			# =============================================
			# KINGSGUARD EVENT FAILED / WAS LOST
			# =============================================

			else_if = {

				limit = {

					has_character_flag = segrellion_daenerys_post_throne_opening_seen

					NOT = {
						has_character_flag = segrellion_daenerys_kingsguard_rebuilt
					}

					NOT = {
						has_character_flag = segrellion_daenerys_kingsguard_event_scheduled
					}


					OR = {

						has_character_flag = segrellion_tyrion_named_hand

						NOT = {
							character:Lannister_8 ?= {
								is_alive = yes
							}
						}
					}
				}


				add_character_flag = {

					flag = segrellion_daenerys_kingsguard_event_scheduled

					days = 30
				}


				trigger_event = {

					id = segrellion_daenerys_post_throne.0003
				}
			}


			# =============================================
			# REGIONAL SETTLEMENT V2 FAILED / WAS LOST
			# =============================================

			else_if = {

				limit = {

					has_character_flag = segrellion_daenerys_kingsguard_rebuilt


					NOT = {
						has_character_flag = segrellion_daenerys_regional_settlement_v2_started
					}


					NOT = {
						has_character_flag = segrellion_daenerys_generic_settlements_complete
					}


					NOT = {
						has_character_flag = segrellion_daenerys_lost_tyranny_revolt
					}
				}


				segrellion_daenerys_start_regional_settlement_v2_effect = yes
			}
		}
	}
}


# =========================================================
# CHANGE PACIFICATION TYRANNY
#
# VALUE = amount to add
# =========================================================

segrellion_change_daenerys_pacification_tyranny_effect = {

	segrellion_daenerys_ensure_pacification_story_effect = yes


	change_variable = {

		name = segrellion_daenerys_pacification_tyranny

		add = $VALUE$
	}


	if = {

		limit = {

			var:segrellion_daenerys_pacification_tyranny > 100
		}


		set_variable = {

			name = segrellion_daenerys_pacification_tyranny

			value = 100
		}
	}


	if = {

		limit = {

			var:segrellion_daenerys_pacification_tyranny < 0
		}


		set_variable = {

			name = segrellion_daenerys_pacification_tyranny

			value = 0
		}
	}


	segrellion_refresh_daenerys_pacification_modifier_effect = yes

	segrellion_daenerys_sync_pacification_story_effect = yes

	segrellion_daenerys_check_tyranny_revolt_effect = yes
}


# =========================================================
# TYRANNY CONSEQUENCES
# =========================================================

segrellion_refresh_daenerys_pacification_modifier_effect = {

	if = {
		limit = {
			has_character_modifier = segrellion_pacification_tyranny_1_modifier
		}
		remove_character_modifier = segrellion_pacification_tyranny_1_modifier
	}

	if = {
		limit = {
			has_character_modifier = segrellion_pacification_tyranny_2_modifier
		}
		remove_character_modifier = segrellion_pacification_tyranny_2_modifier
	}

	if = {
		limit = {
			has_character_modifier = segrellion_pacification_tyranny_3_modifier
		}
		remove_character_modifier = segrellion_pacification_tyranny_3_modifier
	}

	if = {
		limit = {
			has_character_modifier = segrellion_pacification_tyranny_4_modifier
		}
		remove_character_modifier = segrellion_pacification_tyranny_4_modifier
	}


	if = {

		limit = {
			var:segrellion_daenerys_pacification_tyranny >= 80
		}

		add_character_modifier = {
			modifier = segrellion_pacification_tyranny_4_modifier
		}
	}

	else_if = {

		limit = {
			var:segrellion_daenerys_pacification_tyranny >= 60
		}

		add_character_modifier = {
			modifier = segrellion_pacification_tyranny_3_modifier
		}
	}

	else_if = {

		limit = {
			var:segrellion_daenerys_pacification_tyranny >= 40
		}

		add_character_modifier = {
			modifier = segrellion_pacification_tyranny_2_modifier
		}
	}

	else_if = {

		limit = {
			var:segrellion_daenerys_pacification_tyranny >= 20
		}

		add_character_modifier = {
			modifier = segrellion_pacification_tyranny_1_modifier
		}
	}
}

segrellion_daenerys_install_tyrion_hand_effect = {

	character:Lannister_8 ?= {

		if = {

			limit = {
				is_alive = yes
			}


			if = {

				limit = {
					is_ruler = no
				}

				set_employer = character:Targaryen_9
			}


			character:Targaryen_9 = {

				assign_councillor_type = {
					type = councillor_chancellor
					target = character:Lannister_8
					remove_existing_councillor = yes
				}

				add_character_flag = segrellion_tyrion_named_hand
			}


			character:Lannister_8 = {
				add_character_flag = segrellion_tyrion_hand_of_daenerys
			}
		}
	}
}


# =========================================================
# REBUILD THE KINGSGUARD
#
# Jaime remains.
# Barristan becomes Lord Commander.
# Every other living Kingsguard member is sent to the Wall.
# =========================================================

segrellion_daenerys_rebuild_kingsguard_effect = {

	character:Targaryen_9 = {

		save_scope_as = segrellion_daenerys_queen

		remove_character_flag = segrellion_daenerys_kingsguard_purge_happened
	}


	# =====================================================
	# REMOVE THE OLD GUARD
	# =====================================================

	character:Targaryen_9 = {

		every_councillor = {

			limit = {

				is_alive = yes

				has_trait = kingsguard

				this != character:Lannister_7

				this != character:Selmy_3
			}


			character:Targaryen_9 = {

				add_character_flag = segrellion_daenerys_kingsguard_purge_happened
			}


			agot_add_to_nightswatch_effect = yes
		}
	}


	# =====================================================
	# JAIME LANNISTER
	# =====================================================

	character:Lannister_7 ?= {

		if = {

			limit = {
				is_alive = yes
			}


			if = {

				limit = {

					NOT = {
						has_trait = kingsguard
					}
				}


				agot_join_kingsguard_effect = {

					KINGSGUARD = character:Lannister_7

					KING = character:Targaryen_9
				}
			}


			else_if = {

				limit = {

					NOT = {
						is_councillor_of = character:Targaryen_9
					}
				}


				agot_join_kingsguard_effect = {

					KINGSGUARD = character:Lannister_7

					KING = character:Targaryen_9
				}
			}


			add_character_flag = segrellion_jaime_retained_by_daenerys
		}
	}


	# =====================================================
	# BARRISTAN SELMY
	# =====================================================

	character:Selmy_3 ?= {

		if = {

			limit = {
				is_alive = yes
			}


			if = {

				limit = {

					NOT = {
						has_trait = kingsguard
					}
				}


				agot_join_kingsguard_effect = {

					KINGSGUARD = character:Selmy_3

					KING = character:Targaryen_9
				}
			}


			agot_assign_lord_commander_effect = {

				COMMANDER = character:Selmy_3

				KING = character:Targaryen_9
			}


			add_character_flag = segrellion_barristan_lord_commander_daenerys
		}
	}


	character:Targaryen_9 = {

		add_character_flag = segrellion_daenerys_kingsguard_rebuilt


		if = {

			limit = {

				has_character_flag = segrellion_daenerys_kingsguard_purge_happened
			}


			remove_character_flag = segrellion_daenerys_kingsguard_purge_happened


			segrellion_change_daenerys_pacification_tyranny_effect = {

				VALUE = 5
			}
		}
	}
}
