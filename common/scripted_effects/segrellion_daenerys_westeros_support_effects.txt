# =========================================================
# SECURE DAEMON
#
# ROOT = DAENERYS
#
# Daemon has explicitly chosen Daenerys over Aegon.
# If the Pentos Compact still exists, this is a political
# betrayal and must destroy every remaining live commitment
# between Daemon and Young Griff.
# =========================================================

segrellion_daenerys_secure_daemon_support_effect = {

	add_character_flag = segrellion_daenerys_flamelands_support_secured


	character:Segrellion_60 ?= {

		add_character_flag = segrellion_daemon_supports_daenerys_in_throne_war

		add_character_flag = segrellion_support_agreement_reached_with_daenerys

		remove_character_flag = segrellion_support_declined_daenerys


		# =================================================
		# THE DEADLY TRIARCHY STILL EXISTS
		#
		# Daemon is the character breaking the compact.
		# Apply the canonical Pentos oathbreaker consequence
		# BEFORE dissolving the Triarchy for everyone.
		# =================================================

		if = {

			limit = {
				has_character_flag = segrellion_deadly_triarchy_formed
			}


			segrellion_apply_pentos_oathbreaker_effect = yes

			segrellion_dissolve_deadly_triarchy_effect = yes
		}


		# =================================================
		# PENTOS PLEDGE EXISTS WITHOUT AN ACTIVE TRIARCHY
		#
		# This can happen in intermediate / repaired saves.
		# The political betrayal still happened, but we must
		# NOT call dissolve_deadly_triarchy because that
		# effect marks the Triarchy as having existed.
		# =================================================

		else_if = {

			limit = {
				has_character_flag = segrellion_pledged_to_young_griff
			}


			segrellion_apply_pentos_oathbreaker_effect = yes
		}


		# =================================================
		# DESTROY DAEMON'S LIVE AEGON COMMITMENTS
		#
		# CRITICAL:
		# pledged_to_young_griff is used by the war
		# allegiance guard to force Daemon back into
		# Aegon's wars.
		# =================================================

		remove_character_flag = segrellion_pledged_to_young_griff

		remove_character_flag = segrellion_daemon_supported_by_young_griff


		# =================================================
		# BREAK POLITICAL ALLIANCE
		# =================================================

		if = {

			limit = {
				is_allied_to = character:IAF_Young_Griff_1
			}


			break_alliance = character:IAF_Young_Griff_1
		}


		# =================================================
		# DAEMON CANNOT REMAIN AEGON'S VASSAL
		# WHILE FIGHTING FOR DAENERYS
		# =================================================

		if = {

			limit = {

				liege ?= {
					this = character:IAF_Young_Griff_1
				}
			}


			becomes_independent = yes
		}
	}


	# =====================================================
	# AEGON'S SIDE OF THE BROKEN COMPACT
	# =====================================================

	character:IAF_Young_Griff_1 ?= {

		if = {

			limit = {
				is_alive = yes
			}


			# Daemon has abandoned Aegon's cause.
			# Aegon no longer waits for or supports
			# Daemon's restoration.

			segrellion_young_griff_leave_daemon_restoration_effect = yes


			add_opinion = {

				target = character:Segrellion_60

				modifier = segrellion_aegon_daemon_betrayed_for_daenerys_opinion
			}
		}
	}


	# =====================================================
	# DAENERYS / DAEMON ALLIANCE
	# =====================================================

	create_alliance = character:Segrellion_60
}


# =========================================================
# FINISH DAEMON MEETING
#
# ROOT = DAENERYS
# =========================================================

segrellion_daenerys_finish_daemon_support_meeting_effect = {

		add_character_flag = segrellion_daenerys_support_flamelands_resolved

	# =====================================================
	# STORE DAEMON'S FINAL POSITION
	# =====================================================

	if = {

		limit = {
			has_character_flag = segrellion_daenerys_flamelands_support_secured
		}

		character:Segrellion_60 ?= {

			add_character_flag = segrellion_support_agreement_reached_with_daenerys

			remove_character_flag = segrellion_support_declined_daenerys
		}
	}

	else = {

		character:Segrellion_60 ?= {

			add_character_flag = segrellion_support_declined_daenerys

			remove_character_flag = segrellion_support_agreement_reached_with_daenerys
		}
	}

	# =====================================================
	# OLD NEGOTIATION STATE
	# =====================================================

	remove_character_flag = segrellion_daenerys_daemon_support_marriage_pending

	remove_character_flag = segrellion_daenerys_daemon_support_office_pending

	remove_character_flag = segrellion_daenerys_daemon_support_independence_pending


	# =====================================================
	# NEW NEGOTIATION STATE
	# =====================================================

	remove_character_flag = segrellion_daenerys_daemon_offer_common_cause_pending

	remove_character_flag = segrellion_daenerys_daemon_offer_marriage_pending

	remove_character_flag = segrellion_daenerys_daemon_offer_council_pending

	remove_character_flag = segrellion_daenerys_daemon_offer_independence_pending


	if = {

		limit = {
			has_character_flag = segrellion_daenerys_daemon_meeting_from_beacon
		}


		remove_character_flag = segrellion_daenerys_daemon_meeting_from_beacon


		trigger_event = {

			id = segrellion_daenerys_invasion.0020

			days = 1
		}
	}


	else = {

		remove_character_flag = segrellion_daenerys_daemon_meeting_from_gather_support
	}
}

# =========================================================
# CALL ALL PROMISED SUPPORTERS
#
# ROOT = DAENERYS
#
# The war has just been created, and the Declare War
# decision requires Daenerys to have no previous active war.
# Therefore every_character_war contains the newly created
# throne war.
# =========================================================

segrellion_daenerys_call_promised_supporters_effect = {

	every_character_war = {

		save_scope_as = segrellion_daenerys_promised_support_war


		# =================================================
		# NORTH
		# =================================================

		if = {

			limit = {

				root = {
					has_character_flag = segrellion_daenerys_north_support_secured
				}


				title:e_the_north.holder ?= {

					is_alive = yes

					this != root

					this != title:h_the_iron_throne.holder

					has_character_flag = segrellion_jon_supports_daenerys_throne_claim
				}
			}


			set_called_to = title:e_the_north.holder

			add_attacker = title:e_the_north.holder
		}


		# =================================================
		# VALE FOLLOWS THE NORTH
		# =================================================

		if = {

			limit = {

				root = {
					has_character_flag = segrellion_daenerys_vale_support_secured_with_north
				}


				title:e_the_vale.holder ?= {

					is_alive = yes

					this != root

					this != title:e_the_north.holder

					this != title:h_the_iron_throne.holder
				}
			}


			set_called_to = title:e_the_vale.holder

			add_attacker = title:e_the_vale.holder
		}


		# =================================================
		# REACH
		# =================================================

		if = {

			limit = {

				root = {
					has_character_flag = segrellion_daenerys_reach_support_secured
				}


				title:k_the_reach.holder ?= {

					is_alive = yes

					this != root

					this != title:h_the_iron_throne.holder

					has_character_flag = segrellion_reach_secretly_supports_daenerys
				}
			}


			set_called_to = title:k_the_reach.holder

			add_attacker = title:k_the_reach.holder
		}


		# =================================================
		# WESTERLANDS
		# =================================================

		if = {

			limit = {

				root = {
					has_character_flag = segrellion_daenerys_secret_westerlands_support_against_aegon
				}


				title:e_the_westerlands.holder ?= {

					is_alive = yes

					this != root

					this != title:h_the_iron_throne.holder

					has_character_flag = segrellion_westerlands_secretly_supports_daenerys
				}
			}


			set_called_to = title:e_the_westerlands.holder

			add_attacker = title:e_the_westerlands.holder
		}


		# =================================================
		# DORNE
		# =================================================

		if = {

			limit = {

				root = {
					has_character_flag = segrellion_daenerys_secret_dornish_support_against_aegon
				}

				title:e_dorne.holder ?= {

					is_alive = yes

					this != root

					this != title:h_the_iron_throne.holder

					has_character_flag = segrellion_dorne_secretly_supports_daenerys
				}
			}


			set_called_to = title:e_dorne.holder

			add_attacker = title:e_dorne.holder
		}


		# =================================================
		# RIVERLANDS
		# =================================================

		if = {

			limit = {

				root = {
					has_character_flag = segrellion_daenerys_riverlands_support_secured
				}


				title:e_the_riverlands.holder ?= {

					is_alive = yes

					this != root

					this != title:h_the_iron_throne.holder

					has_character_flag = segrellion_riverlands_supports_daenerys
				}
			}


			set_called_to = title:e_the_riverlands.holder

			add_attacker = title:e_the_riverlands.holder
		}

		# =================================================
		# RIVERLANDS - SECRET LORDS AGAINST BAELISH
		# =================================================

		if = {

			limit = {

				root = {
					has_character_flag = segrellion_daenerys_riverlands_rebellion_support_secured
				}

				title:e_the_riverlands.holder ?= {
					is_alive = yes
				}
			}

			title:e_the_riverlands.holder = {

				every_vassal = {

					limit = {

						is_alive = yes

						this != root

						has_character_flag = segrellion_riverlord_secretly_supports_daenerys
					}

					scope:segrellion_daenerys_promised_support_war = {

						set_called_to = prev

						add_attacker = prev
					}
				}
			}
		}


		# =================================================
		# FLAMELANDS
		# =================================================

		if = {

			limit = {

				root = {
					has_character_flag = segrellion_daenerys_flamelands_support_secured
				}


				character:Segrellion_60 ?= {

					is_alive = yes

					this != title:h_the_iron_throne.holder

					has_character_flag = segrellion_daemon_supports_daenerys_in_throne_war
				}
			}


			set_called_to = character:Segrellion_60

			add_attacker = character:Segrellion_60
		}


		# =================================================
		# STORMLANDS
		#
		# Not hard-coded to Joffrey anymore.
		# Mya / Edric / another accepted special ruler
		# may now have promised support.
		# =================================================

		if = {

			limit = {

				root = {
					has_character_flag = segrellion_daenerys_stormlands_support_secured
				}

				title:e_the_stormlands.holder ?= {

					is_alive = yes

					this != root

					this != title:h_the_iron_throne.holder

					OR = {

						has_character_flag = segrellion_joffrey_supports_daenerys_in_throne_war

						has_character_flag = segrellion_stormlands_supports_daenerys
					}
				}
			}

			set_called_to = title:e_the_stormlands.holder

			add_attacker = title:e_the_stormlands.holder
		}


		# =================================================
		# REACH AUTOMATICALLY SUPPORTS AEGON IF HE IS
		# CURRENTLY MARRIED TO MARGAERY
		# =================================================

		if = {

			limit = {

				title:h_the_iron_throne.holder = character:IAF_Young_Griff_1


				character:IAF_Young_Griff_1 = {

					any_spouse = {
						this = character:Tyrell_14
					}
				}


				title:k_the_reach.holder ?= {

					is_alive = yes

					this != character:IAF_Young_Griff_1
				}
			}


			set_called_to = title:k_the_reach.holder

			add_defender = title:k_the_reach.holder
		}


		# =================================================
		# WESTERLANDS CHOSE AEGON
		# =================================================

		if = {

			limit = {

				title:e_the_westerlands.holder ?= {
					has_character_flag = segrellion_westerlands_supports_aegon_against_daenerys
				}
			}


			set_called_to = title:e_the_westerlands.holder

			add_defender = title:e_the_westerlands.holder
		}


		# =================================================
		# DORNE ALREADY COMMITTED TO AEGON
		# =================================================

		if = {

			limit = {

				title:e_dorne.holder ?= {

					has_character_flag = segrellion_dorne_supports_young_griff

					this != title:h_the_iron_throne.holder
				}
			}


			set_called_to = title:e_dorne.holder

			add_defender = title:e_dorne.holder
		}
	}
}

# =========================================================
# CALL THE LESSER LORDS OF WESTEROS
#
# ROOT = DAENERYS
#
# When Daenerys openly claims the Iron Throne, every
# duchy-tier or kingdom-tier ruler in the Seven Kingdoms
# receives an opportunity to abandon the established order
# and join her war.
#
# Lords Paramount / equivalent great rulers are excluded.
# Their allegiance is already handled through Gather Support
# and the existing promised-support system.
# =========================================================

segrellion_daenerys_call_lesser_westerosi_lords_effect = {

	# =====================================================
	# SAVE DAENERYS' NEW THRONE WAR
	#
	# Take What Is Mine requires Daenerys to have no active
	# war before pressing the decision, so this should be
	# the newly-created Iron Throne war.
	# =====================================================

	every_character_war = {

		save_scope_as = segrellion_daenerys_throne_war
	}


	# =====================================================
	# EVERY ELIGIBLE WESTEROSI RULER
	# =====================================================

	every_ruler = {

		limit = {

			is_alive = yes

			is_landed = yes


			# =================================================
			# DUCHY / KINGDOM RULERS ONLY
			# =================================================

			OR = {

				primary_title.tier = tier_duchy

				primary_title.tier = tier_kingdom
			}


			# =================================================
			# WESTEROS
			#
			# Excludes Essos and Beyond the Wall.
			# =================================================

			capital_county.title_province = {

				geographical_region = world_westeros_seven_kingdoms_without_wall
			}


			# =================================================
			# BASIC EXCLUSIONS
			# =================================================

			this != root

			this != title:h_the_iron_throne.holder


			NOT = {
				target_is_vassal_or_below = root
			}


			# =================================================
			# LORDS PARAMOUNT / GREAT REALMS
			#
			# They already have their own political routes.
			# =================================================

			NOT = {
				has_title = title:e_the_north
			}

			NOT = {
				has_title = title:e_the_vale
			}

			NOT = {
				has_title = title:e_the_westerlands
			}

			NOT = {
				has_title = title:e_the_riverlands
			}

			NOT = {
				has_title = title:k_the_reach
			}

			NOT = {
				has_title = title:e_dorne
			}

			NOT = {
				has_title = title:e_the_stormlands
			}

			NOT = {
				has_title = title:e_the_iron_islands
			}

			NOT = {
				has_title = title:e_flamelands
			}


			# =================================================
			# ONE ROLL ONLY
			# =================================================

			NOT = {
				has_character_flag = segrellion_daenerys_lesser_lord_call_resolved
			}
		}


		add_character_flag = segrellion_daenerys_lesser_lord_call_resolved


		# =====================================================
		# BASE CHANCE: 3%
		#
		# The following modifiers are deliberately additive.
		# A genuinely rebellious / pro-Daenerys lord can become
		# substantially more likely to answer the call.
		# =====================================================

		random = {

			chance = 3


			# =================================================
			# OPINION OF DAENERYS
			# =================================================

			modifier = {

				add = 5

				opinion = {
					target = root
					value >= 25
				}
			}


			modifier = {

				add = 8

				opinion = {
					target = root
					value >= 50
				}
			}


			modifier = {

				add = 10

				opinion = {
					target = root
					value >= 75
				}
			}


			modifier = {

				add = -3

				opinion = {
					target = root
					value <= -25
				}
			}


			modifier = {

				add = -7

				opinion = {
					target = root
					value <= -50
				}
			}


			# =================================================
			# PERSONAL RELATIONSHIP WITH DAENERYS
			# =================================================

			modifier = {

				add = 10

				has_relation_friend = root
			}


			modifier = {

				add = 18

				has_relation_best_friend = root
			}


			modifier = {

				add = -15

				has_relation_rival = root
			}


			# =================================================
			# OPINION OF THEIR OWN LIEGE
			#
			# A lord who despises his liege has a much better
			# reason to use Daenerys' invasion as an opportunity.
			# =================================================

			modifier = {

				add = 6

				exists = liege

				opinion = {
					target = liege
					value <= -25
				}
			}


			modifier = {

				add = 8

				exists = liege

				opinion = {
					target = liege
					value <= -50
				}
			}


			modifier = {

				add = 8

				exists = liege

				opinion = {
					target = liege
					value <= -75
				}
			}


			modifier = {

				add = -4

				exists = liege

				opinion = {
					target = liege
					value >= 50
				}
			}


			# =================================================
			# OPINION OF THE CURRENT KING
			# =================================================

			modifier = {

				add = 5

				opinion = {
					target = title:h_the_iron_throne.holder
					value <= -25
				}
			}


			modifier = {

				add = 7

				opinion = {
					target = title:h_the_iron_throne.holder
					value <= -50
				}
			}


			modifier = {

				add = 8

				opinion = {
					target = title:h_the_iron_throne.holder
					value <= -75
				}
			}


			modifier = {

				add = -5

				opinion = {
					target = title:h_the_iron_throne.holder
					value >= 50
				}
			}


			modifier = {

				add = -8

				opinion = {
					target = title:h_the_iron_throne.holder
					value >= 75
				}
			}


			# =================================================
			# PERSONAL RELATIONSHIP WITH THE KING
			# =================================================

			modifier = {

				add = 10

				has_relation_rival = title:h_the_iron_throne.holder
			}


			modifier = {

				add = -10

				has_relation_friend = title:h_the_iron_throne.holder
			}


			modifier = {

				add = -18

				has_relation_best_friend = title:h_the_iron_throne.holder
			}


			# =================================================
			# AEGON REVEALED AS BLACKFYRE
			# =================================================

			modifier = {

				add = 6

				title:h_the_iron_throne.holder = character:IAF_Young_Griff_1

				character:IAF_Young_Griff_1 = {
					has_character_flag = segrellion_young_griff_blackfyre_revealed
				}
			}


			# =================================================
			# PERSONALITY
			# =================================================

			modifier = {

				add = 5

				has_trait = ambitious
			}


			modifier = {

				add = 3

				has_trait = brave
			}


			modifier = {

				add = 3

				has_trait = vengeful
			}


			modifier = {

				add = 2

				has_trait = deceitful
			}


			modifier = {

				add = 2

				has_trait = arbitrary
			}


			modifier = {

				add = -4

				has_trait = content
			}


			modifier = {

				add = -4

				has_trait = craven
			}


			modifier = {

				add = -2

				has_trait = honest
			}


			modifier = {

				add = -2

				has_trait = just
			}


			modifier = {

				add = -1

				has_trait = humble
			}


			# =================================================
			# THE LORD ANSWERS THE DRAGON'S CALL
			# =================================================

			add_character_flag = segrellion_lesser_lord_supports_daenerys_in_throne_war


			save_scope_as = segrellion_daenerys_lesser_lord_supporter


			# =================================================
			# ADD THEM DIRECTLY TO THE THRONE WAR
			# =================================================

			scope:segrellion_daenerys_throne_war = {

				set_called_to = scope:segrellion_daenerys_lesser_lord_supporter

				add_attacker = scope:segrellion_daenerys_lesser_lord_supporter
			}


			# =================================================
			# KEEP COUNT FOR FUTURE EVENTS / DEBUGGING
			# =================================================

			root = {

				change_variable = {

					name = segrellion_daenerys_lesser_lords_joined

					add = 1
				}


				trigger_event = {

					id = segrellion_daenerys_support.0800

					days = { 1 7 }
				}
			}
		}
	}
}
