# =========================================================
# CREATE DAEMON'S AGOT TRUE-PARENTAGE SECRET
#
# Root   = character who learns the secret
# Target = Daemon Segrellion
#
# Daemon's public mother is already his real mother, so the
# Jon Snow-specific secret (which requires both parents to be
# disputed) is not valid here. Use CK3/AGOT's paternal
# illegitimacy secret, which AGOT's royal-bastard events also
# handle natively.
# =========================================================

segrellion_create_daemon_parentage_secret_effect = {

	# Repair older saves before validating or creating the secret.
	character:Segrellion_60 ?= {
		set_real_father = character:Targaryen_1
	}

	if = {

		limit = {

			NOT = {

				any_known_secret = {

					secret_type = secret_unmarried_illegitimate_child

					secret_target = {
						this = character:Segrellion_60
					}
				}
			}
		}

		add_secret = {
			type = secret_unmarried_illegitimate_child
			target = character:Segrellion_60
		}
	}
}

# =========================================================
# DAEMON PARENTAGE - CONTINUOUS PROGRESSION
#
# Handles:
# - public father correction
# - old save recovery
# - personal identity storyline
# =========================================================

segrellion_daemon_parentage_progression_effect = {

	# Recovery for saves in which Hugh already attempted the public
	# reveal while the old, invalid Jon Snow secret type was in use.
	character:Segrellion_43 ?= {

		if = {

			limit = {
				is_alive = yes
				has_character_flag = segrellion_hugh_revealed_daemon_parentage

			character:Segrellion_60 ?= {

				is_alive = yes


				# Hugh's visible revelation must actually have happened.
				has_character_flag = segrellion_daemon_parentage_known_with_hugh


				NOT = {
					has_character_flag = segrellion_daemon_parentage_publicly_revealed
				}


				NOT = {
					has_character_flag = awaiting_bastardry_result
				}


				NOT = {
					owns_story_of_type = agot_royal_bastard_story
				}
			}

			segrellion_create_daemon_parentage_secret_effect = yes

			random_known_secret = {
				limit = {
					secret_type = secret_unmarried_illegitimate_child
					secret_target = { this = character:Segrellion_60 }
				}
				save_scope_as = heritage_secret
			}

			if = {
				limit = { exists = scope:heritage_secret }

				save_scope_as = revealer

				character:Segrellion_60 = {
					add_character_flag = royal_bastard
					add_character_flag = awaiting_bastardry_result
					add_character_flag = segrellion_daemon_parentage_known
					add_character_flag = segrellion_daemon_parentage_known_with_hugh
					segrellion_daemon_try_start_parentage_identity_chain_effect = yes

					trigger_event = {
						id = agot_events_bastard.0950
						days = 1
					}
				}
			}
		}
	}

	character:Segrellion_60 ?= {

		# =====================================================
		# REPAIR DAEMON'S KNOWLEDGE STATE
		#
		# Do not rely exclusively on our custom flag.
		# AGOT may already consider Daemon to know or have
		# publicly revealed his royal parentage.
		# =====================================================

		if = {

			limit = {

				is_alive = yes

				NOT = {
					has_character_flag = segrellion_daemon_parentage_known
				}

				OR = {

					# Daemon mechanically knows his own
					# illegitimate-parentage secret.

					any_known_secret = {

						secret_type = secret_unmarried_illegitimate_child

						secret_target = {
							this = character:Segrellion_60
						}
					}


					# AGOT has already begun/finished the
					# public royal-bastard reveal.

					owns_story_of_type = agot_royal_bastard_story

					has_character_flag = royal_bastard
				}
			}


			add_character_flag = segrellion_daemon_parentage_known
		}

		# =====================================================
		# PUBLIC REVELATION
		#
		# AGOT creates agot_royal_bastard_story when the
		# parentage becomes public, but does NOT replace the
		# displayed father.
		# =====================================================

		if = {

			limit = {

				owns_story_of_type = agot_royal_bastard_story
			}


			# Find Aerys II dynamically through Daenerys.
			character:Targaryen_9 ?= {

				father ?= {
					save_scope_as = segrellion_daemon_aerys
				}
			}


			if = {

				limit = {
					exists = scope:segrellion_daemon_aerys
				}


				# Biological truth.
				set_real_father = scope:segrellion_daemon_aerys


				# PUBLIC genealogy.
				if = {

					limit = {

						NOT = {
							father = scope:segrellion_daemon_aerys
						}
					}

					set_father = scope:segrellion_daemon_aerys
				}


				add_character_flag = segrellion_daemon_parentage_publicly_revealed

				add_character_flag = segrellion_daemon_parentage_known
			}
		}


		# =====================================================
		# PERSONAL STORY
		# =====================================================

		if = {

			limit = {

				has_character_flag = segrellion_daemon_parentage_known
			}

			segrellion_daemon_try_start_parentage_identity_chain_effect = yes
		}
	}
}


# =========================================================
# START DAEMON'S PERSONAL PARENTAGE STORY
# =========================================================

segrellion_daemon_try_start_parentage_identity_chain_effect = {

	if = {

		limit = {

			this = character:Segrellion_60

			is_alive = yes

			has_character_flag = segrellion_daemon_parentage_known

			NOT = {
				has_character_flag = segrellion_daemon_parentage_identity_chain_started
			}
		}


		# Clear the old permanent queue flag.
		# Older versions could leave this flag forever if
		# the delayed event was lost.

		if = {

			limit = {
				has_character_flag = segrellion_daemon_parentage_identity_chain_queued
			}

			remove_character_flag = segrellion_daemon_parentage_identity_chain_queued
		}


		# Fire the identity event directly.
		# 0100 itself marks the chain as started,
		# so a permanent queue flag is unnecessary.

		trigger_event = {
			id = segrellion_daemon_parentage.0100
		}
	}
}

# =========================================================
# DAEMON ACCEPTS HIS TARGARYEN CLAIM
#
# Uses King's Landing's CURRENT de jure hierarchy instead
# of hardcoding Blackwater/Crownlands title IDs.
#
# This automatically covers:
# county -> duchy -> kingdom -> empire
#
# The Iron Throne hegemony is added separately.
# =========================================================

segrellion_daemon_accept_targaryen_claim_effect = {

	add_character_flag = segrellion_daemon_targaryen_claimant_path

	remove_character_flag = segrellion_daemon_renounced_targaryen_claim


	# =====================================================
	# IRON THRONE HEGEMONY
	# =====================================================

	if = {

		limit = {

			NOT = {
				has_claim_on = title:h_the_iron_throne
			}
		}

		add_pressed_claim = title:h_the_iron_throne
	}


	# =====================================================
	# KING'S LANDING HIERARCHY
	# =====================================================

	title:c_kings_landing ?= {

		save_scope_as = segrellion_daemon_claim_county


		de_jure_liege ?= {

			save_scope_as = segrellion_daemon_claim_duchy


			de_jure_liege ?= {

				save_scope_as = segrellion_daemon_claim_kingdom


				de_jure_liege ?= {

					save_scope_as = segrellion_daemon_claim_empire
				}
			}
		}
	}


	if = {

		limit = {

			exists = scope:segrellion_daemon_claim_county

			NOT = {
				has_claim_on = scope:segrellion_daemon_claim_county
			}
		}

		add_pressed_claim = scope:segrellion_daemon_claim_county
	}


	if = {

		limit = {

			exists = scope:segrellion_daemon_claim_duchy

			NOT = {
				has_claim_on = scope:segrellion_daemon_claim_duchy
			}
		}

		add_pressed_claim = scope:segrellion_daemon_claim_duchy
	}


	if = {

		limit = {

			exists = scope:segrellion_daemon_claim_kingdom

			NOT = {
				has_claim_on = scope:segrellion_daemon_claim_kingdom
			}
		}

		add_pressed_claim = scope:segrellion_daemon_claim_kingdom
	}


	if = {

		limit = {

			exists = scope:segrellion_daemon_claim_empire

			NOT = {
				has_claim_on = scope:segrellion_daemon_claim_empire
			}
		}

		add_pressed_claim = scope:segrellion_daemon_claim_empire
	}
}


# =========================================================
# DAEMON REJECTS HIS TARGARYEN CLAIM
# =========================================================

segrellion_daemon_renounce_targaryen_claim_effect = {

	add_character_flag = segrellion_daemon_renounced_targaryen_claim

	remove_character_flag = segrellion_daemon_targaryen_claimant_path


	title:c_kings_landing ?= {

		save_scope_as = segrellion_daemon_claim_county


		de_jure_liege ?= {

			save_scope_as = segrellion_daemon_claim_duchy


			de_jure_liege ?= {

				save_scope_as = segrellion_daemon_claim_kingdom


				de_jure_liege ?= {

					save_scope_as = segrellion_daemon_claim_empire
				}
			}
		}
	}


	if = {

		limit = {
			has_claim_on = title:h_the_iron_throne
		}

		remove_claim = title:h_the_iron_throne
	}


	if = {

		limit = {

			exists = scope:segrellion_daemon_claim_county

			has_claim_on = scope:segrellion_daemon_claim_county
		}

		remove_claim = scope:segrellion_daemon_claim_county
	}


	if = {

		limit = {

			exists = scope:segrellion_daemon_claim_duchy

			has_claim_on = scope:segrellion_daemon_claim_duchy
		}

		remove_claim = scope:segrellion_daemon_claim_duchy
	}


	if = {

		limit = {

			exists = scope:segrellion_daemon_claim_kingdom

			has_claim_on = scope:segrellion_daemon_claim_kingdom
		}

		remove_claim = scope:segrellion_daemon_claim_kingdom
	}


	if = {

		limit = {

			exists = scope:segrellion_daemon_claim_empire

			has_claim_on = scope:segrellion_daemon_claim_empire
		}

		remove_claim = scope:segrellion_daemon_claim_empire
	}
}

# =========================================================
# DAEMON BECOMES A TARGARYEN
# =========================================================

segrellion_daemon_choose_targaryen_dynasty_effect = {

	character:Targaryen_9 ?= {

		house ?= {
			save_scope_as = segrellion_daemon_targaryen_house
		}
	}


	if = {

		limit = {
			exists = scope:segrellion_daemon_targaryen_house
		}


		set_house = scope:segrellion_daemon_targaryen_house


		# Ensure Aerys' bloodline mechanics are inherited even
		# if Daemon learned the truth privately.
		apply_bastard_bloodline_inheritance_effect = yes


		add_character_flag = segrellion_daemon_chose_targaryen_identity
		add_character_flag = segrellion_daemon_parentage_banner_resolved
		add_character_flag = segrellion_daemon_parentage_next_stage_ready
	}
}

	add_character_flag = segrellion_daemon_chose_segrellion_identity

	add_character_flag = segrellion_daemon_parentage_banner_resolved

	add_character_flag = segrellion_daemon_parentage_next_stage_ready
}


# =========================================================
# HYBRID DYNASTY - COMMON FINALIZATION
# =========================================================

segrellion_daemon_finalize_hybrid_dynasty_effect = {

	add_character_flag = segrellion_daemon_hybrid_dynasty

	add_character_flag = segrellion_daemon_parentage_banner_resolved

	add_character_flag = segrellion_daemon_parentage_next_stage_ready


	# Targaryen bloodline inheritance remains part of Daemon
	# even though he founded another dynasty.

	apply_bastard_bloodline_inheritance_effect = yes


	# =====================================================
	# NEW DYNASTY - STARTING RENOWN / SPLENDOR
	# =====================================================

	if = {

		limit = {

			NOT = {
				has_character_flag = segrellion_daemon_hybrid_dynasty_initialized
			}
		}


		dynasty ?= {

			# 6250 spendable Renown.

			add_dynasty_prestige = 6250


			# But the dynasty begins at Splendor level 0.

			while = {

				limit = {

					dynasty_prestige_level > 0
				}

				add_dynasty_prestige_level = -1
			}
		}


		add_character_flag = segrellion_daemon_hybrid_dynasty_initialized
	}


	dynasty ?= {

		add_dynasty_modifier = segrellion_hybrid_dragon_unicorn_dynasty_modifier


		# Copy every Segrellion legacy already unlocked.

		if = {

			limit = {

				dynasty:dynn_Segrellion = {
					has_dynasty_perk = segrellion_legacy_1
				}

				NOT = {
					has_dynasty_perk = segrellion_legacy_1
				}
			}

			add_dynasty_perk = segrellion_legacy_1
		}


		if = {

			limit = {

				dynasty:dynn_Segrellion = {
					has_dynasty_perk = segrellion_legacy_2
				}

				NOT = {
					has_dynasty_perk = segrellion_legacy_2
				}
			}

			add_dynasty_perk = segrellion_legacy_2
		}


		if = {

			limit = {

				dynasty:dynn_Segrellion = {
					has_dynasty_perk = segrellion_legacy_3
				}

				NOT = {
					has_dynasty_perk = segrellion_legacy_3
				}
			}

			add_dynasty_perk = segrellion_legacy_3
		}


		if = {

			limit = {

				dynasty:dynn_Segrellion = {
					has_dynasty_perk = segrellion_legacy_4
				}

				NOT = {
					has_dynasty_perk = segrellion_legacy_4
				}
			}

			add_dynasty_perk = segrellion_legacy_4
		}


		if = {

			limit = {

				dynasty:dynn_Segrellion = {
					has_dynasty_perk = segrellion_legacy_5
				}

				NOT = {
					has_dynasty_perk = segrellion_legacy_5
				}
			}

			add_dynasty_perk = segrellion_legacy_5
		}
	}


	# =====================================================
	# SEGRELLION RELATIONS
	# =====================================================

	dynasty:dynn_Segrellion = {

		every_dynasty_member = {

			limit = {

				is_alive = yes

				this != root
			}


			add_opinion = {

				target = root

				modifier = segrellion_hybrid_dynasty_blood_kinship_opinion
			}


			root = {

				add_opinion = {

					target = prev

					modifier = segrellion_hybrid_dynasty_blood_kinship_opinion
				}
			}
		}
	}


	# =====================================================
	# TARGARYEN RELATIONS
	# =====================================================

	character:Targaryen_9 ?= {

		dynasty ?= {

			every_dynasty_member = {

				limit = {

					is_alive = yes

					this != root
				}


				add_opinion = {

					target = root

					modifier = segrellion_hybrid_dynasty_blood_kinship_opinion
				}


				root = {

					add_opinion = {

						target = prev

						modifier = segrellion_hybrid_dynasty_blood_kinship_opinion
					}
				}
			}
		}
	}
}


segrellion_daemon_become_vhalaryen_effect = {

	# Create a REAL independent dynasty and founding House for Daemon.

	create_dynasty = {
		save_scope_as = segrellion_daemon_new_hybrid_dynasty
	}


	# Dynasty identity.

	scope:segrellion_daemon_new_hybrid_dynasty = {

		set_dynasty_name = dynn_Vhalaryen

		set_coa = dynn_Vhalaryen
	}


	# Founding House uses exactly the same name and arms.

	house ?= {

		set_house_name_from_dynasty = scope:segrellion_daemon_new_hybrid_dynasty

		set_coa = dynn_Vhalaryen
	}


	add_character_flag = segrellion_daemon_founded_vhalaryen


	segrellion_daemon_finalize_hybrid_dynasty_effect = yes
}
