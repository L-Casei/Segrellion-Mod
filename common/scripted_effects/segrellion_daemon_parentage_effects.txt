# DAEMON SEGRELLION - ROYAL BASTARD SECRET
# =========================================================


# =========================================================
# CREATE DAEMON'S AGOT TRUE-PARENTAGE SECRET
#
# Root   = character who learns the secret
# Target = Daemon Segrellion
#
# This is the SAME heritage-secret type used by
# AGOT for Jon Snow.
# =========================================================

segrellion_create_daemon_parentage_secret_effect = {

	if = {

		limit = {

			NOT = {

				any_known_secret = {

					secret_type = secret_agot_disputed_heritage

					secret_target = {
						this = character:Segrellion_60
					}
				}
			}
		}

		add_secret = {
			type = secret_agot_disputed_heritage
			target = character:Segrellion_60
		}
	}
}
