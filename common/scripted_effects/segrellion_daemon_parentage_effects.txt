# =========================================================
# DAEMON SEGRELLION - ROYAL BASTARD SECRET
# =========================================================


# =========================================================
# CREATE DAEMON'S AGOT TRUE-PARENTAGE SECRET
#
# Root   = character who learns the secret
# Target = Daemon Segrellion
#
# This is the SAME heritage-secret type used by
# AGOT for Jon Snow.
# =========================================================

segrellion_create_daemon_parentage_secret_effect = {

	if = {

		limit = {

			NOT = {

				any_known_secret = {

					secret_type = secret_agot_disputed_heritage

					secret_target = {
						this = character:Segrellion_60
					}
				}
			}
		}

		add_secret = {
			type = secret_agot_disputed_heritage
			target = character:Segrellion_60
		}
	}
}

# =========================================================
# PUBLICLY REVEAL DAEMON'S TRUE PARENTAGE
#
# Current scope = the revealer
#
# Revealer may be:
# - Daemon himself
# - Hugh Segrellion
#
# The AGOT heritage Secret is deliberately created only
# here, at the moment the truth actually becomes public.
# =========================================================

segrellion_publicly_reveal_daemon_parentage_effect = {

	# AGOT's bastard event expects this scope.
	save_scope_as = revealer


	# =====================================================
	# GUARANTEE AERYS II IS DAEMON'S REAL FATHER
	# =====================================================

	character:Targaryen_9 ?= {

		father ?= {
			save_scope_as = segrellion_daemon_true_father
		}
	}


	character:Segrellion_60 ?= {

		if = {

			limit = {
				exists = scope:segrellion_daemon_true_father
			}

			set_real_father = scope:segrellion_daemon_true_father
		}
	}


	# =====================================================
	# CREATE THE ACTUAL AGOT HERITAGE SECRET
	#
	# Until this exact moment the truth existed only through
	# our narrative flags.
	# =====================================================

	segrellion_create_daemon_parentage_secret_effect = yes


	random_known_secret = {

		limit = {

			secret_type = secret_agot_disputed_heritage

			secret_target = {
				this = character:Segrellion_60
			}
		}

		save_scope_as = heritage_secret
	}


	# =====================================================
	# PUBLIC REVELATION
	# =====================================================

	if = {

		limit = {
			exists = scope:heritage_secret
		}


		# Hugh no longer needs any delayed-revelation loop.
		character:Segrellion_43 ?= {

			remove_character_flag = segrellion_hugh_waiting_to_reveal_daemon_parentage
			remove_character_flag = segrellion_hugh_withholding_daemon_parentage


			if = {

				limit = {
					this = scope:revealer
				}

				add_character_flag = told_true_parentage
				add_character_flag = segrellion_hugh_revealed_daemon_parentage
			}
		}


		character:Segrellion_60 = {

			add_character_flag = segrellion_daemon_parentage_known

			remove_character_flag = segrellion_daemon_parentage_known_daemon_only

			add_character_flag = segrellion_daemon_parentage_publicly_revealed


			# AGOT's royal-bastard revelation requires this
			# state before entering its public reveal handler.

			add_character_flag = royal_bastard
			add_character_flag = awaiting_bastardry_result


			# Hugh revealing Daemon's secret without Daemon
			# doing it himself carries AGOT's normal opinion
			# consequence.

			if = {

				limit = {
					this != scope:revealer
				}

				add_character_flag = segrellion_daemon_parentage_known_with_hugh

				add_opinion = {
					modifier = revealed_my_royal_secret_without
					target = scope:revealer
				}
			}


			# AGOT native public royal-bastard revelation.
			#
			# No delay. This prevents its generic decision
			# from briefly appearing between creation of the
			# Secret and its exposure.

			trigger_event = {
				id = agot_events_bastard.0950
			}
		}
	}
}
