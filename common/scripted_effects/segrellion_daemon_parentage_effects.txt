# =========================================================
# CREATE DAEMON'S REAL CK3 HERITAGE SECRET
#
# Root = character who learns the secret
# Target = Daemon Segrellion
#
# AGOT recognises secret_disputed_heritage for its
# royal bastard mechanics.
# =========================================================

segrellion_create_daemon_parentage_secret_effect = {

	if = {

		limit = {

			NOT = {

				any_known_secret = {

					secret_type = secret_disputed_heritage

					secret_target = {
						this = character:Segrellion_60
					}
				}
			}
		}

		add_secret = {
			type = secret_disputed_heritage
			target = character:Segrellion_60
		}
	}
}
