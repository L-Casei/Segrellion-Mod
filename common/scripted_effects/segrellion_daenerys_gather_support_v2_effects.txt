# =========================================================
# CLEAR ACTIVE NEGOTIATION
# ROOT = DAENERYS
# =========================================================

segrellion_daenerys_clear_support_negotiation_effect = {

	if = {

		limit = {
			has_variable = segrellion_daenerys_support_negotiator
		}


		# =================================================
		# STORE THE FINAL POLITICAL POSITION
		#
		# If no agreement was reached, this character has
		# explicitly declined Daenerys' cause and cannot
		# later opportunistically betray their liege to
		# join her through Call the Banners.
		# =================================================

		var:segrellion_daenerys_support_negotiator = {

			if = {

				limit = {
					has_character_flag = segrellion_support_agreement_reached_with_daenerys
				}

				remove_character_flag = segrellion_support_declined_daenerys
			}

			else = {

				add_character_flag = segrellion_support_declined_daenerys

				remove_character_flag = segrellion_support_agreement_reached_with_daenerys
			}


			remove_character_flag = segrellion_support_negotiation_reach
			remove_character_flag = segrellion_support_negotiation_westerlands
			remove_character_flag = segrellion_support_negotiation_dorne
			remove_character_flag = segrellion_support_negotiation_riverlands
			remove_character_flag = segrellion_support_negotiation_stormlands

			remove_character_flag = segrellion_support_demand_marriage
			remove_character_flag = segrellion_support_demand_favorable_contract
			remove_character_flag = segrellion_support_demand_hand
			remove_character_flag = segrellion_support_demand_council
			remove_character_flag = segrellion_support_demand_gold
			remove_character_flag = segrellion_support_demand_future_favor
		}
	}


	remove_variable = segrellion_daenerys_support_negotiator

	remove_character_flag = segrellion_daenerys_support_negotiation_pending
}


# =========================================================
# SECURE SUPPORT FROM CURRENT NEGOTIATOR
# ROOT = DAENERYS
# =========================================================

segrellion_daenerys_secure_current_support_negotiator_effect = {

	if = {

		limit = {
			has_variable = segrellion_daenerys_support_negotiator
		}


		# =================================================
		# WESTERLANDS
		# =================================================

		if = {

			limit = {

				var:segrellion_daenerys_support_negotiator = {
					has_character_flag = segrellion_support_negotiation_westerlands
				}
			}

			add_character_flag = segrellion_daenerys_secret_westerlands_support_against_aegon

			var:segrellion_daenerys_support_negotiator = {

				add_character_flag = segrellion_westerlands_secretly_supports_daenerys

				remove_character_flag = segrellion_westerlands_supports_aegon_against_daenerys

				add_character_flag = segrellion_support_agreement_reached_with_daenerys
			}
		}


		# =================================================
		# DORNE
		# =================================================

		else_if = {

			limit = {

				var:segrellion_daenerys_support_negotiator = {
					has_character_flag = segrellion_support_negotiation_dorne
				}
			}

			add_character_flag = segrellion_daenerys_secret_dornish_support_against_aegon

			var:segrellion_daenerys_support_negotiator = {

				add_character_flag = segrellion_dorne_secretly_supports_daenerys

				remove_character_flag = segrellion_dorne_supports_young_griff

				add_character_flag = segrellion_support_agreement_reached_with_daenerys
			}
		}


		# =================================================
		# RIVERLANDS
		# =================================================

		else_if = {

			limit = {

				var:segrellion_daenerys_support_negotiator = {
					has_character_flag = segrellion_support_negotiation_riverlands
				}
			}

			add_character_flag = segrellion_daenerys_riverlands_support_secured

			var:segrellion_daenerys_support_negotiator = {

				add_character_flag = segrellion_riverlands_supports_daenerys

				add_character_flag = segrellion_support_agreement_reached_with_daenerys
			}
		}


		# =================================================
		# STORMLANDS
		# =================================================

		else_if = {

			limit = {

				var:segrellion_daenerys_support_negotiator = {
					has_character_flag = segrellion_support_negotiation_stormlands
				}
			}

			if = {

				limit = {

					var:segrellion_daenerys_support_negotiator = {
						this = character:Vladwyn_5
					}
				}

				segrellion_daenerys_secure_joffrey_support_effect = yes
			}

			else = {

				add_character_flag = segrellion_daenerys_stormlands_support_secured

				var:segrellion_daenerys_support_negotiator = {

					add_character_flag = segrellion_stormlands_supports_daenerys

					add_character_flag = segrellion_support_agreement_reached_with_daenerys
				}
			}
		}

		var:segrellion_daenerys_support_negotiator = {

			add_character_flag = segrellion_support_agreement_reached_with_daenerys

			remove_character_flag = segrellion_support_declined_daenerys
		}
	}
}

# ROOT = DAENERYS

	add_character_flag = segrellion_daenerys_support_marriage_already_promised
