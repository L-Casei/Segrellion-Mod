# =========================================================
# HOUSE STORMFLAME AT THE BEACON
# =========================================================

segrellion_gather_stormflames_at_beacon_effect = {

	# Leo remains personally at the Beacon.
	character:Stormflame_3 ?= {

		if = {
			limit = {
				is_alive = yes
			}

			set_knight_status = forbid

			set_location = title:b_castle_farring.title_province

			# If CK3 somehow assigned Leo to an army,
			# remove him immediately.
			every_army = {

				if = {
					limit = {

						army_commander ?= {
							this = character:Stormflame_3
						}
					}

					remove_commander = yes
				}
			}
		}
	}


	# Gather the entire living Stormflame household.
	house:house_Stormflame = {

		every_house_member = {

			limit = {
				is_alive = yes
			}

			# They are not to be randomly pulled out as knights.
			set_knight_status = forbid

			if = {
				limit = {
					is_ruler = no
					this != character:Stormflame_13
				}

				set_employer = character:Stormflame_3
				set_location = title:b_castle_farring.title_province
			}
		}
	}


	# Clifford is the deliberate exception.
	# He rides out to command Leo's field army.
	character:Stormflame_13 ?= {

		if = {
			limit = {
				is_alive = yes
			}

			set_employer = character:Stormflame_3
		}
	}
}


# =========================================================
# CAPTURE LEO STORMFLAME AND HIS IMMEDIATE FAMILY
#
# Daemon takes:
# - Leo
# - every living child of Leo
# - Clifford explicitly, if still alive
# =========================================================

segrellion_capture_stormflame_court_effect = {


	# =====================================================
	# LEO
	# =====================================================

	character:Stormflame_3 ?= {

		if = {

			limit = {

				is_alive = yes

				is_imprisoned = no
			}


			add_character_flag = {
				flag = block_imprisonment_event
				days = 2
			}


			character:Segrellion_60 = {

				imprison = {

					target = character:Stormflame_3

					type = dungeon
				}
			}
		}
	}


	# =====================================================
	# LEO'S CHILDREN
	# =====================================================

	character:Stormflame_3 ?= {

		every_child = {

			limit = {

				is_alive = yes

				is_imprisoned = no
			}


			save_scope_as = segrellion_stormflame_child_capture


			add_character_flag = {
				flag = block_imprisonment_event
				days = 2
			}


			character:Segrellion_60 = {

				imprison = {

					target = scope:segrellion_stormflame_child_capture

					type = dungeon
				}
			}
		}
	}


	# =====================================================
	# CLIFFORD - EXPLICIT FAILSAFE
	# =====================================================

	character:Stormflame_13 ?= {

		if = {

			limit = {

				is_alive = yes

				is_imprisoned = no
			}


			add_character_flag = {
				flag = block_imprisonment_event
				days = 2
			}


			character:Segrellion_60 = {

				imprison = {

					target = character:Stormflame_13

					type = dungeon
				}
			}
		}
	}
}


# =========================================================
# BREAK THE PROMISE
# DAEMON EXECUTES EVERY SURVIVING STORMFLAME
# =========================================================

segrellion_execute_all_stormflame_captives_effect = {

	character:Segrellion_60 = {
		add_character_flag = segrellion_broke_beacon_mercy_oath
	}

	house:house_Stormflame = {

		every_house_member = {

			limit = {
				is_alive = yes
			}

			death = {
				death_reason = death_execution
				killer = character:Segrellion_60
			}
		}
	}
}


# =========================================================
# HONOR THE PROMISE
# LEO DIES, HIS FAMILY IS RELEASED
# =========================================================

segrellion_execute_leo_spare_stormflame_family_effect = {

	character:Segrellion_60 = {
		add_character_flag = segrellion_honored_beacon_mercy_oath
	}


	# =====================================================
	# SURVIVING STORMFLAMES
	#
	# Adult men take the black.
	# Women and children are released.
	#
	# This happens BEFORE Leo dies so the adult men are
	# removed from normal succession before his death.
	# =====================================================

	house:house_Stormflame = {

		every_house_member = {

			limit = {
				is_alive = yes
				this != character:Stormflame_3
			}


			# -------------------------------------------------
			# ADULT MEN -> NIGHT'S WATCH
			# -------------------------------------------------

			if = {

				limit = {
					is_male = yes
					is_adult = yes
				}


				add_character_flag = segrellion_sent_to_wall_by_daemon


				if = {
					limit = {
						is_imprisoned = yes
					}

					release_from_prison = yes
				}


				if = {
					limit = {
						NOT = {
							has_trait = nightswatch
						}
					}

					add_trait = nightswatch
				}


				if = {
					limit = {
						is_ruler = no

						title:k_the_wall.holder ?= {
							is_alive = yes
						}
					}

					set_employer = title:k_the_wall.holder
				}
			}


			# -------------------------------------------------
			# WOMEN AND CHILDREN -> RELEASED
			# -------------------------------------------------

			else = {

				add_character_flag = segrellion_spared_by_daemon_at_beacon

				if = {
					limit = {
						is_imprisoned = yes
					}

					release_from_prison = yes
				}
			}
		}
	}


	# =====================================================
	# LEO PAYS THE PRICE
	# =====================================================

	character:Stormflame_3 ?= {

		if = {
			limit = {
				is_alive = yes
			}

			death = {
				death_reason = death_execution
				killer = character:Segrellion_60
			}
		}
	}
}
