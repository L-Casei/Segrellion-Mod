# SEGRELLION - RICKON STARK EDUCATION
# =========================================================

# =========================================================
# AGOT RANDOM PERSONALITY LOCK
#
# AGOT's child_personality_gain on_action explicitly refuses
# to roll random personality events while agot_planned_traits
# is above 0. We use that native gate only while Rickon's
# custom education is queued / active.
# =========================================================

segrellion_rickon_lock_agot_random_personality_effect = {

	if = {

		limit = {

			this = character:Stark_Dummy_11
			is_alive = yes
			is_adult = no

			OR = {
				has_character_flag = segrellion_rickon_education_start_queued
				has_character_flag = segrellion_rickon_education_chain_active
			}
		}


		if = {

			limit = {

				OR = {
					NOT = { has_variable = agot_planned_traits }
					var:agot_planned_traits <= 0
				}
			}


			set_variable = {
				name = agot_planned_traits
				value = 99
			}

			add_character_flag = segrellion_rickon_owns_agot_planned_traits_lock
		}


		else_if = {

			limit = {

				has_character_flag = segrellion_rickon_owns_agot_planned_traits_lock
				var:agot_planned_traits < 50
			}


			set_variable = {
				name = agot_planned_traits
				value = 99
			}
		}
	}
}


segrellion_rickon_unlock_agot_random_personality_effect = {

	if = {

		limit = {
			has_character_flag = segrellion_rickon_owns_agot_planned_traits_lock
		}


		if = {

			limit = {
				has_variable = agot_planned_traits
			}

			remove_variable = agot_planned_traits
		}


		remove_character_flag = segrellion_rickon_owns_agot_planned_traits_lock
	}
}


# =========================================================
# TRY TO START THE STORY EDUCATION NOW
#
# This is deliberately separate from the repeating watchdog.
# The old version could already have education_watch_running
# from game start, causing the Beacon handoff to do nothing.
# =========================================================

segrellion_rickon_try_start_story_education_effect = {

	if = {

		limit = {

			this = character:Stark_Dummy_11
			is_alive = yes
			is_adult = no

			NOT = { has_character_flag = segrellion_rickon_education_chain_active }
			NOT = { has_character_flag = segrellion_rickon_education_chain_ended }
			NOT = { has_character_flag = segrellion_rickon_education_start_queued }
		}


		if = {

			limit = {

				has_character_flag = segrellion_rickon_under_daemon_protection

				character:Segrellion_60 = {
					is_alive = yes
					is_adult = yes
					is_imprisoned = no
					is_incapable = no
				}
			}


			add_character_flag = segrellion_rickon_education_start_queued

			segrellion_rickon_lock_agot_random_personality_effect = yes


			character:Segrellion_60 = {

				trigger_event = {
					id = segrellion_rickon_education.0001
					days = 1
				}
			}
		}


		else_if = {

			limit = {

				has_character_flag = segrellion_rickon_under_visenya_protection

				OR = {
					NOT = { exists = character:Segrellion_60 }
					character:Segrellion_60 = { is_alive = no }
				}

				character:Black_Unicorn_1 = {
					is_alive = yes
					is_adult = yes
					is_imprisoned = no
					is_incapable = no
				}
			}


			add_character_flag = segrellion_rickon_education_start_queued

			segrellion_rickon_lock_agot_random_personality_effect = yes


			character:Black_Unicorn_1 = {

				trigger_event = {
					id = segrellion_rickon_education.0001
					days = 1
				}
			}
		}
	}
}


segrellion_rickon_ensure_education_watch_effect = {

	if = {

		limit = {

			this = character:Stark_Dummy_11
			is_alive = yes
			is_adult = no

			NOT = { has_character_flag = segrellion_rickon_education_chain_ended }
		}


		# Always attempt the immediate handoff. This is the key fix:
		# an already-running watchdog must never swallow the start event.
		segrellion_rickon_try_start_story_education_effect = yes


		if = {

			limit = {
				NOT = { has_character_flag = segrellion_rickon_education_watch_running }
			}


			add_character_flag = segrellion_rickon_education_watch_running

			trigger_event = {
				id = segrellion_rickon_education.0000
				days = 30
			}
		}


		if = {

			limit = {

				OR = {
					has_character_flag = segrellion_rickon_education_start_queued
					has_character_flag = segrellion_rickon_education_chain_active
				}
			}


			segrellion_rickon_lock_agot_random_personality_effect = yes
		}
	}
}

segrellion_rickon_award_story_trait_effect = {
	add_trait_force_tooltip = $TRAIT$
	add_character_flag = $FLAG$
}

segrellion_rickon_trigger_lesson_event_effect = {
	if = {
		limit = {
			has_character_flag = segrellion_rickon_tutor_daemon
			has_relation_guardian = character:Segrellion_60
			character:Segrellion_60 = { is_alive = yes is_imprisoned = no is_incapable = no }
		}
		character:Segrellion_60 = { trigger_event = { id = $EVENT$ days = 1 } }
	}
	else_if = {
		limit = {
			has_character_flag = segrellion_rickon_tutor_visenya
			has_relation_guardian = character:Black_Unicorn_1
			character:Black_Unicorn_1 = { is_alive = yes is_imprisoned = no is_incapable = no }
		}
		character:Black_Unicorn_1 = { trigger_event = { id = $EVENT$ days = 1 } }
	}
	else = {
		trigger_event = { id = $RETRY_EVENT$ days = 90 }
	}
}

segrellion_rickon_queue_first_lesson_effect = {
	if = {
		limit = { number_of_personality_traits >= 3 }
		add_character_flag = segrellion_rickon_personality_lessons_completed
	}
	else_if = {
		limit = { number_of_personality_traits >= 2 }
		add_character_flag = segrellion_rickon_lesson_stage_1_completed
		add_character_flag = segrellion_rickon_lesson_stage_2_completed
		trigger_event = { id = segrellion_rickon_education.0300 days = { 180 300 } }
	}
	else_if = {
		limit = { number_of_personality_traits >= 1 }
		add_character_flag = segrellion_rickon_lesson_stage_1_completed
		trigger_event = { id = segrellion_rickon_education.0200 days = { 120 240 } }
	}
	else = {
		trigger_event = { id = segrellion_rickon_education.0100 days = { 120 240 } }
	}
}

segrellion_rickon_end_education_chain_effect = {

	segrellion_rickon_unlock_agot_random_personality_effect = yes

	remove_character_flag = segrellion_rickon_education_chain_active
	remove_character_flag = segrellion_rickon_education_start_queued
	remove_character_flag = segrellion_rickon_tutor_daemon
	remove_character_flag = segrellion_rickon_tutor_visenya
	remove_character_flag = segrellion_rickon_education_watch_running
	add_character_flag = segrellion_rickon_education_chain_ended
}
