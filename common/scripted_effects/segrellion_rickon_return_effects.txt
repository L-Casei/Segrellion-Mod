# =========================================================
# SEGRELLION - RICKON RETURN TO THE NORTH
# =========================================================

# Monthly global progression. This is deliberately character-explicit
# so it repairs existing saves as well as new games.
segrellion_rickon_return_progression = {
	character:Stark_Dummy_11 ?= {
		if = {
			limit = {
				is_alive = yes
				OR = {
					has_character_flag = segrellion_rickon_under_daemon_protection
					has_character_flag = segrellion_rickon_under_visenya_protection
					has_character_flag = segrellion_rickon_black_unicorn_route
					has_character_flag = segrellion_rickon_sheltered_with_sansa_at_beacon
					has_character_flag = segrellion_rickon_with_sansa_riverlands
					has_character_flag = segrellion_rickon_kept_as_beacon_ward
					has_character_flag = segrellion_rickon_travelling_north
					has_character_flag = segrellion_rickon_waiting_white_harbor
				}
				NOT = { has_character_flag = segrellion_rickon_north_return_resolved }
				NOT = { has_character_flag = segrellion_rickon_return_watch_queued }
			}

			add_character_flag = {
				flag = segrellion_rickon_return_watch_queued
				days = 40
			}

			trigger_event = {
				id = segrellion_rickon_return.0000
				days = 1
			}
		}
	}
}

# ROOT = Rickon
segrellion_rickon_pause_story_education_effect = {
	if = {
		limit = {
			OR = {
				has_character_flag = segrellion_rickon_education_chain_active
				has_character_flag = segrellion_rickon_education_start_queued
				has_character_flag = segrellion_rickon_education_watch_running
			}
		}
		segrellion_rickon_end_education_chain_effect = yes
	}

	add_character_flag = segrellion_rickon_education_paused_by_departure

	if = {
		limit = { has_relation_guardian = character:Segrellion_60 }
		remove_relation_guardian = character:Segrellion_60
	}
	if = {
		limit = { has_relation_guardian = character:Black_Unicorn_1 }
		remove_relation_guardian = character:Black_Unicorn_1
	}
}

# ROOT = Rickon
segrellion_rickon_resume_daemon_education_effect = {
	if = {
		limit = {
			is_alive = yes
			is_adult = no
			character:Segrellion_60 = {
				is_alive = yes
				has_title = title:e_flamelands
			}
		}

		remove_character_flag = segrellion_rickon_education_chain_ended
		remove_character_flag = segrellion_rickon_education_paused_by_departure
		remove_character_flag = segrellion_rickon_tutor_visenya
		add_character_flag = segrellion_rickon_under_daemon_protection

		set_relation_guardian = character:Segrellion_60
		segrellion_rickon_ensure_education_watch_effect = yes
	}
}

# ROOT = Rickon
segrellion_rickon_resume_visenya_education_effect = {
	if = {
		limit = {
			is_alive = yes
			is_adult = no
			character:Black_Unicorn_1 = {
				is_alive = yes
				has_title = title:e_flamelands
			}
		}

		remove_character_flag = segrellion_rickon_education_chain_ended
		remove_character_flag = segrellion_rickon_education_paused_by_departure
		remove_character_flag = segrellion_rickon_tutor_daemon
		add_character_flag = segrellion_rickon_under_visenya_protection

		set_relation_guardian = character:Black_Unicorn_1
		segrellion_rickon_ensure_education_watch_effect = yes
	}
}

# ROOT = Rickon
# SENDER = character:Segrellion_60 or character:Black_Unicorn_1
segrellion_rickon_begin_north_journey_effect = {
	segrellion_rickon_pause_story_education_effect = yes

	remove_character_flag = segrellion_rickon_kept_as_beacon_ward
	remove_character_flag = segrellion_rickon_sheltered_with_sansa_at_beacon
	remove_character_flag = segrellion_rickon_with_sansa_riverlands
	remove_character_flag = segrellion_rickon_publicly_with_daemon

	add_character_flag = segrellion_rickon_travelling_north
	add_character_flag = segrellion_rickon_return_accepted_by_jon

	if = {
		limit = { $SENDER$ = character:Segrellion_60 }
		add_character_flag = segrellion_rickon_sent_north_by_daemon
		remove_character_flag = segrellion_rickon_sent_north_by_visenya
	}
	else = {
		add_character_flag = segrellion_rickon_sent_north_by_visenya
		remove_character_flag = segrellion_rickon_sent_north_by_daemon
	}

	character:Stark_hh_13 ?= {
		if = {
			limit = { is_alive = yes }
			add_character_flag = segrellion_osha_escorting_rickon_north
		}
	}

	trigger_event = {
		id = segrellion_rickon_return.0040
		days = { 45 75 }
	}
}

# ROOT = Rickon
segrellion_rickon_arrive_white_harbor_effect = {
	remove_character_flag = segrellion_rickon_travelling_north
	add_character_flag = segrellion_rickon_waiting_white_harbor

	set_location = title:c_white_harbor.title_province

	title:c_white_harbor.holder ?= {
		if = {
			limit = { is_alive = yes }
			add_courtier = root
		}
	}

	character:Stark_hh_13 ?= {
		if = {
			limit = { is_alive = yes }
			set_location = title:c_white_harbor.title_province
			title:c_white_harbor.holder ?= {
				if = {
					limit = { is_alive = yes }
					add_courtier = character:Stark_hh_13
				}
			}
		}
	}
}

# ROOT = Rickon
segrellion_rickon_arrive_winterfell_effect = {
	remove_character_flag = segrellion_rickon_waiting_white_harbor
	remove_character_flag = segrellion_rickon_travelling_north

	title:e_the_north.holder = {
		add_courtier = root
	}

	set_location = title:c_winterfell.title_province

	character:Stark_hh_13 ?= {
		if = {
			limit = { is_alive = yes }
			title:e_the_north.holder = {
				add_courtier = character:Stark_hh_13
			}
			set_location = title:c_winterfell.title_province
			remove_character_flag = segrellion_osha_escorting_rickon_north
			add_character_flag = segrellion_osha_arrived_winterfell_with_rickon
		}
	}

	add_character_flag = segrellion_rickon_arrived_winterfell
	add_character_flag = segrellion_rickon_north_return_resolved
}

# ROOT = Rickon
segrellion_rickon_join_sansa_riverlands_effect = {
	segrellion_rickon_pause_story_education_effect = yes

	remove_character_flag = segrellion_rickon_sheltered_with_sansa_at_beacon
	add_character_flag = segrellion_rickon_with_sansa_riverlands

	character:Stark_Dummy_8 = {
		add_courtier = root
	}

	if = {
		limit = { exists = character:Stark_Dummy_8.capital_province }
		set_location = character:Stark_Dummy_8.capital_province
	}

	character:Stark_hh_13 ?= {
		if = {
			limit = { is_alive = yes }
			character:Stark_Dummy_8 = {
				add_courtier = character:Stark_hh_13
			}
			if = {
				limit = { exists = character:Stark_Dummy_8.capital_province }
				set_location = character:Stark_Dummy_8.capital_province
			}
		}
	}
}

# ROOT = Rickon
segrellion_rickon_return_to_beacon_from_sansa_effect = {
	remove_character_flag = segrellion_rickon_with_sansa_riverlands
	add_character_flag = segrellion_rickon_sheltered_with_sansa_at_beacon

	set_location = title:b_castle_farring.title_province

	if = {
		limit = {
			character:Segrellion_60 = {
				is_alive = yes
				has_title = title:e_flamelands
			}
		}

		character:Segrellion_60 = {
			add_courtier = root
		}
		segrellion_rickon_resume_daemon_education_effect = yes

		character:Stark_hh_13 ?= {
			if = {
				limit = { is_alive = yes }
				character:Segrellion_60 = {
					add_courtier = character:Stark_hh_13
				}
				set_location = title:b_castle_farring.title_province
			}
		}
	}

	else_if = {
		limit = {
			character:Black_Unicorn_1 = {
				is_alive = yes
				has_title = title:e_flamelands
			}
		}

		character:Black_Unicorn_1 = {
			add_courtier = root
		}
		segrellion_rickon_resume_visenya_education_effect = yes

		character:Stark_hh_13 ?= {
			if = {
				limit = { is_alive = yes }
				character:Black_Unicorn_1 = {
					add_courtier = character:Stark_hh_13
				}
				set_location = title:b_castle_farring.title_province
			}
		}
	}
}
