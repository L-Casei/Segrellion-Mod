# =========================================================
# BREAK THE POLITICAL ALLIANCE WITH DAEMON
# =========================================================

segrellion_young_griff_break_daemon_alliance_effect = {

    if = {

        limit = {

            character:Segrellion_60 ?= {
                is_alive = yes
            }

            is_allied_to = character:Segrellion_60
        }


        break_alliance = character:Segrellion_60
    }


    # The Pentos Compact is over.
    # The Deadly Triarchy cannot survive it.

    segrellion_dissolve_deadly_triarchy_effect = yes
}


# =========================================================
# LEAVE DAEMON'S RESTORATION
#
# If The Unicorn Returns has not begun yet:
# removing the support flag prevents Aegon from joining.
#
# If the war is already active:
# Aegon is removed from the war immediately.
# =========================================================

segrellion_young_griff_leave_daemon_restoration_effect = {

	remove_character_flag = segrellion_young_griff_supports_daemon
	remove_character_flag = segrellion_young_griff_waiting_for_daemon

	add_character_flag = segrellion_young_griff_free_for_iron_throne


	# If Aegon is already participating in The Unicorn Returns,
	# remove him from that war.

	character:Segrellion_60 ?= {

		var:segrellion_unicorn_returns_war ?= {

			remove_participant = root
		}
	}
}


# =========================================================
# REMAIN WITH DAEMON
#
# Aegon keeps his military commitment and therefore waits
# for The Unicorn Returns to end before pursuing his throne.
# =========================================================

segrellion_young_griff_stay_with_daemon_restoration_effect = {

	remove_character_flag = segrellion_young_griff_free_for_iron_throne

	add_character_flag = segrellion_young_griff_supports_daemon
	add_character_flag = segrellion_young_griff_waiting_for_daemon
}

# =========================================================
# THE UNICORN RETURNS HAS ENDED
#
# IMPORTANT:
#
# PATH 1 does NOT care about this anymore.
# Aegon abandoned Daemon and follows his own six-month
# throne route through 7900 -> 7910.
#
# PATH 2 waited for Daemon and therefore continues here.
# =========================================================

segrellion_young_griff_after_unicorn_returns_effect = {

	# Persistent world-state marker on Daemon.

	if = {

		limit = {
			this = character:Segrellion_60
		}

		add_character_flag = segrellion_unicorn_returns_resolved
	}


	character:IAF_Young_Griff_1 ?= {

		if = {

			limit = {

				is_alive = yes

				has_character_flag = segrellion_young_griff_canon_path_2
			}


			add_character_flag = segrellion_young_griff_daemon_war_resolved

			remove_character_flag = segrellion_young_griff_waiting_for_daemon

			add_character_flag = segrellion_young_griff_free_for_iron_throne


			# =================================================
			# DAEMON WON
			#
			# Aegon helped him, so resolve the blood debt first.
			# =================================================

			if = {

				limit = {

					has_character_flag = segrellion_young_griff_supports_daemon


					character:Segrellion_60 ?= {

						is_alive = yes

						has_character_flag = segrellion_unicorn_returns_completed

						has_title = title:e_flamelands
					}
				}


				if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_young_griff_blood_debt_event_scheduled
						}
					}


					add_character_flag = segrellion_young_griff_blood_debt_event_scheduled


					trigger_event = {

						id = segrellion_young_griff.7000

						days = 1
					}
				}
			}


			# =================================================
			# DAEMON LOST OR DIED
			#
			# There is no blood debt to resolve.
			# Varys immediately takes over the storyline.
			# =================================================

			else_if = {

				limit = {

					OR = {

						NOT = {

							character:Segrellion_60 = {
								is_alive = yes
							}
						}


						character:Segrellion_60 = {

							NOT = {
								has_title = title:e_flamelands
							}
						}
					}
				}


				segrellion_young_griff_try_schedule_path_2_varys_effect = yes
			}
		}
	}
}

# =========================================================
# YOUNG GRIFF - THRONE STORY REPAIR
#
# PATH 1:
# Independent six-month route after abandoning Daemon.
#
# PATH 2:
# Waits for The Unicorn Returns and then Varys.
# =========================================================

segrellion_young_griff_force_iron_throne_progression_effect = {


	# =====================================================
	# PATH 2 - REPAIR STATE AFTER UNICORN RETURNS
	# =====================================================

	if = {

		limit = {

			character:Segrellion_60 ?= {

				has_character_flag = segrellion_unicorn_returns_resolved
			}


			character:IAF_Young_Griff_1 ?= {

				is_alive = yes

				has_character_flag = segrellion_young_griff_canon_path_2


				NOT = {
					has_character_flag = segrellion_young_griff_daemon_war_resolved
				}
			}
		}


		character:IAF_Young_Griff_1 = {

			add_character_flag = segrellion_young_griff_daemon_war_resolved

			remove_character_flag = segrellion_young_griff_waiting_for_daemon

			add_character_flag = segrellion_young_griff_free_for_iron_throne
		}
	}


	character:IAF_Young_Griff_1 ?= {


		# =================================================
		# PATH 1
		#
		# Stannis/Daenerys split.
		# If the original 7900 schedule was lost, restore it.
		# =================================================

		if = {

			limit = {

				is_alive = yes
				is_imprisoned = no

				has_character_flag = segrellion_young_griff_canon_path_1

				has_character_flag = segrellion_young_griff_free_for_iron_throne


				OR = {

					has_character_flag = segrellion_young_griff_left_daemon_stannis

					has_character_flag = segrellion_young_griff_left_daemon_daenerys
				}


				NOT = {
					has_character_flag = segrellion_young_griff_independent_throne_path_scheduled
				}

				NOT = {
					has_character_flag = segrellion_young_griff_independent_throne_path_started
				}

				NOT = {
					has_character_flag = segrellion_young_griff_iron_throne_war_started
				}

				NOT = {
					has_title = title:h_the_iron_throne
				}
			}


			add_character_flag = segrellion_young_griff_independent_throne_path_scheduled


			trigger_event = {

				id = segrellion_young_griff.7900

				months = 6
			}
		}


		# =================================================
		# PATH 2
		# =================================================

		if = {

			limit = {

				is_alive = yes

				has_character_flag = segrellion_young_griff_canon_path_2

				has_character_flag = segrellion_young_griff_daemon_war_resolved
			}


			segrellion_young_griff_try_schedule_path_2_varys_effect = yes
		}
	}
}


# =========================================================
# YOUNG GRIFF FIGHTS FOR DAENERYS' THRONE
# =========================================================

segrellion_start_daenerys_throne_war_effect = {

	if = {
		limit = {

			this = character:IAF_Young_Griff_1

			is_alive = yes

			character:Targaryen_9 ?= {
				is_alive = yes
			}

			title:h_the_iron_throne ?= {
				exists = holder
				holder = {
					is_alive = yes
				}
			}

			NOT = {
				title:h_the_iron_throne.holder = character:Targaryen_9
			}
		}


		# Daenerys' normal scripted invasion removes this claim
		# to prevent vanilla wars. Restore it for Aegon's pact.

		character:Targaryen_9 = {

			if = {
				limit = {
					NOT = {
						has_claim_on = title:h_the_iron_throne
					}
				}

				add_pressed_claim = title:h_the_iron_throne
			}

			if = {
				limit = {
					has_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
				}

				remove_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
			}
		}


		title:h_the_iron_throne.holder = {
			save_scope_as = young_griff_daenerys_throne_defender
		}


		start_war = {
			cb = claim_cb
			target = scope:young_griff_daenerys_throne_defender
			claimant = character:Targaryen_9
			target_title = title:h_the_iron_throne
		}


		add_character_flag = segrellion_young_griff_fighting_for_daenerys
	}


	else_if = {
		limit = {
			title:h_the_iron_throne.holder = character:Targaryen_9
		}

		segrellion_complete_young_griff_daenerys_pact_effect = yes
	}
}


# =========================================================
# MARRY ONLY AFTER DAENERYS HAS THE IRON THRONE
# =========================================================

segrellion_complete_young_griff_daenerys_pact_effect = {

	if = {
		limit = {

			title:h_the_iron_throne.holder = character:Targaryen_9

			character:IAF_Young_Griff_1 ?= {
				is_alive = yes
				is_married = no
				has_character_flag = segrellion_young_griff_daenerys_throne_pact
			}

			character:Targaryen_9 ?= {
				is_alive = yes
				is_married = no
				has_character_flag = segrellion_daenerys_young_griff_throne_pact
			}
		}

		character:IAF_Young_Griff_1 = {

			marry = character:Targaryen_9

			add_character_flag = segrellion_young_griff_daenerys_pact_fulfilled
			remove_character_flag = segrellion_young_griff_fighting_for_daenerys
		}

		character:Targaryen_9 = {

			add_character_flag = segrellion_daenerys_young_griff_pact_fulfilled
		}
	}
}

# =========================================================
# VARYS' FINAL PLAN - BEGIN THE IRON THRONE WAR
#
# Shared by Path 1 and Path 2 despite the legacy effect name.
# =========================================================

segrellion_young_griff_start_path_1_throne_war_effect = {

	if = {

		limit = {

			this = character:IAF_Young_Griff_1

			is_alive = yes
			is_imprisoned = no


			NOT = {
				has_title = title:h_the_iron_throne
			}


			title:h_the_iron_throne ?= {

				exists = holder

				holder = {

					is_alive = yes

					this != root
				}
			}
		}


		# =============================================
		# CLAIM
		# =============================================

		if = {

			limit = {

				NOT = {
					has_claim_on = title:h_the_iron_throne
				}
			}


			add_pressed_claim = title:h_the_iron_throne
		}


		# =============================================
		# DEFENDER
		# =============================================

		title:h_the_iron_throne.holder = {

			save_scope_as = segrellion_young_griff_throne_defender
		}


		# =============================================
		# START WAR
		# =============================================

		start_war = {

			cb = claim_cb

			target = scope:segrellion_young_griff_throne_defender

			claimant = root

			target_title = title:h_the_iron_throne
		}


		# =============================================
		# PATH MARKER
		# =============================================

		if = {

			limit = {
				has_character_flag = segrellion_young_griff_canon_path_1
			}

			add_character_flag = segrellion_young_griff_path_1_throne_war_declared
		}


		else_if = {

			limit = {
				has_character_flag = segrellion_young_griff_canon_path_2
			}

			add_character_flag = segrellion_young_griff_path_2_throne_war_declared
		}


		add_character_flag = segrellion_young_griff_varys_final_plan_active


		trigger_event = {

			id = segrellion_young_griff.6010

			days = 1
		}
	}
}

# =========================================================
# YOUNG GRIFF - 300 AC WESTEROS CORE SETUP
#
# Permanent starting entourage:
# - Jon Connington at Storm's End
# - Golden Company already landed in Westeros
#
# Idempotent and safe for existing saves.
# =========================================================

segrellion_young_griff_westeros_core_setup_effect = {

	if = {

		limit = {

			this = character:IAF_Young_Griff_1

			is_alive = yes

			game_start_date >= 8300.8.20
			game_start_date < 8301.1.1
		}


		# =================================================
		# JON CONNINGTON
		# =================================================

		segrellion_young_griff_place_jon_connington_effect = yes


		character:Connington_4 ?= {

			if = {

				limit = {
					is_alive = yes
				}


				set_location = title:c_storms_end.title_province

				add_character_flag = segrellion_jon_connington_at_storms_end
			}
		}


		# =================================================
		# THE GOLDEN COMPANY
		#
		# 10,000 permanent special troops belonging to Aegon.
		#
		# They are NOT tied to one specific war, so they can
		# fight for him in The Unicorn Returns, the Iron
		# Throne war and every future conflict.
		# =================================================

		if = {

			limit = {

				NOT = {
					has_character_flag = segrellion_young_griff_golden_company_setup
				}
			}


			spawn_army = {

				name = segrellion_golden_company_host


				men_at_arms = {
					type = armored_footmen
					men = 3500
				}


				men_at_arms = {
					type = pikemen_unit
					men = 2500
				}


				men_at_arms = {
					type = bowmen
					men = 2500
				}


				men_at_arms = {
					type = armored_horsemen
					men = 1500
				}


				location = title:c_storms_end.title_province

				origin = title:c_storms_end.title_province


				# Not tied to a particular war.
				inheritable = no

				uses_supply = yes
			}


			add_character_flag = segrellion_young_griff_golden_company_setup

			add_character_flag = segrellion_young_griff_golden_company_in_westeros
		}
	}
}
