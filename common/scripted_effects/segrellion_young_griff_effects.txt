# =========================================================
# BREAK THE POLITICAL ALLIANCE WITH DAEMON
# =========================================================

segrellion_young_griff_break_daemon_alliance_effect = {

    if = {

        limit = {

            character:Segrellion_60 ?= {
                is_alive = yes
            }

            is_allied_to = character:Segrellion_60
        }


        break_alliance = character:Segrellion_60
    }


    # The Pentos Compact is over.
    # The Deadly Triarchy cannot survive it.

    segrellion_dissolve_deadly_triarchy_effect = yes
}


# =========================================================
# LEAVE DAEMON'S RESTORATION
#
# If The Unicorn Returns has not begun yet:
# removing the support flag prevents Aegon from joining.
#
# If the war is already active:
# Aegon is removed from the war immediately.
# =========================================================

segrellion_young_griff_leave_daemon_restoration_effect = {

	remove_character_flag = segrellion_young_griff_supports_daemon
	remove_character_flag = segrellion_young_griff_waiting_for_daemon

	add_character_flag = segrellion_young_griff_free_for_iron_throne


	# If Aegon is already participating in The Unicorn Returns,
	# remove him from that war.

	character:Segrellion_60 ?= {

		var:segrellion_unicorn_returns_war ?= {

			remove_participant = root
		}
	}
}


# =========================================================
# REMAIN WITH DAEMON
#
# Aegon keeps his military commitment and therefore waits
# for The Unicorn Returns to end before pursuing his throne.
# =========================================================

segrellion_young_griff_stay_with_daemon_restoration_effect = {

	remove_character_flag = segrellion_young_griff_free_for_iron_throne

	add_character_flag = segrellion_young_griff_supports_daemon
	add_character_flag = segrellion_young_griff_waiting_for_daemon
}

# =========================================================
# THE UNICORN RETURNS HAS ENDED
#
# IMPORTANT:
#
# PATH 1 does NOT care about this anymore.
# Aegon abandoned Daemon and follows his own six-month
# throne route through 7900 -> 7910.
#
# PATH 2 waited for Daemon and therefore continues here.
# =========================================================

segrellion_young_griff_after_unicorn_returns_effect = {

	# =====================================================
	# PERSISTENT WORLD STATE
	# =====================================================

	if = {

		limit = {
			this = character:Segrellion_60
		}

		add_character_flag = segrellion_unicorn_returns_resolved
	}


	character:IAF_Young_Griff_1 ?= {

		if = {

			limit = {
				is_alive = yes
			}


			# =================================================
			# REPAIR MISSING PATH
			#
			# If Aegon is still supporting Daemon when
			# The Unicorn Returns ends, then politically he
			# followed Path 2.
			#
			# Some reinforcement routes, especially the Vale,
			# previously forgot to set this flag.
			# =================================================

			if = {

				limit = {

					has_character_flag = segrellion_young_griff_supports_daemon

					NOR = {
						has_character_flag = segrellion_young_griff_canon_path_1
						has_character_flag = segrellion_young_griff_canon_path_2
					}
				}


				add_character_flag = segrellion_young_griff_canon_path_2

				remove_character_flag = segrellion_young_griff_canon_path_1
			}


			# =================================================
			# PATH 2
			# =================================================

			if = {

				limit = {
					has_character_flag = segrellion_young_griff_canon_path_2
				}


				# Path 2 means Aegon remained committed to
				# Daemon's restoration.

				if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_young_griff_supports_daemon
						}
					}


					add_character_flag = segrellion_young_griff_supports_daemon
				}


				add_character_flag = segrellion_young_griff_daemon_war_resolved

				remove_character_flag = segrellion_young_griff_waiting_for_daemon

				add_character_flag = segrellion_young_griff_free_for_iron_throne


				# =================================================
				# DAEMON WON
				# =================================================

				if = {

					limit = {

						character:Segrellion_60 ?= {

							is_alive = yes

							has_character_flag = segrellion_unicorn_returns_completed

							has_title = title:e_flamelands
						}
					}


					# ---------------------------------------------
					# BLOOD DEBT NOT YET RESOLVED
					# ---------------------------------------------

					if = {

						limit = {

							NOT = {
								has_character_flag = segrellion_young_griff_blood_debt_resolved
							}

							NOT = {
								has_character_flag = segrellion_young_griff_blood_debt_event_scheduled
							}
						}


						add_character_flag = {
							flag = segrellion_young_griff_blood_debt_event_scheduled
							days = 10
						}


						trigger_event = {

							id = segrellion_young_griff.7000

							days = 1
						}
					}


					# ---------------------------------------------
					# BLOOD DEBT ALREADY RESOLVED
					# ---------------------------------------------

					else_if = {

						limit = {
							has_character_flag = segrellion_young_griff_blood_debt_resolved
						}


						segrellion_young_griff_try_schedule_path_2_varys_effect = yes
					}
				}


				# =================================================
				# DAEMON LOST / DIED
				# =================================================

				else = {

					segrellion_young_griff_try_schedule_path_2_varys_effect = yes
				}
			}
		}
	}
}


# =========================================================
# YOUNG GRIFF FIGHTS FOR DAENERYS' THRONE
# =========================================================

segrellion_start_daenerys_throne_war_effect = {

	if = {
		limit = {

			this = character:IAF_Young_Griff_1

			is_alive = yes

			character:Targaryen_9 ?= {
				is_alive = yes
			}

			title:h_the_iron_throne ?= {
				exists = holder
				holder = {
					is_alive = yes
				}
			}

			NOT = {
				title:h_the_iron_throne.holder = character:Targaryen_9
			}
		}


		# Daenerys' normal scripted invasion removes this claim
		# to prevent vanilla wars. Restore it for Aegon's pact.

		character:Targaryen_9 = {

			if = {
				limit = {
					NOT = {
						has_claim_on = title:h_the_iron_throne
					}
				}

				add_pressed_claim = title:h_the_iron_throne
			}

			if = {
				limit = {
					has_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
				}

				remove_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
			}
		}


		title:h_the_iron_throne.holder = {
			save_scope_as = young_griff_daenerys_throne_defender
		}


		segrellion_young_griff_open_scripted_war_window_effect = yes


		start_war = {
			cb = claim_cb
			target = scope:young_griff_daenerys_throne_defender
			claimant = character:Targaryen_9
			target_title = title:h_the_iron_throne
		}


		segrellion_young_griff_apply_scripted_war_lock_effect = yes


		add_character_flag = segrellion_young_griff_fighting_for_daenerys
	}


	else_if = {
		limit = {
			title:h_the_iron_throne.holder = character:Targaryen_9
		}

		segrellion_complete_young_griff_daenerys_pact_effect = yes
	}
}


# =========================================================
# MARRY ONLY AFTER DAENERYS HAS THE IRON THRONE
# =========================================================

segrellion_complete_young_griff_daenerys_pact_effect = {

	if = {
		limit = {

			title:h_the_iron_throne.holder = character:Targaryen_9

			character:IAF_Young_Griff_1 ?= {
				is_alive = yes
				is_married = no
				has_character_flag = segrellion_young_griff_daenerys_throne_pact
			}

			character:Targaryen_9 ?= {
				is_alive = yes
				is_married = no
				has_character_flag = segrellion_daenerys_young_griff_throne_pact
			}
		}

		character:IAF_Young_Griff_1 = {

			marry = character:Targaryen_9

			add_character_flag = segrellion_young_griff_daenerys_pact_fulfilled
			remove_character_flag = segrellion_young_griff_fighting_for_daenerys

			segrellion_young_griff_release_scripted_war_lock_effect = yes
		}

		character:Targaryen_9 = {

			add_character_flag = segrellion_daenerys_young_griff_pact_fulfilled
		}
	}
}

# =========================================================
# YOUNG GRIFF - THRONE STORY REPAIR
#
# PATH 1:
# Aegon abandoned Daemon.
# Six months later -> 7900 -> 7910 -> Iron Throne.
#
# PATH 2:
# Aegon remained with Daemon.
# Wait for Unicorn Returns -> debt -> Varys -> Iron Throne.
# =========================================================

segrellion_young_griff_force_iron_throne_progression_effect = {

	character:IAF_Young_Griff_1 ?= {

		if = {

			limit = {
				is_alive = yes
			}


			# =================================================
			# REPAIR MISSING CANON PATH
			#
			# If Aegon remained with Daemon until the
			# restoration was resolved, he belongs to Path 2.
			# =================================================

			if = {

				limit = {

					has_character_flag = segrellion_young_griff_supports_daemon

					NOR = {
						has_character_flag = segrellion_young_griff_canon_path_1
						has_character_flag = segrellion_young_griff_canon_path_2
					}

					character:Segrellion_60 ?= {

						OR = {
							has_character_flag = segrellion_unicorn_returns_completed
							has_character_flag = segrellion_unicorn_returns_resolved
						}
					}
				}


				add_character_flag = segrellion_young_griff_canon_path_2

				remove_character_flag = segrellion_young_griff_canon_path_1
			}


			# =================================================
			# PATH 2 - REPAIR POSTWAR STATE
			# =================================================

			if = {

				limit = {

					has_character_flag = segrellion_young_griff_canon_path_2

					character:Segrellion_60 ?= {

						OR = {
							has_character_flag = segrellion_unicorn_returns_completed
							has_character_flag = segrellion_unicorn_returns_resolved
						}
					}
				}


				if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_young_griff_supports_daemon
						}
					}


					add_character_flag = segrellion_young_griff_supports_daemon
				}


				if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_young_griff_daemon_war_resolved
						}
					}


					add_character_flag = segrellion_young_griff_daemon_war_resolved

					remove_character_flag = segrellion_young_griff_waiting_for_daemon

					add_character_flag = segrellion_young_griff_free_for_iron_throne
				}
			}


			# =================================================
			# PATH 1 - RECOVER 7900
			# =================================================

			if = {

				limit = {

					is_imprisoned = no

					has_character_flag = segrellion_young_griff_canon_path_1

					has_character_flag = segrellion_young_griff_free_for_iron_throne


					OR = {

						has_character_flag = segrellion_young_griff_left_daemon_stannis

						has_character_flag = segrellion_young_griff_left_daemon_daenerys

						has_character_flag = segrellion_young_griff_left_daemon_dorne

						has_character_flag = segrellion_young_griff_left_daemon_vale
					}


					NOT = {
						has_character_flag = segrellion_young_griff_independent_throne_path_scheduled
					}

					NOT = {
						has_character_flag = segrellion_young_griff_independent_throne_path_started
					}

					NOT = {
						has_character_flag = segrellion_young_griff_iron_throne_war_started
					}

					NOT = {
						has_title = title:h_the_iron_throne
					}
				}


				add_character_flag = {
					flag = segrellion_young_griff_independent_throne_path_scheduled
					days = 210
				}


				trigger_event = {

					id = segrellion_young_griff.7900

					months = 6
				}
			}


			# =================================================
			# PATH 1 - RECOVER 7910
			# =================================================

			if = {

				limit = {

					is_imprisoned = no

					has_character_flag = segrellion_young_griff_canon_path_1

					has_character_flag = segrellion_young_griff_independent_throne_path_started

					has_character_flag = segrellion_young_griff_varys_contacted


					NOT = {
						has_character_flag = segrellion_young_griff_independent_varys_seen
					}

					NOT = {
						has_character_flag = segrellion_young_griff_independent_varys_scheduled
					}

					NOT = {
						has_character_flag = segrellion_young_griff_iron_throne_war_started
					}

					NOT = {
						has_title = title:h_the_iron_throne
					}
				}


				add_character_flag = {
					flag = segrellion_young_griff_independent_varys_scheduled
					days = 30
				}


				trigger_event = {

					id = segrellion_young_griff.7910

					days = 1
				}
			}


			# =================================================
			# PATH 2 - DEBT / VARYS HANDOFF
			# =================================================

			if = {

				limit = {

					is_imprisoned = no

					has_character_flag = segrellion_young_griff_canon_path_2

					has_character_flag = segrellion_young_griff_daemon_war_resolved


					NOT = {
						has_character_flag = segrellion_young_griff_iron_throne_war_started
					}

					NOT = {
						has_title = title:h_the_iron_throne
					}
				}


				# ---------------------------------------------
				# DAEMON WON
				# ---------------------------------------------

				if = {

					limit = {

						character:Segrellion_60 ?= {
							is_alive = yes
							has_title = title:e_flamelands
						}
					}


					# Blood Debt somehow disappeared.
					# Restore it.

					if = {

						limit = {

							NOT = {
								has_character_flag = segrellion_young_griff_blood_debt_resolved
							}

							NOT = {
								has_character_flag = segrellion_young_griff_blood_debt_event_scheduled
							}
						}


						add_character_flag = {
							flag = segrellion_young_griff_blood_debt_event_scheduled
							days = 10
						}


						trigger_event = {

							id = segrellion_young_griff.7000

							days = 1
						}
					}


					else_if = {

						limit = {
							has_character_flag = segrellion_young_griff_blood_debt_resolved
						}


						segrellion_young_griff_try_schedule_path_2_varys_effect = yes
					}
				}


				# ---------------------------------------------
				# DAEMON LOST / DIED
				# ---------------------------------------------

				else = {

					segrellion_young_griff_try_schedule_path_2_varys_effect = yes
				}
			}


			# =================================================
			# WAR CREATION FAILED
			#
			# .6010 detected that start_war did not actually
			# create a war.
			#
			# Do not attempt to replay a Varys event that has
			# already been marked as seen. Retry the actual war.
			# =================================================

			if = {

				limit = {

					is_imprisoned = no

					has_character_flag = segrellion_young_griff_invasion_event_failed


					NOT = {
						has_character_flag = segrellion_young_griff_iron_throne_war_started
					}

					NOT = {
						has_title = title:h_the_iron_throne
					}
				}


				remove_character_flag = segrellion_young_griff_invasion_event_failed


				segrellion_young_griff_start_path_1_throne_war_effect = yes
			}
		}
	}
}

				is_alive = yes
			}
		}

		#
		# Do not interrupt the blood-debt scene.

		# to call upon Daemon.

			}

				flag = segrellion_young_griff_path_2_varys_scheduled
				days = 30
			}


			trigger_event = {

				id = segrellion_young_griff.8300

				days = { 7 14 }
			}
		}


		# =============================================
		# DAEMON LOST / DIED
		# =============================================

		else_if = {

			limit = {

				OR = {

					character:Segrellion_60 = {
						NOT = {
							has_title = title:e_flamelands
						}
					}

					NOT = {
						character:Segrellion_60 = {
							is_alive = yes
						}
					}
				}
			}


			add_character_flag = {
				flag = segrellion_young_griff_path_2_varys_scheduled
				days = 30
			}


			trigger_event = {

				id = segrellion_young_griff.8310

				days = { 2 5 }
			}
		}
	}
}

# VARYS' FINAL PLAN - BEGIN THE IRON THRONE WAR
#
# Shared by Path 1 and Path 2 despite the legacy effect name.
# =========================================================

		limit = {

			this = character:IAF_Young_Griff_1

			is_alive = yes
			is_imprisoned = no


			NOT = {
				has_title = title:h_the_iron_throne
			}

				exists = holder

				holder = {

					is_alive = yes

					this != root

			has_title = title:h_the_iron_throne
		}


		# Always repair the political consolidation.

		segrellion_young_griff_post_throne_conquest_effect = yes


		# =================================================
		# CERSEI
		# =================================================

		if = {

			limit = {

				has_character_flag = segrellion_young_griff_tommen_fate_resolved

				NOT = {
					has_character_flag = segrellion_young_griff_cersei_settlement_resolved
				}
			}


			segrellion_young_griff_schedule_cersei_settlement_effect = yes
		}


		# =================================================
		# AFTER CERSEI
		# =================================================

		if = {

			limit = {

				has_character_flag = segrellion_young_griff_cersei_settlement_resolved
			}


			# =============================================
			# PATH 2: STORMLANDS FIRST
			# =============================================

			if = {

				limit = {

					has_character_flag = segrellion_young_griff_canon_path_2

					has_title = title:e_the_stormlands

					NOT = {
						has_character_flag = segrellion_young_griff_stormlands_settlement_resolved
					}
				}


				# Repair old saves where this permanent flag
				# survived after the actual event was lost.

				if = {

					limit = {

						has_character_flag = segrellion_young_griff_stormlands_settlement_scheduled
					}

					remove_character_flag = segrellion_young_griff_stormlands_settlement_scheduled
				}


				segrellion_young_griff_after_cersei_settlement_effect = yes
			}


			# =============================================
			# PATH 1 OR STORMLANDS ALREADY DONE:
			# MARGAERY
			# =============================================

			else = {

				# Repair saves in which the delayed event was lost while
				# the old permanent scheduled flag survived.

				if = {

					limit = {

						has_character_flag = segrellion_young_griff_margaery_story_scheduled

						NOT = {
							has_character_flag = segrellion_young_griff_margaery_story_started
						}

						NOT = {
							has_character_flag = segrellion_young_griff_margaery_story_resolved
						}
					}

					remove_character_flag = segrellion_young_griff_margaery_story_scheduled
				}

				segrellion_young_griff_schedule_margaery_effect = yes
			}
		}
	}
}

					is_alive = yes

					is_adult = yes

			# monthly progression eventually repairs it.

				flag = segrellion_young_griff_blackfyre_reveal_scheduled

				days = 120
			}


			character:Baelish_2 = {

				trigger_event = {

					id = segrellion_young_griff.5000
