# =========================================================
# LIGHTBRINGER
# =========================================================

segrellion_transform_icebringer_into_lightbringer_effect = {

	$OWNER$ = {
		save_scope_as = owner
	}


	# =====================================================
	# DESTROY CURRENT FORM
	# =====================================================

	if = {
		limit = {
			exists = global_var:segrellion_icebringer
		}

		global_var:segrellion_icebringer = {
			save_scope_as = old_icebringer
		}

		remove_global_variable = segrellion_icebringer

		destroy_artifact = scope:old_icebringer
	}


	# =====================================================
	# ARTIFACT VALUES
	# =====================================================

	save_scope_value_as = {
		name = quality
		value = 100
	}

	save_scope_value_as = {
		name = wealth
		value = 100
	}

	set_artifact_rarity_illustrious = yes


	# =====================================================
	# CREATE LIGHTBRINGER
	# =====================================================

	create_artifact = {

		name = segrellion_lightbringer_name
		description = segrellion_lightbringer_desc

		type = sword

		visuals = segrellion_lightbringer_visuals
		template = segrellion_lightbringer_template

		wealth = scope:wealth
		quality = scope:quality

		modifier = segrellion_lightbringer_modifier

		history = {
			type = reforged
			actor = scope:owner
			recipient = scope:owner
		}

		save_scope_as = newly_created_artifact

		decaying = no
	}


	scope:newly_created_artifact = {

		equip_artifact_to_owner_replace = yes


		# Still occupies Icebringer's narrative/systemic identity.
		set_global_variable = {
			name = segrellion_icebringer
			value = this
		}

		set_global_variable = {
			name = segrellion_lightbringer
			value = this
		}


		set_variable = {
			name = valyrian_steel
			value = yes
		}

		set_variable = {
			name = historical_unique_artifact
			value = yes
		}

		set_variable = {
			name = traditional_house_owner
			value = scope:owner.house
		}


		# Keeps the Awaken Icebringer decision working.
		set_variable = {
			name = segrellion_icebringer_artifact
			value = yes
		}

		set_variable = {
			name = segrellion_lightbringer_artifact
			value = yes
		}
	}
}



# =========================================================
# BROKEN ICE SWORDS
# =========================================================

segrellion_create_broken_ice_sword_effect = {

	$OWNER$ = {
		save_scope_as = owner
	}

	save_scope_value_as = {
		name = quality
		value = 75
	}

	save_scope_value_as = {
		name = wealth
		value = 60
	}

	# Blue / Famed artifact.
	set_artifact_rarity_famed = yes

	create_artifact = {

		name = segrellion_broken_ice_sword_name
		description = segrellion_broken_ice_sword_desc

		type = miscellaneous

		visuals = segrellion_broken_ice_sword_visuals
		template = segrellion_broken_ice_sword_template

		wealth = scope:wealth
		quality = scope:quality

		modifier = segrellion_broken_ice_sword_modifier

		save_scope_as = newly_created_artifact

		decaying = no
	}

	scope:newly_created_artifact = {

		set_variable = {
			name = historical_unique_artifact
			value = yes
		}

		set_variable = {
			name = segrellion_broken_ice_sword_artifact
			value = yes
		}

		set_variable = {
			name = traditional_house_owner
			value = scope:owner.house
		}
	}
}



# =========================================================
# IMPROVE MARTIAL EDUCATION
#
# Only normal Martial education.
#
# 1 -> 2
# 2 -> 3
# 3 -> 4
# 4 -> 5
#
# Non-Martial education: nothing.
# Level 5: nothing.
# =========================================================

segrellion_upgrade_martial_education_effect = {

	if = {
		limit = {
			has_trait = education_martial_1
		}

		remove_trait = education_martial_1
		add_trait = education_martial_2
	}

	else_if = {
		limit = {
			has_trait = education_martial_2
		}

		remove_trait = education_martial_2
		add_trait = education_martial_3
	}

	else_if = {
		limit = {
			has_trait = education_martial_3
		}

		remove_trait = education_martial_3
		add_trait = education_martial_4
	}

	else_if = {
		limit = {
			has_trait = education_martial_4
		}

		remove_trait = education_martial_4
		add_trait = education_martial_5
	}
}
