segrellion_daenerys_start_regional_settlement_v2_effect = {

	if = {

		limit = {

			this = character:Targaryen_9

			is_alive = yes

			has_title = title:h_the_iron_throne


			NOT = {
				has_character_flag = segrellion_daenerys_regional_settlement_v2_started
			}


			NOT = {
				has_character_flag = segrellion_daenerys_lost_tyranny_revolt
			}
		}


		add_character_flag = segrellion_daenerys_regional_settlement_v2_started


		trigger_event = {

			id = segrellion_daenerys_settlement.0001
		}
	}
}

# =========================================================
# ROUTE CURRENT REGION
# =========================================================

segrellion_daenerys_route_current_settlement_effect = {

	segrellion_daenerys_refresh_current_settlement_holder_effect = yes


	if = {

		limit = {

			NOT = {
				has_variable = segrellion_daenerys_current_settlement_holder
			}
		}


		segrellion_daenerys_finish_current_settlement_effect = yes
	}


	else_if = {

		limit = {

			var:segrellion_daenerys_current_settlement_holder = root
		}


		segrellion_daenerys_finish_current_settlement_effect = yes
	}


	# =====================================================
	# ALLY
	# =====================================================

	else_if = {

		limit = {

			var:segrellion_daenerys_current_settlement_holder = {

				OR = {

					has_character_flag = segrellion_supported_daenerys_invasion

					has_character_flag = segrellion_support_agreement_reached_with_daenerys
				}
			}
		}


		trigger_event = {

			id = segrellion_daenerys_settlement.0100
		}
	}


	# =====================================================
	# ENEMY
	# =====================================================

	else_if = {

		limit = {

			var:segrellion_daenerys_current_settlement_holder = {

				has_character_flag = segrellion_opposed_daenerys_invasion
			}
		}


		trigger_event = {

			id = segrellion_daenerys_settlement.0200
		}
	}


	# =====================================================
	# NEUTRAL
	# =====================================================

	else = {

		var:segrellion_daenerys_current_settlement_holder = {

			trigger_event = {

				id = segrellion_daenerys_settlement.0400
			}
		}
	}
}


# =========================================================
# SWEAR FEALTY
# =========================================================

segrellion_daenerys_current_region_swears_fealty_effect = {

	segrellion_daenerys_refresh_current_settlement_holder_effect = yes


	if = {

		limit = {

			has_variable = segrellion_daenerys_current_settlement_holder


			var:segrellion_daenerys_current_settlement_holder != root


			var:segrellion_daenerys_current_settlement_holder = {

				NOT = {
					top_liege = character:Targaryen_9
				}
			}
		}


		create_title_and_vassal_change = {

			type = swear_fealty

			save_scope_as = segrellion_daenerys_regional_fealty_change

			add_claim_on_loss = no
		}


		var:segrellion_daenerys_current_settlement_holder = {

			change_liege = {

				liege = character:Targaryen_9

				change = scope:segrellion_daenerys_regional_fealty_change
			}
		}


		resolve_title_and_vassal_change = scope:segrellion_daenerys_regional_fealty_change
	}


	var:segrellion_daenerys_current_settlement_holder ?= {

		add_character_flag = segrellion_swore_fealty_to_daenerys
	}
}

# =========================================================
# MERCY FOR A DEFEATED ENEMY
# =========================================================

segrellion_daenerys_pardon_enemy_house_effect = {

	segrellion_daenerys_refresh_current_settlement_holder_effect = yes


	if = {

		limit = {
			has_variable = segrellion_daenerys_current_settlement_holder
		}


		add_hook = {

			target = var:segrellion_daenerys_current_settlement_holder

			type = favor_hook
		}


		var:segrellion_daenerys_current_settlement_holder = {

			save_scope_as = segrellion_pardoned_enemy_lord


			if = {

				limit = {
					is_imprisoned = yes
				}

				release_from_prison = yes
			}


			if = {

				limit = {
					has_trait = segrellion_traitor
				}

				remove_trait = segrellion_traitor
			}


			add_character_flag = segrellion_pardoned_by_daenerys


			add_opinion = {

				target = character:Targaryen_9

				modifier = segrellion_daenerys_granted_full_pardon_opinion

				years = 20
			}


			house = {

				every_house_member = {

					limit = {
						is_alive = yes
					}


					if = {

						limit = {
							has_trait = segrellion_traitor
						}

						remove_trait = segrellion_traitor
					}


					hidden_effect = {

						add_opinion = {

							target = character:Targaryen_9

							modifier = segrellion_daenerys_pardoned_house_opinion

							years = 20
						}
					}
				}
			}
		}


		segrellion_daenerys_current_region_swears_fealty_effect = yes
	}
}

#

# =========================================================

segrellion_daenerys_purge_enemy_house_effect = {

	segrellion_daenerys_refresh_current_settlement_holder_effect = yes


	title:c_pentos.holder ?= {

		save_scope_as = segrellion_pentos_refuge_holder
	}


	var:segrellion_daenerys_current_settlement_holder ?= {

		house = {

			every_house_member = {

				limit = {
					is_alive = yes
				}


				random_list = {

					10 = {

						trigger = {

							is_landed = no
						}


						add_character_flag = segrellion_escaped_daenerys_house_purge


						if = {

							limit = {
								exists = scope:segrellion_pentos_refuge_holder
							}

							set_employer = scope:segrellion_pentos_refuge_holder
						}


						set_location = title:c_pentos.title_province
					}


					90 = {

						death = {

							death_reason = death_execution

							killer = character:Targaryen_9
						}
					}
				}
			}
		}
	}


	add_character_flag = segrellion_daenerys_destroyed_enemy_house


	segrellion_daenerys_take_current_paramountcy_effect = yes


	segrellion_change_daenerys_pacification_tyranny_effect = {

		VALUE = 40
	}
