# =========================================================
# FLAMELANDER CULTURE
# =========================================================

# Converts the territories that formed the original Flamelands.
# House Segrellion and the de jure nobility adopt the regional identity.
segrellion_apply_flamelander_culture_effect = {

	# The new culture begins with the technological knowledge of the two
	# principal Westerosi cultures from which the region developed.
	culture:flamelander = {
		get_all_innovations_from = culture:crownlander
		get_all_innovations_from = culture:stormlander_main
	}

	culture:high_flamelander = {
		get_all_innovations_from = culture:flamelander
	}

	# Vhalerion
	title:d_vhalerion = {
		every_de_jure_county = {
			set_county_culture = culture:flamelander
		}
	}

	# Massey's Hook
	title:d_masseys_hook = {
		every_de_jure_county = {
			set_county_culture = culture:flamelander
		}
	}

	# Wendwater Forest / the central Flamelands
	title:d_wendwater_forest = {
		every_de_jure_county = {
			set_county_culture = culture:flamelander
		}
	}

	# Baywater Woods, excluding The Gauntlet.
	title:d_the_baywater_woods = {
		every_de_jure_county = {
			limit = {
				this != title:c_the_gauntlet
			}
			set_county_culture = culture:flamelander
		}
	}

	if = {
		limit = { NOT = { has_global_variable = segrellion_flamelander_nobility_adopted } }
		segrellion_apply_flamelander_nobility_effect = yes
	}

	set_global_variable = {
		name = segrellion_flamelander_culture_created
		value = yes
	}

	title:k_flamelands = {
		remove_variable = segrellion_flamelander_culture_waiting
	}
}


# Dynamic formation path.
# Used when the player actually takes the decision to found the Flamelands.
# The creation date becomes the real date on which the five-year process ends.
segrellion_create_flamelander_culture_dynamic_effect = {

	culture:flamelander = {

		reset_culture_creation_date = yes
	}


	culture:high_flamelander = {

		reset_culture_creation_date = yes
	}


	segrellion_apply_flamelander_culture_effect = yes
}


# Historical formation path.
# Aeryn founded the Flamelands on 8144.1.28, so the culture exists from
# 8144.1.28 in starts where the kingdom was founded through title history.
# We deliberately do not reset the culture creation date here, because the
# culture definition already records the historical date 8144.1.28.
segrellion_create_flamelander_culture_historical_effect = {

	segrellion_apply_flamelander_culture_effect = yes
}


# Main progression check. It is intentionally independent of Aeryn's life,
# so succession or an early death can never destroy the five-year timer.
segrellion_flamelander_culture_progression_effect = {

    # Historical adoption and migration for saves where the culture already exists.
    if = {
        limit = {
            current_date >= 8144.1.28
            title:k_flamelands = { exists = holder }
            NOT = { has_global_variable = segrellion_flamelander_nobility_adopted }
        }
        segrellion_apply_flamelander_nobility_effect = yes
    }

	# =====================================================
	# HIGH FLAMELANDER SAVE UPGRADE
	#
	# Old saves may already have completed the original
	# Flamelander adoption before High Flamelander existed.
	# =====================================================

	if = {

		limit = {

			current_date >= 8144.1.28


			title:k_flamelands = {
				exists = holder
			}


			has_global_variable = segrellion_flamelander_nobility_adopted


			NOT = {
				has_global_variable = segrellion_high_flamelander_adopted
			}
		}


		segrellion_apply_high_flamelander_segrellion_effect = yes
	}

	if = {
		limit = {
			NOT = {
				has_global_variable = segrellion_flamelander_culture_created
			}

			title:k_flamelands = {
				exists = holder
			}
		}

		# =====================================================
		# DYNAMIC FOUNDATION
		# =====================================================
		# segrellion_flamelands_founded is set by the actual
		# Forge the Flamelands decision. On the first monthly
		# pulse after foundation we start a five-year title timer.
		if = {
			limit = {
				has_global_variable = segrellion_flamelands_founded

				title:k_flamelands = {
					NOT = {
						has_variable = segrellion_flamelander_culture_clock_started
					}
				}
			}

			title:k_flamelands = {
				set_variable = {
					name = segrellion_flamelander_culture_clock_started
					value = yes
				}

				set_variable = {
					name = segrellion_flamelander_culture_waiting
					value = yes
					years = 5
				}
			}
		}

		# Five years have elapsed since the dynamic foundation.
		if = {
			limit = {
				has_global_variable = segrellion_flamelands_founded

				title:k_flamelands = {
					has_variable = segrellion_flamelander_culture_clock_started

					NOT = {
						has_variable = segrellion_flamelander_culture_waiting
					}
				}
			}

			segrellion_create_flamelander_culture_dynamic_effect = yes
		}

		# =====================================================
		# HISTORICAL FOUNDATION
		# =====================================================
		# When loading a bookmark after Aeryn's historical
		# foundation, the decision itself never ran. In that case
		# the fixed historical emergence date is used instead.
		else_if = {
			limit = {
				NOT = {
					has_global_variable = segrellion_flamelands_founded
				}

				current_date >= 8144.1.28
			}

			segrellion_create_flamelander_culture_historical_effect = yes
		}
	}
}

# Character scope: include children so the first succession preserves the culture.
segrellion_convert_flamelander_lord_effect = {
    set_culture = culture:flamelander
    every_child = {
        limit = { is_alive = yes }
        set_culture = culture:flamelander
    }
}

# =========================================================
# HIGH FLAMELANDER
#
# The ordinary population and nobility of the Flamelands
# remain Flamelander.
#
# Only House Segrellion adopts High Flamelander.
# =========================================================

segrellion_apply_high_flamelander_segrellion_effect = {

	# Technologically, High Flamelander begins from the
	# regional Flamelander culture rather than gaining
	# additional free technologies.

	culture:high_flamelander = {

		get_all_innovations_from = culture:flamelander
	}


	# =====================================================
	# HOUSE SEGRELLION
	# =====================================================

	dynasty:dynn_Segrellion = {

		every_dynasty_member = {

			limit = {
				is_alive = yes
			}


			set_culture = culture:high_flamelander
		}
	}


	set_global_variable = {

		name = segrellion_high_flamelander_adopted

		value = yes
	}
}

# =========================================================
# FLAMELANDER NOBILITY
#
# Regional nobility becomes Flamelander.
# House Segrellion becomes High Flamelander afterwards.
# =========================================================

segrellion_apply_flamelander_nobility_effect = {

	# =====================================================
	# REGIONAL NOBILITY
	# =====================================================

	title:k_flamelands = {

		every_de_jure_county = {

			if = {

				limit = {
					exists = holder
				}


				holder = {

					segrellion_convert_flamelander_lord_effect = yes
				}
			}


			de_jure_liege = {

				if = {

					limit = {
						exists = holder
					}


					holder = {

						segrellion_convert_flamelander_lord_effect = yes
					}
				}
			}
		}
	}


	# =====================================================
	# HOUSE SEGRELLION
	#
	# Must run AFTER the regional lord conversion so a
	# Segrellion ruler is not overwritten back to ordinary
	# Flamelander.
	# =====================================================

	segrellion_apply_high_flamelander_segrellion_effect = yes


	set_global_variable = {

		name = segrellion_flamelander_nobility_adopted

		value = yes
	}
}
