segrellion_create_artifact_icebringer_effect = {
	$OWNER$ = { save_scope_as = owner }
	set_artifact_rarity_illustrious = yes

	create_artifact = {
		name = segrellion_icebringer_name
		visuals = segrellion_icebringer_visuals
		description = segrellion_icebringer_desc
		type = sword
		wealth = scope:wealth
		quality = scope:quality
		template = segrellion_icebringer_template
		history = {
			type = created_before_history
			date = 7600.1.1
		}
		modifier = segrellion_icebringer_modifier
		save_scope_as = newly_created_artifact
		decaying = no
	}

	scope:newly_created_artifact ?= {
		equip_artifact_to_owner_replace = yes
		set_global_variable = {
			name = segrellion_icebringer
			value = this
		}
		save_scope_as = epic
		set_variable = { name = valyrian_steel value = yes }
		set_variable = { name = historical_unique_artifact value = yes }
		set_variable = { name = traditional_house_owner value = scope:owner.house }
		set_variable = { name = segrellion_icebringer_artifact value = yes }
		agot_add_artifact_history = {
			ARTIFACT = this
			TYPE = inherited
			DATE = 8061.1.1
			RECIPIENT = character:Segrellion_4
		}
		agot_add_artifact_history = {
			ARTIFACT = this
			TYPE = inherited
			DATE = 8106.4.18
			RECIPIENT = character:Segrellion_1
		}
        
        if = {
    limit = { current_date >= 8139.4.18 }

        agot_add_artifact_history = {
            ARTIFACT = this
            TYPE = inherited
            DATE = 8139.4.18
            RECIPIENT = character:Segrellion_12
        }
    }

    if = {
        limit = { current_date >= 8168.6.13 }

        agot_add_artifact_history = {
            ARTIFACT = this
            TYPE = inherited
            DATE = 8168.6.13
            RECIPIENT = character:Segrellion_19
        }
    }

    if = {
        limit = { current_date >= 8174.9.17 }

        agot_add_artifact_history = {
            ARTIFACT = this
            TYPE = inherited
            DATE = 8174.9.17
            RECIPIENT = character:Segrellion_21
        }
    }
	}
}

segrellion_find_icebringer_effect = {
	if = {
		limit = { NOT = { exists = global_var:segrellion_icebringer } }
		character:Segrellion_1 ?= {
			any_character_artifact = {
				limit = { has_variable = segrellion_icebringer_artifact }
				set_global_variable = {
					name = segrellion_icebringer
					value = this
				}
			}
		}
	}
}

segrellion_refresh_icebringer_affinity_effect = {
	segrellion_find_icebringer_effect = yes
	global_var:segrellion_icebringer ?= {
		artifact_owner ?= {
			remove_character_modifier = segrellion_icebringer_rejection_modifier
			if = {
				limit = {
					any_equipped_character_artifact = {
						has_variable = segrellion_icebringer_artifact
					}
					segrellion_icebringer_rejected_trigger = yes
				}
				add_character_modifier = {
					modifier = segrellion_icebringer_rejection_modifier
					days = 40
				}
			}
		}
	}
}

segrellion_setup_trueflame_history_effect = {

    if = {
        limit = {
            current_date >= 8150.1.1
            NOT = {
                exists = global_var:segrellion_trueflame
            }
        }

        # Aeryn discovers Trueflame around 150 AC.
        if = {
            limit = {
                current_date < 8161.10.21

                character:Segrellion_12 ?= {
                    is_alive = yes
                }
            }

            character:Segrellion_12 = {
                segrellion_create_artifact_trueflame_effect = {
                    OWNER = this
                }
            }
        }

        # Aeryn gives Trueflame to his eldest son Rhaegar
        # when Rhaegar comes of age.
        else_if = {
            limit = {
                current_date < 8174.9.17

                character:Segrellion_18 ?= {
                    is_alive = yes
                }
            }

            character:Segrellion_18 = {
                segrellion_create_artifact_trueflame_effect = {
                    OWNER = this
                }
            }
        }

        # After the Segrellion civil war and Rhaegar's fall,
        # his son Maelor becomes the bearer of Trueflame.
        else_if = {
            limit = {
                current_date < 8196.4.12

                character:Segrellion_38 ?= {
                    is_alive = yes
                }
            }

            character:Segrellion_38 = {
                segrellion_create_artifact_trueflame_effect = {
                    OWNER = this
                }
            }
        }

        # Maelor dies in 196 AC.
        # Trueflame returns to Aegon Dragonspear,
        # the head of House Segrellion.
        else_if = {
            limit = {
                current_date < 8200.4.12

                character:Segrellion_21 ?= {
                    is_alive = yes
                }
            }

            character:Segrellion_21 = {
                segrellion_create_artifact_trueflame_effect = {
                    OWNER = this
                }
            }
        }

        # Four years after Maelor's death,
        # Dragonspear entrusts Trueflame to Aethan.
        else_if = {
            limit = {
                current_date < 8218.9.26

                character:Segrellion_24 ?= {
                    is_alive = yes
                }
            }

            character:Segrellion_24 = {
                segrellion_create_artifact_trueflame_effect = {
                    OWNER = this
                }
            }
        }

        # After Aethan's death, Trueflame passes
        # to his brother Rhaegar.
        else_if = {
            limit = {
                current_date < 8245.2.21

                character:Segrellion_25 ?= {
                    is_alive = yes
                }
            }

            character:Segrellion_25 = {
                segrellion_create_artifact_trueflame_effect = {
                    OWNER = this
                }
            }
        }

        # Rhaegar's heir Aenys becomes the next bearer.
        else = {
            character:Segrellion_30 ?= {
                if = {
                    limit = {
                        is_alive = yes
                    }

                    segrellion_create_artifact_trueflame_effect = {
                        OWNER = this
                    }
                }
            }
        }
    }
}
