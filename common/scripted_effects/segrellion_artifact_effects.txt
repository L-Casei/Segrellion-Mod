segrellion_find_icebringer_effect = {
	if = {
		limit = { NOT = { exists = global_var:segrellion_icebringer } }
		character:Segrellion_1 ?= {
			any_character_artifact = {
				limit = { has_variable = segrellion_icebringer_artifact }
				set_global_variable = {
					name = segrellion_icebringer
					value = this
				}
			}
		}
	}
}

segrellion_refresh_icebringer_affinity_effect = {
	segrellion_find_icebringer_effect = yes

	global_var:segrellion_icebringer ?= {
		artifact_owner ?= {

			# -------------------------------------------------
			# Clear previous state
			# -------------------------------------------------

			remove_character_flag = segrellion_icebringer_state_rejected
			remove_character_flag = segrellion_icebringer_state_partial
			remove_character_flag = segrellion_icebringer_state_accepted


			# -------------------------------------------------
			# Determine current relationship with the blade
			# -------------------------------------------------

			if = {
				limit = {
					segrellion_icebringer_accepted_trigger = yes
				}

				add_character_flag = segrellion_icebringer_state_accepted
			}

			else_if = {
				limit = {
					segrellion_icebringer_partial_trigger = yes
				}

				add_character_flag = segrellion_icebringer_state_partial
			}

			else = {
				add_character_flag = segrellion_icebringer_state_rejected
			}


			# -------------------------------------------------
			# Rejection penalty only matters while equipped
			# -------------------------------------------------

			if = {
				limit = {
					has_character_flag = segrellion_icebringer_state_rejected

					any_equipped_character_artifact = {
						has_variable = segrellion_icebringer_artifact
					}
				}

				if = {
					limit = {
						NOT = {
							has_character_modifier = segrellion_icebringer_rejection_modifier
						}
					}

					add_character_modifier = {
						modifier = segrellion_icebringer_rejection_modifier
					}
				}
			}

			else = {
				remove_character_modifier = segrellion_icebringer_rejection_modifier
			}
		}
	}
}

segrellion_icebringer_monthly_event_pulse_effect = {
	segrellion_find_icebringer_effect = yes

	global_var:segrellion_icebringer ?= {
		artifact_owner ?= {

			if = {
				limit = {
					is_alive = yes

					any_equipped_character_artifact = {
						has_variable = segrellion_icebringer_artifact
					}
				}

				# Keep the state up to date.
				segrellion_refresh_icebringer_affinity_effect = yes


				# ---------------------------------------------
				# REJECTED
				# Roughly 1.2 events/year on average.
				# ---------------------------------------------

				if = {
					limit = {
						has_character_flag = segrellion_icebringer_state_rejected
					}

					random = {
						chance = 10

						trigger_event = {
							id = segrellion_icebringer.0001
							days = 1
						}
					}
				}


				# ---------------------------------------------
				# PARTIAL
				# Roughly 0.8 events/year.
				# ---------------------------------------------

				else_if = {
					limit = {
						has_character_flag = segrellion_icebringer_state_partial
					}

					random = {
						chance = 7

						trigger_event = {
							id = segrellion_icebringer.0002
							days = 1
						}
					}
				}


				# ---------------------------------------------
				# ACCEPTED
				# Roughly 0.6 events/year.
				# ---------------------------------------------

				else_if = {
					limit = {
						has_character_flag = segrellion_icebringer_state_accepted
					}

					random = {
						chance = 5

						trigger_event = {
							id = segrellion_icebringer.0003
							days = 1
						}
					}
				}
			}

			else = {
				remove_character_modifier = segrellion_icebringer_rejection_modifier
			}
		}
	}
}

        # After Aenys' death it passes to Maegor.

                segrellion_create_artifact_trueflame_effect = {

# =========================================================

segrellion_create_crown_of_void_effect = {

	$OWNER$ = {
		save_scope_as = owner
	}

	save_scope_value_as = {
		name = quality
		value = 10
	}

	save_scope_value_as = {
		name = wealth
		value = 10
	}

	create_artifact = {
		name = segrellion_crown_of_void_name
		description = segrellion_crown_of_void_desc

		type = helmet

		# Reuse AGOT's Valyrian crown model.
		visuals = valyrian_crown1_visuals
		template = valyrian_crown1_template

		wealth = scope:wealth
		quality = scope:quality

		modifier = segrellion_crown_of_void_modifier

		save_scope_as = newly_created_artifact

		decaying = no
	}

	scope:newly_created_artifact ?= {
		set_variable = {
			name = segrellion_crown_of_void_artifact
			value = yes
		}
	}
}

# ICEBRINGER - VANISH FOREVER
# =========================================================

segrellion_icebringer_vanish_forever_effect = {

	set_global_variable = {
		name = segrellion_icebringer_lost_forever
		value = yes
	}

	if = {
		limit = {
			exists = global_var:segrellion_icebringer
		}

		global_var:segrellion_icebringer = {
			save_scope_as = icebringer_to_destroy
		}

		remove_global_variable = segrellion_icebringer

		destroy_artifact = scope:icebringer_to_destroy
	}

	remove_character_modifier = segrellion_icebringer_rejection_modifier
	remove_character_modifier = segrellion_beyond_waking_veil_modifier

	remove_character_flag = segrellion_icebringer_awakening_chain_active

	remove_character_flag = segrellion_icebringer_state_rejected
	remove_character_flag = segrellion_icebringer_state_partial
	remove_character_flag = segrellion_icebringer_state_accepted
}

# =========================================================
# AWAKEN ICEBRINGER
# =========================================================

segrellion_awaken_icebringer_effect = {

	# =====================================================
	# THE GUARDIAN
	# =====================================================

	add_trait = segrellion_the_guardian_trait

	add_character_flag = segrellion_icebringer_awakening_succeeded
	add_character_flag = segrellion_icebringer_awakened_bearer

	remove_character_flag = segrellion_icebringer_awakening_chain_active

	remove_character_modifier = segrellion_beyond_waking_veil_modifier
	remove_character_modifier = segrellion_icebringer_rejection_modifier


	# =====================================================
	# OLD RELATIONSHIP STATE ENDS
	# =====================================================

	remove_character_flag = segrellion_icebringer_state_rejected
	remove_character_flag = segrellion_icebringer_state_partial
	remove_character_flag = segrellion_icebringer_state_accepted

	add_character_flag = segrellion_icebringer_state_awakened


	# =====================================================
	# ONLY ONCE IN THE ENTIRE GAME
	# =====================================================

	set_global_variable = {
		name = segrellion_icebringer_has_awakened
		value = yes
	}


	# =====================================================
	# AWAKEN THE SAME HISTORICAL ARTIFACT
	# =====================================================

	global_var:segrellion_icebringer = {

		remove_artifact_modifier = segrellion_icebringer_modifier

		add_artifact_modifier = segrellion_awakened_icebringer_modifier

		set_artifact_name = segrellion_awakened_icebringer_name

		set_artifact_description = segrellion_awakened_icebringer_desc

		set_variable = {
			name = segrellion_icebringer_awakened
			value = yes
		}
	}
}
