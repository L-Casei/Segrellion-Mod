segrellion_find_icebringer_effect = {

	# Elimina una referencia global inválida.
	if = {
		limit = {
			has_global_variable = segrellion_icebringer
			NOT = {
				exists = global_var:segrellion_icebringer
			}
		}

		remove_global_variable = segrellion_icebringer
	}

	# Reconstruye la referencia buscando Icebringer
	# entre todos los artefactos existentes.
	if = {
		limit = {
			NOT = {
				exists = global_var:segrellion_icebringer
			}
		}

		every_artifact = {
			limit = {
				has_variable = segrellion_icebringer_artifact
			}

			set_global_variable = {
				name = segrellion_icebringer
				value = this
			}
		}
	}
}

segrellion_icebringer_exists_anywhere_trigger = {
	OR = {
		exists = global_var:segrellion_icebringer

		any_artifact = {
			has_variable = segrellion_icebringer_artifact
		}
	}
}


segrellion_can_search_legendary_relic_trigger = {
	OR = {

		# House Segrellion and its cadet houses.
		dynasty = dynasty:dynn_Segrellion

		# Segrellion historical knowledge.
		AND = {
			exists = var:segrellion_lore
			var:segrellion_lore >= 5
		}

		# General draconic knowledge.
		AND = {
			exists = var:dragonlore
			var:dragonlore >= 5
		}
	}
}

segrellion_refresh_icebringer_affinity_effect = {
	segrellion_find_icebringer_effect = yes

	global_var:segrellion_icebringer ?= {
		artifact_owner ?= {

			# -------------------------------------------------
			# Clear previous state
			# -------------------------------------------------

			remove_character_flag = segrellion_icebringer_state_rejected
			remove_character_flag = segrellion_icebringer_state_partial
			remove_character_flag = segrellion_icebringer_state_accepted


			# -------------------------------------------------
			# Determine current relationship with the blade
			# -------------------------------------------------

			if = {
				limit = {
					segrellion_icebringer_accepted_trigger = yes
				}

				add_character_flag = segrellion_icebringer_state_accepted
			}

			else_if = {
				limit = {
					segrellion_icebringer_partial_trigger = yes
				}

				add_character_flag = segrellion_icebringer_state_partial
			}

			else = {
				add_character_flag = segrellion_icebringer_state_rejected
			}


			# -------------------------------------------------
			# Rejection penalty only matters while equipped
			# -------------------------------------------------

			if = {
				limit = {
					has_character_flag = segrellion_icebringer_state_rejected

					any_equipped_character_artifact = {
						has_variable = segrellion_icebringer_artifact
					}
				}

				if = {
					limit = {
						NOT = {
							has_character_modifier = segrellion_icebringer_rejection_modifier
						}
					}

					add_character_modifier = {
						modifier = segrellion_icebringer_rejection_modifier
					}
				}
			}

			else = {
				remove_character_modifier = segrellion_icebringer_rejection_modifier
			}
		}
	}
}

segrellion_icebringer_monthly_event_pulse_effect = {
	segrellion_find_icebringer_effect = yes

	global_var:segrellion_icebringer ?= {
		artifact_owner ?= {

			if = {
				limit = {
					is_alive = yes

					any_equipped_character_artifact = {
						has_variable = segrellion_icebringer_artifact
					}
				}

				# Keep the state up to date.
				segrellion_refresh_icebringer_affinity_effect = yes


				# ---------------------------------------------
				# REJECTED
				# Roughly 1.2 events/year on average.
				# ---------------------------------------------

				if = {
					limit = {
						has_character_flag = segrellion_icebringer_state_rejected
					}

					random = {
						chance = 10

						trigger_event = {
							id = segrellion_icebringer.0001
							days = 1
						}
					}
				}


				# ---------------------------------------------
				# PARTIAL
				# Roughly 0.8 events/year.
				# ---------------------------------------------

				else_if = {
					limit = {
						has_character_flag = segrellion_icebringer_state_partial
					}

					random = {
						chance = 7

						trigger_event = {
							id = segrellion_icebringer.0002
							days = 1
						}
					}
				}


				# ---------------------------------------------
				# ACCEPTED
				# Roughly 0.6 events/year.
				# ---------------------------------------------

				else_if = {
					limit = {
						has_character_flag = segrellion_icebringer_state_accepted
					}

					random = {
						chance = 5

						trigger_event = {
							id = segrellion_icebringer.0003
							days = 1
						}
					}
				}
			}

			else = {
				remove_character_modifier = segrellion_icebringer_rejection_modifier
			}
		}
	}
}

# =========================================================
# CORRUPT ICEBRINGER
#
# Stage V Darkness transformation.
# Destroys the previous form and replaces it with
# Corrupted Icebringer.
# =========================================================

segrellion_corrupt_icebringer_effect = {


	# =====================================================
	# THE DARKNESS HAS CHOSEN A BEARER
	# =====================================================

	if = {
		limit = {
			NOT = {
				has_trait = segrellion_bearer_of_darkness_trait
			}
		}

		add_trait = segrellion_bearer_of_darkness_trait
	}


	# =====================================================
	# MARK ICEBRINGER AS HAVING PASSED BEYOND DORMANCY
	#
	# This also prevents the legendary relic decision
	# from ever treating the sword as merely "lost".
	# =====================================================

	set_global_variable = {
		name = segrellion_icebringer_has_awakened
		value = yes
	}


	# =====================================================
	# DESTROY DORMANT ICEBRINGER
	# =====================================================

	if = {
		limit = {
			exists = global_var:segrellion_icebringer
		}

		global_var:segrellion_icebringer = {
			save_scope_as = corrupted_old_icebringer
		}

		remove_global_variable = segrellion_icebringer

		destroy_artifact = scope:corrupted_old_icebringer
	}


	# =====================================================
	# DEFENSIVE SUPPORT:
	# DESTROY AWAKENED ICEBRINGER IF THIS EFFECT IS EVER
	# REACHED THROUGH DEBUG / FUTURE ALTERNATE CONTENT
	# =====================================================

	if = {
		limit = {
			exists = global_var:segrellion_awakened_icebringer
		}

		global_var:segrellion_awakened_icebringer = {
			save_scope_as = corrupted_old_awakened_icebringer
		}

		remove_global_variable = segrellion_awakened_icebringer

		destroy_artifact = scope:corrupted_old_awakened_icebringer
	}


	# =====================================================
	# OLD ICEBRINGER STATES END
	# =====================================================

	remove_character_modifier = segrellion_icebringer_rejection_modifier
	remove_character_modifier = segrellion_beyond_waking_veil_modifier

	remove_character_flag = segrellion_icebringer_state_rejected
	remove_character_flag = segrellion_icebringer_state_partial
	remove_character_flag = segrellion_icebringer_state_accepted


	# =====================================================
	# DARKNESS STATE
	# =====================================================

	add_character_flag = segrellion_icebringer_corrupted_bearer
	add_character_flag = segrellion_icebringer_state_corrupted


	# =====================================================
	# CREATE THE NEW BLADE
	# =====================================================

	segrellion_create_artifact_corrupted_icebringer_effect = {
		OWNER = this
	}
}

		# Icebringer object, it vanishes with Dragonspear.

		if = {
			limit = {
				exists = global_var:segrellion_icebringer
			}

			destroy_artifact = global_var:segrellion_icebringer
		}


		# Clear the global pointer as well.

		if = {
			limit = {
				has_global_variable = segrellion_icebringer
			}

			remove_global_variable = segrellion_icebringer
		}


		# Never perform the canonical 205 AC disappearance again.
		# If Icebringer is recovered afterwards it remains real.

		set_global_variable = {
			name = segrellion_icebringer_historical_loss_processed
			value = yes
		}
	}
}
