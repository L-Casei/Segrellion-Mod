segrellion_remove_legacy_canon_traits_effect = {
	if = { limit = { has_trait = segrellion_canon_balerion } remove_trait = segrellion_canon_balerion }
	if = { limit = { has_trait = segrellion_canon_shiera } remove_trait = segrellion_canon_shiera }
	if = { limit = { has_trait = segrellion_canon_aegor } remove_trait = segrellion_canon_aegor }
	if = { limit = { has_trait = segrellion_canon_saera } remove_trait = segrellion_canon_saera }
	if = { limit = { has_trait = segrellion_canon_alex } remove_trait = segrellion_canon_alex }
	if = { limit = { has_trait = segrellion_canon_alysanne } remove_trait = segrellion_canon_alysanne }
	if = { limit = { has_trait = segrellion_canon_daenerys } remove_trait = segrellion_canon_daenerys }
	if = { limit = { has_trait = vladwyn_canon_tristan } remove_trait = vladwyn_canon_tristan }
}

segrellion_add_canon_modifier_effect = {
	segrellion_remove_legacy_canon_traits_effect = yes
	if = {
		limit = {
			has_game_rule = segrellion_canon_traits_enabled
			NOT = { has_character_modifier = $MODIFIER$ }
		}
		add_character_modifier = $MODIFIER$
	}
}

segrellion_remove_canon_modifier_effect = {
	segrellion_remove_legacy_canon_traits_effect = yes
	if = {
		limit = { has_character_modifier = $MODIFIER$ }
		remove_character_modifier = $MODIFIER$
	}
}
