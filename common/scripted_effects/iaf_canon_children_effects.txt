iaf_canon_children_add_identity_effect = {
	if = {
		limit = {
			is_alive = yes
			NOT = { has_character_flag = is_$ID$ }
		}
		add_character_flag = is_$ID$
	}
	make_trait_inactive = is_$ID$
}
