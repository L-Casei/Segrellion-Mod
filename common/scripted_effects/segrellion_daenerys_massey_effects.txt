# =========================================================
# DAENERYS - MARIC MASSEY INFORMANT NETWORK
# =========================================================


segrellion_daenerys_massey_news_progression_effect = {

	character:Targaryen_9 ?= {

		if = {

			limit = {

				is_alive = yes

				game_start_date = 8300.8.20
			}


			# =================================================
			# FIRST CONTACT
			#
			# This DOES require Daenerys to be in Pentos.
			# =================================================

			if = {

				limit = {

					has_character_flag = segrellion_daenerys_camped_at_pentos

					NOT = {
						has_character_flag = segrellion_daenerys_massey_network_established
					}

					NOT = {
						has_character_flag = segrellion_daenerys_massey_contact_scheduled
					}
				}


				add_character_flag = {
					flag = segrellion_daenerys_massey_contact_scheduled
					days = 45
				}


				trigger_event = {

					id = segrellion_daenerys_massey.0001

					days = { 5 12 }
				}
			}


			# =================================================
			# INFORMATION NETWORK
			#
			# This does NOT require Pentos.
			# Massey continues sending information after
			# Daenerys has crossed to Westeros.
			# =================================================

			if = {

				limit = {

					has_character_flag = segrellion_daenerys_massey_network_established

					has_game_rule = segrellion_story_news_events_enabled
				}


				# =============================================
				# DAEMON TAKES THE FLAMELANDS
				# =============================================

				if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_daenerys_news_daemon_flamelands
						}


						character:Segrellion_60 ?= {

							is_alive = yes

							OR = {

								has_character_flag = segrellion_unicorn_returns_completed

								has_title = title:e_flamelands

								has_title = title:k_flamelands
							}
						}
					}


					add_character_flag = segrellion_daenerys_news_daemon_flamelands


					trigger_event = {

						id = segrellion_daenerys_massey.0010

						days = { 2 6 }
					}
				}


				# =============================================
				# AEGON TAKES THE IRON THRONE
				# =============================================

				else_if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_daenerys_news_aegon_throne
						}


						title:h_the_iron_throne.holder ?= {
							this = character:IAF_Young_Griff_1
						}
					}


					add_character_flag = segrellion_daenerys_news_aegon_throne


					trigger_event = {

						id = segrellion_daenerys_massey.0020

						days = { 2 6 }
					}
				}


				# =============================================
				# STANNIS TAKES THE NORTH
				# =============================================

				else_if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_daenerys_news_stannis_north
						}


						title:e_the_north.holder ?= {
							this = character:Baratheon_3
						}
					}


					add_character_flag = segrellion_daenerys_news_stannis_north


					trigger_event = {

						id = segrellion_daenerys_massey.0030

						days = { 2 6 }
					}
				}


				# =============================================
				# JON TAKES THE NORTH
				# =============================================

				else_if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_daenerys_news_jon_north
						}


						OR = {

							title:e_the_north.holder ?= {
								this = character:Stark_7
							}


							AND = {

								has_global_variable = segrellion_resurrected_jon

								title:e_the_north.holder ?= {
									this = global_var:segrellion_resurrected_jon
								}
							}
						}
					}


					add_character_flag = segrellion_daenerys_news_jon_north


					trigger_event = {

						id = segrellion_daenerys_massey.0040

						days = { 2 6 }
					}
				}


				# =============================================
				# AEGON REVEALED AS BLACKFYRE
				# =============================================

				else_if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_daenerys_news_aegon_blackfyre
						}


						character:IAF_Young_Griff_1 ?= {

							is_alive = yes

							has_character_flag = segrellion_young_griff_blackfyre_revealed
						}
					}


					add_character_flag = segrellion_daenerys_news_aegon_blackfyre


					trigger_event = {

						id = segrellion_daenerys_massey.0050

						days = { 2 6 }
					}
				}
			}
		}
	}
}

# =========================================================
# DAENERYS - START THE BEACON CROSSING
#
# ROOT = Daenerys
#
# Used both by the normal Massey -> Daemon negotiation
# chain and by the save-compatible recovery.
# =========================================================

segrellion_daenerys_start_beacon_crossing_effect = {

	if = {

		limit = {

			this = character:Targaryen_9

			is_alive = yes


			NOT = {
				exists = current_travel_plan
			}


			segrellion_daemon_controls_flamelands_trigger = yes
		}


		start_travel_plan = {

			destination = title:c_farring_cross.title_province

			players_use_planner = no

			return_trip = no

			travel_with_domicile = yes

			on_arrival_event = segrellion_daenerys_invasion.0018

			on_arrival_destinations = last

			on_travel_planner_cancel_event = segrellion_daenerys_invasion.0091
		}
	}
}


# =========================================================
# DAENERYS - BEACON NEGOTIATION / TRAVEL RECOVERY
#
# Repairs two possible broken states:
#
# 1. Daemon accepted, but "The Beacon Is Open" never fired.
#
# 2. Daenerys selected the Beacon, but the travel plan
#    failed to start.
#
# Save-compatible.
# =========================================================

segrellion_daenerys_beacon_negotiation_recovery_effect = {

	character:Targaryen_9 ?= {

		if = {

			limit = {

				is_alive = yes

				has_character_flag = segrellion_daenerys_camped_at_pentos

				has_character_flag = segrellion_daenerys_invasion_committed

				has_character_flag = segrellion_daenerys_crossing_committed


				character:Segrellion_60 ?= {

					is_alive = yes

					has_character_flag = segrellion_daemon_accepts_daenerys_beacon_negotiation
				}


				segrellion_daemon_controls_flamelands_trigger = yes
			}


			# =================================================
			# DAEMON ACCEPTED BUT THE FINAL EVENT NEVER ARRIVED
			# =================================================

			if = {

				limit = {

					has_character_flag = segrellion_daenerys_beacon_negotiation_started


					NOT = {
						has_character_flag = segrellion_daenerys_beacon_invitation_ready
					}


					NOT = {
						has_character_flag = segrellion_daenerys_landing_chosen
					}


					NOT = {
						has_character_flag = segrellion_daenerys_beacon_open_event_pending
					}
				}


				add_character_flag = {

					flag = segrellion_daenerys_beacon_open_event_pending

					days = 15
				}


				trigger_event = {

					id = segrellion_daenerys_massey.0150

					days = 1
				}
			}


			# =================================================
			# THE BEACON WAS CHOSEN BUT TRAVEL NEVER STARTED
			#
			# camped_at_pentos disappears when .0018 fires,
			# so this cannot send Daenerys back after arrival.
			# =================================================

			if = {

				limit = {

					has_character_flag = segrellion_daenerys_landing_chosen

					has_character_flag = segrellion_daenerys_landing_flamelands

					has_character_flag = segrellion_daenerys_landing_beacon


					NOT = {
						exists = current_travel_plan
					}
				}


				segrellion_daenerys_start_beacon_crossing_effect = yes
			}
		}
	}
}
