# =========================================================
# FORCE CRAVEN
# =========================================================

segrellion_force_craven_personality_effect = {

	if = {
		limit = {
			has_trait = craven
		}
	}


	else_if = {
		limit = {
			has_trait = brave
		}

		remove_trait = brave
		add_trait = craven
	}


	else_if = {
		limit = {

			calc_true_if = {
				amount >= 4

				has_trait = ambitious
				has_trait = arrogant
				has_trait = calm
				has_trait = chaste
				has_trait = compassionate
				has_trait = content
				has_trait = cynical
				has_trait = deceitful
				has_trait = diligent
				has_trait = eccentric
				has_trait = fickle
				has_trait = forgiving
				has_trait = generous
				has_trait = greedy
				has_trait = gregarious
				has_trait = honest
				has_trait = humble
				has_trait = impatient
				has_trait = just
				has_trait = lazy
				has_trait = lustful
				has_trait = paranoid
				has_trait = patient
				has_trait = sadistic
				has_trait = shy
				has_trait = stubborn
				has_trait = temperate
				has_trait = trusting
				has_trait = vengeful
				has_trait = wrathful
				has_trait = zealous
				has_trait = arbitrary
				has_trait = callous
				has_trait = authoritative
				has_trait = rude
				has_trait = loyal
				has_trait = disloyal
			}
		}

		segrellion_remove_one_random_personality_trait_effect = yes
		add_trait = craven
	}


	else = {
		add_trait = craven
	}
}
