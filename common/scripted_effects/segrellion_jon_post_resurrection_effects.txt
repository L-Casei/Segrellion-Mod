segrellion_jon_normalize_personality_effect = {

	# Remove the old Jon traits we no longer want.

	if = {
		limit = { has_trait = honest }
		remove_trait = honest
	}

	if = {
		limit = { has_trait = honorable }
		remove_trait = honorable
	}

	if = {
		limit = { has_trait = compassionate }
		remove_trait = compassionate
	}

	if = {
		limit = { has_trait = cynical }
		remove_trait = cynical
	}


	# Canon Segrellion Jon personality.

	if = {
		limit = {
			NOT = { has_trait = brave }
		}

		add_trait = brave
	}

	if = {
		limit = {
			NOT = { has_trait = just }
		}

		add_trait = just
	}

	if = {
		limit = {
			NOT = { has_trait = callous }
		}

		add_trait = callous
	}

	if = {
		limit = {
			NOT = { has_trait = vengeful }
		}

		add_trait = vengeful
	}


	# Jon is a Dreamer.

	if = {
		limit = {
			NOT = {
				has_trait = segrellion_dreamer_trait
			}
		}

		add_trait = segrellion_dreamer_trait
	}
}

segrellion_jon_execute_surviving_mutineers_effect = {

	# ROOT = Jon

	every_character = {
		limit = {
			is_alive = yes
			segrellion_jon_is_murder_mutineer_trigger = yes
		}

		add_character_flag = segrellion_jon_mutineer_judged

		death = {
			death_reason = death_execution
			killer = root
		}
	}
}

segrellion_jon_close_mutiny_storyline_effect = {

	# ROOT = Jon

	remove_character_flag = segrellion_jon_mutiny_active
	remove_character_flag = segrellion_jon_post_resurrection_chain_running

	add_character_flag = segrellion_jon_mutiny_closed

	# Compatibility with the original resurrection storyline.

	add_character_flag = segrellion_jon_mutineers_purged
	add_character_flag = segrellion_jon_waiting_for_stannis_northern_result


	# Continue toward the Stannis / Northern storyline.

	trigger_event = {
		id = segrellion_jon_resurrection.0220
		days = { 2 5 }
	}
}

segrellion_jon_post_resurrection_dreamer_progression_effect = {

	# ROOT = Jon

	if = {
		limit = {
			NOT = {
				has_trait = segrellion_dreamer_trait
			}
		}

		add_trait = segrellion_dreamer_trait
	}

	add_trait_xp = {
		trait = segrellion_dreamer_trait
		value = 25
	}

	segrellion_sync_dreamer_visual_level_effect = yes
}

segrellion_jon_apply_awakened_appearance_effect = {

	# ROOT = Jon
	# El portrait modifier conserva la cara de Jon y sustituye
	# únicamente los colores de pelo y ojos.

	if = {
		limit = {
			has_inactive_trait = purple_eyes
		}

		make_trait_active = purple_eyes
		remove_trait = purple_eyes
	}

	make_trait_inactive = cool_silver_hair

	add_character_flag = segrellion_jon_awakened_appearance
}
