# =========================================================
# JON SNOW - RESURRECTION
#
# The historical Stark_7 remains dead.
#
# A new runtime character is created as the resurrected
# continuation of Jon Snow.
# =========================================================

# =========================================================
# DENYS MALLISTER - SHADOW TOWER
#
# Denys is canonically commander of the Shadow Tower.
# This also guarantees that he remains landed while
# the player temporarily controls him after Jon's death.
# =========================================================

segrellion_setup_denys_shadow_tower_effect = {

	if = {
		limit = {

			game_start_date = 8300.8.20

			character:Mallister_6 ?= {
				is_alive = yes
			}
		}


		create_title_and_vassal_change = {

			type = granted

			save_scope_as = denys_shadow_tower_change

			add_claim_on_loss = no
		}


		# =================================================
		# SHADOW TOWER
		# =================================================

		title:d_shadow_tower = {

			change_title_holder_include_vassals = {

				holder = character:Mallister_6

				change = scope:denys_shadow_tower_change

				government_base = title:k_the_wall.holder
			}
		}


		title:c_shadow_tower = {

			change_title_holder = {

				holder = character:Mallister_6

				change = scope:denys_shadow_tower_change
			}
		}


		resolve_title_and_vassal_change = scope:denys_shadow_tower_change

		# =====================================================
		# DENYS MUST REMAIN UNDER THE LORD COMMANDER
		# =====================================================

		title:k_the_wall.holder ?= {

			save_scope_as = segrellion_denys_watch_liege
		}


		if = {

			limit = {

				exists = scope:segrellion_denys_watch_liege


				scope:segrellion_denys_watch_liege != character:Mallister_6


				character:Mallister_6 ?= {

					is_alive = yes

					has_title = title:d_shadow_tower


					top_liege != scope:segrellion_denys_watch_liege
				}
			}


			create_title_and_vassal_change = {

				type = returned

				save_scope_as = segrellion_denys_shadow_tower_liege_change

				add_claim_on_loss = no
			}


			character:Mallister_6 = {

				change_liege = {

					liege = scope:segrellion_denys_watch_liege

					change = scope:segrellion_denys_shadow_tower_liege_change
				}
			}


			resolve_title_and_vassal_change = scope:segrellion_denys_shadow_tower_liege_change
		}

		# =================================================
		# DENYS REMAINS A BLACK BROTHER
		# =================================================

		character:Mallister_6 = {

			if = {
				limit = {
					NOT = {
						has_trait = nightswatch
					}
				}

				add_trait = nightswatch
			}

			change_government = nights_watch_government

			set_primary_title_to = title:d_shadow_tower

			set_realm_capital = title:c_shadow_tower

			set_location = title:c_shadow_tower.title_province

			add_character_flag = segrellion_denys_shadow_tower_setup
		}
	}
}

# =========================================================
# JON SNOW - RESURRECTION
# =========================================================

segrellion_create_resurrected_jon_effect = {

	if = {

		limit = {

			character:Stark_7 = {
				is_alive = no
			}

			NOT = {
				has_global_variable = segrellion_resurrected_jon
			}
		}

		# =========================================================
		# THE MAN CURRENTLY COMMANDING THE WATCH
		# =========================================================

		root = {
			save_scope_as = interim_lord_commander
		}


		# =================================================
		# CREATE THE RETURNED JON
		# =================================================

		create_character = {

			name = "Jon"

			age = 17

			gender = male


			faith = character:Stark_7.faith

			culture = character:Stark_7.culture


			father = character:Stark_7.father

			real_father = character:Stark_7.real_father

			mother = character:Stark_7.mother


			dynasty_house = character:Stark_7.house


			location = title:c_castle_black.title_province


			random_traits = no


			diplomacy = 8
			martial = 14
			stewardship = 8
			intrigue = 10
			learning = 9
			prowess = 15


			save_scope_as = resurrected_jon


			after_creation = {

				# =========================================
				# FACE / GENETICS
				# =========================================

				copy_inheritable_appearance_from = character:Stark_7


				# =========================================
				# ORIGINAL TRAITS
				# =========================================

				copy_traits = character:Stark_7


				set_house = character:Stark_7.house

				set_sexuality = heterosexual


				add_character_flag = canon_status_canon

				add_character_flag = segrellion_resurrected_jon

				add_character_flag = segrellion_canonical_current_jon

				add_character_flag = segrellion_resurrected_jon_appearance

				add_character_flag = segrellion_jon_force_no_headgear

				add_character_flag = has_scripted_appearance

				add_character_flag = hair_trait_m_hair_ep3_byzantine_01

				add_character_flag = beard_trait_no_beard	


				# =========================================
				# HE RETURNS AS LORD COMMANDER
				# =========================================

				if = {

					limit = {
						NOT = {
							has_trait = nightswatch
						}
					}

					add_trait = nightswatch
				}


				# =========================================
				# REMOVE THE WOUNDS THAT KILLED HIM
				# =========================================

				if = {
					limit = { has_trait = wounded_1 }
					remove_trait = wounded_1
				}

				if = {
					limit = { has_trait = wounded_2 }
					remove_trait = wounded_2
				}

				if = {
					limit = { has_trait = wounded_3 }
					remove_trait = wounded_3
				}


				# =========================================
				# DEATH CHANGED HIM
				# =========================================

				segrellion_jon_normalize_personality_effect = yes


				# =========================================
				# PHYSICAL CHANGE
				# =========================================

				make_trait_inactive = cool_silver_hair

				make_trait_inactive = purple_eyes


				# =========================================
				# RESURRECTED
				# =========================================

				add_trait = segrellion_resurrected_trait


				if = {

					limit = {

						NOT = {
							has_trait = refusing_marriage
						}
					}

					add_trait = refusing_marriage
				}


				add_character_modifier = {
					modifier = segrellion_jon_returned_from_death_modifier
				}


				# =========================================
				# NORTHERN CLAIMS
				# =========================================

				add_pressed_claim = title:e_the_north

				add_pressed_claim = title:c_winterfell
			}
		}


		# =================================================
		# STORE NEW JON
		# =================================================

		set_global_variable = {

			name = segrellion_resurrected_jon

			value = scope:resurrected_jon
		}


		set_global_variable = {

			name = segrellion_jon_resurrection_completed

			value = yes
		}


		# =================================================
		# JON RECLAIMS THE NIGHT'S WATCH
		# =================================================

		create_title_and_vassal_change = {

			type = granted

			save_scope_as = jon_returns_to_watch_change

			add_claim_on_loss = no
		}


		title:k_the_wall = {

			change_title_holder_include_vassals = {

				holder = scope:resurrected_jon

				change = scope:jon_returns_to_watch_change

				government_base = scope:interim_lord_commander
			}
		}


		if = {

			limit = {

				title:d_the_wall.holder ?= {
					this = scope:interim_lord_commander
				}
			}


			title:d_the_wall = {

				change_title_holder = {

					holder = scope:resurrected_jon

					change = scope:jon_returns_to_watch_change
				}
			}
		}


		if = {

			limit = {

				title:c_castle_black.holder ?= {
					this = scope:interim_lord_commander
				}
			}


			title:c_castle_black = {

				change_title_holder = {

					holder = scope:resurrected_jon

					change = scope:jon_returns_to_watch_change
				}
			}
		}


		resolve_title_and_vassal_change = scope:jon_returns_to_watch_change

		# =========================================================
		# JON IS FULLY RESTORED AS LORD COMMANDER FIRST
		# =========================================================

		scope:resurrected_jon = {

			if = {

				limit = {

					NOT = {
						has_trait = nightswatch
					}
				}

				add_trait = nightswatch
			}


			change_government = nights_watch_government

			set_primary_title_to = title:k_the_wall

			set_realm_capital = title:c_castle_black

			set_location = title:c_castle_black.title_province


			# =================================================
			# CRITICAL HANDOFF
			#
			# 1. Switch human player back to Jon immediately.
			# 2. Put Denys beneath Jon.
			# 3. Return the Night's Watch court to Jon.
			# =================================================

			segrellion_repair_resurrected_jon_watch_handoff_effect = yes
		}


		# =========================================================
		# ONLY AFTER THE PLAYER HAS RETURNED TO JON,
		# DENYS RETURNS TO THE SHADOW TOWER
		# =========================================================

		character:Mallister_6 ?= {

			if = {

				limit = {

					is_alive = yes

					has_title = title:d_shadow_tower
				}


				if = {

					limit = {

						NOT = {
							has_trait = nightswatch
						}
					}

					add_trait = nightswatch
				}


				change_government = nights_watch_government

				set_primary_title_to = title:d_shadow_tower

				set_realm_capital = title:c_shadow_tower

				set_location = title:c_shadow_tower.title_province
			}
		}

		# =========================================================
		# LONGCLAW RETURNS TO JON
		#
		# Arya / fake Arya may have inherited it after Jon's death.
		# We take the SAME artifact object back.
		# =========================================================

		if = {

			limit = {

				has_global_variable = segrellion_jon_longclaw_artifact
			}


			global_var:segrellion_jon_longclaw_artifact ?= {

				set_owner = {

					target = scope:resurrected_jon


					history = {

						type = inherited

						recipient = scope:resurrected_jon
					}
				}


				equip_artifact_to_owner_replace = yes
			}
		}

		# =========================================================
		# EVERY OTHER HUMAN PLAYER
		#
		# Gets the major phase event instead.
		# =========================================================

		every_player = {

			limit = {

				is_ai = no

				this != scope:resurrected_jon

				this != character:Mallister_6
			}


			trigger_event = {

				id = segrellion_jon_resurrection.0205

				days = 1
			}
		}
	}
}


# =========================================================
# JON LEAVES THE NIGHT'S WATCH
#
# This is NOT used when he resurrects.
#
# It is used only AFTER he judges the mutineers and chooses
# to march against Winterfell.
# =========================================================

segrellion_jon_leave_watch_for_winterfell_effect = {

	# =====================================================
	# PREFER THE SAME MAN WHO HELD THE WALL WHILE JON WAS DEAD
	# =====================================================

	if = {

		limit = {

			has_global_variable = segrellion_jon_loyal_lord_commander


			global_var:segrellion_jon_loyal_lord_commander ?= {

				is_alive = yes

				has_trait = nightswatch

				this != root
			}
		}


		global_var:segrellion_jon_loyal_lord_commander = {

			save_scope_as = jon_next_lord_commander
		}
	}


	# =====================================================
	# FALLBACK
	# =====================================================

	if = {

		limit = {

			NOT = {
				exists = scope:jon_next_lord_commander
			}
		}


		ordered_vassal_or_below = {

			limit = {

				is_alive = yes
				is_adult = yes

				has_trait = nightswatch

				this != root


				opinion = {
					target = root
					value >= 0
				}
			}

			order_by = diplomacy

			position = 0

			check_range_bounds = no

			save_scope_as = jon_next_lord_commander
		}
	}


	# =====================================================
	# TRANSFER THE WATCH
	# =====================================================

	if = {

		limit = {
			exists = scope:jon_next_lord_commander
		}


		create_title_and_vassal_change = {

			type = granted

			save_scope_as = jon_leaves_watch_change

			add_claim_on_loss = no
		}


		title:k_the_wall = {

			change_title_holder_include_vassals = {

				holder = scope:jon_next_lord_commander

				change = scope:jon_leaves_watch_change

				government_base = root
			}
		}


		if = {

			limit = {

				title:d_the_wall.holder ?= {
					this = root
				}
			}


			title:d_the_wall = {

				change_title_holder = {

					holder = scope:jon_next_lord_commander

					change = scope:jon_leaves_watch_change
				}
			}
		}


		if = {

			limit = {

				title:c_castle_black.holder ?= {
					this = root
				}
			}


			title:c_castle_black = {

				change_title_holder = {

					holder = scope:jon_next_lord_commander

					change = scope:jon_leaves_watch_change
				}
			}
		}


		resolve_title_and_vassal_change = scope:jon_leaves_watch_change


		scope:jon_next_lord_commander = {
			set_realm_capital = title:c_castle_black
		}


		# =================================================
		# JON'S VOWS ENDED WITH HIS FIRST DEATH
		# =================================================


		add_character_flag = segrellion_jon_watch_ended_in_death

		add_character_flag = segrellion_jon_left_nights_watch



		if = {

			limit = {

				NOT = {
					has_realm_law = camp_purpose_legitimists
				}
			}


			add_realm_law_skip_effects = camp_purpose_legitimists
		}


		set_location = title:c_castle_black.title_province
	}
}


# =========================================================
# RESURRECTED JON
# RESTORE PLAYER + WATCH COURT + DENYS VASSALAGE
#
# ROOT = resurrected Jon
# =========================================================

segrellion_repair_resurrected_jon_watch_handoff_effect = {

	if = {

		limit = {

			is_alive = yes

			has_character_flag = segrellion_canonical_current_jon

			has_title = title:k_the_wall
		}


		# =================================================
		# PLAYER HANDOFF
		#
		# NEVER leave the human player on Denys after Jon
		# has reclaimed the Wall.
		# =================================================

		if = {

			limit = {

				is_playable_character = yes


				NOT = {
					has_character_flag = segrellion_jon_player_restored_after_resurrection
				}
			}


			if = {

				limit = {

					character:Mallister_6 ?= {

						is_alive = yes

						is_ai = no

						has_character_flag = segrellion_jon_resurrection_chain_started
					}
				}


				character:Mallister_6 = {

					set_player_character = root
				}
			}


			add_character_flag = segrellion_jon_player_restored_after_resurrection
		}


		# =================================================
		# DENYS RETURNS UNDER THE LORD COMMANDER
		# =================================================

		if = {

			limit = {

				character:Mallister_6 ?= {

					is_alive = yes

					has_title = title:d_shadow_tower

					top_liege != root
				}
			}


			create_title_and_vassal_change = {

				type = returned

				save_scope_as = segrellion_denys_returns_under_watch_change

				add_claim_on_loss = no
			}


			character:Mallister_6 = {

				change_liege = {

					liege = root

					change = scope:segrellion_denys_returns_under_watch_change
				}
			}


			resolve_title_and_vassal_change = scope:segrellion_denys_returns_under_watch_change
		}


		# =================================================
		# RETURN THE LORD COMMANDER'S COURT TO JON
		#
		# Title transfer does NOT transfer courtiers.
		#
		# Capture them first, then move them, so modifying
		# employment does not mutate the iterator itself.
		# =================================================

		if = {

			limit = {

				NOT = {
					has_character_flag = segrellion_jon_watch_court_restored
				}
			}


			character:Mallister_6 ?= {

				every_courtier = {

					limit = {

						is_alive = yes

						is_landed = no
					}


					add_to_temporary_list = segrellion_watch_courtiers_to_restore
				}
			}


			every_in_list = {

				list = segrellion_watch_courtiers_to_restore


				set_employer = root
			}


			add_character_flag = segrellion_jon_watch_court_restored
		}
	}
}
