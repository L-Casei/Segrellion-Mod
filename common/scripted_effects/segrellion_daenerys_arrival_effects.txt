# =========================================================
# DAENERYS TARGARYEN
# GLOBAL WESTEROS ARRIVAL ANNOUNCEMENT
# =========================================================


# =========================================================
# ANNOUNCE DAENERYS' ARRIVAL IN WESTEROS
#
# Expected scope: Daenerys Targaryen
#
# This effect can safely be called from ANY future landing
# route. It will only ever announce her arrival once.
# =========================================================

segrellion_daenerys_announce_westeros_arrival_effect = {

	if = {

		limit = {

			this = character:Targaryen_9

			is_alive = yes

			has_character_flag = segrellion_daenerys_landed_in_westeros

			NOT = {
				has_character_flag = segrellion_daenerys_westeros_arrival_announced
			}
		}


		# Prevent duplicate announcements if several
		# landing systems call this effect.
		add_character_flag = segrellion_daenerys_westeros_arrival_announced


		# =================================================
		# HUMAN PLAYERS ONLY
		#
		# Daenerys herself is explicitly excluded.
		# =================================================

		every_player = {

			limit = {

				this != character:Targaryen_9
			}

			trigger_event = segrellion_daenerys_arrival.0001
		}
	}
}
