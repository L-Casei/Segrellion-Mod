# =========================================================
# DAENERYS TARGARYEN
# GLOBAL WESTEROS ARRIVAL ANNOUNCEMENT
# =========================================================

segrellion_daenerys_announce_westeros_arrival_effect = {

    if = {

        limit = {

            this = character:Targaryen_9

            is_alive = yes

            has_character_flag = segrellion_daenerys_landed_in_westeros

            NOT = {
                has_character_flag = segrellion_daenerys_westeros_arrival_announced
            }
        }


        add_character_flag = segrellion_daenerys_westeros_arrival_announced


        every_player = {

            limit = {

                this != character:Targaryen_9
            }


            trigger_event = {

                id = segrellion_daenerys_arrival.0001

                days = 1
            }
        }
    }
}
