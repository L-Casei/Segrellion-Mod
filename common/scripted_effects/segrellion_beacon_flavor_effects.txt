# =========================================================
# REPEATABLE SEGRELLION LORE
#
# The existing segrellion_add_lore_effect also marks a POI
# as visited. Castle events must not do that, so they use
# this smaller repeatable helper instead.
# =========================================================

segrellion_beacon_add_lore_effect = {

	if = {
		limit = {
			exists = var:segrellion_lore
		}

		change_variable = {
			name = segrellion_lore
			add = $AMOUNT$
		}
	}

	else = {

		set_variable = {
			name = segrellion_lore
			value = $AMOUNT$
		}
	}
}

segrellion_beacon_flavor_pool_effect = {
