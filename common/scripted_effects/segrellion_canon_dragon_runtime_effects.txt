# ============================================================
# ICE AND FIRE - CANON DRAGON RUNTIME DNA
#
# Converts the supplied persistent Dragon DNA into AGOT's
# runtime story_dragon_variable_storage system.
#
# IMPORTANT:
# Only the numeric template IDs for horns/body/wings are
# provisional. Every other value comes from the supplied DNA.
# ============================================================

# ============================================================
# NATIVE AGOT HISTORICAL DRAGON INITIALIZATION
#
# Mirrors the exact architecture used by AGOT canon dragons:
#
# 1. Store the dragon's current historical age on the dragon.
# 2. Start size at 0.
# 3. Back-apply every year of historical growth.
# 4. Apply the canonical appearance variables to the dragon.
# 5. Transfer everything into story_dragon_variable_storage.
#
# This happens immediately at game start.
# NO day advancement.
# NO delayed event.
# NO monthly initialization.
# ============================================================

segrellion_native_historical_dragon_setup_effect = {

	$DRAGON$ = {
		save_scope_as = dragon
	}

	# ========================================================
	# DOES THIS DRAGON ALREADY HAVE A VALID AGOT STORAGE?
	#
	# Runtime hatchlings created through
	# agot_dragon_hatchling_template already have one.
	# We must NOT create a second one.
	# ========================================================

	if = {
		limit = {
			any_in_global_list = {
				variable = gl_dragon_variable_storage

				var:dragon_id ?= scope:dragon

				exists = var:dragon_size_base

				OR = {
					var:dragon_size_base > 0

					scope:dragon = {
						age < 1
					}
				}
			}
		}

		# Existing runtime storage.
		# Only replace our canonical appearance variables.
		# DO NOT touch historical growth or create another story.

		every_in_global_list = {
			variable = gl_dragon_variable_storage

			limit = {
				var:dragon_id ?= scope:dragon
			}

			set_variable = {
				name = gene_dragon_primary_color_value
				value = $gene_dragon_primary_color_value$
			}
			set_variable = {
				name = gene_dragon_primary_color_hue
				value = $gene_dragon_primary_color_hue$
			}

			set_variable = {
				name = gene_dragon_secondary_value
				value = $gene_dragon_secondary_value$
			}
			set_variable = {
				name = gene_dragon_secondary_hue
				value = $gene_dragon_secondary_hue$
			}

			set_variable = {
				name = gene_dragon_tertiary_value
				value = $gene_dragon_tertiary_value$
			}
			set_variable = {
				name = gene_dragon_tertiary_hue
				value = $gene_dragon_tertiary_hue$
			}

			set_variable = {
				name = gene_dragon_horn_color_value
				value = $gene_dragon_horn_color_value$
			}
			set_variable = {
				name = gene_dragon_horn_color_hue
				value = $gene_dragon_horn_color_hue$
			}

			set_variable = {
				name = gene_dragon_eye_color_value

			}
			set_variable = {
				name = gene_dragon_eye_color_hue
				value = $gene_dragon_eye_color_hue$
			}

			set_variable = {
				name = gene_dragon_main_horn_shape_template
				value = $gene_dragon_main_horn_shape_template$
			}
			set_variable = {
				name = gene_dragon_main_horn_shape_value
				value = $gene_dragon_main_horn_shape_value$
			}

			set_variable = {
				name = gene_dragon_body_shading_template
				value = $gene_dragon_body_shading_template$
			}
			set_variable = {
				name = gene_dragon_body_shading_value
				value = $gene_dragon_body_shading_value$
			}

			set_variable = {
				name = gene_dragon_wings_shading_template
				value = $gene_dragon_wings_shading_template$
			}
			set_variable = {
				name = gene_dragon_wings_shading_value
				value = $gene_dragon_wings_shading_value$
			}

			set_variable = {
				name = gene_dragon_neck_length
				value = $gene_dragon_neck_length$
			}
			set_variable = {
				name = gene_dragon_metallic_scales_strength
				value = $gene_dragon_metallic_scales_strength$
			}
			set_variable = {
				name = gene_dragon_wounded
				value = $gene_dragon_wounded$
			}
			set_variable = {
				name = gene_dragon_old_neck
				value = $gene_dragon_old_neck$
			}
			set_variable = {
				name = gene_dragon_tail_length
				value = $gene_dragon_tail_length$
			}

			set_variable = {
				name = gene_dragon_center_fin_size
				value = $gene_dragon_center_fin_size$
			}
			set_variable = {
				name = gene_dragon_side_fin_size
				value = $gene_dragon_side_fin_size$
			}
			set_variable = {
				name = gene_dragon_crest_depth
				value = $gene_dragon_crest_depth$
			}
			set_variable = {
				name = gene_dragon_back_spike_size
				value = $gene_dragon_back_spike_size$
			}
			set_variable = {
				name = gene_dragon_neck_spike_size
				value = $gene_dragon_neck_spike_size$
			}

			set_variable = {
				name = gene_dragon_horns_eyebrow_length
				value = $gene_dragon_horns_eyebrow_length$
			}
			set_variable = {
				name = gene_dragon_horns_eyebrow
				value = $gene_dragon_horns_eyebrow$
			}

			set_variable = {
				name = gene_dragon_snout_length
				value = $gene_dragon_snout_length$
			}
			set_variable = {
				name = gene_dragon_snout_height

				name = gene_dragon_snout_profile
				value = $gene_dragon_snout_profile$
			}

				name = gene_dragon_chin_profile
				value = $gene_dragon_chin_profile$
			}
			set_variable = {
				name = gene_dragon_jaw_width
				value = $gene_dragon_jaw_width$
			}
			set_variable = {
				name = gene_dragon_upper_jaw_width
				value = $gene_dragon_upper_jaw_width$

			}
			set_variable = {
				name = gene_dragon_lower_jaw_height
				value = $gene_dragon_lower_jaw_height$
			}

				name = gene_dragon_cheek_width
				value = $gene_dragon_cheek_width$
			}
			set_variable = {
				name = gene_dragon_head_roundness
				value = $gene_dragon_head_roundness$
			}
			set_variable = {
				name = gene_dragon_brow_width
				value = $gene_dragon_brow_width$
			}
			set_variable = {
				name = gene_dragon_outer_brow_height
				value = $gene_dragon_outer_brow_height$
			}
		}
	}

	else = {

		# ====================================================
		# NO VALID STORAGE
		#
		# Remove references to any broken size-0 storage.
		# Do not end_story here.
		# ====================================================

		every_in_global_list = {
			variable = gl_dragon_variable_storage

			limit = {
				var:dragon_id ?= scope:dragon
			}

			remove_list_global_variable = {
				name = gl_dragon_variable_storage
				target = this
			}
		}


		# ====================================================
		# THIS IS AGOT'S REAL HISTORICAL DRAGON SYSTEM
		# ====================================================

		agot_historical_dragon_transfer_vars_to_story_cycle_effect = {

			DRAGON = scope:dragon
			dragon_age = $DRAGON_AGE$

			gene_dragon_primary_color_value = $gene_dragon_primary_color_value$

			gene_dragon_horn_color_value = $gene_dragon_horn_color_value$
			gene_dragon_horn_color_hue = $gene_dragon_horn_color_hue$

			gene_dragon_eye_color_value = $gene_dragon_eye_color_value$

			gene_dragon_main_horn_shape_template = $gene_dragon_main_horn_shape_template$
			gene_dragon_main_horn_shape_value = $gene_dragon_main_horn_shape_value$

			gene_dragon_body_shading_template = $gene_dragon_body_shading_template$
			gene_dragon_body_shading_value = $gene_dragon_body_shading_value$

			gene_dragon_wings_shading_template = $gene_dragon_wings_shading_template$
			gene_dragon_wings_shading_value = $gene_dragon_wings_shading_value$

			gene_dragon_neck_length = $gene_dragon_neck_length$
			gene_dragon_metallic_scales_strength = $gene_dragon_metallic_scales_strength$
			gene_dragon_wounded = $gene_dragon_wounded$
			gene_dragon_old_neck = $gene_dragon_old_neck$
			gene_dragon_tail_length = $gene_dragon_tail_length$

			gene_dragon_crest_depth = $gene_dragon_crest_depth$
