segrellion_refresh_heir_of_the_unicorn_effect = {
	every_living_character = {
		limit = {
			has_character_modifier = heir_of_the_unicorn_modifier
			has_variable = segrellion_unicorn_designator
		}

		if = {
			limit = {
				var:segrellion_unicorn_designator = { is_alive = no }
			}
			remove_character_modifier = heir_of_the_unicorn_modifier
			remove_variable = segrellion_unicorn_designator
		}
		else_if = {
			limit = {
				var:segrellion_unicorn_designator = {
					NOR = {
						primary_heir ?= prev
						designated_heir ?= prev
					}
				}
			}
			remove_character_modifier = heir_of_the_unicorn_modifier
			var:segrellion_unicorn_designator ?= {
				remove_variable = segrellion_designated_unicorn_heir
			}
			remove_variable = segrellion_unicorn_designator
		}
	}
}

segrellion_convert_inherited_unicorn_heir_effect = {
	if = {
		limit = {
			has_character_modifier = heir_of_the_unicorn_modifier
			has_variable = segrellion_unicorn_designator
			var:segrellion_unicorn_designator ?= scope:previous_holder
		}
		remove_character_modifier = heir_of_the_unicorn_modifier
		remove_variable = segrellion_unicorn_designator
		add_character_modifier = {
			modifier = chosen_by_the_unicorn_modifier
			years = 10
		}
	}
}


# =========================================================
# LEGACY V
# CALL UPON THE GREAT HOUSES
# =========================================================

segrellion_call_upon_great_houses_effect = {

	save_scope_as = segrellion_great_house_caller


	# =====================================================
	# NORMAL WESTEROS
	#
	# If the Iron Throne exists, ask every major direct
	# vassal of the Iron Throne.
	#
	# This naturally includes the Lords Paramount and also
	# custom Paramountcies such as the Flamelands.
	# =====================================================

	if = {

		limit = {

			title:h_the_iron_throne.holder ?= {
				is_alive = yes
			}
		}

		title:h_the_iron_throne.holder = {

			every_vassal = {

				limit = {

					segrellion_legacy_v_valid_great_house_target_trigger = {
						CALLER = scope:segrellion_great_house_caller
					}
				}

				save_scope_as = segrellion_great_house_recipient


				# Only send the letter if there is at least
				# one current war this ruler could join.

				if = {

					limit = {

						scope:segrellion_great_house_caller = {

							any_character_war = {

								segrellion_legacy_v_callable_war_trigger = {
									CALLER = scope:segrellion_great_house_caller
									TARGET = scope:segrellion_great_house_recipient
								}
							}
						}
					}

					trigger_event = segrellion_legacy.5110
				}
			}
		}
	}


	# =====================================================
	# FALLBACK
	#
	# For shattered realms / Iron Throne destroyed.
	# If the caller has a liege, call the other great
	# vassals beneath that liege.
	# =====================================================

	else_if = {

		limit = {
			exists = liege
		}

		liege = {

			every_vassal = {

				limit = {

					segrellion_legacy_v_valid_great_house_target_trigger = {
						CALLER = scope:segrellion_great_house_caller
					}
				}

				save_scope_as = segrellion_great_house_recipient

				if = {

					limit = {

						scope:segrellion_great_house_caller = {

							any_character_war = {

								segrellion_legacy_v_callable_war_trigger = {
									CALLER = scope:segrellion_great_house_caller
									TARGET = scope:segrellion_great_house_recipient
								}
							}
						}
					}

					trigger_event = segrellion_legacy.5110
				}
			}
		}
	}


	# =====================================================
	# INDEPENDENT / NO IRON THRONE
	# =====================================================

	else = {

		every_neighboring_top_liege_realm_owner = {

			limit = {

				segrellion_legacy_v_valid_great_house_target_trigger = {
					CALLER = scope:segrellion_great_house_caller
				}
			}

			save_scope_as = segrellion_great_house_recipient

			if = {

				limit = {

					scope:segrellion_great_house_caller = {

						any_character_war = {

							segrellion_legacy_v_callable_war_trigger = {
								CALLER = scope:segrellion_great_house_caller
								TARGET = scope:segrellion_great_house_recipient
							}
						}
					}
				}

				trigger_event = segrellion_legacy.5110
			}
		}
	}
}
