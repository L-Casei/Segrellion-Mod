segrellion_refresh_heir_of_the_unicorn_effect = {
	every_living_character = {
		limit = {
			has_character_modifier = heir_of_the_unicorn_modifier
			has_variable = segrellion_unicorn_designator
		}

		if = {
			limit = {
				var:segrellion_unicorn_designator = { is_alive = no }
			}
			remove_character_modifier = heir_of_the_unicorn_modifier
			remove_variable = segrellion_unicorn_designator
		}
		else_if = {
			limit = {
				var:segrellion_unicorn_designator = {
					NOR = {
						primary_heir ?= prev
						designated_heir ?= prev
					}
				}
			}
			remove_character_modifier = heir_of_the_unicorn_modifier
			var:segrellion_unicorn_designator ?= {
				remove_variable = segrellion_designated_unicorn_heir
			}
			remove_variable = segrellion_unicorn_designator
		}
	}
}

segrellion_convert_inherited_unicorn_heir_effect = {
	if = {
		limit = {
			has_character_modifier = heir_of_the_unicorn_modifier
			has_variable = segrellion_unicorn_designator
			var:segrellion_unicorn_designator ?= scope:previous_holder
		}
		remove_character_modifier = heir_of_the_unicorn_modifier
		remove_variable = segrellion_unicorn_designator
		add_character_modifier = {
			modifier = chosen_by_the_unicorn_modifier
			years = 10
		}
	}
}
