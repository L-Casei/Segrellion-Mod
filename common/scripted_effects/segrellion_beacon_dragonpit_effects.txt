# ---------------------------------------------------------
# Keep the Dragonpit barony personal to the Lord of Beacon
# and permanently use its post-construction name.
# ---------------------------------------------------------

segrellion_repair_great_dragonpit_title_effect = {

	if = {
		limit = {
			province:7263 = {
				OR = {
					has_building_or_higher = dragonpit_01

					# Compatibility with saves from older mod versions.
					has_building_or_higher = segrellion_beacon_dragonpit_01
					has_building_or_higher = generic_dragon_pit_01
				}
			}

			exists = title:c_farring_cross.holder
		}

		# Keep the original AGOT place name. This also repairs old saves
		# where an earlier mod version renamed the barony.
		title:b_knights_bridge = {
			reset_title_name = yes
		}

		if = {
			limit = {
				title:b_knights_bridge = {
					NOT = {
						holder = title:c_farring_cross.holder
					}
				}
			}

			title:c_farring_cross.holder = {
				save_scope_as = segrellion_beacon_holder

				create_title_and_vassal_change = {
					type = revoked
					save_scope_as = segrellion_great_dragonpit_title_change
					add_claim_on_loss = no
				}

				title:b_knights_bridge = {
					change_title_holder_include_vassals = {
						holder = scope:segrellion_beacon_holder
						change = scope:segrellion_great_dragonpit_title_change
					}
				}

				resolve_title_and_vassal_change = scope:segrellion_great_dragonpit_title_change
			}
		}
	}
}


# ---------------------------------------------------------
# AGOT Dragonkeeper state
# ---------------------------------------------------------

segrellion_activate_beacon_dragonpit_effect = {

	if = {

		limit = {

			title:c_farring_cross = {
				any_county_province = {
					segrellion_beacon_dragonpit_building_active_trigger = yes
				}
			}
		}

		title:c_farring_cross = {

			if = {
				limit = {
					NOT = {
						has_variable = has_dragonkeeper_order
					}
				}

				set_variable = {
					name = has_dragonkeeper_order
				}
			}
		}

		if = {
			limit = {
				agot_has_any_dragonpit_access_law = no
			}

			agot_dragonpit_access_law_close_family_effect = yes
		}

		if = {
			limit = {
				has_ep1_dlc_trigger = yes
			}

			hidden_effect = {
				set_amenity_level = {
					type = agot_dragonpit_amenities
					value = 3
				}
			}
		}

		if = {

			limit = {

				NOT = {
					employs_court_position = head_dragonkeeper_court_position
				}

				NOT = {
					has_character_flag = segrellion_beacon_head_dragonkeeper_event_pending
				}
			}

			add_character_flag = {
				flag = segrellion_beacon_head_dragonkeeper_event_pending
				days = 30
			}

			trigger_event = {
				id = agot_dragonkeepers.0001
				days = 1
			}
		}
	}
}


# ---------------------------------------------------------
# Canon Segrellion dragons move to the Great Dragonpit in 160 AC.
#
# This fires once. It deliberately does not keep forcing dragons back
# into the pit after the player later releases or relocates them.
#
# The rider-dynasty check also catches dynamically-created Segrellion
# dragons in campaigns played forward from early bookmarks.
# ---------------------------------------------------------

segrellion_seed_great_dragonpit_dragons_effect = {

	if = {
		limit = {
			current_date >= 8160.1.1

			NOT = {
				has_global_variable = segrellion_great_dragonpit_dragons_initialized
			}

			province:7263 = {
				has_building_or_higher = dragonpit_01
			}

			title:c_farring_cross = {
				has_variable = has_dragonkeeper_order
			}
		}

		every_living_character = {
			limit = {
				has_trait = dragon

				OR = {
					# Canon Ice and Fire dragon characters.
					this = character:dragon_vhalyrax
					this = character:dragon_maenor
					this = character:dragon_valor
					this = character:dragon_azuron
					this = character:dragon_silverwing

					# Dynamic early-bookmark versions.
					has_character_flag = segrellion_vhalyrax_dragon
					has_character_flag = segrellion_maenor_dragon
					has_character_flag = segrellion_nightdancer_dragon

					# Any other dragon actually ridden by a Segrellion in 160.
					AND = {
						exists = var:current_rider
						var:current_rider = {
							dynasty = dynasty:dynn_Segrellion
						}
					}
				}
			}

			save_scope_as = segrellion_dragon_to_move

			# Remove cleanly from Dragonstone, King's Landing, Vhal or any
			# other pit before adding it to the Great Dragonpit.
			if = {
				limit = {
					has_character_flag = in_dragonpit
					exists = var:pitted_dragon_location
				}

				var:pitted_dragon_location = {
					save_scope_as = segrellion_previous_dragonpit
				}

				agot_remove_from_dragonpit_no_event = {
					DRAGON = scope:segrellion_dragon_to_move
					DRAGONPIT_COUNTY = scope:segrellion_previous_dragonpit
				}
			}

			agot_send_to_dragonpit_no_event = {
				DRAGON = scope:segrellion_dragon_to_move
				DRAGONPIT_COUNTY = title:c_farring_cross
			}
		}

		set_global_variable = {
			name = segrellion_great_dragonpit_dragons_initialized
			value = yes
		}
	}
}


# ---------------------------------------------------------
# Historical / save-compatible state repair
# ---------------------------------------------------------

segrellion_beacon_dragonpit_historical_progression_effect = {

	# Canon inauguration: 160 AC. New bookmarks at or after this date
	# receive the castle and AGOT's fully completed native Dragonpit.
	if = {
		limit = {
			current_date >= 8160.1.1

			NOT = {
				has_global_variable = segrellion_beacon_dragonpit_project_active
			}

			province:7263 = {
				OR = {
					has_holding = no
					has_holding_type = castle_holding
				}
			}
		}

		province:7263 = {

			if = {
				limit = {
					has_holding = no
				}

				set_holding_type = castle_holding
				add_building = castle_01
			}

			if = {
				limit = {
					has_special_building_slot = no
				}

				add_special_building_slot = dragonpit_03
			}

			if = {
				limit = {
					NOT = {
						has_building_or_higher = dragonpit_01
					}
				}

				add_special_building = dragonpit_03
			}
		}
	}


	# Migrate earlier Ice and Fire versions that used a custom Great
	# Building plus generic_dragon_pit_01 as a backend. The result is
	# now one visible Special Building: AGOT's native dragonpit_03.
	if = {
		limit = {
			province:7263 = {
				OR = {
					has_building_or_higher = segrellion_beacon_dragonpit_01
					has_building_or_higher = generic_dragon_pit_01
				}

				NOT = {
					has_building_or_higher = dragonpit_01
				}
			}
		}

		province:7263 = {

			if = {
				limit = {
					has_special_building_slot = no
				}

				add_special_building_slot = dragonpit_03
			}

			if = {
				limit = {
					has_building_or_higher = generic_dragon_pit_01
				}

				remove_building = generic_dragon_pit_01
			}

			if = {
				limit = {
					has_building_or_higher = segrellion_beacon_dragonpit_01
				}

				remove_building = segrellion_beacon_dragonpit_01
			}

			add_special_building = dragonpit_03
		}
	}


	# Never leave the obsolete generic backend beside the real Special
	# Building, otherwise the holding would display two Dragonpits.
	if = {
		limit = {
			province:7263 = {
				has_building_or_higher = dragonpit_01
				has_building_or_higher = generic_dragon_pit_01
			}
		}

		province:7263 = {
			remove_building = generic_dragon_pit_01
		}
	}


	if = {
		limit = {
			province:7263 = {
				has_building_or_higher = dragonpit_01
			}
		}

		if = {
			limit = {
				has_global_variable = segrellion_beacon_dragonpit_project_active
			}

			remove_global_variable = segrellion_beacon_dragonpit_project_active
		}

		set_global_variable = {
			name = segrellion_beacon_dragonpit_completed
			value = yes
		}

		segrellion_repair_great_dragonpit_title_effect = yes

		title:c_farring_cross.holder ?= {
			segrellion_activate_beacon_dragonpit_effect = yes
		}

		segrellion_seed_great_dragonpit_dragons_effect = yes
	}
}
