# =========================================================
# KNIGHT'S BRIDGE DRAGONPIT
# AGOT DRAGONKEEPER INTEGRATION
#
# ROOT = holder of c_farring_cross
# =========================================================

segrellion_activate_beacon_dragonpit_effect = {

	if = {

		limit = {

			title:c_farring_cross = {

				any_county_province = {
					segrellion_beacon_dragonpit_building_active_trigger = yes
				}
			}
		}

		# AGOT keys its dragonpit systems from this county variable.
		title:c_farring_cross = {

			if = {
				limit = {
					NOT = {
						has_variable = has_dragonkeeper_order
					}
				}

				set_variable = {
					name = has_dragonkeeper_order
				}
			}
		}

		if = {
			limit = {
				agot_has_any_dragonpit_access_law = no
			}

			agot_dragonpit_access_law_close_family_effect = yes
		}

		if = {
			limit = {
				has_ep1_dlc_trigger = yes
			}

			hidden_effect = {
				set_amenity_level = {
					type = agot_dragonpit_amenities
					value = 3
				}
			}
		}

		# One ruler only needs one Head Dragonkeeper court position,
		# even if that ruler owns several active dragonpits.
		if = {

			limit = {

				NOT = {
					employs_court_position = head_dragonkeeper_court_position
				}

				NOT = {
					has_character_flag = segrellion_beacon_head_dragonkeeper_event_pending
				}
			}

			add_character_flag = {
				flag = segrellion_beacon_head_dragonkeeper_event_pending
				days = 14
			}

			trigger_event = {
				id = agot_dragonkeepers.0001
				days = 1
			}
		}
	}
}


# =========================================================
# HISTORICAL INAUGURATION - 160 AC
#
# Later bookmarks receive the castle from province history.
# This effect supplies the Great Building itself, repairs old
# saves, and also makes campaigns played forward reach the
# historical inauguration in 160 AC.
# =========================================================

segrellion_beacon_dragonpit_historical_progression_effect = {

	# In a campaign played forward from before 160 AC, construct the
	# historical Dragonpit if the site has not diverged and no player/
	# AI Great Project is already actively building it.
	if = {
		limit = {
			current_date >= 8160.1.1

			NOT = {
				has_global_variable = segrellion_beacon_dragonpit_project_active
			}

			province:7263 = {
				NOT = {
					has_building_or_higher = segrellion_beacon_dragonpit_01
				}

				OR = {
					has_holding = no
					has_holding_type = castle_holding
				}
			}
		}

		province:7263 = {

			if = {
				limit = {
					has_holding = no
				}

				set_holding_type = castle_holding
				add_building = castle_01
			}

			add_great_building = segrellion_beacon_dragonpit_01
		}
	}

	# Once the real building exists, it is the source of truth.
	# Repair flags and initialize AGOT's Dragonkeeper integration.
	if = {
		limit = {
			current_date >= 8160.1.1

			province:7263 = {
				has_building_or_higher = segrellion_beacon_dragonpit_01
			}
		}

		if = {
			limit = {
				has_global_variable = segrellion_beacon_dragonpit_project_active
			}

			remove_global_variable = segrellion_beacon_dragonpit_project_active
		}

		set_global_variable = {
			name = segrellion_beacon_dragonpit_completed
			value = yes
		}

		title:c_farring_cross.holder ?= {
			segrellion_activate_beacon_dragonpit_effect = yes
		}
	}
}
