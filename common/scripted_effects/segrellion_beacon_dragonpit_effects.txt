# =========================================================
# THE GREAT DRAGONPIT
# AGOT DRAGONKEEPER + TITLE INTEGRATION
# =========================================================


# ---------------------------------------------------------
# Keep the Dragonpit barony personal to the Lord of Beacon
# and permanently use its post-construction name.
# ---------------------------------------------------------

segrellion_repair_great_dragonpit_title_effect = {

	if = {
		limit = {
			province:7263 = {
				OR = {
					has_building_or_higher = segrellion_beacon_dragonpit_01
					has_building_or_higher = generic_dragon_pit_01
				}
			}

			exists = title:c_farring_cross.holder
		}

		# The internal barony key remains b_knights_bridge for map
		# compatibility; its displayed title becomes The Great Dragonpit.
		title:b_knights_bridge = {
			set_title_name = b_segrellion_great_dragonpit
		}

		# This fortress is an extension of the Beacon itself, never an
		# independently held barony.
		if = {
			limit = {
				title:b_knights_bridge = {
					NOT = {
						holder = title:c_farring_cross.holder
					}
				}
			}

			title:c_farring_cross.holder = {
				save_scope_as = segrellion_beacon_holder

				create_title_and_vassal_change = {
					type = revoked
					save_scope_as = segrellion_great_dragonpit_title_change
					add_claim_on_loss = no
				}

				title:b_knights_bridge = {
					change_title_holder_include_vassals = {
						holder = scope:segrellion_beacon_holder
						change = scope:segrellion_great_dragonpit_title_change
					}
				}

				resolve_title_and_vassal_change = scope:segrellion_great_dragonpit_title_change
			}
		}
	}
}


# ---------------------------------------------------------
# AGOT Dragonkeeper state
#
# The actual native generic_dragon_pit_01 building is installed
# at province 7263. This effect initializes the county/order/law
# pieces which AGOT normally sets when a generic pit completes.
# ---------------------------------------------------------

segrellion_activate_beacon_dragonpit_effect = {

	if = {

		limit = {

			title:c_farring_cross = {

				any_county_province = {
					segrellion_beacon_dragonpit_building_active_trigger = yes
				}
			}
		}

		# AGOT keys most Dragonpit systems from this county variable.
		title:c_farring_cross = {

			if = {
				limit = {
					NOT = {
						has_variable = has_dragonkeeper_order
					}
				}

				set_variable = {
					name = has_dragonkeeper_order
				}
			}
		}

		if = {
			limit = {
				agot_has_any_dragonpit_access_law = no
			}

			agot_dragonpit_access_law_close_family_effect = yes
		}

		if = {
			limit = {
				has_ep1_dlc_trigger = yes
			}

			hidden_effect = {
				set_amenity_level = {
					type = agot_dragonpit_amenities
					value = 3
				}
			}
		}

		# Historical/history-script construction does not fire the normal
		# generic building completion event, so ensure a Head Dragonkeeper
		# exists. The flag prevents repeated monthly event spam.
		if = {

			limit = {

				NOT = {
					employs_court_position = head_dragonkeeper_court_position
				}

				NOT = {
					has_character_flag = segrellion_beacon_head_dragonkeeper_event_pending
				}
			}

			add_character_flag = {
				flag = segrellion_beacon_head_dragonkeeper_event_pending
				days = 30
			}

			trigger_event = {
				id = agot_dragonkeepers.0001
				days = 1
			}
		}
	}
}


# ---------------------------------------------------------
# Historical / save-compatible state repair
# ---------------------------------------------------------

segrellion_beacon_dragonpit_historical_progression_effect = {

	# Canon inauguration: 160 AC.
	# If no divergent Great Project is currently underway, ensure the
	# historical site exists in all bookmarks and played-forward games.
	if = {
		limit = {
			current_date >= 8160.1.1

			NOT = {
				has_global_variable = segrellion_beacon_dragonpit_project_active
			}

			province:7263 = {
				OR = {
					has_holding = no
					has_holding_type = castle_holding
				}
			}
		}

		province:7263 = {

			if = {
				limit = {
					has_holding = no
				}

				set_holding_type = castle_holding
				add_building = castle_01
			}

			if = {
				limit = {
					NOT = {
						has_building_or_higher = generic_dragon_pit_01
					}
				}

				add_building = generic_dragon_pit_01
			}

			if = {
				limit = {
					NOT = {
						has_building_or_higher = segrellion_beacon_dragonpit_01
					}
				}

				add_great_building = segrellion_beacon_dragonpit_01
			}
		}
	}


	# Save repair for Dragonpits constructed before this update.
	# A custom Great Dragonpit always receives AGOT's native backend,
	# even if it was completed before 160 AC in a live campaign.
	if = {
		limit = {
			province:7263 = {
				has_building_or_higher = segrellion_beacon_dragonpit_01

				NOT = {
					has_building_or_higher = generic_dragon_pit_01
				}
			}
		}

		province:7263 = {
			add_building = generic_dragon_pit_01
		}
	}


	# Conversely, historical 160+ bookmarks with the native backend
	# receive our unique Great Building if an older save lacks it.
	if = {
		limit = {
			current_date >= 8160.1.1

			province:7263 = {
				has_building_or_higher = generic_dragon_pit_01

				NOT = {
					has_building_or_higher = segrellion_beacon_dragonpit_01
				}
			}
		}

		province:7263 = {
			add_great_building = segrellion_beacon_dragonpit_01
		}
	}


	# Once either canonical layer is present, repair ownership, name and
	# every AGOT Dragonkeeper state dependency.
	if = {
		limit = {
			province:7263 = {
				OR = {
					has_building_or_higher = segrellion_beacon_dragonpit_01

					AND = {
						this = province:7263
						has_building_or_higher = generic_dragon_pit_01
						current_date >= 8160.1.1
					}
				}
			}
		}

		if = {
			limit = {
				has_global_variable = segrellion_beacon_dragonpit_project_active
			}

			remove_global_variable = segrellion_beacon_dragonpit_project_active
		}

		set_global_variable = {
			name = segrellion_beacon_dragonpit_completed
			value = yes
		}

		segrellion_repair_great_dragonpit_title_effect = yes

		title:c_farring_cross.holder ?= {
			segrellion_activate_beacon_dragonpit_effect = yes
		}
	}
}
