# =========================================================
# SEGRELLION - SECOND WIFE RELIGIOUS CONSEQUENCES
# =========================================================

# =========================================================
# EXPAND EXCEPTIONALISM - ENGINE SUPPORT
#
# IMPORTANT:
# CK3 re-validates marriages after the marry effect. If the
# culture immediately drops back to one spouse, the new wife
# is invalidated on the next tick.
#
# Therefore this hidden cultural parameter MUST PERSIST once
# the exceptional second-wife right has been opened.
# =========================================================

segrellion_enable_expanded_exceptionalism_polygamy_effect = {

	culture = {

		if = {

			limit = {
				NOT = {
					has_cultural_tradition = tradition_segrellion_temporary_two_spouses
				}
			}

			add_culture_tradition = tradition_segrellion_temporary_two_spouses

			# Scripted engine support must not place the culture on the
			# normal tradition-reform cooldown.
			remove_variable = tradition_cooldown
		}
	}
}


# =========================================================
# CREATE THE ENGINE-VALID SECOND MARRIAGE
#
# THIS = prospective husband
# SECOND_WIFE = prospective second wife
# =========================================================

segrellion_force_second_wife_engine_marriage_effect = {

	$SECOND_WIFE$ = {
		save_scope_as = segrellion_engine_second_wife
	}


	# Keep the two-spouse cultural parameter active permanently.
	# The personal permission flags still decide WHO may use it.
	segrellion_enable_expanded_exceptionalism_polygamy_effect = yes


	marry = scope:segrellion_engine_second_wife
}


# =========================================================
# CREATE A REAL SECOND MARRIAGE
#
# THIS = prospective husband
# SECOND_WIFE = prospective second wife
# =========================================================

segrellion_create_second_wife_marriage_effect = {

	# Stable scope used by the backlash effects below.
	save_scope_as = segrellion_second_wife_husband

	$SECOND_WIFE$ = {
		save_scope_as = segrellion_new_second_wife
	}


	primary_spouse ?= {
		save_scope_as = segrellion_first_wife
	}


	# Daemon and Young Griff have hard story marriage guards.
	# Open their scripted window for the exact tick in which
	# CK3 creates the real spouse relation.

	if = {

		limit = {
			this = character:Segrellion_60
		}

		segrellion_daemon_open_story_marriage_window_effect = yes
	}


	else_if = {

		limit = {
			this = character:IAF_Young_Griff_1
		}

		segrellion_young_griff_open_story_marriage_window_effect = yes
	}


	# Create the actual second-spouse relation with the engine cap opened.
	segrellion_force_second_wife_engine_marriage_effect = {
		SECOND_WIFE = scope:segrellion_new_second_wife
	}


	if = {

		limit = {
			this = character:Segrellion_60
		}

		segrellion_daemon_close_story_marriage_window_effect = yes
	}


	else_if = {

		limit = {
			this = character:IAF_Young_Griff_1
		}

		segrellion_young_griff_close_story_marriage_window_effect = yes
	}


	# Never set success flags or show a success toast unless
	# CK3 confirms that the second spouse relation really exists.

	if = {

		limit = {

			is_spouse_of = scope:segrellion_new_second_wife
			number_of_spouses >= 2
		}


		if = {

			limit = {
				exists = scope:segrellion_first_wife
			}


			hidden_effect = {

				set_primary_spouse = scope:segrellion_first_wife
			}
		}


		add_character_flag = segrellion_two_wives_active


		# -------------------------------------------------
		# SOCIAL / RELIGIOUS BACKLASH
		#
		# Permission makes the marriage legal for this
		# mechanic; it does not make the arrangement popular.
		# Both families resent it, the first wife is displeased,
		# and the High Septon still dislikes seeing the exception
		# actually used.
		# -------------------------------------------------

		scope:segrellion_new_second_wife = {

			every_close_family_member = {

				limit = {
					is_alive = yes
					this != scope:segrellion_second_wife_husband
				}

				add_opinion = {
					target = scope:segrellion_second_wife_husband
					modifier = segrellion_second_wife_family_offended_opinion
				}
			}
		}


		if = {

			limit = {
				exists = scope:segrellion_first_wife
			}

			scope:segrellion_first_wife = {

				add_opinion = {
					target = scope:segrellion_second_wife_husband
					modifier = segrellion_second_wife_first_wife_displeased_opinion
				}

				every_close_family_member = {

					limit = {
						is_alive = yes
						this != scope:segrellion_second_wife_husband
					}

					add_opinion = {
						target = scope:segrellion_second_wife_husband
						modifier = segrellion_second_wife_family_offended_opinion
					}
				}
			}
		}


		title:k_the_most_devout.holder ?= {

			if = {

				limit = {
					is_alive = yes
					this != scope:segrellion_second_wife_husband
				}

				add_opinion = {
					target = scope:segrellion_second_wife_husband
					modifier = segrellion_second_wife_high_septon_disapproval_opinion
				}
			}
		}


		send_interface_toast = {

			type = event_toast_effect_good

			title = segrellion_second_wife_marriage_complete_toast

			left_icon = scope:segrellion_new_second_wife
		}
	}


	else = {

		remove_character_flag = segrellion_two_wives_active


		send_interface_toast = {

			type = event_toast_effect_bad

			title = segrellion_second_wife_marriage_failed_toast

			left_icon = scope:segrellion_new_second_wife
		}
	}
}



# The High Septon rejected a formal imperial petition.
# No war. The petitioner is excommunicated and suffers a
# realm-wide reputational penalty.

segrellion_second_wife_rejected_petition_censure_effect = {

	if = {

		limit = {
			religion = religion:the_seven_religion
		}


		if = {

			limit = {
				NOT = {
					has_trait = excommunicated
				}
			}

			add_excommunication_actual_effect = yes
		}


		add_character_modifier = {

			modifier = segrellion_rejected_exceptionalism_petition_modifier

			years = 5
		}
	}
}


# A storyline marriage may deliberately ignore the Faith.
# Once two wives become public, the High Septon responds.

segrellion_second_wife_public_illegal_union_censure_effect = {

	if = {

		limit = {

			religion = religion:the_seven_religion

			NOT = {
				has_character_flag = segrellion_two_wives_authorized
			}
		}


		if = {

			limit = {
				NOT = {
					has_trait = excommunicated
				}
			}

			add_excommunication_actual_effect = yes
		}


		add_character_modifier = {

			modifier = segrellion_public_second_wife_scandal_modifier

			years = 15
		}
	}
}
