# =========================================================
# VLADWYN LEGACY V
# TRUE SENTINEL VETERANS
#
# Special event army raised through the
# "Raise the True Sentinels" decision.
#
# This does NOT unlock a new Men-at-Arms type.
# These are Special Soldiers assembled from existing troops.
# =========================================================

vladwyn_raise_true_sentinel_veterans_effect = {

	spawn_army = {

		name = vladwyn_true_sentinel_veterans

		# -------------------------------------------------
		# CORE OF THE HOST
		# Veteran Vladwyn heavy infantry.
		# -------------------------------------------------

		men_at_arms = {
			type = black_sentinels
			men = 500
		}

		# -------------------------------------------------
		# ARMORED HORSEMEN
		# Retainers, exiles and mounted veterans.
		# -------------------------------------------------

		men_at_arms = {
			type = armored_horsemen
			men = 150
		}

		# -------------------------------------------------
		# SPEARMEN
		# -------------------------------------------------

		men_at_arms = {
			type = pikemen_unit
			men = 150
		}

		# -------------------------------------------------
		# BOWMEN
		# -------------------------------------------------

		men_at_arms = {
			type = bowmen
			men = 150
		}

		# -------------------------------------------------
		# SPECIAL TROOPS
		# -------------------------------------------------

		inheritable = no
		uses_supply = yes

		location = root.location
		origin = root.location

		# The veterans answer for the current war.
		# When that war ends, the host disperses again.
		war = scope:vladwyn_true_sentinels_war
	}
}

# =========================================================
# CALL THE BANNERS
#
# ROOT = ruler calling the banners
# =========================================================

segrellion_call_the_banners_effect = {

	save_scope_as = segrellion_banner_liege

	# =====================================================
	# RESET RESPONSE STATE FOR THIS CALL
	# =====================================================

	every_vassal = {

		remove_character_flag = segrellion_banner_answered_current_call
		remove_character_flag = segrellion_banner_refused_current_call
	}

	# =====================================================
	# UNICORN RETURNS
	#
	# Gaunt and Pyle initially support Leo Stormflame,
	# but if Daemon formally calls his banners they answer.
	# =====================================================

	if = {

		limit = {

			this = character:Segrellion_60

			has_character_flag = segrellion_unicorn_returns_war_active

			has_variable = segrellion_unicorn_returns_war
		}

		segrellion_unicorn_returns_gaunt_pyle_answer_call_effect = yes
	}

	# =====================================================
	# DIRECT VASSALS ALREADY FIGHTING AGAINST THEIR LIEGE
	# =====================================================

	every_vassal = {

		limit = {

			is_alive = yes
			is_landed = yes

			is_imprisoned = no

			highest_held_title_tier > tier_barony

			NOT = {
				government_has_flag = government_is_nw
			}
		}

		save_temporary_scope_as = segrellion_banner_enemy_candidate


		scope:segrellion_banner_liege = {

			every_character_war = {

				limit = {

					OR = {

						# Liege is attacker, vassal joined defenders.
						AND = {

							is_attacker = scope:segrellion_banner_liege

							any_war_defender = {
								this = scope:segrellion_banner_enemy_candidate
							}
						}


						# Liege is defender, vassal joined attackers.
						AND = {

							NOT = {
								is_attacker = scope:segrellion_banner_liege
							}

							any_war_attacker = {
								this = scope:segrellion_banner_enemy_candidate
							}
						}
					}
				}


				scope:segrellion_banner_enemy_candidate = {
					segrellion_banner_joined_enemy_effect = yes
				}
			}
		}
	}


	every_vassal = {

		limit = {

			is_alive = yes
			is_landed = yes

			is_imprisoned = no
			is_incapable = no

			highest_held_title_tier > tier_barony

			NOT = {
				government_has_flag = government_is_nw
			}


			# The caller only needs to participate in the war.
			# They do not need to be the primary war leader.

			save_temporary_scope_as = segrellion_banner_candidate

			scope:segrellion_banner_liege = {

				any_character_war = {

					NOR = {

						any_war_attacker = {
							this = scope:segrellion_banner_candidate
						}

						any_war_defender = {
							this = scope:segrellion_banner_candidate
						}
					}

					save_temporary_scope_as = segrellion_banner_candidate_war

					scope:segrellion_banner_candidate = {

						NOT = {
							is_at_war_with = scope:segrellion_banner_candidate_war.primary_attacker
						}

						NOT = {
							is_at_war_with = scope:segrellion_banner_candidate_war.primary_defender
						}
					}
				}
			}
		}


		save_scope_as = segrellion_banner_vassal


		# Human-controlled vassals choose for themselves.
		if = {

			limit = {
				is_ai = no
			}

			trigger_event = {
				id = segrellion_banners.0010
			}
		}


		# AI vassals answer according to loyalty.
		else = {

			segrellion_resolve_banner_vassal_response_effect = yes
		}
	}
}


# =========================================================
# BANNER OATHBREAKER
#
# ROOT = vassal who refused the call
# scope:segrellion_banner_liege = liege
# =========================================================

segrellion_banner_refuse_liege_effect = {

	save_scope_as = segrellion_banner_refuser


	# =====================================================
	# RESPONSE STATE
	# =====================================================

	add_character_flag = {
		flag = segrellion_banner_refused_current_call
		years = 1
	}

	remove_character_flag = segrellion_banner_answered_current_call


	# =====================================================
	# OATHBREAKER STATUS
	#
	# A simple refusal is the lesser form of oathbreaking.
	# =====================================================

	if = {

		limit = {

			NOT = {
				has_character_modifier = segrellion_banner_oathbreaker_modifier
			}
		}

		remove_character_modifier = segrellion_banner_oathbreaker_minor_modifier

		add_character_modifier = {
			modifier = segrellion_banner_oathbreaker_minor_modifier
			years = 5
		}
	}


	add_character_flag = {
		flag = segrellion_banner_oathbreaker_unpunished
		years = 5
	}


	# =====================================================
	# STORE THE LORD WHOSE CALL WAS REFUSED
	# =====================================================

	set_variable = {
		name = segrellion_banner_oathbreaker_liege
		value = scope:segrellion_banner_liege
		years = 5
	}


	# =====================================================
	# STORE THE LIEGE'S PRIMARY TITLE
	#
	# Unlike the character, the title survives succession.
	# This lets the heir deal with the matter if the lord dies.
	# =====================================================

	set_variable = {
		name = segrellion_banner_oathbreaker_liege_title
		value = scope:segrellion_banner_liege.primary_title
		years = 5
	}


	# =====================================================
	# REGISTER OATHBREAKER ON THE LIEGE'S TITLE
	# =====================================================

	scope:segrellion_banner_liege.primary_title = {

		add_to_variable_list = {
			name = segrellion_banner_pending_oathbreakers
			target = scope:segrellion_banner_refuser
		}

		if = {

			limit = {
				NOT = {
					has_variable = segrellion_banner_original_liege
				}
			}

			set_variable = {
				name = segrellion_banner_original_liege
				value = scope:segrellion_banner_liege
			}
		}
	}


	# =====================================================
	# LIEGE OPINION
	# =====================================================

	scope:segrellion_banner_liege = {

		add_opinion = {
			target = scope:segrellion_banner_refuser
			modifier = segrellion_banner_refused_call_opinion
		}


		# =================================================
		# ONLY VASSALS WHO HAVE ALREADY ANSWERED THE CALL
		# DESPISE THIS OATHBREAKER
		# =================================================

		every_vassal = {

			limit = {

				this != scope:segrellion_banner_refuser

				is_alive = yes
				is_landed = yes

				highest_held_title_tier > tier_barony

				has_character_flag = segrellion_banner_answered_current_call
			}

			add_opinion = {
				target = scope:segrellion_banner_refuser
				modifier = segrellion_banner_oathbreaker_fellow_vassal_opinion
			}
		}


		# =================================================
		# RESPONSE LETTER
		# =================================================

		trigger_event = {
			id = segrellion_banners.0021
		}
	}
}

# =========================================================
# BANNER TURNS AGAINST THE LIEGE
#
# ROOT = traitorous vassal
# scope:segrellion_banner_liege = current liege
#
# The vassal joins the opposite side of every valid war
# in which the liege is participating.
# =========================================================

segrellion_banner_join_enemy_wars_effect = {

	save_scope_as = segrellion_banner_traitor


	# =====================================================
	# RESPONSE STATE
	# =====================================================

	remove_character_flag = segrellion_banner_answered_current_call
	remove_character_flag = segrellion_banner_refused_current_call

	add_character_flag = {
		flag = segrellion_banner_betrayed_current_call
		years = 1
	}


	# =====================================================
	# JOIN THE OPPOSITE SIDE
	# =====================================================

	scope:segrellion_banner_liege = {

		every_character_war = {

			save_scope_as = segrellion_banner_betrayal_war


			if = {

				limit = {

					NOR = {

						any_war_attacker = {
							this = scope:segrellion_banner_traitor
						}

						any_war_defender = {
							this = scope:segrellion_banner_traitor
						}
					}


					scope:segrellion_banner_traitor = {

						NOT = {
							is_at_war_with = scope:segrellion_banner_betrayal_war.primary_attacker
						}

						NOT = {
							is_at_war_with = scope:segrellion_banner_betrayal_war.primary_defender
						}
					}
				}


				hidden_effect = {

					if = {

						limit = {

							NOT = {
								was_called = scope:segrellion_banner_traitor
							}
						}

						set_called_to = scope:segrellion_banner_traitor
					}
				}


				if = {

					limit = {
						is_attacker = scope:segrellion_banner_liege
					}

					add_defender = scope:segrellion_banner_traitor
				}


				else = {

					add_attacker = scope:segrellion_banner_traitor
				}
			}
		}
	}


	# =====================================================
	# APPLY THE BETRAYAL CONSEQUENCES
	# =====================================================

	scope:segrellion_banner_traitor = {

		segrellion_banner_joined_enemy_effect = yes
	}
}


# =========================================================
# BANNER JOINS THE ENEMY
#
# ROOT = treacherous vassal
# scope:segrellion_banner_liege = betrayed liege
# =========================================================

segrellion_banner_joined_enemy_effect = {

	save_scope_as = segrellion_banner_traitor


	# =====================================================
	# TRAITOR REPUTATION
	# =====================================================

	if = {

		limit = {

			NOT = {
				has_trait = segrellion_traitor
			}
		}

		add_trait = segrellion_traitor

		add_prestige_level = -1
	}


	# =====================================================
	# FULL OATHBREAKER
	# =====================================================

	if = {

		limit = {
			has_character_modifier = segrellion_banner_oathbreaker_minor_modifier
		}

		remove_character_modifier = segrellion_banner_oathbreaker_minor_modifier
	}


	add_character_modifier = {
		modifier = segrellion_banner_oathbreaker_modifier
		years = 20
	}


	add_character_flag = segrellion_banner_oathbreaker_unpunished

	add_character_flag = segrellion_banner_fought_against_liege


	# =====================================================
	# STORE THE BETRAYED LIEGE
	# =====================================================

	set_variable = {
		name = segrellion_banner_betrayed_liege
		value = scope:segrellion_banner_liege
	}


	set_variable = {
		name = segrellion_banner_betrayed_liege_title
		value = scope:segrellion_banner_liege.primary_title
	}


	# =====================================================
	# REGISTER TRAITOR ON THE LIEGE TITLE
	#
	# Needed later for postwar judgment.
	# =====================================================

	scope:segrellion_banner_liege.primary_title = {

		add_to_variable_list = {
			name = segrellion_banner_pending_traitors
			target = scope:segrellion_banner_traitor
		}
	}


	# =====================================================
	# UNICORN RETURNS COMPATIBILITY
	# =====================================================

	if = {

		limit = {

			scope:segrellion_banner_liege = {

				this = character:Segrellion_60

				has_character_flag = segrellion_unicorn_returns_war_active
			}
		}

		add_character_flag = segrellion_unicorn_returns_traitor
	}


	# =====================================================
	# BETRAYED LIEGE OPINION
	# =====================================================

	scope:segrellion_banner_liege = {

		add_opinion = {
			target = scope:segrellion_banner_traitor
			modifier = segrellion_banner_joined_enemy_opinion
		}


		# =================================================
		# VASSALS WHO ANSWERED THE CALL DESPISE THE TRAITOR
		# =================================================

		every_vassal = {

			limit = {

				this != scope:segrellion_banner_traitor

				is_alive = yes
				is_landed = yes

				highest_held_title_tier > tier_barony

				has_character_flag = segrellion_banner_answered_current_call
			}


			add_opinion = {

				target = scope:segrellion_banner_traitor

				modifier = segrellion_banner_traitor_fellow_vassal_opinion
			}
		}
	}


	# =====================================================
	# THE TRAITOR'S OWN BANNERS MUST NOW CHOOSE
	# =====================================================

	segrellion_banner_resolve_traitor_vassals_effect = yes


	# =====================================================
	# REPUTATION EVENT
	# =====================================================

	trigger_event = {
		id = segrellion_banners.0023
	}


	# =====================================================
	# BETRAYAL LETTER TO THE ORIGINAL LIEGE
	# =====================================================

	if = {

		limit = {

			NOT = {
				has_character_flag = segrellion_banner_betrayal_letter_sent
			}
		}

		add_character_flag = {
			flag = segrellion_banner_betrayal_letter_sent
			days = 30
		}


		scope:segrellion_banner_liege = {

			trigger_event = {
				id = segrellion_banners.0022
			}
		}
	}
}

# =========================================================
# TRAITOR'S OWN BANNERS
#
# ROOT = traitor
# scope:segrellion_banner_liege = original upper liege
# =========================================================

segrellion_banner_resolve_traitor_vassals_effect = {

	save_scope_as = segrellion_banner_rebel_liege


	every_vassal = {

		limit = {

			is_alive = yes
			is_landed = yes

			is_imprisoned = no
			is_incapable = no

			highest_held_title_tier > tier_barony

			NOT = {
				government_has_flag = government_is_nw
			}
		}


		save_scope_as = segrellion_banner_traitor_vassal


		if = {

			limit = {
				is_ai = no
			}

			trigger_event = {
				id = segrellion_banners.0024
			}
		}


		else = {

			segrellion_resolve_traitor_vassal_ai_effect = yes
		}
	}
}

# =========================================================
# AI RESPONSE TO A TRAITOROUS DIRECT LIEGE
#
# ROOT = vassal of the traitor
#
# Following the traitor remains the most common result.
# Refusing is common.
# Actively turning against the traitor is possible but rare.
# =========================================================

segrellion_resolve_traitor_vassal_ai_effect = {


	# =====================================================
	# LOYAL
	# Strongly inclined to follow their direct liege.
	# =====================================================

	if = {

		limit = {
			has_trait = loyal
		}

		random_list = {

			75 = {
				segrellion_banner_follow_traitor_liege_effect = yes
			}

			23 = {
				segrellion_banner_refuse_traitor_liege_effect = yes
			}

			2 = {
				segrellion_banner_remain_loyal_to_upper_liege_effect = yes
			}
		}
	}


	# =====================================================
	# DISLOYAL
	# Much more willing to abandon their direct liege.
	# =====================================================

	else_if = {

		limit = {
			has_trait = disloyal
		}

		random_list = {

			25 = {
				segrellion_banner_follow_traitor_liege_effect = yes
			}

			50 = {
				segrellion_banner_refuse_traitor_liege_effect = yes
			}

			25 = {
				segrellion_banner_remain_loyal_to_upper_liege_effect = yes
			}
		}
	}


	# =====================================================
	# HONORABLE PERSONALITIES
	# =====================================================

	else_if = {

		limit = {

			OR = {
				has_trait = honest
				has_trait = just
				has_trait = trusting
			}
		}

		random_list = {

			45 = {
				segrellion_banner_follow_traitor_liege_effect = yes
			}

			40 = {
				segrellion_banner_refuse_traitor_liege_effect = yes
			}

			15 = {
				segrellion_banner_remain_loyal_to_upper_liege_effect = yes
			}
		}
	}


	# =====================================================
	# AMBITIOUS / TREACHEROUS PERSONALITIES
	# =====================================================

	else_if = {

		limit = {

			OR = {
				has_trait = ambitious
				has_trait = deceitful
				has_trait = arbitrary
			}
		}

		random_list = {

			35 = {
				segrellion_banner_follow_traitor_liege_effect = yes
			}

			45 = {
				segrellion_banner_refuse_traitor_liege_effect = yes
			}

			20 = {
				segrellion_banner_remain_loyal_to_upper_liege_effect = yes
			}
		}
	}


	# =====================================================
	# DEFAULT
	# =====================================================

	else = {

		random_list = {

			60 = {
				segrellion_banner_follow_traitor_liege_effect = yes
			}

			33 = {
				segrellion_banner_refuse_traitor_liege_effect = yes
			}

			7 = {
				segrellion_banner_remain_loyal_to_upper_liege_effect = yes
			}
		}
	}
}

# =========================================================
# FOLLOW THE TRAITOR
#
# ROOT = vassal
# scope:segrellion_banner_rebel_liege = traitorous lord
#
# They are obeying their direct liege.
# They DO NOT receive the Traitor trait.
# =========================================================

segrellion_banner_follow_traitor_liege_effect = {

	add_character_flag = {
		flag = segrellion_banner_followed_traitor_liege
		years = 1
	}


	scope:segrellion_banner_rebel_liege = {

		every_character_war = {

			if = {

				limit = {

					OR = {

						any_war_attacker = {
							this = scope:segrellion_banner_rebel_liege
						}

						any_war_defender = {
							this = scope:segrellion_banner_rebel_liege
						}
					}


					NOR = {

						any_war_attacker = {
							this = root
						}

						any_war_defender = {
							this = root
						}
					}
				}


				if = {

					limit = {

						any_war_attacker = {
							this = scope:segrellion_banner_rebel_liege
						}
					}

					add_attacker = root
				}


				else = {

					add_defender = root
				}
			}
		}
	}
}

					limit = {

# =========================================================
# RESOLVE AI VASSAL RESPONSE
#
# ROOT = vassal
# scope:segrellion_banner_liege = liege
# =========================================================

segrellion_resolve_banner_vassal_response_effect = {


	# =====================================================
	# LOYAL TRAIT
	# 95% answers the call
	# =====================================================

	if = {

		limit = {
			has_trait = loyal
		}

		random_list = {

			95 = {
				segrellion_banner_join_liege_wars_effect = yes
			}

			5 = {
				segrellion_banner_refuse_liege_effect = yes
			}
		}
	}


	# =====================================================
	# DISLOYAL TRAIT
	# 20% answers the call
	# =====================================================

	else_if = {

		limit = {
			has_trait = disloyal
		}

		random_list = {

			20 = {
				segrellion_banner_join_liege_wars_effect = yes
			}

			80 = {
				segrellion_banner_refuse_liege_effect = yes
			}
		}
	}


	# =====================================================
	# VERY LOYAL
	# Opinion 60+
	# 100% joins
	# =====================================================

	else_if = {

		limit = {

			opinion = {
				target = scope:segrellion_banner_liege
				value >= 60
			}
		}

		segrellion_banner_join_liege_wars_effect = yes
	}


	# =====================================================
	# LOYAL
	# Opinion 20 to 59
	# 85% joins
	# =====================================================

	else_if = {

		limit = {

			opinion = {
				target = scope:segrellion_banner_liege
				value >= 20
			}
		}

		random_list = {

			85 = {
				segrellion_banner_join_liege_wars_effect = yes
			}

			15 = {
				segrellion_banner_refuse_liege_effect = yes
			}
		}
	}


	# =====================================================
	# UNCERTAIN
	# Opinion -20 to 19
	# 60% joins
	# =====================================================

	else_if = {

		limit = {

			opinion = {
				target = scope:segrellion_banner_liege
				value >= -20
			}
		}

		random_list = {

			60 = {
				segrellion_banner_join_liege_wars_effect = yes
			}

			40 = {
				segrellion_banner_refuse_liege_effect = yes
			}
		}
	}


	# =====================================================
	# RELUCTANT
	# Opinion -50 to -21
	# 30% joins
	# =====================================================

	else_if = {

		limit = {

			opinion = {
				target = scope:segrellion_banner_liege
				value >= -50
			}
		}

		random_list = {

			30 = {
				segrellion_banner_join_liege_wars_effect = yes
			}

			70 = {
				segrellion_banner_refuse_liege_effect = yes
			}
		}
	}


	# =====================================================
	# HOSTILE
	# Below -50
	# 10% joins
	# =====================================================

	else = {

		random_list = {

			10 = {
				segrellion_banner_join_liege_wars_effect = yes
			}

			90 = {
				segrellion_banner_refuse_liege_effect = yes
			}
		}
	}
}


# =========================================================
# ADD A VASSAL TO THE LIEGE'S VALID WARS
#
# ROOT = joining vassal
# scope:segrellion_banner_liege = liege
# =========================================================

segrellion_banner_join_liege_wars_effect = {

	save_scope_as = segrellion_banner_joiner

	# =====================================================
	# RESPONSE STATE
	# =====================================================

	add_character_flag = {
		flag = segrellion_banner_answered_current_call
		years = 1
	}

	remove_character_flag = segrellion_banner_refused_current_call


	# =====================================================
	# THIS LOYAL VASSAL RESENTS OATHBREAKERS WHO ALREADY
	# REFUSED THE SAME CALL
	# =====================================================

	scope:segrellion_banner_liege = {

		every_vassal = {

			limit = {

				this != scope:segrellion_banner_joiner

				is_alive = yes
				is_landed = yes

				has_character_flag = segrellion_banner_refused_current_call
			}

			scope:segrellion_banner_joiner = {

				add_opinion = {
					target = prev
					modifier = segrellion_banner_oathbreaker_fellow_vassal_opinion
				}
			}
		}
	}


	scope:segrellion_banner_liege = {

		every_character_war = {

			save_scope_as = segrellion_banner_war


			if = {

				limit = {

					# Joiner is not already participating.
					NOR = {

						any_war_attacker = {
							this = scope:segrellion_banner_joiner
						}

						any_war_defender = {
							this = scope:segrellion_banner_joiner
						}
					}


					# Joiner is not currently fighting either
					# primary side of the war.
					scope:segrellion_banner_joiner = {

						NOT = {
							is_at_war_with = scope:segrellion_banner_war.primary_attacker
						}

						NOT = {
							is_at_war_with = scope:segrellion_banner_war.primary_defender
						}
					}
				}


				hidden_effect = {

					if = {

						limit = {
							NOT = {
								was_called = scope:segrellion_banner_joiner
							}
						}

						set_called_to = scope:segrellion_banner_joiner
					}
				}


				if = {

					limit = {
						is_attacker = scope:segrellion_banner_liege
					}

					add_attacker = scope:segrellion_banner_joiner
				}


				else = {

					add_defender = scope:segrellion_banner_joiner
				}
			}
		}
	}

	# =====================================================
	# RESPONSE LETTER TO THE LIEGE
	# =====================================================

	scope:segrellion_banner_liege = {

		trigger_event = {
			id = segrellion_banners.0020
		}
	}
}

# =========================================================
# POSTWAR OATHBREAKER CLEANUP
#
# ROOT = lord judging the oathbreakers
# scope:segrellion_banner_review_title = title that stored them
# =========================================================

segrellion_banner_pardon_refusers_effect = {

	save_scope_as = segrellion_banner_pardoning_liege


	scope:segrellion_banner_review_title = {

		every_in_list = {

			variable = segrellion_banner_pending_oathbreakers

			limit = {

				is_alive = yes

				has_character_flag = segrellion_banner_oathbreaker_unpunished

				NOT = {
					has_character_flag = segrellion_banner_fought_against_liege
				}
			}


			save_scope_as = segrellion_banner_pardoned_refuser


			# ---------------------------------------------
			# The oathbreaker appreciates being spared.
			# ---------------------------------------------

			add_opinion = {
				target = scope:segrellion_banner_pardoning_liege
				modifier = segrellion_banner_pardoned_oathbreaker_opinion
			}


			# ---------------------------------------------
			# The matter is considered legally settled.
			#
			# The temporary Oathbreaker modifier remains.
			# It disappears naturally after five years.
			# ---------------------------------------------

			remove_character_flag = segrellion_banner_oathbreaker_unpunished
			remove_variable = segrellion_banner_oathbreaker_liege
			remove_variable = segrellion_banner_oathbreaker_liege_title
		}


		# =================================================
		# LOYAL VASSALS REACT TO THE LORD'S MERCY
		# =================================================

		scope:segrellion_banner_pardoning_liege = {

			every_vassal = {

				limit = {

					is_alive = yes

					has_character_flag = segrellion_banner_answered_current_call
				}


				# -----------------------------------------
				# Forgiving / compassionate personalities
				# tend to respect mercy.
				# -----------------------------------------

				if = {

					limit = {

						OR = {
							has_trait = forgiving
							has_trait = compassionate
							has_trait = content
						}
					}

					add_opinion = {
						target = scope:segrellion_banner_pardoning_liege
						modifier = segrellion_banner_liege_pardoned_oathbreakers_opinion
					}
				}


				# -----------------------------------------
				# Harsh / honor-conscious personalities
				# see it as weakness.
				# -----------------------------------------

				else_if = {

					limit = {

						OR = {
							has_trait = wrathful
							has_trait = vengeful
							has_trait = sadistic
							has_trait = callous
							has_trait = just
						}
					}

					add_opinion = {
						target = scope:segrellion_banner_pardoning_liege
						modifier = segrellion_banner_liege_abandoned_loyal_vassals_opinion
					}
				}
			}
		}


		clear_variable_list = segrellion_banner_pending_oathbreakers
		remove_variable = segrellion_banner_original_liege
	}
}
