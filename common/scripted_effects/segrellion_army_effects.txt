# =========================================================
# VLADWYN LEGACY V
# TRUE SENTINEL VETERANS
#
# Special event army raised through the
# "Raise the True Sentinels" decision.
#
# This does NOT unlock a new Men-at-Arms type.
# These are Special Soldiers assembled from existing troops.
# =========================================================

vladwyn_raise_true_sentinel_veterans_effect = {

	spawn_army = {

		name = vladwyn_true_sentinel_veterans

		# -------------------------------------------------
		# CORE OF THE HOST
		# Veteran Vladwyn heavy infantry.
		# -------------------------------------------------

		men_at_arms = {
			type = black_sentinels
			men = 500
		}

		# -------------------------------------------------
		# ARMORED HORSEMEN
		# Retainers, exiles and mounted veterans.
		# -------------------------------------------------

		men_at_arms = {
			type = armored_horsemen
			men = 150
		}

		# -------------------------------------------------
		# SPEARMEN
		# -------------------------------------------------

		men_at_arms = {
			type = pikemen_unit
			men = 150
		}

		# -------------------------------------------------
		# BOWMEN
		# -------------------------------------------------

		men_at_arms = {
			type = bowmen
			men = 150
		}

		# -------------------------------------------------
		# SPECIAL TROOPS
		# -------------------------------------------------

		inheritable = no
		uses_supply = yes

		location = root.location
		origin = root.location

		# The veterans answer for the current war.
		# When that war ends, the host disperses again.
		war = scope:vladwyn_true_sentinels_war
	}
}

# =========================================================
# CALL THE BANNERS
#
# ROOT = ruler calling the banners
# =========================================================

segrellion_call_the_banners_effect = {

	save_scope_as = segrellion_banner_liege

	# =====================================================
	# DIRECT VASSALS ALREADY FIGHTING AGAINST THEIR LIEGE
	# =====================================================

	every_vassal = {

		limit = {

			is_alive = yes
			is_landed = yes

			is_imprisoned = no

			highest_held_title_tier > tier_barony

			NOT = {
				government_has_flag = government_is_nw
			}
		}

		save_temporary_scope_as = segrellion_banner_enemy_candidate


		scope:segrellion_banner_liege = {

			every_character_war = {

				limit = {

					is_civil_war = no

					is_war_leader = scope:segrellion_banner_liege


					OR = {

						# Liege is attacker, vassal joined defenders.
						AND = {

							is_attacker = scope:segrellion_banner_liege

							any_war_defender = {
								this = scope:segrellion_banner_enemy_candidate
							}
						}


						# Liege is defender, vassal joined attackers.
						AND = {

							NOT = {
								is_attacker = scope:segrellion_banner_liege
							}

							any_war_attacker = {
								this = scope:segrellion_banner_enemy_candidate
							}
						}
					}
				}


				scope:segrellion_banner_enemy_candidate = {
					segrellion_banner_joined_enemy_effect = yes
				}
			}
		}
	}


	every_vassal = {

		limit = {

			is_alive = yes
			is_landed = yes

			is_imprisoned = no
			is_incapable = no

			highest_held_title_tier > tier_barony

			NOT = {
				government_has_flag = government_is_nw
			}


			# Only summon vassals who can join at least one current war.
			save_temporary_scope_as = segrellion_banner_candidate
			scope:segrellion_banner_liege = {
				any_character_war = {
					is_civil_war = no
					is_war_leader = scope:segrellion_banner_liege
					NOR = {
						any_war_attacker = { this = scope:segrellion_banner_candidate }
						any_war_defender = { this = scope:segrellion_banner_candidate }
					}
					save_temporary_scope_as = segrellion_banner_candidate_war
					scope:segrellion_banner_candidate = {
						NOT = { is_at_war_with = scope:segrellion_banner_candidate_war.primary_attacker }
						NOT = { is_at_war_with = scope:segrellion_banner_candidate_war.primary_defender }
					}
				}
			}
		}


		save_scope_as = segrellion_banner_vassal


		# Human-controlled vassals choose for themselves.
		if = {

			limit = {
				is_ai = no
			}

			trigger_event = {
				id = segrellion_banners.0010
			}
		}


		# AI vassals answer according to loyalty.
		else = {

			segrellion_resolve_banner_vassal_response_effect = yes
		}
	}
}


# =========================================================
# BANNER OATHBREAKER
#
# ROOT = vassal who refused the call
# scope:segrellion_banner_liege = liege
# =========================================================

segrellion_banner_refuse_liege_effect = {

	save_scope_as = segrellion_banner_refuser


	add_character_modifier = {
		modifier = segrellion_banner_oathbreaker_modifier
		years = 10
	}

	add_character_flag = segrellion_banner_oathbreaker_unpunished


	# =====================================================
	# LIEGE OPINION
	# =====================================================

	scope:segrellion_banner_liege = {

		add_opinion = {
			target = scope:segrellion_banner_refuser
			modifier = segrellion_banner_refused_call_opinion
		}


		# =================================================
		# LOYAL CO-VASSALS DESPISE THE OATHBREAKER
		# =================================================

		every_vassal = {

			limit = {

				this != scope:segrellion_banner_refuser

				is_alive = yes

				OR = {

					has_trait = loyal

					opinion = {
						target = scope:segrellion_banner_liege
						value >= 50
					}
				}
			}

			add_opinion = {
				target = scope:segrellion_banner_refuser
				modifier = segrellion_banner_oathbreaker_fellow_vassal_opinion
			}
		}
	}
}


# =========================================================
# BANNER JOINS THE ENEMY
#
# ROOT = treacherous vassal
# scope:segrellion_banner_liege = liege
# =========================================================

segrellion_banner_joined_enemy_effect = {

	save_scope_as = segrellion_banner_traitor


	add_character_modifier = {
		modifier = segrellion_banner_oathbreaker_modifier
		years = 20
	}

	add_character_flag = segrellion_banner_oathbreaker_unpunished


	# =====================================================
	# LIEGE OPINION
	# =====================================================

	scope:segrellion_banner_liege = {

		add_opinion = {
			target = scope:segrellion_banner_traitor
			modifier = segrellion_banner_joined_enemy_opinion
		}


		# =================================================
		# LOYAL CO-VASSALS DESPISE THE TRAITOR
		# =================================================

		every_vassal = {

			limit = {

				this != scope:segrellion_banner_traitor

				is_alive = yes

				OR = {

					has_trait = loyal

					opinion = {
						target = scope:segrellion_banner_liege
						value >= 50
					}
				}
			}

			add_opinion = {
				target = scope:segrellion_banner_traitor
				modifier = segrellion_banner_traitor_fellow_vassal_opinion
			}
		}
	}
}

# =========================================================
# RESOLVE AI VASSAL RESPONSE
#
# ROOT = vassal
# scope:segrellion_banner_liege = liege
# =========================================================

segrellion_resolve_banner_vassal_response_effect = {


	# =====================================================
	# LOYAL TRAIT
	# 95% answers the call
	# =====================================================

	if = {

		limit = {
			has_trait = loyal
		}

		random_list = {

			95 = {
				segrellion_banner_join_liege_wars_effect = yes
			}

			5 = {
				segrellion_banner_refuse_liege_effect = yes
			}
		}
	}


	# =====================================================
	# DISLOYAL TRAIT
	# 20% answers the call
	# =====================================================

	else_if = {

		limit = {
			has_trait = disloyal
		}

		random_list = {

			20 = {
				segrellion_banner_join_liege_wars_effect = yes
			}

			80 = {
				segrellion_banner_refuse_liege_effect = yes
			}
		}
	}


	# =====================================================
	# VERY LOYAL
	# Opinion 60+
	# 100% joins
	# =====================================================

	else_if = {

		limit = {

			opinion = {
				target = scope:segrellion_banner_liege
				value >= 60
			}
		}

		segrellion_banner_join_liege_wars_effect = yes
	}


	# =====================================================
	# LOYAL
	# Opinion 20 to 59
	# 85% joins
	# =====================================================

	else_if = {

		limit = {

			opinion = {
				target = scope:segrellion_banner_liege
				value >= 20
			}
		}

		random_list = {

			85 = {
				segrellion_banner_join_liege_wars_effect = yes
			}

			15 = {
				segrellion_banner_refuse_liege_effect = yes
			}
		}
	}


	# =====================================================
	# UNCERTAIN
	# Opinion -20 to 19
	# 60% joins
	# =====================================================

	else_if = {

		limit = {

			opinion = {
				target = scope:segrellion_banner_liege
				value >= -20
			}
		}

		random_list = {

			60 = {
				segrellion_banner_join_liege_wars_effect = yes
			}

			40 = {
				segrellion_banner_refuse_liege_effect = yes
			}
		}
	}


	# =====================================================
	# RELUCTANT
	# Opinion -50 to -21
	# 30% joins
	# =====================================================

	else_if = {

		limit = {

			opinion = {
				target = scope:segrellion_banner_liege
				value >= -50
			}
		}

		random_list = {

			30 = {
				segrellion_banner_join_liege_wars_effect = yes
			}

			70 = {
				segrellion_banner_refuse_liege_effect = yes
			}
		}
	}


	# =====================================================
	# HOSTILE
	# Below -50
	# 10% joins
	# =====================================================

	else = {

		random_list = {

			10 = {
				segrellion_banner_join_liege_wars_effect = yes
			}

			90 = {
				segrellion_banner_refuse_liege_effect = yes
			}
		}
	}
}


# =========================================================
# ADD A VASSAL TO THE LIEGE'S VALID WARS
#
# ROOT = joining vassal
# scope:segrellion_banner_liege = liege
# =========================================================

segrellion_banner_join_liege_wars_effect = {

	save_scope_as = segrellion_banner_joiner


	scope:segrellion_banner_liege = {

		every_character_war = {

			save_scope_as = segrellion_banner_war


			if = {

				limit = {

					# Only external wars.
					is_civil_war = no


					# CRITICAL:
					# The liege must actually lead this war.
					# Secondary participants cannot propagate
					# Call the Banners further downward.
					is_war_leader = scope:segrellion_banner_liege


					# Joiner is not already participating.
					NOR = {

						any_war_attacker = {
							this = scope:segrellion_banner_joiner
						}

						any_war_defender = {
							this = scope:segrellion_banner_joiner
						}
					}


					# Joiner is not currently fighting either
					# primary side of the war.
					scope:segrellion_banner_joiner = {

						NOT = {
							is_at_war_with = scope:segrellion_banner_war.primary_attacker
						}

						NOT = {
							is_at_war_with = scope:segrellion_banner_war.primary_defender
						}
					}
				}


				hidden_effect = {

					if = {

						limit = {
							NOT = {
								was_called = scope:segrellion_banner_joiner
							}
						}

						set_called_to = scope:segrellion_banner_joiner
					}
				}


				if = {

					limit = {
						is_attacker = scope:segrellion_banner_liege
					}

					add_attacker = scope:segrellion_banner_joiner
				}


				else = {

					add_defender = scope:segrellion_banner_joiner
				}
			}
		}
	}
}
