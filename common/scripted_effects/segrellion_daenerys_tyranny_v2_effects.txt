# =========================================================
# ENSURE THE VISIBLE STORY EXISTS
# ROOT = DAENERYS
# =========================================================

segrellion_daenerys_ensure_pacification_story_effect = {

	if = {

		limit = {

			NOT = {
				has_variable = segrellion_daenerys_pacification_tyranny
			}
		}


		set_variable = {

			name = segrellion_daenerys_pacification_tyranny

			value = 0
		}
	}


	if = {

		limit = {

			NOT = {

				any_owned_story = {

					story_type = story_segrellion_daenerys_pacification
				}
			}
		}


		create_story = {

			type = story_segrellion_daenerys_pacification

			save_scope_as = segrellion_daenerys_pacification_story
		}
	}


	segrellion_daenerys_sync_pacification_story_effect = yes
}


# =========================================================
# SYNC CHARACTER VARIABLE -> STORY BAR
# =========================================================

segrellion_daenerys_sync_pacification_story_effect = {

	every_owned_story = {

		limit = {

			story_type = story_segrellion_daenerys_pacification
		}


		set_variable = {

			name = segrellion_daenerys_pacification_tyranny

			value = root.var:segrellion_daenerys_pacification_tyranny
		}
	}
}


# =========================================================
# KING'S LANDING WAS BURNED
#
# The meter can never start below 50 after this.
# =========================================================

segrellion_daenerys_set_pacification_tyranny_minimum_50_effect = {

	segrellion_daenerys_ensure_pacification_story_effect = yes


	if = {

		limit = {

			var:segrellion_daenerys_pacification_tyranny < 50
		}


		set_variable = {

			name = segrellion_daenerys_pacification_tyranny

			value = 50
		}
	}


	segrellion_refresh_daenerys_pacification_modifier_effect = yes

	segrellion_daenerys_sync_pacification_story_effect = yes

	segrellion_daenerys_check_tyranny_revolt_effect = yes
}


# =========================================================
# CHECK FOR THE GREAT TYRANNY REVOLT
# =========================================================

segrellion_daenerys_check_tyranny_revolt_effect = {

	if = {

		limit = {

			this = character:Targaryen_9

			is_alive = yes

			has_title = title:h_the_iron_throne

			has_character_flag = segrellion_daenerys_post_throne_started

			var:segrellion_daenerys_pacification_tyranny >= 100


			NOT = {
				has_character_flag = segrellion_daenerys_tyranny_revolt_pending
			}


			NOT = {
				has_character_flag = segrellion_daenerys_tyranny_revolt_active
			}


			NOT = {
				has_character_flag = segrellion_daenerys_tyranny_revolt_cooldown
			}


			NOT = {
				has_character_flag = segrellion_daenerys_lost_tyranny_revolt
			}
		}


		add_character_flag = {

			flag = segrellion_daenerys_tyranny_revolt_pending

			days = 30
		}


		trigger_event = {

			id = segrellion_daenerys_tyranny.0001

			days = 1
		}
	}
}


# =========================================================
# DAENERYS SURVIVES THE REVOLT
# =========================================================

segrellion_daenerys_survived_tyranny_revolt_effect = {

	remove_character_flag = segrellion_daenerys_tyranny_revolt_pending

	remove_character_flag = segrellion_daenerys_tyranny_revolt_active


	add_character_flag = {

		flag = segrellion_daenerys_tyranny_revolt_cooldown

		years = 3
	}


	set_variable = {

		name = segrellion_daenerys_pacification_tyranny

		value = 70
	}


	if = {

		limit = {

			has_character_modifier = segrellion_daenerys_abandoned_by_lords_modifier
		}


		remove_character_modifier = segrellion_daenerys_abandoned_by_lords_modifier
	}


	remove_variable = segrellion_daenerys_tyranny_revolt_war


	segrellion_refresh_daenerys_pacification_modifier_effect = yes

	segrellion_daenerys_sync_pacification_story_effect = yes
}
