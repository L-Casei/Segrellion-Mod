# =========================================================
# BEGIN HUGH'S JOURNEY
#
# ROOT = Hugh
# =========================================================

segrellion_hugh_black_unicorn_begin_journey_effect = {

	if = {

		limit = {

			this = character:Segrellion_43

			is_alive = yes

			NOT = {
				has_character_flag = hugh_black_unicorn_journey_queued
			}

			NOT = {
				has_character_flag = hugh_black_unicorn_journey_started
			}
		}


		add_character_flag = hugh_black_unicorn_journey_queued


		trigger_event = {

			id = segrellion_hugh_black_unicorn.1000

			days = 1
		}
	}
}


# =========================================================
# RETURN HUGH TO WHERE HIS JOURNEY BEGAN
#
# ROOT = Hugh
# =========================================================

segrellion_hugh_black_unicorn_return_home_effect = {

	current_travel_plan ?= {
		abort_travel_plan = yes
	}


	var:hugh_black_unicorn_origin ?= {

		save_scope_as = hugh_black_unicorn_return_destination
	}


	if = {

		limit = {
			exists = scope:hugh_black_unicorn_return_destination
		}


		start_travel_plan = {

			destination = scope:hugh_black_unicorn_return_destination

			return_trip = no

			on_arrival_event = segrellion_hugh_black_unicorn.1090

			on_arrival_destinations = first

			on_travel_planner_cancel_event = segrellion_hugh_black_unicorn.1098
		}
	}


	else = {

		trigger_event = {

			id = segrellion_hugh_black_unicorn.1090

			days = 1
		}
	}
}

# =========================================================
# MOVE VISENYA + HER PEOPLE TO VHAL
#
# ROOT = Hugh
# =========================================================

segrellion_hugh_black_unicorn_move_visenya_to_vhal_effect = {

	capital_province ?= {
		save_scope_as = hugh_vhal_destination
	}


	if = {

		limit = {
			NOT = {
				exists = scope:hugh_vhal_destination
			}
		}


		location ?= {
			save_scope_as = hugh_vhal_destination
		}
	}


	character:Black_Unicorn_1 ?= {

		every_courtier_or_guest = {

			limit = {
				is_alive = yes
			}


			add_to_temporary_list = segrellion_hugh_visenya_followers
		}


		current_travel_plan ?= {
			abort_travel_plan = yes
		}


		remove_character_flag = segrellion_black_unicorn_lock_camp_skagos
		remove_character_flag = segrellion_black_unicorn_camp_lock_started


		if = {

			limit = {
				is_landless_adventurer = yes
			}


			destroy_title = primary_title
		}


		else_if = {

			limit = {

				title:d_laamp_red_unicorn ?= {

					holder = {
						this = character:Black_Unicorn_1
					}
				}
			}


			destroy_title = title:d_laamp_red_unicorn
		}


		set_employer = root
		return_to_court = yes


		if = {

			limit = {
				exists = scope:hugh_vhal_destination
			}


			set_location = scope:hugh_vhal_destination
		}


		add_character_flag = segrellion_visenya_hidden_at_vhal_with_hugh
	}


	every_in_list = {

		list = segrellion_hugh_visenya_followers


		limit = {
			is_alive = yes
		}


		set_employer = root
		return_to_court = yes


		if = {

			limit = {
				exists = scope:hugh_vhal_destination
			}


			set_location = scope:hugh_vhal_destination
		}


		add_character_flag = segrellion_black_unicorn_followed_visenya_to_vhal
	}
}

		save_scope_as = hugh_vhal_dragon_destination
	}

		limit = {

			exists = global_var:segrellion_historical_rhaegon

		}


		global_var:segrellion_historical_rhaegon = {

			if = {

				limit = {
					is_alive = yes
				}


				set_location = scope:hugh_vhal_dragon_destination

				add_character_flag = segrellion_rhaegon_housed_at_vhal_dragonpit
				add_character_flag = segrellion_rhaegon_under_hugh_protection
			}
		}
	}
}

# =========================================================
# SUCCESSFUL HUGH + VISENYA COUP
#
# ROOT = Hugh
# =========================================================

segrellion_hugh_black_unicorn_execute_vhal_coup_effect = {

	character:Black_Unicorn_1 ?= {

		if = {

			limit = {
				is_alive = yes
			}


			segrellion_black_unicorn_visenya_take_flamelands_effect = yes

			add_character_flag = segrellion_visenya_flamelands_civil_war_pending
			add_character_flag = segrellion_visenya_took_flamelands_with_hugh
		}
	}


	character:Segrellion_60 ?= {

		spouse ?= {

			if = {

				limit = {
					is_alive = yes
				}


				death = {

					death_reason = segrellion_death_vhal_coup

					killer = character:Black_Unicorn_1
				}
			}
		}


		every_child = {

			limit = {
				is_alive = yes
			}


			death = {

				death_reason = segrellion_death_vhal_coup

				killer = character:Black_Unicorn_1
			}
		}


		if = {

			limit = {
				is_alive = yes
			}


			death = {

				death_reason = segrellion_death_vhal_coup

				killer = character:Black_Unicorn_1
			}
		}
	}


	character:Black_Unicorn_1 ?= {

		remove_character_flag = segrellion_hugh_route_coup_vhal

		add_character_flag = segrellion_hugh_route_coup_succeeded
	}


	add_character_flag = hugh_black_unicorn_coup_succeeded
}
