# =========================================================
# DAENERYS - 300 AC STARTUP
#
# 10,000 total:
# - 1,500 Dothraki Screamers
# - 8,500 levies
# =========================================================

segrellion_setup_daenerys_300_effect = {

	if = {
		limit = {
			current_date >= 8300.1.1
			current_date < 8301.1.1

			character:Targaryen_9 ?= {
				is_alive = yes
				
				NOT = {
					has_character_flag = segrellion_daenerys_300_host_created
				}
			}
		}

		character:Targaryen_9 = {

			# Place Daenerys at Ghoyan Drohe.

			set_location = title:c_ghoyan_drohe.title_province

			# =================================================
			# DAENERYS' GREAT CAMP
			#
			# She begins with a fully developed Great Pavilion
			# and a strong military/logistical camp.
			# =================================================

			domicile ?= {

				# ---------------------------------------------
				# MAIN PAVILION
				#
				# Level I already exists on normal camps.
				# Upgrade through II -> III.
				# ---------------------------------------------

				add_domicile_building = camp_main_02
				add_domicile_building = camp_main_03


				# ---------------------------------------------
				# SUPPLY TENT
				# Strong logistical infrastructure.
				# ---------------------------------------------

				add_domicile_building = supply_tent_01
				add_domicile_building = supply_tent_02


				# ---------------------------------------------
				# PROVING GROUNDS
				# Appropriate for an enormous professional host.
				# ---------------------------------------------

				add_domicile_building = proving_grounds_01
				add_domicile_building = proving_grounds_02
			}

			# =================================================
			# DAENERYS' MEN-AT-ARMS
			#
			# 2 Dothraki regiments
			# 3 Unsullied regiments
			#
			# These are REAL permanent MaA regiments.
			# =================================================


			# -------------------------------------------------
			# DOTHRAKI
			# -------------------------------------------------

			create_maa_regiment = {
				type = dothraki_screamers
				check_can_recruit = no
				size = 7
			}

			create_maa_regiment = {
				type = dothraki_screamers
				check_can_recruit = no
				size = 7
			}


			# -------------------------------------------------
			# UNSULLIED
			# -------------------------------------------------

			create_maa_regiment = {
				type = unsullied
				check_can_recruit = no
				size = 6
			}

			create_maa_regiment = {
				type = unsullied
				check_can_recruit = no
				size = 6
			}

			create_maa_regiment = {
				type = unsullied
				check_can_recruit = no
				size = 6
			}

			spawn_army = {

				name = segrellion_daenerys_300_army

				# 5 regiments
				men_at_arms = {
					type = dothraki_screamers
					men = 600
				}

				men_at_arms = {
					type = dothraki_screamers
					men = 600
				}

				men_at_arms = {
					type = dothraki_screamers
					men = 600
				}

				men_at_arms = {
					type = unsullied
					men = 1000
				}

				men_at_arms = {
					type = unsullied
					men = 1000
				}

				levies = 7500

				inheritable = yes

				location = title:c_ghoyan_drohe.title_province
				origin = title:c_ghoyan_drohe.title_province
			}

			# =================================================
			# HISTORICAL FOLLOWERS
			#
			# Same startup path that creates Daenerys' camp
			# and army. No delayed event required.
			# =================================================

			if = {
				limit = {
					game_start_date = 8300.8.20
				}

				segrellion_setup_daenerys_historical_followers_effect = yes
			}

			add_character_flag = segrellion_daenerys_300_host_created

			# =================================================
			# THE QUEEN ACROSS THE SEA
			#
			# AI timing is controlled by the Daenerys
			# Iron Throne Aggression Game Rule.
			# =================================================

			if = {

				limit = {
					game_start_date = 8300.8.20
				}

				segrellion_initialize_daenerys_aggression_effect = yes
			}
		}
	}
}

# =========================================================
# DAENERYS - HISTORICAL FOLLOWERS
#
# Runs directly from Daenerys' 300 AC startup setup.
# No event. No delay. No scheduler.
#
# ROOT / THIS = Daenerys
# =========================================================

segrellion_setup_daenerys_historical_followers_effect = {

	if = {
		limit = {

			this = character:Targaryen_9

			is_alive = yes

			game_start_date = 8300.8.20

			NOT = {
				has_character_flag = segrellion_daenerys_followers_delayed_completed
			}
		}


		# =================================================
		# REMEMBER THE SECOND SONS OWNER
		#
		# Capture Tyrion's employer BEFORE moving him.
		# This identifies which Jorah ID belongs to the
		# same adventurer camp.
		# =================================================

		character:Lannister_8 ?= {

			employer ?= {
				save_scope_as = segrellion_second_sons_owner
			}
		}


		# =================================================
		# JORAH MORMONT
		# =================================================

		if = {
			limit = {

				exists = scope:segrellion_second_sons_owner

				character:Mormont_3 ?= {
					is_alive = yes
					employer = scope:segrellion_second_sons_owner
				}
			}

			character:Mormont_3 = {
				move_to_pool = yes
			}

			add_courtier = character:Mormont_3

			character:Mormont_3 = {

				if = {
					limit = {
						is_courtier_of = character:Targaryen_9
					}

					return_to_court = yes
				}
			}
		}


		if = {
			limit = {

				exists = scope:segrellion_second_sons_owner

				character:Mormont_3 ?= {
					is_alive = yes
					employer = scope:segrellion_second_sons_owner
				}
			}

			character:Mormont_3 = {
				move_to_pool = yes
			}

			add_courtier = character:Mormont_3

			character:Mormont_3 = {

				if = {
					limit = {
						is_courtier_of = character:Targaryen_9
					}

					return_to_court = yes
				}
			}
		}


		# =================================================
		# TYRION LANNISTER
		# =================================================

		character:Lannister_8 ?= {

			if = {
				limit = {
					is_alive = yes
					exists = employer
				}

				move_to_pool = yes
			}
		}

		if = {
			limit = {
				character:Lannister_8 ?= {
					is_alive = yes
				}
			}

			add_courtier = character:Lannister_8
		}

		character:Lannister_8 ?= {

			if = {
				limit = {
					is_alive = yes
					is_courtier_of = character:Targaryen_9
				}

				return_to_court = yes
			}
		}


		# =================================================
		# BARRISTAN SELMY
		# =================================================

		character:Selmy_3 ?= {

			if = {
				limit = {
					is_alive = yes
					exists = employer
				}

				move_to_pool = yes
			}
		}

		if = {
			limit = {
				character:Selmy_3 ?= {
					is_alive = yes
				}
			}

			add_courtier = character:Selmy_3
		}

		character:Selmy_3 ?= {

			if = {
				limit = {
					is_alive = yes
					is_courtier_of = character:Targaryen_9
				}

				return_to_court = yes
			}
		}


		# =================================================
		# MISSANDEI
		# =================================================

		character:IAF_Missandei_1 ?= {

			if = {
				limit = {
					is_alive = yes
					exists = employer
				}

				move_to_pool = yes
			}
		}

		if = {
			limit = {
				character:IAF_Missandei_1 ?= {
					is_alive = yes
				}
			}

			add_courtier = character:IAF_Missandei_1
		}

		character:IAF_Missandei_1 ?= {

			if = {
				limit = {
					is_alive = yes
					is_courtier_of = character:Targaryen_9
				}

				return_to_court = yes
			}
		}


		# =================================================
		# TORGO NUDHO
		# =================================================

		character:IAF_Torgo_Nudho_1 ?= {

			if = {
				limit = {
					is_alive = yes
					exists = employer
				}

				move_to_pool = yes
			}
		}

		if = {
			limit = {
				character:IAF_Torgo_Nudho_1 ?= {
					is_alive = yes
				}
			}

			add_courtier = character:IAF_Torgo_Nudho_1
		}

		character:IAF_Torgo_Nudho_1 ?= {

			if = {
				limit = {
					is_alive = yes
					is_courtier_of = character:Targaryen_9
				}

				return_to_court = yes
			}
		}


		# =================================================
		# TYRION BECOMES "THE SECOND"
		# =================================================

		if = {
			limit = {

				character:Lannister_8 ?= {
					is_alive = yes
					is_courtier_of = character:Targaryen_9
				}

				NOT = {
					any_court_position_holder = {
						type = second_camp_officer
					}
				}

				is_valid_to_hire_court_position_type = second_camp_officer
			}

			appoint_court_position = {
				recipient = character:Lannister_8
				court_position = second_camp_officer
			}
		}


		add_character_flag = segrellion_daenerys_followers_delayed_completed
	}
}

# =========================================================
# DAENERYS - HISTORICAL DRAGONS AT THE 300 AC START
#
# AGOT's history leaves the location of Drogon, Rhaegal and
# Viserion unresolved. Put all three in Daenerys' camp at zero
# tick and restore Drogon's rider relation only if a submod has
# prevented the historical relation from being applied.
# =========================================================

segrellion_setup_daenerys_dragons_300_effect = {

	if = {
		limit = {
			game_start_date = 8300.8.20
			character:Targaryen_9 ?= { is_alive = yes }
		}

		character:Targaryen_9 = {

			if = {
				limit = {
					character:dragon_drogon ?= {
						is_alive = yes
						NOT = { is_courtier_of = character:Targaryen_9 }
					}
				}
				add_courtier = character:dragon_drogon
			}

			if = {
				limit = {
					character:dragon_rhaegal ?= {
						is_alive = yes
						NOT = { is_courtier_of = character:Targaryen_9 }
					}
				}
				add_courtier = character:dragon_rhaegal
			}

			if = {
				limit = {
					character:dragon_viserion ?= {
						is_alive = yes
						NOT = { is_courtier_of = character:Targaryen_9 }
					}
				}
				add_courtier = character:dragon_viserion
			}

			if = {
				limit = {
					character:dragon_drogon ?= { is_alive = yes }
					NOT = { has_relation_agot_dragon = character:dragon_drogon }
				}
				agot_tame_dragon = {
					TAMER = character:Targaryen_9
					DRAGON = character:dragon_drogon
				}
			}
		}

		character:dragon_drogon ?= {
			if = {
				limit = { is_alive = yes }
				set_location = title:c_ghoyan_drohe.title_province
			}
		}
		character:dragon_rhaegal ?= {
			if = {
				limit = { is_alive = yes }
				set_location = title:c_ghoyan_drohe.title_province
			}
		}
		character:dragon_viserion ?= {
			if = {
				limit = { is_alive = yes }
				set_location = title:c_ghoyan_drohe.title_province
			}
		}
	}
}

# DAENERYS TRANSFERS PART OF HER HOST TO DAEMON
#
# Daenerys' starting host:
#   8,500 levies
#   1,500 Dothraki (5 x 300)
#
# Transfer:
#   -3,000 levies
#   -600 Dothraki
#
# Remaining:
#   5,500 levies
#   900 Dothraki
# =========================================================

segrellion_daenerys_transfer_reinforcements_to_daemon_effect = {

	character:Targaryen_9 = {

		# -------------------------------------------------
		# 3,000 LEVIES
		#
		# The special levy regiment uses 100 men per size.
		# 30 sizes = 3,000 men.
		# -------------------------------------------------

		random_maa_regiment = {

			limit = {
				is_maa_type = levies
			}

			change_maa_regiment_size = {
				size = -30
				reinforce = no
			}
		}


		# -------------------------------------------------
		# 600 DOTHRAKI
		#
		# Daenerys starts with five separate 300-man
		# Dothraki regiments. Destroy two of them.
		# -------------------------------------------------

		random_maa_regiment = {

			limit = {
				is_maa_type = dothraki_screamers
			}

			destroy_maa_regiment = yes
		}

		random_maa_regiment = {

			limit = {
				is_maa_type = dothraki_screamers
			}

			destroy_maa_regiment = yes
		}


		add_character_flag = segrellion_daenerys_sent_daemon_reinforcements
	}
}

# =========================================================
# DAENERYS - RUSHED CROSSING
#
# No fleet means abandoning the Dothraki and most levies.
# Unsullied are preserved.
#
# Target remaining force:
# roughly 4,000 men.
# =========================================================

segrellion_daenerys_abandon_army_for_early_crossing_effect = {

	# =====================================================
	# ABANDON ALL DOTHRAKI
	# =====================================================

	every_maa_regiment = {

		limit = {
			is_maa_type = dothraki_screamers
		}

		destroy_maa_regiment = yes
	}


	# =====================================================
	# REDUCE LEVIES
	#
	# Normal:
	# 7,500 -> 2,000
	#
	# If she already gave Daemon troops:
	# 5,500 -> 2,000
	#
	# Unsullied remain untouched.
	# =====================================================

	if = {

		limit = {
			has_character_flag = segrellion_daenerys_sent_daemon_reinforcements
		}

		random_maa_regiment = {

			limit = {
				is_maa_type = levies
			}

			change_maa_regiment_size = {
				size = -35
				reinforce = no
			}
		}
	}

	else = {

		random_maa_regiment = {

			limit = {
				is_maa_type = levies
			}

			change_maa_regiment_size = {
				size = -55
				reinforce = no
			}
		}
	}

	add_character_flag = segrellion_daenerys_rushed_crossing_army_reduced
}
