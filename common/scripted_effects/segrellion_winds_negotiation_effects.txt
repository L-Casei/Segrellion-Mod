# =========================================================
# DAENERYS - 300 AC STARTUP
#
# 10,000 total:
# - 1,500 Dothraki Screamers
# - 8,500 levies
# =========================================================

segrellion_setup_daenerys_300_effect = {

	if = {
		limit = {
			current_date >= 8300.1.1
			current_date < 8301.1.1

			character:Targaryen_9 ?= {
				is_alive = yes
				
				NOT = {
					has_character_flag = segrellion_daenerys_300_host_created
				}
			}
		}

		character:Targaryen_9 = {

			# Place Daenerys at Ghoyan Drohe.

			set_location = title:c_ghoyan_drohe.title_province

			# =================================================
			# DAENERYS' GREAT CAMP
			#
			# She begins with a fully developed Great Pavilion
			# and a strong military/logistical camp.
			# =================================================

			domicile ?= {

				# ---------------------------------------------
				# MAIN PAVILION
				#
				# Level I already exists on normal camps.
				# Upgrade through II -> III.
				# ---------------------------------------------

				add_domicile_building = camp_main_02
				add_domicile_building = camp_main_03


				# ---------------------------------------------
				# SUPPLY TENT
				# Strong logistical infrastructure.
				# ---------------------------------------------

				add_domicile_building = supply_tent_01
				add_domicile_building = supply_tent_02


				# ---------------------------------------------
				# PROVING GROUNDS
				# Appropriate for an enormous professional host.
				# ---------------------------------------------

				add_domicile_building = proving_grounds_01
				add_domicile_building = proving_grounds_02
			}

			# =================================================
			# DAENERYS' MEN-AT-ARMS
			#
			# 2 Dothraki regiments
			# 3 Unsullied regiments
			#
			# These are REAL permanent MaA regiments.
			# =================================================


			# -------------------------------------------------
			# DOTHRAKI
			# -------------------------------------------------

			create_maa_regiment = {
				type = dothraki_screamers
				check_can_recruit = no
				size = 7
			}

			create_maa_regiment = {
				type = dothraki_screamers
				check_can_recruit = no
				size = 7
			}


			# -------------------------------------------------
			# UNSULLIED
			# -------------------------------------------------

			create_maa_regiment = {
				type = unsullied
				check_can_recruit = no
				size = 6
			}

			create_maa_regiment = {
				type = unsullied
				check_can_recruit = no
				size = 6
			}

			create_maa_regiment = {
				type = unsullied
				check_can_recruit = no
				size = 6
			}

			spawn_army = {

				name = segrellion_daenerys_300_army

				# 5 regiments
				men_at_arms = {
					type = dothraki_screamers
					men = 600
				}

				men_at_arms = {
					type = dothraki_screamers
					men = 600
				}

				men_at_arms = {
					type = dothraki_screamers
					men = 600
				}

				men_at_arms = {
					type = unsullied
					men = 1000
				}

				men_at_arms = {
					type = unsullied
					men = 1000
				}

				levies = 7500

				inheritable = yes

				location = title:c_ghoyan_drohe.title_province
				origin = title:c_ghoyan_drohe.title_province
			}

			add_character_flag = segrellion_daenerys_300_host_created


			# =================================================
			# INITIALIZE SCRIPTED INVASION ROUTE
			#
			# Do this during the normal 300 AC setup as well as
			# after the lobby.
			#
			# The initializer is protected by its own scheduled
			# flag, so the after-lobby call cannot duplicate it.
			#
			# Most importantly, AI Daenerys receives her war
			# lock immediately and cannot bypass the scripted
			# invasion by declaring an ordinary AGOT war.
			# =================================================

			if = {

				limit = {

					game_start_date = 8300.8.20
				}

				segrellion_initialize_daenerys_aggression_effect = yes
			}

		}
	}
}


# =========================================================
# DAENERYS - HISTORICAL FOLLOWERS
#
# Runs directly from Daenerys' 300 AC startup setup.
# No event. No delay. No scheduler.
#
# ROOT / THIS = Daenerys
# =========================================================

# =========================================================
# DAENERYS - MISSING HISTORICAL FOLLOWERS
#
# Missandei and Torgo Nudho already exist as historical
# characters in the 300 AC bookmark, but can spawn without
# a proper employer/camp.
#
# This effect repairs them and places both permanently in
# Daenerys' landless court and entourage.
#
# It is safe to run repeatedly from the monthly fallback.
# =========================================================

segrellion_setup_daenerys_followers_effect = {

	if = {

		limit = {

			game_start_date = 8300.8.20

			current_date < 8301.1.1


			character:Targaryen_9 ?= {

				is_alive = yes
			}
		}


		# =================================================
		# DAENERYS
		# =================================================

		character:Targaryen_9 = {

			save_scope_as = daenerys_camp_owner
		}


		# =================================================
		# MISSANDEI
		# =================================================

		character:IAF_Missandei_1 ?= {

			if = {

				limit = {

					is_alive = yes


					NOT = {

						is_courtier_of = character:Targaryen_9
					}
				}


				# Daenerys becomes Missandei's employer.

				set_employer = scope:daenerys_camp_owner


				# Roads to Power:
				# explicitly register her in Daenerys'
				# court / camp entourage.

				add_to_court_and_entourage_effect = yes


				# Physically move her back to Daenerys'
				# current camp rather than leaving her
				# stranded where AGOT created her.

				return_home = yes
			}
		}


		# =================================================
		# TORGO NUDHO - GREY WORM
		# =================================================

		character:IAF_Torgo_Nudho_1 ?= {

			if = {

				limit = {

					is_alive = yes


					NOT = {

						is_courtier_of = character:Targaryen_9
					}
				}


				# Daenerys becomes Torgo Nudho's employer.

				set_employer = scope:daenerys_camp_owner


				# Explicitly attach him to the camp
				# entourage.

				add_to_court_and_entourage_effect = yes


				# Bring him physically to Daenerys.

				return_home = yes
			}
		}
	}
}

# =========================================================
# DAENERYS - HISTORICAL DRAGONS AT THE 300 AC START
#
# AGOT's history leaves the location of Drogon, Rhaegal and
# Viserion unresolved. Put all three in Daenerys' camp at zero
# tick and restore Drogon's rider relation only if a submod has
# prevented the historical relation from being applied.
# =========================================================

segrellion_setup_daenerys_dragons_300_effect = {

	if = {
		limit = {

			game_start_date = 8300.8.20
			current_date < 8301.1.1

			character:Targaryen_9 ?= {
				is_alive = yes
			}
		}

		character:Targaryen_9 = {

			location ?= {

				save_scope_as = segrellion_daenerys_dragon_location
			}
		}


		# =====================================================
		# DROGON
		# =====================================================

		character:dragon_drogon ?= {

			if = {
				limit = {
					is_alive = yes
				}

				# ---------------------------------------------
				# Canon dragon traits
				# ---------------------------------------------

				if = {
					limit = {
						NOT = {
							has_trait = dragon_destined
						}
					}

					add_trait = dragon_destined
				}

				# Drogon is canonically the strongest of the
				# three: Tyrannic Leviathan.
				if = {
					limit = {
						NOT = {
							has_trait = dragon_physique_good_3
						}
					}

					remove_trait = dragon_physique_good_1
					remove_trait = dragon_physique_good_2
					add_trait = dragon_physique_good_3
				}


				# ---------------------------------------------
				# Daenerys' camp
				# ---------------------------------------------

				if = {
					limit = {
						NOT = {
							is_courtier_of = character:Targaryen_9
						}
					}

					set_variable = {
						name = target_employer
						value = character:Targaryen_9
					}

					set_employer = character:Targaryen_9
					return_to_court = yes
				}

				if = {

					limit = {
						exists = scope:segrellion_daenerys_dragon_location
					}

					set_location = scope:segrellion_daenerys_dragon_location
				}
			}
		}


		# =====================================================
		# RHAEGAL
		# =====================================================

		character:dragon_rhaegal ?= {

			if = {

				limit = {

					NOT = {
						has_character_flag = owned_dragon
					}
				}

				add_character_flag = owned_dragon
			}

			if = {
				limit = {
					is_alive = yes
				}

				# ---------------------------------------------
				# Canon dragon traits
				# ---------------------------------------------

				if = {
					limit = {
						NOT = {
							has_trait = dragon_destined
						}
					}

					add_trait = dragon_destined
				}

				# Canonical physique tier for Rhaegal.
				if = {
					limit = {
						NOR = {
							has_trait = dragon_physique_good_2
							has_trait = dragon_physique_good_3
						}
					}

					remove_trait = dragon_physique_good_1
					add_trait = dragon_physique_good_2
				}


				# ---------------------------------------------
				# Daenerys' camp
				# ---------------------------------------------

				if = {
					limit = {
						NOT = {
							is_courtier_of = character:Targaryen_9
						}
					}

					set_variable = {
						name = target_employer
						value = character:Targaryen_9
					}

					set_employer = character:Targaryen_9
					return_to_court = yes
				}

				set_location = title:c_ghoyan_drohe.title_province
			}
		}


		# =====================================================
		# VISERION
		# =====================================================

		character:dragon_viserion ?= {

			if = {

				limit = {

					NOT = {
						has_character_flag = owned_dragon
					}
				}

				add_character_flag = owned_dragon
			}

			if = {
				limit = {
					is_alive = yes
				}

				# ---------------------------------------------
				# Canon dragon traits
				# ---------------------------------------------

				if = {
					limit = {
						NOT = {
							has_trait = dragon_destined
						}
					}

					add_trait = dragon_destined
				}

				# Canonical physique tier for Viserion.
				if = {
					limit = {
						NOR = {
							has_trait = dragon_physique_good_2
							has_trait = dragon_physique_good_3
						}
					}

					remove_trait = dragon_physique_good_1
					add_trait = dragon_physique_good_2
				}


				# ---------------------------------------------
				# Daenerys' camp
				# ---------------------------------------------

				if = {
					limit = {
						NOT = {
							is_courtier_of = character:Targaryen_9
						}
					}

					set_variable = {
						name = target_employer
						value = character:Targaryen_9
					}

					set_employer = character:Targaryen_9
					return_to_court = yes
				}

				set_location = title:c_ghoyan_drohe.title_province
			}
		}


		# =====================================================
		# DROGON IS DAENERYS' RIDER DRAGON
		# =====================================================

		character:Targaryen_9 = {

			if = {
				limit = {

					character:dragon_drogon ?= {
						is_alive = yes
					}

					NOT = {
						has_relation_agot_dragon = character:dragon_drogon
					}
				}

				agot_tame_dragon = {
					TAMER = character:Targaryen_9
					DRAGON = character:dragon_drogon
				}
			}
		}
	}
}

# =========================================================
# MYA STONE - DEFERRED MARRIAGE
#
# Mya remains in the Vale while she serves as regent.
#
# The promise comes due if:
#
# - Robin reaches adulthood;
# - Robin dies;
# - Mya ceases to be Robin's regent.
#
# If Daemon has restored the Flamelands when that happens,
# Mya travels to him and the marriage is completed.
#
# Otherwise the marriage pact is broken.
#
# The Robin death condition is intentionally handled
# silently as a failsafe.
# =========================================================

segrellion_check_mya_deferred_marriage_effect = {

	character:Segrellion_60 ?= {

		if = {

			limit = {

				has_character_flag = segrellion_mya_marriage_deferred

				has_variable = segrellion_mya_promised_bride
			}


			# Preserve Mya as a proper scope for this check.

			var:segrellion_mya_promised_bride = {

				save_scope_as = segrellion_mya_deferred_bride
			}


			# =================================================
			# THE PROMISE HAS BECOME IMPOSSIBLE
			#
			# Daemon died, married somebody else,
			# or Mya died / married somebody else.
			# =================================================

			if = {

				limit = {

					OR = {

						is_alive = no

						is_married = yes


						scope:segrellion_mya_deferred_bride = {

							is_alive = no
						}


						scope:segrellion_mya_deferred_bride = {

							is_married = yes
						}
					}
				}


				scope:segrellion_mya_deferred_bride = {

					remove_character_flag = segrellion_mya_promised_to_daemon_after_regency


					if = {

						limit = {

							has_character_flag = segrellion_mya_pact_added_refusing_marriage
						}


						if = {

							limit = {

								has_trait = refusing_marriage
							}


							remove_trait = refusing_marriage
						}


						remove_character_flag = segrellion_mya_pact_added_refusing_marriage
					}
				}


				remove_character_flag = segrellion_mya_marriage_deferred

				remove_character_flag = segrellion_vale_marriage_pact

				add_character_flag = segrellion_mya_marriage_pact_broken


				remove_variable = segrellion_mya_promised_bride

				remove_variable = segrellion_mya_regency_ward
			}


			# =================================================
			# CHECK WHETHER MYA'S REGENCY IS OVER
			# =================================================

			else_if = {

				limit = {

					OR = {


						# Old-save safety:
						# if the ward reference vanished,
						# resolve the pact rather than
						# leaving it frozen forever.

						NOT = {

							has_variable = segrellion_mya_regency_ward
						}


						# Robin / the Vale ruler died.

						var:segrellion_mya_regency_ward ?= {

							is_alive = no
						}


						# Robin has reached adulthood.

						var:segrellion_mya_regency_ward ?= {

							is_adult = yes
						}


						# Mya is no longer the regent of
						# the ruler she promised to remain
						# in the Vale for.

						var:segrellion_mya_regency_ward ?= {

							NOT = {

								diarch = {

									this = scope:segrellion_mya_deferred_bride
								}
							}
						}
					}
				}


				# =================================================
				# DAEMON RESTORED THE FLAMELANDS
				#
				# The promise is fulfilled.
				# =================================================

				if = {

					limit = {

						is_alive = yes

						is_married = no

						has_title = title:e_flamelands


						scope:segrellion_mya_deferred_bride = {

							is_alive = yes

							is_married = no
						}
					}


					scope:segrellion_mya_deferred_bride = {

						remove_character_flag = segrellion_mya_promised_to_daemon_after_regency


						if = {

							limit = {

								has_character_flag = segrellion_mya_pact_added_refusing_marriage
							}


							if = {

								limit = {

									has_trait = refusing_marriage
								}


								remove_trait = refusing_marriage
							}


							remove_character_flag = segrellion_mya_pact_added_refusing_marriage
						}


						# Mya leaves the Vale and joins
						# Daemon's court / camp.

						set_employer = character:Segrellion_60

						return_to_court = yes

						set_location = character:Segrellion_60.location
					}


					# Complete the promised marriage.

					marry = scope:segrellion_mya_deferred_bride


					add_character_flag = segrellion_mya_marriage_completed
				}


				# =================================================
				# DAEMON FAILED TO RESTORE THE FLAMELANDS
				#
				# Mya owes a restored lord a marriage,
				# not a permanent exile.
				# =================================================

				else = {

					scope:segrellion_mya_deferred_bride = {

						remove_character_flag = segrellion_mya_promised_to_daemon_after_regency


						if = {

							limit = {

								has_character_flag = segrellion_mya_pact_added_refusing_marriage
							}


							if = {

								limit = {

									has_trait = refusing_marriage
								}


								remove_trait = refusing_marriage
							}


							remove_character_flag = segrellion_mya_pact_added_refusing_marriage
						}
					}


					remove_character_flag = segrellion_vale_marriage_pact

					add_character_flag = segrellion_mya_marriage_pact_broken
				}


				# =================================================
				# CLEANUP
				# =================================================

				remove_character_flag = segrellion_mya_marriage_deferred

				remove_variable = segrellion_mya_promised_bride

				remove_variable = segrellion_mya_regency_ward
			}
		}
	}
}
