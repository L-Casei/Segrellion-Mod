# =========================================================
# DAENERYS - 300 AC STARTUP
#
# 10,000 total:
# - 1,500 Dothraki Screamers
# - 8,500 levies
# =========================================================

segrellion_setup_daenerys_300_effect = {

	if = {
		limit = {
			current_date >= 8300.1.1
			current_date < 8301.1.1

			character:Targaryen_9 ?= {
				is_alive = yes
				
				NOT = {
					has_character_flag = segrellion_daenerys_300_host_created
				}
			}
		}

		character:Targaryen_9 = {

			# Place Daenerys at Ghoyan Drohe.

			set_location = title:c_ghoyan_drohe.title_province

			# =================================================
			# DAENERYS' GREAT CAMP
			#
			# She begins with a fully developed Great Pavilion
			# and a strong military/logistical camp.
			# =================================================

			domicile ?= {

				# ---------------------------------------------
				# MAIN PAVILION
				#
				# Level I already exists on normal camps.
				# Upgrade through II -> III.
				# ---------------------------------------------

				add_domicile_building = camp_main_02
				add_domicile_building = camp_main_03


				# ---------------------------------------------
				# SUPPLY TENT
				# Strong logistical infrastructure.
				# ---------------------------------------------

				add_domicile_building = supply_tent_01
				add_domicile_building = supply_tent_02


				# ---------------------------------------------
				# PROVING GROUNDS
				# Appropriate for an enormous professional host.
				# ---------------------------------------------

				add_domicile_building = proving_grounds_01
				add_domicile_building = proving_grounds_02
			}

			# =================================================
			# DAENERYS' MEN-AT-ARMS
			#
			# 2 Dothraki regiments
			# 3 Unsullied regiments
			#
			# These are REAL permanent MaA regiments.
			# =================================================


			# -------------------------------------------------
			# DOTHRAKI
			# -------------------------------------------------

			create_maa_regiment = {
				type = dothraki_screamers
				size = 7
			}

			create_maa_regiment = {
				type = dothraki_screamers
				size = 7
			}


			# -------------------------------------------------
			# UNSULLIED
			# -------------------------------------------------

			create_maa_regiment = {
				type = unsullied
				size = 6
			}

			create_maa_regiment = {
				type = unsullied
				size = 6
			}

			create_maa_regiment = {
				type = unsullied
				size = 6
			}

			spawn_army = {

				name = segrellion_daenerys_300_army

				# 5 regiments
				men_at_arms = {
					type = dothraki_screamers
					men = 600
				}

				men_at_arms = {
					type = dothraki_screamers
					men = 600
				}

				men_at_arms = {
					type = dothraki_screamers
					men = 600
				}

				men_at_arms = {
					type = unsullied
					men = 1000
				}

				men_at_arms = {
					type = unsullied
					men = 1000
				}

				levies = 7500

				inheritable = yes

				location = title:c_ghoyan_drohe.title_province
				origin = title:c_ghoyan_drohe.title_province
			}

			add_character_flag = segrellion_daenerys_300_host_created

			# =================================================
			# THE QUEEN ACROSS THE SEA
			#
			# AI timing is controlled by the Daenerys
			# Iron Throne Aggression Game Rule.
			# =================================================

			if = {

				limit = {
					game_start_date = 8300.8.20
				}

				segrellion_initialize_daenerys_aggression_effect = yes
			}
		}
	}
}

# =========================================================
# DAENERYS TRANSFERS PART OF HER HOST TO DAEMON
#
# Daenerys' starting host:
#   8,500 levies
#   1,500 Dothraki (5 x 300)
#
# Transfer:
#   -3,000 levies
#   -600 Dothraki
#
# Remaining:
#   5,500 levies
#   900 Dothraki
# =========================================================

segrellion_daenerys_transfer_reinforcements_to_daemon_effect = {

	character:Targaryen_9 = {

		# -------------------------------------------------
		# 3,000 LEVIES
		#
		# The special levy regiment uses 100 men per size.
		# 30 sizes = 3,000 men.
		# -------------------------------------------------

		random_maa_regiment = {

			limit = {
				is_maa_type = levies
			}

			change_maa_regiment_size = {
				size = -30
				reinforce = no
			}
		}


		# -------------------------------------------------
		# 600 DOTHRAKI
		#
		# Daenerys starts with five separate 300-man
		# Dothraki regiments. Destroy two of them.
		# -------------------------------------------------

		random_maa_regiment = {

			limit = {
				is_maa_type = dothraki_screamers
			}

			destroy_maa_regiment = yes
		}

		random_maa_regiment = {

			limit = {
				is_maa_type = dothraki_screamers
			}

			destroy_maa_regiment = yes
		}


		add_character_flag = segrellion_daenerys_sent_daemon_reinforcements
	}
}

# =========================================================
# VHALERION REUNIFICATION CHECK
# =========================================================

segrellion_check_vhal_reunification_effect = {

	if = {
		limit = {

			character:Segrellion_60 ?= {
				is_alive = yes
				has_title = title:e_flamelands
			}

			title:d_vhalerion = {
				exists = holder
			}

			title:d_vhalerion.holder = {

				NOT = {
					has_character_flag = segrellion_vhal_reunification_resolved
				}

				NOT = {
					liege = character:Segrellion_60
				}
			}
		}

		title:d_vhalerion.holder = {

			add_character_flag = segrellion_vhal_reunification_event_pending

			trigger_event = {
				id = segrellion_support.5000
				days = 2
			}
		}
	}
}


# =========================================================
# DAENERYS MARRIAGE AFTER DAEMON TAKES THE FLAMELANDS
# =========================================================

segrellion_check_daenerys_marriage_effect = {

	if = {
		limit = {

			character:Segrellion_60 ?= {
				is_alive = yes

				has_title = title:e_flamelands

				has_character_flag = segrellion_daenerys_marriage_promised

				NOT = {
					has_character_flag = segrellion_daenerys_marriage_event_pending
				}

				is_married = no
			}

			character:Targaryen_9 ?= {
				is_alive = yes
				is_married = no
			}
		}

		character:Segrellion_60 = {

			add_character_flag = segrellion_daenerys_marriage_event_pending

			trigger_event = {
				id = segrellion_support.4000
				days = 3
			}
		}
	}
}


# =========================================================
# FUTURE HOOK
#
# Call this whenever Daemon explicitly chooses to support
# Young Griff in a future event.
# =========================================================

segrellion_break_stannis_oath_for_young_griff_effect = {

	if = {
		limit = {

			character:Segrellion_60 ?= {
				has_character_flag = segrellion_pledged_to_stannis
			}

			character:Baratheon_3 ?= {
				is_alive = yes
			}
		}

		character:Baratheon_3 = {

			add_opinion = {
				target = character:Segrellion_60
				modifier = segrellion_broke_stannis_oath_opinion
			}
		}

		character:Segrellion_60 = {
			remove_character_flag = segrellion_pledged_to_stannis
			add_character_flag = segrellion_broke_stannis_oath
		}
	}
}
