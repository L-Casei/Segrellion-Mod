# ============================================================
# INITIALIZE CUSTOM HISTORICAL DRAGON
#
# Static history dragons do not pass through AGOT's normal
# create_character hatchling initialization.
#
# This creates the dragon story storage and reconstructs
# dragon size according to the dragon's current age.
#
# The check for dragon_size means dynamically-hatched dragons
# are left completely untouched.
# ============================================================

segrellion_initialize_historical_dragon_effect = {

	$DRAGON$ = {
		save_scope_as = segrellion_dragon_to_initialize
	}

	scope:segrellion_dragon_to_initialize = {

		if = {
			limit = {
				is_alive = yes
				has_trait = dragon

				NOT = {
					has_variable = dragon_size
				}
			}


			# AGOT dragon runtime storage.
			create_story = story_dragon_alive


			# Required internal dragon variables.
			set_variable = {
				name = dragon_age
				value = age
			}

			set_variable = {
				name = dragon_size
				value = 0
			}

			set_variable = {
				name = dragon_size_base
				value = 0
			}

			# Reconstruct growth for every year already lived.
			agot_back_apply_dragon_aging_effect = yes


			# Reapply the exact canonical colors from the supplied DNA
			# before AGOT stores the dragon's visual variables.
			segrellion_apply_canon_dragon_colors_effect = yes


			# Push size + appearance into AGOT's dragon storage.
			agot_dragon_transfer_vars_to_story_cycle_effect = {
				DRAGON = this
			}	
		}
	}
}

				has_variable = current_dragon
				var:current_dragon ?= character:dragon_silverwing
			}

			agot_untame_dragon = {
				OWNER = this
				DRAGON = character:dragon_silverwing
			}
		}
	}


	# 146 A.C. - Rhaenys claims Silverwing.

	if = {
		limit = {
			current_date >= 8146.1.1
			current_date < 8173.1.1

			character:dragon_silverwing ?= {
				is_alive = yes
			}

			character:Segrellion_17 ?= {
				is_alive = yes
			}
		}

		character:Segrellion_17 = {

			if = {
				limit = {
					has_variable = current_dragon

					NOT = {
						var:current_dragon ?= character:dragon_silverwing
					}
				}

				var:current_dragon = {
					save_scope_as = rhaenys_previous_dragon
				}
			}
		}

		if = {
			limit = {
				exists = scope:rhaenys_previous_dragon
			}

			agot_untame_dragon = {
				OWNER = character:Segrellion_17
				DRAGON = scope:rhaenys_previous_dragon
			}
		}

		character:Segrellion_17 = {
			if = {
				limit = {
					OR = {
						NOT = {
							has_variable = current_dragon
						}

						NOT = {
							var:current_dragon ?= character:dragon_silverwing
						}
					}
				}

				agot_tame_dragon = {
					TAMER = this
					DRAGON = character:dragon_silverwing
				}
			}
		}
	}


	# 173 A.C. - Vhalyrax burns Silverwing to death.

	if = {
		limit = {
			current_date >= 8173.1.1

			character:dragon_silverwing ?= {
				is_alive = yes
			}

			exists = global_var:segrellion_historical_vhalyrax

			global_var:segrellion_historical_vhalyrax = {
				is_alive = yes
			}
		}

		global_var:segrellion_historical_vhalyrax = {
			save_scope_as = vhalyrax
		}

		character:dragon_silverwing = {
			death = {
				death_reason = death_segrellion_burned_by_vhalyrax
				killer = scope:vhalyrax
			}
		}
	}
}
