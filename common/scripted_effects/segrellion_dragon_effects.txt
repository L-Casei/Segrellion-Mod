	$OWNER$ = { save_scope_as = owner }
	$EGG$ = { save_scope_as = egg }

	create_character = {
		name = Vhalyrax
		location = $OWNER$.location
		template = agot_dragon_hatchling_template
		gender = female
		random_traits = no
		save_scope_as = dragon
	}

	hidden_effect = {
		agot_set_as_owned_dragon = {
			OWNER = scope:owner
			DRAGON = scope:dragon
		}
	}

	scope:dragon = {
		change_first_name = Vhalyrax

		remove_trait = education_dragon_3
		remove_trait = education_dragon_4
		add_trait = education_dragon_5
		add_prowess_skill = 10
		change_dragon_size = { VALUE = 8 }
	}

	agot_set_dragon_parents = { EGG = scope:egg }
	agot_dragon_tree_creation_effect = { DRAGON = scope:dragon }

	agot_historical_dragon_transfer_vars_to_story_cycle_effect = {

		gene_dragon_main_horn_shape_value = 0.55
		gene_dragon_body_shading_template = 15
		gene_dragon_body_shading_value = 0.72

segrellion_vhalyrax_growth_effect = {
	every_living_character = {
		limit = {
			has_character_flag = segrellion_vhalyrax_dragon
			is_alive = yes
			age < 8
		}
		change_dragon_size = { VALUE = 5 }
		add_prowess_skill = 2
	}
}
