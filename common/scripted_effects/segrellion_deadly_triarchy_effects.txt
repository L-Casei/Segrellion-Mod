# =========================================================
# THE DEADLY TRIARCHY
#
# Daemon Segrellion
# Young Griff
# Joffrey Vladwyn
#
# Called after lobby and from the monthly failsafe.
# Safe for both new games and existing saves.
# =========================================================

segrellion_form_deadly_triarchy_effect = {

    # =====================================================
    # ONLY THE 300 AC BOOKMARK
    # =====================================================

    if = {

        limit = {

            game_start_date >= 8300.8.20
            game_start_date < 8301.1.1


            character:Segrellion_60 ?= {
                is_alive = yes
            }


            character:IAF_Young_Griff_1 ?= {
                is_alive = yes
	            has_character_flag = segrellion_young_griff_supports_daemon
            }


            character:Vladwyn_5 ?= {
                is_alive = yes
            }

            title:e_the_stormlands.holder = character:IAF_Young_Griff_1
        }


        # =================================================
        # REPAIR THE PENTOS COMPACT
        #
        # Do not trust only the zero-tick setup.
        # This also repairs old saves.
        # =================================================

        character:IAF_Young_Griff_1 = {

            if = {

                limit = {

                    NOT = {
                        has_character_flag = segrellion_young_griff_supports_daemon
                    }
                }


                add_character_flag = segrellion_young_griff_supports_daemon
            }
        }


        character:Segrellion_60 = {

            if = {

                limit = {

                    NOT = {
                        has_character_flag = segrellion_daemon_supported_by_young_griff
                    }
                }


                add_character_flag = segrellion_daemon_supported_by_young_griff
            }
        }


        # =================================================
        # FORM THE TRIARCHY
        # =================================================

        if = {

            limit = {

               character:Segrellion_60 = {

                    has_character_flag = segrellion_daemon_supported_by_young_griff

                    NOT = {
                        has_character_flag = segrellion_deadly_triarchy_formed
                    }

                    NOT = {
                        has_character_flag = segrellion_deadly_triarchy_ever_formed
                    }
                }


                character:IAF_Young_Griff_1 = {
                    has_character_flag = segrellion_young_griff_supports_daemon
                }
            }


            # =============================================
            # DAEMON
            # =============================================

            character:Segrellion_60 = {

                add_character_flag = segrellion_deadly_triarchy_formed
                add_character_flag = segrellion_deadly_triarchy_ever_formed


                add_character_modifier = {
                    modifier = segrellion_deadly_triarchy_modifier
                }


                add_opinion = {
                    target = character:IAF_Young_Griff_1
                    modifier = segrellion_deadly_triarchy_opinion
                }


                add_opinion = {
                    target = character:Vladwyn_5
                    modifier = segrellion_deadly_triarchy_opinion
                }


                trigger_event = {
                    id = segrellion_deadly_triarchy.0001
                    days = 3
                }
            }


            # =============================================
            # JOFFREY
            # =============================================

            character:Vladwyn_5 = {

                add_character_flag = segrellion_deadly_triarchy_formed
                add_character_flag = segrellion_deadly_triarchy_ever_formed


                add_character_modifier = {
                    modifier = segrellion_deadly_triarchy_modifier
                }


                add_opinion = {
                    target = character:Segrellion_60
                    modifier = segrellion_deadly_triarchy_opinion
                }


                add_opinion = {
                    target = character:IAF_Young_Griff_1
                    modifier = segrellion_deadly_triarchy_opinion
                }


                trigger_event = {
                    id = segrellion_deadly_triarchy.0002
                    days = 3
                }
            }


            # =============================================
            # YOUNG GRIFF
            # =============================================

            character:IAF_Young_Griff_1 = {

                add_character_flag = segrellion_deadly_triarchy_formed
                add_character_flag = segrellion_deadly_triarchy_ever_formed


                add_character_modifier = {
                    modifier = segrellion_deadly_triarchy_modifier
                }


                add_opinion = {
                    target = character:Segrellion_60
                    modifier = segrellion_deadly_triarchy_opinion
                }


                add_opinion = {
                    target = character:Vladwyn_5
                    modifier = segrellion_deadly_triarchy_opinion
                }


                trigger_event = {
                    id = segrellion_deadly_triarchy.0003
                    days = 3
                }
            }
        }
    }
}

# =========================================================
# DEADLY TRIARCHY - FINAL TRIUMPH
#
# Aegon = Iron Throne
# Daemon = Flamelands
# Joffrey = Stormlands or Westerlands
# =========================================================

segrellion_deadly_triarchy_check_final_triumph_effect = {

	if = {

		limit = {

			character:IAF_Young_Griff_1 ?= {

				is_alive = yes

				has_character_flag = segrellion_deadly_triarchy_formed

				has_title = title:h_the_iron_throne


				NOT = {
					has_character_flag = segrellion_deadly_triarchy_final_triumph_seen
				}
			}


			character:Segrellion_60 ?= {

				is_alive = yes

				has_character_flag = segrellion_deadly_triarchy_formed


				OR = {

					has_title = title:e_flamelands

					has_title = title:k_flamelands
				}


				NOT = {
					has_character_flag = segrellion_deadly_triarchy_final_triumph_seen
				}
			}


			character:Vladwyn_5 ?= {

				is_alive = yes

				has_character_flag = segrellion_deadly_triarchy_formed


				OR = {

					has_title = title:e_the_stormlands

					has_title = title:e_the_westerlands
				}


				NOT = {
					has_character_flag = segrellion_deadly_triarchy_final_triumph_seen
				}
			}
		}


		character:Segrellion_60 = {

			add_character_flag = segrellion_deadly_triarchy_final_triumph_seen


			trigger_event = {

				id = segrellion_deadly_triarchy.0010

				days = 1
			}
		}


		character:Vladwyn_5 = {

			add_character_flag = segrellion_deadly_triarchy_final_triumph_seen


			trigger_event = {

				id = segrellion_deadly_triarchy.0011

				days = 2
			}
		}


		character:IAF_Young_Griff_1 = {

			add_character_flag = segrellion_deadly_triarchy_final_triumph_seen


			trigger_event = {

				id = segrellion_deadly_triarchy.0012

				days = 3
			}
		}
	}
}

# =========================================================
# DISSOLVE THE DEADLY TRIARCHY
#
# Called whenever the Pentos Compact between
# Young Griff and Daemon is broken.
# =========================================================

segrellion_dissolve_deadly_triarchy_effect = {

    # =====================================================
    # DAEMON
    # =====================================================

    character:Segrellion_60 ?= {

        remove_character_flag = segrellion_deadly_triarchy_formed

		add_character_flag = segrellion_deadly_triarchy_ever_formed

        if = {
            limit = {
                has_character_modifier = segrellion_deadly_triarchy_modifier
            }

            add_character_flag = segrellion_deadly_triarchy_ever_formed

            remove_character_modifier = segrellion_deadly_triarchy_modifier
        }


        remove_opinion = {
            target = character:IAF_Young_Griff_1
            modifier = segrellion_deadly_triarchy_opinion
        }


        remove_opinion = {
            target = character:Vladwyn_5
            modifier = segrellion_deadly_triarchy_opinion
        }
    }


    # =====================================================
    # YOUNG GRIFF
    # =====================================================

    character:IAF_Young_Griff_1 ?= {

        remove_character_flag = segrellion_deadly_triarchy_formed

		add_character_flag = segrellion_deadly_triarchy_ever_formed

        if = {
            limit = {
                has_character_modifier = segrellion_deadly_triarchy_modifier
            }

            add_character_flag = segrellion_deadly_triarchy_ever_formed

    # JOFFREY

            target = character:Segrellion_60

}


# =========================================================
# OATHBREAKER OF PENTOS
#
# ROOT = the character who broke the Pentos Compact
#
# Valid for:
# - Aegon
# - Daemon
# - Joffrey
#
# This is the canonical punishment used by BOTH:
# - political betrayal
# - military betrayal
# =========================================================

segrellion_apply_pentos_oathbreaker_effect = {

	if = {

		limit = {

			OR = {

				this = character:IAF_Young_Griff_1

				this = character:Segrellion_60

				this = character:Vladwyn_5
			}
		}


		# =================================================
		# REMOVE THE BENEFITS OF THE COMPACT
		# =================================================

		remove_character_flag = segrellion_deadly_triarchy_formed

		add_character_flag = segrellion_deadly_triarchy_oathbreaker


		if = {

			limit = {
				has_character_modifier = segrellion_deadly_triarchy_modifier
			}


			remove_character_modifier = segrellion_deadly_triarchy_modifier
		}


		# =================================================
		# REMOVE OLD / LEGACY OATHBREAKER MODIFIERS
		# =================================================

		if = {

			limit = {
				has_character_modifier = segrellion_young_griff_oathbreaker_modifier
			}


			remove_character_modifier = segrellion_young_griff_oathbreaker_modifier
		}


		if = {

			limit = {
				has_character_modifier = segrellion_oathbreaker_of_the_triarchy_modifier
			}


			remove_character_modifier = segrellion_oathbreaker_of_the_triarchy_modifier
		}


		# =================================================
		# FIRST-TIME PUNISHMENT
		# =================================================

		if = {

			limit = {

				NOT = {
					has_character_flag = segrellion_oathbreaker_of_pentos
				}
			}


			add_character_flag = segrellion_oathbreaker_of_pentos


			add_legitimacy = -400

			add_prestige = -750


			add_character_modifier = {

				modifier = segrellion_oathbreaker_of_pentos_modifier

				years = 20
			}
		}


		# =================================================
		# SAVE / MODIFIER RECOVERY
		# =================================================

		else_if = {

			limit = {

				NOT = {
					has_character_modifier = segrellion_oathbreaker_of_pentos_modifier
				}
			}


			add_character_modifier = {

				modifier = segrellion_oathbreaker_of_pentos_modifier

				years = 20
			}
		}
	}
}

# =========================================================
# OATHBREAKER OF THE TRIARCHY
#
# ROOT = aggressor
#
# scope:segrellion_triarchy_oathbreaker
# scope:segrellion_triarchy_defender
# scope:segrellion_triarchy_third_member
# scope:segrellion_triarchy_betrayal_war
# =========================================================

segrellion_deadly_triarchy_oathbreaker_effect = {

	# =====================================================
	# BRAND THE AGGRESSOR AS AN OATHBREAKER OF PENTOS
	# =====================================================

	segrellion_apply_pentos_oathbreaker_effect = yes

	# =====================================================
	# REMOVE OLD TRIARCHY FRIENDSHIP
	# =====================================================

	remove_opinion = {
		target = scope:segrellion_triarchy_defender
		modifier = segrellion_deadly_triarchy_opinion
	}


	scope:segrellion_triarchy_defender = {

		remove_opinion = {
			target = scope:segrellion_triarchy_oathbreaker
			modifier = segrellion_deadly_triarchy_opinion
		}
	}


	scope:segrellion_triarchy_third_member ?= {

		remove_opinion = {
			target = scope:segrellion_triarchy_oathbreaker
			modifier = segrellion_deadly_triarchy_opinion
		}

		scope:segrellion_triarchy_oathbreaker = {

			remove_opinion = {
				target = scope:segrellion_triarchy_third_member
				modifier = segrellion_deadly_triarchy_opinion
			}
		}
	}


	# =====================================================
	# -100 BETWEEN OATHBREAKER AND BOTH OTHER MEMBERS
	# =====================================================

	add_opinion = {
		target = scope:segrellion_triarchy_defender
		modifier = segrellion_deadly_triarchy_oathbreaker_opinion
	}


	scope:segrellion_triarchy_defender = {

		add_opinion = {
			target = scope:segrellion_triarchy_oathbreaker
			modifier = segrellion_deadly_triarchy_oathbreaker_opinion
		}
	}


	scope:segrellion_triarchy_third_member ?= {

		add_opinion = {
			target = scope:segrellion_triarchy_oathbreaker
			modifier = segrellion_deadly_triarchy_oathbreaker_opinion
		}


		scope:segrellion_triarchy_oathbreaker = {

			add_opinion = {
				target = scope:segrellion_triarchy_third_member
				modifier = segrellion_deadly_triarchy_oathbreaker_opinion
			}
		}
	}


	# =====================================================
	# THIRD MEMBER MUST DEFEND THE VICTIM
	# =====================================================

	scope:segrellion_triarchy_third_member ?= {

		if = {

			limit = {

				is_alive = yes
				is_landed = yes

				has_character_flag = segrellion_deadly_triarchy_formed

				NOT = {
					is_at_war_with = scope:segrellion_triarchy_defender
				}
			}


			scope:segrellion_triarchy_betrayal_war = {

				if = {

					limit = {

						NOT = {
							any_war_attacker = {
								this = scope:segrellion_triarchy_third_member
							}
						}

						NOT = {
							any_war_defender = {
								this = scope:segrellion_triarchy_third_member
							}
						}
					}


					hidden_effect = {

						if = {

							limit = {

								NOT = {
									was_called = scope:segrellion_triarchy_third_member
								}
							}

							set_called_to = scope:segrellion_triarchy_third_member
						}
					}


					add_defender = scope:segrellion_triarchy_third_member
				}
			}
		}
	}

    # =====================================================
	# THE PENTOS COMPACT IS DEAD
	#
	# The third member was already given the chance to
	# defend the betrayed member above.
	# Now remove the Triarchy benefits from everyone.
	# =====================================================

	segrellion_dissolve_deadly_triarchy_effect = yes
}
