# ============================================================
# PLAYER DRAGON WINDOW
#
# Root = Dragon
# scope:owner = Player
# ============================================================

agot_change_dragonpit_status = {

	save_scope_as = dragon

	scope:owner = {

		# --------------------------------------------------------
		# MULTIPLE DRAGONPITS
		# --------------------------------------------------------

		if = {
			limit = {
				agot_has_multiple_active_dragonpits = yes
			}

			trigger_event = agot_choose_dragonpit_event.0001
		}

		# --------------------------------------------------------
		# SINGLE DRAGONPIT
		# --------------------------------------------------------

		else = {

			random_held_county = {
				limit = {

					OR = {

						# Generic constructed Dragonpit
						any_county_province = {
							has_holding_type = castle_holding
							has_building_or_higher = generic_dragon_pit_01
						}

						# Historical / special Dragonpits
						any_county_province = {
							OR = {
								has_building_or_higher = dragonpit_01
								has_building = dragonpit_ruins_03
								has_building = agot_dragonmont_01

								# Caverns of Vhal
								has_building_or_higher = segrellion_vhal_dragonpit_02
							}
						}
					}

					has_variable = has_dragonkeeper_order
				}

				save_scope_as = dragon_pit_location
			}


			# ----------------------------------------------------
			# ACT ON THE DRAGON
			# ----------------------------------------------------

			scope:dragon = {

				# Dragon is already inside a pit
				if = {
					limit = {
						has_character_flag = in_dragonpit
					}

					agot_remove_from_dragonpit_effect = {
						DRAGON = root
						DRAGONPIT_HOLDER = scope:owner
						DRAGONPIT_COUNTY = scope:dragon_pit_location
					}
				}


				# Owned dragon -> send normally
				else_if = {
					limit = {
						has_character_flag = owned_dragon
					}

					agot_send_to_dragonpit_effect = {
						DRAGON = root
						DRAGONPIT_HOLDER = scope:owner
						DRAGONPIT_COUNTY = scope:dragon_pit_location
					}
				}


				# Wild / unowned dragon -> force into pit
				else_if = {
					limit = {
						NOT = {
							has_character_flag = owned_dragon
						}
					}

					agot_force_wild_dragon_into_dragonpit_effect = {
						DRAGON = root
						DRAGONPIT_HOLDER = scope:owner
						DRAGONPIT_COUNTY = scope:dragon_pit_location
					}
				}
			}
		}
	}
}


# ============================================================
# AI / CHARACTER INTERACTION VERSION
# ============================================================

agot_change_dragonpit_status_ai = {

	$OWNER$ = {

		random_held_county = {
			limit = {

				OR = {

					# Generic constructed Dragonpit
					any_county_province = {
						has_holding_type = castle_holding
						has_building_or_higher = generic_dragon_pit_01
					}

					# Historical / special Dragonpits
					any_county_province = {
						OR = {
							has_building_or_higher = dragonpit_01
							has_building = dragonpit_ruins_03
							has_building = agot_dragonmont_01

							# Caverns of Vhal
							has_building_or_higher = segrellion_vhal_dragonpit_02
						}
					}
				}

				has_variable = has_dragonkeeper_order
			}

			save_scope_as = dragon_pit_location
		}
	}


	$DRAGON_REC$ = {

		# Already in a Dragonpit
		if = {
			limit = {
				has_character_flag = in_dragonpit
			}

			agot_remove_from_dragonpit_effect = {
				DRAGON = $DRAGON_REC$
				DRAGONPIT_HOLDER = $OWNER$
				DRAGONPIT_COUNTY = scope:dragon_pit_location
			}
		}


		# Owned dragon
		else_if = {
			limit = {
				has_character_flag = owned_dragon
			}

			agot_send_to_dragonpit_effect = {
				DRAGON = $DRAGON_REC$
				DRAGONPIT_HOLDER = $OWNER$
				DRAGONPIT_COUNTY = scope:dragon_pit_location
			}
		}


		# Wild / unowned dragon
		else_if = {
			limit = {
				NOT = {
					has_character_flag = owned_dragon
				}
			}

			agot_force_wild_dragon_into_dragonpit_effect = {
				DRAGON = $DRAGON_REC$
				DRAGONPIT_HOLDER = $OWNER$
				DRAGONPIT_COUNTY = scope:dragon_pit_location
			}
		}
	}
}
