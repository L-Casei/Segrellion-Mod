# =========================================================
# 300 AC - WINDS OF WINTER
# HISTORICAL ADVENTURER HOSTS
# =========================================================

segrellion_setup_winds_of_winter_hosts_effect = {

	if = {
		limit = {
			current_date >= 8300.1.1
			current_date < 8301.1.1
		}


		# =====================================================
		# DAEMON SEGRELLION
		# THE BASTARD UNICORN HOST
		#
		# Roughly 2,000 professional soldiers.
		# Daemon must gather another thousand men before returning home.
		# =====================================================

		character:Segrellion_60 ?= {

			if = {
				limit = {
					is_alive = yes

					NOT = {
						has_character_flag = iaf_8300_daemon_host_created
					}
				}


				# Daemon must use The Unicorn Returns rather
				# than bypassing the storyline with claim wars.

				if = {
					limit = {
						has_claim_on = title:e_flamelands
					}

					remove_claim = title:e_flamelands
				}

				if = {
					limit = {
						has_claim_on = title:k_flamelands
					}

					remove_claim = title:k_flamelands
				}


				# -------------------------------------------------
				# PERMANENT CAMP MEN-AT-ARMS
				# -------------------------------------------------

				create_maa_regiment = {
					type = armored_horsemen
					check_can_recruit = no
					size = 4
				}

				create_maa_regiment = {
					type = armored_footmen
					check_can_recruit = no
					size = 6
				}

				create_maa_regiment = {
					type = pikemen_unit
					check_can_recruit = no
					size = 6
				}

				create_maa_regiment = {
					type = bowmen
					check_can_recruit = no
					size = 6
				}


				add_character_flag = iaf_8300_daemon_host_created
			}
		}


		# =====================================================
		# JOFFREY VLADWYN
		# THE TRUE SENTINELS
		#
		# Roughly 4,000 professional soldiers.
		# =====================================================

		character:Vladwyn_5 ?= {

			if = {
				limit = {
					is_alive = yes

					NOT = {
						has_character_flag = iaf_8300_true_sentinels_host_created
					}
				}


				create_maa_regiment = {
					type = black_sentinels
					check_can_recruit = no
					size = 12
				}

				create_maa_regiment = {
					type = armored_footmen
					check_can_recruit = no
					size = 11
				}

				create_maa_regiment = {
					type = pikemen_unit
					check_can_recruit = no
					size = 10
				}

				create_maa_regiment = {
					type = bowmen
					check_can_recruit = no
					size = 10
				}


				add_character_flag = iaf_8300_true_sentinels_host_created
			}
		}
	}
}

# =========================================================
# 300 AC - MASSEY'S HOOK
# DEFIANCE AGAINST THE STORMFLAMES
# =========================================================

segrellion_setup_massey_defiance_effect = {

	if = {
		limit = {
			current_date >= 8298.10.24

			title:d_masseys_hook.holder ?= {
				is_alive = yes

				NOT = {
					has_character_flag = segrellion_massey_defiance_initialized
				}
			}
		}

		title:d_masseys_hook.holder = {

			save_scope_as = massey_lord

			add_character_modifier = {
				modifier = segrellion_defiant_against_stormflames_modifier
			}

			# =================================================
			# HATRED OF HOUSE STORMFLAME
			# =================================================

			house:house_Stormflame = {

				every_house_member = {
					limit = {
						is_alive = yes
					}

					save_scope_as = stormflame_member

					scope:massey_lord = {
						add_opinion = {
							target = scope:stormflame_member
							modifier = segrellion_massey_despises_stormflames_opinion
						}
					}
				}
			}

			# Leo personally bears the greatest blame.
			add_opinion = {
				target = character:Segrellion_36
				modifier = segrellion_massey_despises_leo_opinion
			}

			# Daemon represents the restoration of the old order.
			add_opinion = {
				target = character:Segrellion_60
				modifier = segrellion_massey_supports_daemon_opinion
			}

			# The True Sentinels are respected as fellow enemies
			# of the Stormflame regime.
			add_opinion = {
				target = character:Vladwyn_5
				modifier = segrellion_massey_respects_true_sentinels_opinion
			}

			add_character_flag = segrellion_massey_defiance_initialized
		}
	}
}

# Stage II  - Its chain of keepers is reconstructed.

	}
}

# =========================================================
# 300 AC - HISTORICAL YOUNG GRIFF
# =========================================================

segrellion_setup_young_griff_300_effect = {

	if = {
		limit = {

			current_date >= 8300.8.20

			character:IAF_Young_Griff_1 ?= {
				is_alive = yes
			}
		}


		character:IAF_Young_Griff_1 = {

			save_scope_as = iaf_young_griff

			#remove_character_modifier = shade_of_the_evening_hair_dye TO REVEAL WHITE HAIR

			if = {
				limit = {
					NOT = {
						has_character_modifier = shade_of_the_evening_hair_dye
					}
				}

				add_character_modifier = {
					modifier = shade_of_the_evening_hair_dye
				}
			}

			if = {
				limit = {

					NOT = {
						has_character_flag = segrellion_young_griff_supports_daemon
					}

					character:Segrellion_60 = {

						NOT = {
							has_character_flag = segrellion_pledged_to_stannis
						}
					}
				}

				add_character_flag = segrellion_young_griff_supports_daemon
			}
		}

			# =================================================
			# ARIANNE MARTELL
			# One year after the 300 AC start.
			# =================================================

			if = {
				limit = {

					current_date < 8301.1.1

					NOT = {
						has_character_flag = segrellion_arianne_marriage_event_scheduled
					}

					is_married = no

					character:Martell_9 ?= {
						is_alive = yes
						is_married = no
					}
				}

				add_character_flag = segrellion_arianne_marriage_event_scheduled

				trigger_event = {
					id = segrellion_winds.0201
					days = 365
				}
			}


		# Prevent AGOT's normal dynamic Faegon invasion
		# from creating a second Young Griff.
		if = {
			limit = {
				NOT = {
					exists = global_var:faegon_invasion
				}
			}

			set_global_variable = {
				name = faegon_invasion
				value = scope:iaf_young_griff
			}
		}
	}
}

# =========================================================
# 300 AC - LEO STORMFLAME
# DEFENCE OF THE FLAMELANDS
# =========================================================

segrellion_setup_stormflame_300_effect = {

	if = {
		limit = {

			current_date >= 8300.1.1
			current_date < 8301.1.1

			character:Segrellion_36 ?= {

				is_alive = yes

				NOT = {
					has_character_flag = iaf_8300_leo_stormflame_setup
				}
			}
		}


		character:Segrellion_36 = {

			save_scope_as = leo_stormflame_300


			# =================================================
			# THE STORMFLAME HOST
			# 3,000 levies + 2,000 professional soldiers
			# =================================================

			spawn_army = {

				name = iaf_leo_stormflame_host

				men_at_arms = {
					type = armored_footmen
					men = 800
				}

				men_at_arms = {
					type = pikemen_unit
					men = 600
				}

				men_at_arms = {
					type = bowmen
					men = 600
				}

				levies = 3000

				inheritable = yes

				location = title:c_farring_cross.title_province
				origin = title:c_farring_cross.title_province
			}


			# =================================================
			# WALDER FREY
			# =================================================

			if = {
				limit = {
					character:Frey_1 ?= {
						is_alive = yes
					}
				}

				create_alliance = {
					target = character:Frey_1
					allied_through_owner = scope:leo_stormflame_300
					allied_through_target = character:Frey_1
				}
			}


			# =================================================
			# LORD OF DRIFTMARK
			# =================================================

			title:d_driftmark.holder ?= {
				save_scope_as = driftmark_lord_300
			}

			if = {
				limit = {

					exists = scope:driftmark_lord_300

					scope:driftmark_lord_300 = {
						is_alive = yes
						this != scope:leo_stormflame_300
					}
				}

				create_alliance = {
					target = scope:driftmark_lord_300
					allied_through_owner = scope:leo_stormflame_300
					allied_through_target = scope:driftmark_lord_300
				}
			}


			add_character_flag = iaf_8300_leo_stormflame_setup
		}
	}
}

# =========================================================
# THE UNICORN RETURNS
# CAPTURE THE STORMFLAME COURT
# =========================================================

segrellion_capture_stormflame_court_effect = {

	character:Segrellion_60 ?= {
		save_scope_as = daemon_stormflame_judgement
	}

	if = {
		limit = {
			exists = scope:daemon_stormflame_judgement
		}

		house:house_Stormflame = {

			every_house_member = {

				limit = {
					is_alive = yes
				}

				save_temporary_scope_as = stormflame_captive

				add_character_flag = segrellion_stormflame_captive

				scope:daemon_stormflame_judgement = {

					imprison = {
						target = scope:stormflame_captive
						type = dungeon
					}
				}
			}
		}
	}
}

# =========================================================
# EXECUTE LEO, SEND THE MEN TO THE NIGHT'S WATCH
# =========================================================

segrellion_execute_leo_spare_stormflame_family_effect = {

	character:Segrellion_36 ?= {

		if = {
			limit = {

				is_alive = yes

				has_character_flag = segrellion_stormflame_captive
			}

			remove_character_flag = segrellion_stormflame_captive

			death = {
				death_reason = death_execution
			}
		}
	}


	house:house_Stormflame = {

		every_house_member = {

			limit = {

				is_alive = yes

				has_character_flag = segrellion_stormflame_captive
			}


			# Adult Stormflame men take the black.
			if = {
				limit = {

					is_female = no
					is_adult = yes
				}

				release_from_prison = yes

				add_trait = nightswatch
			}


			# Women and boys are released.
			else = {

				release_from_prison = yes
			}


			remove_character_flag = segrellion_stormflame_captive
		}
	}
}
