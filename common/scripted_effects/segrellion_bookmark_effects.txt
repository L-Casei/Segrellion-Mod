segrellion_setup_petty_rebellion_characters_effect = {
	# AGOT's historical Joffrey Arryn is born in 100 A.C. This alternate
	# timeline uses the same character and titles, but makes him forty-three at
	# the 150 A.C. start instead of creating a duplicate Arryn.
	character:Arryn_90 ?= {
		if = {
			limit = {
				is_alive = yes
				NOT = { has_character_flag = segrellion_joffrey_age_adjusted }
			}
			set_age = 43
			add_character_flag = segrellion_joffrey_age_adjusted
		}
	}

	# House Buckwell has been removed from the alternate history before the
	# bookmark. death_vanished keeps the one-time cleanup silent, while the
	# explicit title transfer below ensures Deerfield cannot pass to a survivor.
	character:Vladwyn_2 ?= {
		if = {
			limit = { NOT = { has_character_flag = segrellion_buckwell_cleanup_done } }
			dynasty:dynn_Buckwell = {
				every_dynasty_member = {
					limit = { is_alive = yes }
					death = { death_reason = death_vanished }
				}
			}
			add_character_flag = segrellion_buckwell_cleanup_done
		}
	}

	# Title history grants Antlers (whose capital is Deerfield) in 130 A.C.
	# This fallback protects the intended holder from later-loading submods.
	if = {
		limit = {
			character:Vladwyn_2 ?= { is_alive = yes }
			OR = {
				title:c_antlers = { NOT = { exists = holder } }
				title:c_antlers.holder ?= { NOT = { this = character:Vladwyn_2 } }
			}
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = segrellion_deerfield_title_change
			add_claim_on_loss = no
		}
		title:c_antlers = {
			change_title_holder = {
				holder = character:Vladwyn_2
				change = scope:segrellion_deerfield_title_change
			}
		}
		resolve_title_and_vassal_change = scope:segrellion_deerfield_title_change
	}

	# Tristan is part of Aegon III's historical Kingsguard from 131 A.C. The
	# runtime setup uses AGOT's public Kingsguard effects so its council slots,
	# title variables, White Book and Lord Commander title stay synchronized.
	title:d_kingsguard.holder ?= {
		if = {
			limit = { NOT = { this = character:Vladwyn_1 } }
			agot_remove_kingsguard_effect = { KINGSGUARD = this }
		}
	}

	$KING$ = {
		if = {
			limit = {
				character:Vladwyn_1 = { NOT = { has_trait = kingsguard } }
				exists = cp:kingsguard_1
				exists = cp:kingsguard_2
				exists = cp:kingsguard_3
				exists = cp:kingsguard_4
				exists = cp:kingsguard_5
				exists = cp:kingsguard_6
			}
			cp:kingsguard_6 = {
				agot_remove_kingsguard_effect = { KINGSGUARD = this }
			}
		}
	}

	if = {
		limit = {
			character:Vladwyn_1 = {
				is_alive = yes
				NOT = { has_trait = kingsguard }
			}
		}
		agot_join_kingsguard_effect = {
			KINGSGUARD = character:Vladwyn_1
			KING = $KING$
		}
	}

	if = {
		limit = {
			character:Vladwyn_1 = {
				is_alive = yes
				has_trait = kingsguard
				NOT = { has_character_flag = lord_commander }
			}
		}
		agot_assign_lord_commander_effect = {
			COMMANDER = character:Vladwyn_1
			KING = $KING$
		}
	}
}

segrellion_setup_petty_rebellion_bookmark_effect = {
	if = {
		limit = {
			current_date >= 8150.1.1
			current_date < 8151.1.1
			character:Segrellion_12 ?= { is_alive = yes }
		}

		title:k_flamelands = { set_capital_county = title:c_farring_cross }
		title:d_vhalerion = { set_de_jure_liege_title = title:k_flamelands }
		title:d_masseys_hook = { set_de_jure_liege_title = title:k_flamelands }
		title:d_wendwater_forest = {
			set_title_name = segrellion_blue_forest_name
			set_de_jure_liege_title = title:k_flamelands
		}
		title:d_the_baywater_woods = { set_de_jure_liege_title = title:k_flamelands }

		# History should already create Baywater Woods in 8145. This fallback
		# makes the bookmark setup deterministic if another mod overrides that
		# title history later in the load order.
		if = {
			limit = { title:d_the_baywater_woods = { NOT = { exists = holder } } }
			create_title_and_vassal_change = {
				type = created
				save_scope_as = segrellion_baywater_title_change
				add_claim_on_loss = no
			}
			title:d_the_baywater_woods = {
				change_title_holder = {
					holder = character:Segrellion_12
					change = scope:segrellion_baywater_title_change
				}
			}
			resolve_title_and_vassal_change = scope:segrellion_baywater_title_change
		}

		# Preserve every historical count holder and make each one Aeryn's
		# vassal. The duchy remains in Aeryn's own hands.
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = segrellion_baywater_vassal_change
			add_claim_on_loss = no
		}
		title:c_the_gauntlet.holder ?= {
			if = {
				limit = {
					NOT = { this = character:Segrellion_12 }
					NOT = { liege = character:Segrellion_12 }
				}
				change_liege = {
					liege = character:Segrellion_12
					change = scope:segrellion_baywater_vassal_change
				}
			}
		}
		title:c_brushmill.holder ?= {
			if = {
				limit = {
					NOT = { this = character:Segrellion_12 }
					NOT = { liege = character:Segrellion_12 }
				}
				change_liege = {
					liege = character:Segrellion_12
					change = scope:segrellion_baywater_vassal_change
				}
			}
		}
		title:c_bywater.holder ?= {
			if = {
				limit = {
					NOT = { this = character:Segrellion_12 }
					NOT = { liege = character:Segrellion_12 }
				}
				change_liege = {
					liege = character:Segrellion_12
					change = scope:segrellion_baywater_vassal_change
				}
			}
		}
		title:c_greyhelm.holder ?= {
			if = {
				limit = {
					NOT = { this = character:Segrellion_12 }
					NOT = { liege = character:Segrellion_12 }
				}
				change_liege = {
					liege = character:Segrellion_12
					change = scope:segrellion_baywater_vassal_change
				}
			}
		}
		resolve_title_and_vassal_change = scope:segrellion_baywater_vassal_change
		title:c_farring_cross = { set_title_name = segrellion_the_beacon_name }
		title:b_castle_farring = { set_title_name = segrellion_the_beacon_name }
		title:b_castle_farring.title_province = {
			if = {
				limit = { NOT = { has_building_or_higher = castle_01 } }
				add_building = castle_01
			}
			if = {
				limit = { has_special_building_slot = no }
				add_special_building_slot = segrellion_beacon_01
			}
			if = {
				limit = { NOT = { has_building = segrellion_beacon_01 } }
				add_special_building = segrellion_beacon_01
			}
		}

		# Alternate-history royal succession for this bookmark. At the normal
		# 8150 date Aegon III still rules in AGOT history; here he was killed by
		# Vhalyrax in 8148 and his primary heir, Daeron, inherits the throne.
		title:h_the_iron_throne.holder ?= {
			if = {
				limit = { exists = primary_heir }
				save_scope_as = petty_rebellion_aegon
				primary_heir = { save_scope_as = petty_rebellion_daeron }
			}
		}
		scope:petty_rebellion_aegon ?= {
			death = { death_reason = death_dragon }
		}
		title:h_the_iron_throne.holder ?= {
			save_scope_as = petty_rebellion_king
			add_character_flag = segrellion_petty_rebellion_young_king
		}
		segrellion_setup_petty_rebellion_characters_effect = {
			KING = scope:petty_rebellion_king
		}

		# Cregan Stark and Joffrey Arryn are the two active rebel leaders at
		# bookmark start. Optional title scopes keep the setup compatible with
		# submods that rename or remove either lord-paramountcy.
		title:k_the_north.holder ?= {
			if = {
				limit = {
					is_alive = yes
					scope:petty_rebellion_king ?= { is_alive = yes }
					NOT = { this = scope:petty_rebellion_king }
					NOT = { is_at_war_with = scope:petty_rebellion_king }
				}
				add_character_flag = segrellion_petty_rebellion_leader
				start_war = {
					casus_belli = independence_war
					target = scope:petty_rebellion_king
				}
			}
		}
		title:k_the_vale.holder ?= {
			if = {
				limit = {
					is_alive = yes
					scope:petty_rebellion_king ?= { is_alive = yes }
					NOT = { this = scope:petty_rebellion_king }
					NOT = { is_at_war_with = scope:petty_rebellion_king }
				}
				add_character_flag = segrellion_petty_rebellion_leader
				start_war = {
					casus_belli = independence_war
					target = scope:petty_rebellion_king
				}
			}
		}

		character:Segrellion_12 = {
			save_scope_as = petty_rebellion_aeryn
			add_character_flag = segrellion_founded_the_flamelands
			add_character_flag = segrellion_established_the_beacon
			add_character_flag = segrellion_icebringer_authorized
			if = {
				limit = { has_title = title:k_flamelands }
				set_primary_title_to = title:k_flamelands
			}
			if = {
				limit = { has_title = title:c_farring_cross }
				set_realm_capital = title:c_farring_cross
			}
		}

		character:Segrellion_12 = {
			if = {
				limit = {
					is_current_dragonrider = no
					NOT = { exists = global_var:segrellion_historical_maenor }
				}
				segrellion_spawn_petty_rebellion_maenor_effect = { OWNER = this }
			}
		}

		global_var:segrellion_icebringer ?= {
			set_owner = scope:petty_rebellion_aeryn
			equip_artifact_to_owner_replace = yes
			agot_add_artifact_history = {
				ARTIFACT = this
				TYPE = inherited
				DATE = 8150.1.1
				RECIPIENT = character:Segrellion_12
			}
		}

		character:Segrellion_1 ?= {
			every_character_artifact = {
				limit = { has_variable = segrellion_plate_of_segrell_artifact }
				set_owner = scope:petty_rebellion_aeryn
				equip_artifact_to_owner_replace = yes
				agot_add_artifact_history = {
					ARTIFACT = this
					TYPE = inherited
					DATE = 8150.1.1
					RECIPIENT = character:Segrellion_12
				}
			}
		}

		character:Segrellion_12 = {
			if = {
				limit = {
					NOT = { any_character_artifact = { has_variable = segrellion_icebringer_artifact } }
				}
				segrellion_create_artifact_icebringer_effect = { OWNER = this }
			}
			if = {
				limit = {
					NOT = { any_character_artifact = { has_variable = segrellion_plate_of_segrell_artifact } }
				}
				segrellion_create_artifact_plate_of_segrell_effect = { OWNER = this }
			}
		}
	}
}
