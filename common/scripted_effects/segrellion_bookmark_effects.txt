# =========================================================
# 300 AC - WINDS OF WINTER
# PENTOS EXILE HOSTS
# =========================================================

segrellion_setup_winds_of_winter_hosts_effect = {

	if = {
		limit = {
			current_date >= 8300.1.1
			current_date < 8301.1.1
		}

		# =====================================================
		# DAEMON SEGRELLION
		# 4,000 MEN
		# =====================================================

		character:Segrellion_60 ?= {

			if = {
				limit = {
					is_alive = yes

					NOT = {
						has_character_flag = iaf_8300_daemon_host_created
					}
				}

				# Daemon's restoration of the Flamelands must go through
				# The Unicorn Returns rather than an ordinary claim war.

				if = {
					limit = {
						has_claim_on = title:e_flamelands
					}

					remove_claim = title:e_flamelands
				}

				if = {
					limit = {
						has_claim_on = title:k_flamelands
					}

					remove_claim = title:k_flamelands
				}

				

				spawn_army = {

					name = iaf_daemon_winds_host

					# 4,250 men total.
					men_at_arms = {
						type = armored_horsemen
						men = 300
					}

					men_at_arms = {
						type = armored_footmen
						men = 1000
					}

					men_at_arms = {
						type = pikemen
						men = 900
					}

					men_at_arms = {
						type = bowmen
						men = 1050
					}

					levies = 1000

					inheritable = yes

					location = title:c_storms_end.title_province
					origin = title:c_storms_end.title_province
				}

				add_character_flag = iaf_8300_daemon_host_created
			}
		}


		# =====================================================
		# JOFFREY VLADWYN
		# THE TRUE SENTINELS
		# 4,000 PROFESSIONAL SOLDIERS
		# =====================================================

		character:Vladwyn_5 ?= {

			if = {
				limit = {
					is_alive = yes

					NOT = {
						has_character_flag = iaf_8300_true_sentinels_host_created
					}
				}

				spawn_army = {

					name = iaf_true_sentinels_host

					men_at_arms = {
						type = black_sentinels
						stacks = 12
					}

					men_at_arms = {
						type = armored_footmen
						stacks = 11
					}

					men_at_arms = {
						type = pikemen
						stacks = 10
					}

					men_at_arms = {
						type = bowmen
						stacks = 10
					}

					inheritable = yes

					location = title:c_storms_end.title_province
					origin = title:c_storms_end.title_province
				}

				add_character_flag = iaf_8300_true_sentinels_host_created
			}
		}
	}
}

# =========================================================
# 300 AC - MASSEY'S HOOK
# DEFIANCE AGAINST THE STORMFLAMES
# =========================================================

segrellion_setup_massey_defiance_effect = {

	if = {
		limit = {
			current_date >= 8298.10.24

			title:d_masseys_hook.holder ?= {
				is_alive = yes

				NOT = {
					has_character_flag = segrellion_massey_defiance_initialized
				}
			}
		}

		title:d_masseys_hook.holder = {

			save_scope_as = massey_lord

			add_character_modifier = {
				modifier = segrellion_defiant_against_stormflames_modifier
			}

			# =================================================
			# HATRED OF HOUSE STORMFLAME
			# =================================================

			house:house_Stormflame = {

				every_house_member = {
					limit = {
						is_alive = yes
					}

					save_scope_as = stormflame_member

					scope:massey_lord = {
						add_opinion = {
							target = scope:stormflame_member
							modifier = segrellion_massey_despises_stormflames_opinion
						}
					}
				}
			}

			# Leo personally bears the greatest blame.
			add_opinion = {
				target = character:Segrellion_36
				modifier = segrellion_massey_despises_leo_opinion
			}

			# Daemon represents the restoration of the old order.
			add_opinion = {
				target = character:Segrellion_60
				modifier = segrellion_massey_supports_daemon_opinion
			}

			# The True Sentinels are respected as fellow enemies
			# of the Stormflame regime.
			add_opinion = {
				target = character:Vladwyn_5
				modifier = segrellion_massey_respects_true_sentinels_opinion
			}

			add_character_flag = segrellion_massey_defiance_initialized
		}
	}
}
