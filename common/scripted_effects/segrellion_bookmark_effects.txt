# Keep Arianne available for Daemon's and Young Griff's scripted
# negotiations during the first five years of the 300 AC bookmark.
# This is intentionally save-compatible and idempotent because AGOT
# events can remove refusing_marriage for their own story routes.
segrellion_arianne_marriage_lock_effect = {
	# Repair saves in which an unrelated event or AI action already married
	# Arianne before this protection was installed.
	if = {
		limit = {
			game_start_date >= 8300.8.20
			game_start_date < 8301.1.1
			current_date < 8305.8.20
			character:Martell_9 ?= {
				is_alive = yes

				NOT = {
					has_character_flag = segrellion_arianne_young_griff_route_resolved
				}

				any_spouse = {
					this != character:Segrellion_60
					this != character:IAF_Young_Griff_1
				}
			}
		}

		character:Martell_9 = {
			every_spouse = {
				limit = {
					this != character:Segrellion_60
					this != character:IAF_Young_Griff_1
				}
				divorce = prev
			}
		}
	}

	if = {
		limit = {
			game_start_date >= 8300.8.20
			game_start_date < 8301.1.1
			current_date < 8305.8.20
			character:Martell_9 ?= {
				is_alive = yes
				is_married = no
				
				NOT = { 
					has_character_flag = segrellion_arianne_event_marriage_completed 
					}

				NOT = {
					has_character_flag = segrellion_arianne_young_griff_route_resolved
				}
			}
		}

		character:Martell_9 = {
			if = {
				limit = { NOT = { has_trait = refusing_marriage } }
				add_trait = refusing_marriage
			}
			add_character_flag = segrellion_arianne_marriage_lock_active
		}
	}

	if = {
		limit = {
			game_start_date >= 8300.8.20
			game_start_date < 8301.1.1
			current_date >= 8305.8.20
			character:Martell_9 ?= {
				has_character_flag = segrellion_arianne_marriage_lock_active
			}
		}

		character:Martell_9 = {
			if = {
				limit = { has_trait = refusing_marriage }
				remove_trait = refusing_marriage
			}
			remove_character_flag = segrellion_arianne_marriage_lock_active
			add_character_flag = segrellion_arianne_marriage_lock_expired
		}
	}
}

segrellion_setup_massey_defiance_effect = {

	if = {
		limit = {
			current_date >= 8298.10.24

			title:d_masseys_hook.holder ?= {
				is_alive = yes

				NOT = {
					has_character_flag = segrellion_massey_defiance_initialized
				}
			}
		}

		title:d_masseys_hook.holder = {

			save_scope_as = massey_lord

			add_character_modifier = {
				modifier = segrellion_defiant_against_stormflames_modifier
			}

			# =================================================
			# HATRED OF HOUSE STORMFLAME
			# =================================================

			house:house_Stormflame = {

				every_house_member = {
					limit = {
						is_alive = yes
					}

					save_scope_as = stormflame_member

					scope:massey_lord = {
						add_opinion = {
							target = scope:stormflame_member
							modifier = segrellion_massey_despises_stormflames_opinion
						}
					}
				}
			}

			# Leo personally bears the greatest blame.
			add_opinion = {
				target = character:Stormflame_3
				modifier = segrellion_massey_despises_leo_opinion
			}

			# Daemon represents the restoration of the old order.
			add_opinion = {
				target = character:Segrellion_60
				modifier = segrellion_massey_supports_daemon_opinion
			}

			# The True Sentinels are respected as fellow enemies
			# of the Stormflame regime.
			add_opinion = {
				target = character:Vladwyn_5
				modifier = segrellion_massey_respects_true_sentinels_opinion
			}

			add_character_flag = segrellion_massey_defiance_initialized
		}
	}
}

# 300 AC - DAEMON'S TRUE PARENTAGE
#
# Hugh discovers the letter Daena entrusted to Symon.
#
# This setup is deliberately independent from the
# Plate of Segrell storyline.
# =========================================================

segrellion_setup_daemon_parentage_story_effect = {

	# =========================================================
	# DAEMON'S HIDDEN BIOLOGICAL FATHER
	#
	# Public father remains Maegor the Meek.
	# Biological father is Aerys II.
	#
	# Daenerys' father is Aerys II, so we use her father
	# rather than hardcoding AGOT's Aerys character ID.
	# =========================================================

	character:Targaryen_9 ?= {

		father ?= {
			save_temporary_scope_as = segrellion_daemon_true_father
		}
	}


	if = {
		limit = {

			current_date >= 8300.8.20

			exists = scope:segrellion_daemon_true_father

			character:Segrellion_60 ?= {
				is_alive = yes
			}
		}

		character:Segrellion_60 = {
			set_real_father = scope:segrellion_daemon_true_father
		}
	}

	if = {

		limit = {

			current_date >= 8300.8.20

			character:Segrellion_43 ?= {
				is_alive = yes

				NOT = {
					has_character_flag = segrellion_daemon_parentage_story_initialized
				}
			}

			character:Segrellion_60 ?= {
				is_alive = yes

				NOT = {
					has_character_flag = segrellion_daemon_parentage_known
				}
			}
		}

				id = segrellion_daemon_parentage.0001

# =========================================================
# 300 AC - HISTORICAL YOUNG GRIFF
# =========================================================

# AGOT keeps Blackfyre hidden with Illyrio after 281 AC and normally
# materializes it through the standard fAegon invasion. Our historical
# Young Griff uses a custom route, so give him that same unique artifact
# directly and clear the hidden-artifact state to prevent a later duplicate.
segrellion_setup_young_griff_blackfyre_effect = {
	if = {
		limit = {
			game_start_date >= 8300.8.20
			game_start_date < 8301.1.1
			character:IAF_Young_Griff_1 ?= { is_alive = yes }
		}

		if = {
			limit = {
				character:IAF_Young_Griff_1 = {
					NOT = { has_character_flag = segrellion_young_griff_blackfyre_setup }
				}
			}

			if = {
				limit = {
					any_artifact = {
						count >= 1
						has_artifact_modifier = vs_blackfyre_modifier
					}
				}

				random_artifact = {
					limit = { has_artifact_modifier = vs_blackfyre_modifier }
					set_owner = character:IAF_Young_Griff_1
					equip_artifact_to_owner_replace = yes
				}
			}
			else = {
				agot_create_artifact_vs_blackfyre_effect = {
					OWNER = character:IAF_Young_Griff_1
				}
			}

			character:IAF_Young_Griff_1 = {
				add_character_flag = segrellion_young_griff_blackfyre_setup
			}
		}

		# AGOT's scenario-artifact setup may run after our zero-tick effect.
		# Repeat this cleanup after the lobby and on the monthly fallback.
		if = {
			limit = { exists = global_var:blackfyre_hidden }
			remove_global_variable = blackfyre_hidden
		}

		character:Mopatis_1 ?= {
			if = {
				limit = {
					any_secret = { secret_type = secret_agot_hidden_artifact }
				}
				random_secret = {
					limit = { secret_type = secret_agot_hidden_artifact }
					remove_secret = yes
				}
			}
		}
	}
}

segrellion_setup_young_griff_300_effect = {

	if = {
		limit = {

			current_date >= 8300.8.20

			character:IAF_Young_Griff_1 ?= {
				is_alive = yes
			}
		}


		character:IAF_Young_Griff_1 = {

			save_scope_as = iaf_young_griff

			segrellion_initialize_young_griff_aggression_effect = yes
			segrellion_setup_young_griff_blackfyre_effect = yes

			# =================================================
			# TRUE IDENTITY
			#
			# Publicly Aegon Targaryen.
			# In truth, the Blackfyre identity remains buried.
			# =================================================

			if = {

				limit = {

					NOT = {
						has_character_flag = segrellion_young_griff_true_blackfyre
					}
				}

				add_character_flag = segrellion_young_griff_true_blackfyre
			}

			# =================================================
			# STARTING FAME
			# Young Griff begins at Distinguished.
			# =================================================

			if = {

				limit = {

					NOT = {
						has_character_flag = segrellion_young_griff_starting_fame_setup
					}
				}

				if = {

					limit = {
						prestige_level < 2
					}

					# Established -> Distinguished.
					# Also leaves him the prestige required for his claim.
					add_prestige = 1000
				}

				add_character_flag = segrellion_young_griff_starting_fame_setup
			}

			#remove_character_modifier = shade_of_the_evening_hair_dye TO REVEAL WHITE HAIR

			if = {
				limit = {
					NOT = {
						has_character_modifier = shade_of_the_evening_hair_dye
					}
				}

				add_character_modifier = {
					modifier = shade_of_the_evening_hair_dye
				}
			}

			if = {
				limit = {

					NOT = {
						has_character_flag = segrellion_young_griff_supports_daemon
					}

					character:Segrellion_60 = {

						NOT = {
							has_character_flag = segrellion_pledged_to_stannis
						}
					}
				}

				add_character_flag = segrellion_young_griff_supports_daemon
			}

			# =================================================
			# DAEMON KNOWS YOUNG GRIFF SUPPORTS HIM
			# =================================================

			if = {
				limit = {
					has_character_flag = segrellion_young_griff_supports_daemon
				}

				character:Segrellion_60 ?= {

					if = {
						limit = {

							is_alive = yes

							NOT = {
								has_character_flag = segrellion_daemon_supported_by_young_griff
							}
						}

						add_character_flag = segrellion_daemon_supported_by_young_griff
					}
				}
			}
		}

		# =================================================
		# YOUNG GRIFF - STORYLINE
		#
		# The old automatic 4100 marriage opening has been
		# replaced by:
		#
		# Varys -> Storm's End -> Arianne -> Dorne
		# =================================================

		# Prevent AGOT's normal dynamic Faegon invasion
		# from creating a second Young Griff.
		if = {
			limit = {
				NOT = {
					exists = global_var:faegon_invasion
				}
			}

			set_global_variable = {
				name = faegon_invasion
				value = scope:iaf_young_griff
			}
		}
	}
}
