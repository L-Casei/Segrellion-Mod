# =========================================================
# REVEAL TRUE PARENTS
# =========================================================

vladwyn_reveal_true_parentage_effect = {

	set_father = character:Baratheon_34
	set_mother = character:Stark_79

	if = {

		limit = {

			NOT = {
				has_trait = bastard
			}
		}

		add_trait = bastard
	}

	add_character_flag = vladwyn_public_parentage
}



# =========================================================
# NORTH - PRESSED CLAIMS
# COUNTY -> DUCHY -> KINGDOM -> EMPIRE
# =========================================================

vladwyn_add_north_pressed_claims_effect = {

	save_scope_as = vladwyn_claimant


	title:c_winterfell = {

		save_scope_as = vladwyn_north_county


		de_jure_liege = {

			save_scope_as = vladwyn_north_duchy


			de_jure_liege = {

				save_scope_as = vladwyn_north_kingdom


				de_jure_liege = {

					save_scope_as = vladwyn_north_empire
				}
			}
		}
	}


	scope:vladwyn_claimant = {

		add_pressed_claim = scope:vladwyn_north_county
		add_pressed_claim = scope:vladwyn_north_duchy
		add_pressed_claim = scope:vladwyn_north_kingdom
		add_pressed_claim = scope:vladwyn_north_empire
	}


	clear_saved_scope = vladwyn_claimant
	clear_saved_scope = vladwyn_north_county
	clear_saved_scope = vladwyn_north_duchy
	clear_saved_scope = vladwyn_north_kingdom
	clear_saved_scope = vladwyn_north_empire
}
