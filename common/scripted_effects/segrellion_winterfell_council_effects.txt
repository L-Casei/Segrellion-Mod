# =========================================================
# ENSURE WINTERFELL COUNCIL ATTENDEES
#
# ROOT = Council host
#
# Used both immediately after AI spawn and by recovery
# pulses. This means an already-running empty Council in an
# existing save repairs itself the next time Jon is checked.
# =========================================================

segrellion_winterfell_council_ensure_attendees_effect = {

	# =====================================================
	# GET THE ACTIVITY RELIABLY
	#
	# on_start already supplies scope:activity, while the
	# monthly/save-repair route reaches it through the host's
	# involved_activity. Support both so there is no one-tick
	# race during activity creation.
	# =====================================================

	save_scope_as = segrellion_winterfell_council_runtime_host


	if = {

		limit = {

			exists = scope:activity

			scope:activity = {
				has_activity_type = activity_segrellion_winterfell_council
			}
		}


		scope:activity = {
			save_scope_as = segrellion_winterfell_council_runtime_activity
		}
	}


	else_if = {

		limit = {

			exists = involved_activity

			in_activity_type = activity_segrellion_winterfell_council
		}


		involved_activity = {
			save_scope_as = segrellion_winterfell_council_runtime_activity
		}
	}


	if = {

		limit = {

			exists = scope:segrellion_winterfell_council_runtime_activity
		}


		# =================================================
		# LANDED NORTHERN LORDS
		#
		# Invite the political North, not merely whoever happens
		# to be a direct vassal at this exact post-war moment.
		# =================================================

		every_ruler = {

			limit = {

				this != scope:segrellion_winterfell_council_runtime_host

				is_alive = yes
				is_adult = yes

				highest_held_title_tier >= tier_county

				segrellion_is_northern_lord_for_council_trigger = yes

				in_diplomatic_range = scope:segrellion_winterfell_council_runtime_host

				NOT = {
					is_participant_in_activity = scope:segrellion_winterfell_council_runtime_activity
				}

				can_join_activity = scope:segrellion_winterfell_council_runtime_activity
			}


			invite_to_activity = scope:segrellion_winterfell_council_runtime_activity

			accept_activity_invite = scope:segrellion_winterfell_council_runtime_activity
		}


		# =================================================
		# STARK FAMILY
		# =================================================

		every_close_family_member = {

			limit = {

				this != scope:segrellion_winterfell_council_runtime_host

				is_alive = yes

				OR = {
					is_adult = yes
					dynasty = character:Stark_Dummy_8.dynasty
				}

				in_diplomatic_range = scope:segrellion_winterfell_council_runtime_host

				NOT = {
					is_participant_in_activity = scope:segrellion_winterfell_council_runtime_activity
				}

				can_join_activity = scope:segrellion_winterfell_council_runtime_activity
			}


			invite_to_activity = scope:segrellion_winterfell_council_runtime_activity

			accept_activity_invite = scope:segrellion_winterfell_council_runtime_activity
		}


		# =================================================
		# LORD COMMANDER OF THE NIGHT'S WATCH
		# =================================================

		title:k_the_wall.holder ?= {

			if = {

				limit = {

					this != scope:segrellion_winterfell_council_runtime_host

					is_alive = yes
					is_adult = yes

					in_diplomatic_range = scope:segrellion_winterfell_council_runtime_host

					NOT = {
						is_participant_in_activity = scope:segrellion_winterfell_council_runtime_activity
					}

					can_join_activity = scope:segrellion_winterfell_council_runtime_activity
				}


				invite_to_activity = scope:segrellion_winterfell_council_runtime_activity

				accept_activity_invite = scope:segrellion_winterfell_council_runtime_activity
			}
		}
	}
}


# =========================================================
# JON SNOW - START THE WINTERFELL COUNCIL
#
# ROOT = current Jon
#
# The Council is story-critical. Once Stannis is dead and
# Jon has called the lords, AI Jon must actually host it.
# =========================================================

segrellion_jon_try_start_winterfell_council_effect = {

	# Save-upgrade / recovery path: if the Council already exists,
	# make sure it cannot remain an empty hall.
	if = {

		limit = {

			segrellion_is_current_jon_trigger = yes

			is_alive = yes

			in_activity_type = activity_segrellion_winterfell_council
		}


		segrellion_winterfell_council_ensure_attendees_effect = yes
	}


	if = {

		limit = {

			segrellion_is_current_jon_trigger = yes

			is_alive = yes
			is_ai = yes

			has_title = title:e_the_north

			character:Baratheon_3 ?= {
				is_alive = no
			}

			has_character_flag = segrellion_jon_winterfell_council_called

			NOT = {
				has_character_flag = segrellion_jon_winterfell_council_completed
			}

			NOT = {
				has_character_flag = segrellion_jon_winterfell_council_ai_spawn_cooldown
			}

			NOT = {
				in_activity_type = activity_segrellion_winterfell_council
			}
		}


		# AI Jon now pays 0 Prestige in the activity definition.
		# No artificial Prestige injection is needed here.


		add_character_flag = segrellion_jon_winterfell_council_war_locked


		if = {

			limit = {

				NOT = {
					has_character_flag = sworn_peace
				}
			}

			add_character_flag = sworn_peace
		}


		save_scope_as = segrellion_winterfell_council_ai_host


		every_character_war = {

			if = {

				limit = {

					primary_attacker != scope:segrellion_winterfell_council_ai_host

					primary_defender != scope:segrellion_winterfell_council_ai_host
				}

				remove_participant = scope:segrellion_winterfell_council_ai_host
			}
		}


		add_character_flag = {
			flag = segrellion_jon_winterfell_council_ai_spawn_cooldown
			days = 3
		}


		title:c_winterfell.title_province = {

			spawn_activity = {

				owner = scope:segrellion_winterfell_council_ai_host

				type = activity_segrellion_winterfell_council

				days = 365
			}
		}


		# spawn_activity bypasses the normal planner path, so attach
		# the political attendees immediately.
		scope:segrellion_winterfell_council_ai_host = {
			segrellion_winterfell_council_ensure_attendees_effect = yes
		}
	}
}

# =========================================================
# CALCULATE ONE LORD'S POLITICAL INFLUENCE
#
# Current scope = voting character.
#
# Base:
# Unlanded      1
# Baron         3
# Count         5
# Duke          8
# King         10
# Emperor      12
#
# Bonuses:
# Powerful Vassal +2
# Prestige        +1/+1/+1
#
# Maximum = 15
# =========================================================

segrellion_winterfell_council_calculate_influence_effect = {

	set_variable = {
		name = segrellion_council_influence
		value = 1
	}


	if = {
		limit = {
			is_landed = yes
		}

		set_variable = {
			name = segrellion_council_influence
			value = 3
		}
	}


	if = {
		limit = {
			highest_held_title_tier >= tier_county
		}

		set_variable = {
			name = segrellion_council_influence
			value = 5
		}
	}


	if = {
		limit = {
			highest_held_title_tier >= tier_duchy
		}

		set_variable = {
			name = segrellion_council_influence
			value = 8
		}
	}


	if = {
		limit = {
			highest_held_title_tier >= tier_kingdom
		}

		set_variable = {
			name = segrellion_council_influence
			value = 10
		}
	}


	if = {
		limit = {
			highest_held_title_tier >= tier_empire
		}

		set_variable = {
			name = segrellion_council_influence
			value = 12
		}
	}


	if = {
		limit = {
			is_powerful_vassal = yes
		}

		change_variable = {
			name = segrellion_council_influence
			add = 2
		}
	}


	if = {
		limit = {
			prestige_level >= 3
		}

		change_variable = {
			name = segrellion_council_influence
			add = 1
		}
	}


	if = {
		limit = {
			prestige_level >= 4
		}

		change_variable = {
			name = segrellion_council_influence
			add = 1
		}
	}


	if = {
		limit = {
			prestige_level >= 5
		}

		change_variable = {
			name = segrellion_council_influence
			add = 1
		}
	}


	if = {
		limit = {
			var:segrellion_council_influence > 15
		}

		set_variable = {
			name = segrellion_council_influence
			value = 15
		}
	}
}


# =========================================================
# APPLY ONE CHARACTER'S CURRENT INTENT
#
# This does NOT permanently stack every week.
# The whole intent tally is rebuilt every time.
# =========================================================

segrellion_winterfell_council_apply_single_voter_effect = {

	segrellion_winterfell_council_calculate_influence_effect = yes

	save_temporary_scope_value_as = {
		name = segrellion_council_vote_weight
		value = var:segrellion_council_influence
	}


	# -----------------------------------------------------
	# LONG NIGHT
	# -----------------------------------------------------

	if = {
		limit = {
			has_activity_intent = segrellion_winterfell_council_long_night_intent
		}

		scope:activity = {

			change_variable = {
				name = segrellion_council_wall_intent_support
				add = scope:segrellion_council_vote_weight
			}
		}
	}


	# -----------------------------------------------------
	# RED WEDDING
	# -----------------------------------------------------

	if = {
		limit = {
			has_activity_intent = segrellion_winterfell_council_revenge_intent
		}

		scope:activity = {

			change_variable = {
				name = segrellion_council_revenge_intent_support
				add = scope:segrellion_council_vote_weight
			}
		}
	}


	# -----------------------------------------------------
	# INDEPENDENCE
	# -----------------------------------------------------

	if = {
		limit = {
			has_activity_intent = segrellion_winterfell_council_independence_intent
		}

		scope:activity = {

			change_variable = {
				name = segrellion_council_independence_intent_support
				add = scope:segrellion_council_vote_weight
			}
		}
	}


	# -----------------------------------------------------
	# DEFEND / CONSOLIDATE
	# -----------------------------------------------------

	if = {
		limit = {
			has_activity_intent = segrellion_winterfell_council_defend_north_intent
		}

		scope:activity = {

			change_variable = {
				name = segrellion_council_defend_intent_support
				add = scope:segrellion_council_vote_weight
			}
		}
	}


	# -----------------------------------------------------
	# DRAGON ALLIANCE
	# -----------------------------------------------------

	if = {
		limit = {
			has_activity_intent = segrellion_winterfell_council_dragon_alliance_intent
		}

		scope:activity = {

			change_variable = {
				name = segrellion_council_dragon_intent_support
				add = scope:segrellion_council_vote_weight
			}
		}
	}


	# -----------------------------------------------------
	# KINGMAKER
	#
	# IMPORTANT:
	# This ONLY votes for a leader.
	# It no longer gives Independence points.
	# -----------------------------------------------------

	if = {
		limit = {
			has_activity_intent = segrellion_winterfell_council_kingmaker_intent
		}

		intent_target ?= {

			change_variable = {
				name = segrellion_council_king_intent_support
				add = scope:segrellion_council_vote_weight
			}
		}
	}


	# Hear the North intentionally contributes nothing.

	remove_variable = segrellion_council_influence
}


# =========================================================
# RECOUNT ALL CURRENT INTENTS
# =========================================================

segrellion_winterfell_council_recount_intent_support_effect = {

	scope:activity = {

		set_variable = {
			name = segrellion_council_wall_intent_support
			value = 0
		}

		set_variable = {
			name = segrellion_council_revenge_intent_support
			value = 0
		}

		set_variable = {
			name = segrellion_council_independence_intent_support
			value = 0
		}

		set_variable = {
			name = segrellion_council_defend_intent_support
			value = 0
		}

		set_variable = {
			name = segrellion_council_dragon_intent_support
			value = 0
		}


		every_attending_character = {

			set_variable = {
				name = segrellion_council_king_intent_support
				value = 0
			}
		}


		every_attending_character = {

			limit = {
				this != scope:host
			}

			segrellion_winterfell_council_apply_single_voter_effect = yes
		}
	}


	# Host vote is handled explicitly so it can never be missed.

	scope:host = {

		set_variable = {
			name = segrellion_council_king_intent_support
			value = 0
		}

		segrellion_winterfell_council_apply_single_voter_effect = yes
	}
}

# =========================================================
# SELECT KING CANDIDATE
# =========================================================

segrellion_winterfell_council_select_candidate_effect = {

	clear_saved_scope = segrellion_council_winning_candidate


	# =====================================================
	# BUILD FINAL CANDIDATE TOTALS
	# =====================================================

	scope:activity = {

		every_attending_character = {

			set_variable = {
				name = segrellion_council_king_total_support
				value = 0
			}

			if = {
				limit = {
					exists = var:segrellion_council_king_support
				}

				change_variable = {
					name = segrellion_council_king_total_support
					add = var:segrellion_council_king_support
				}
			}

			if = {
				limit = {
					exists = var:segrellion_council_king_intent_support
				}

				change_variable = {
					name = segrellion_council_king_total_support
					add = var:segrellion_council_king_intent_support
				}
			}
		}
	}


	# Safety for host.

	scope:host = {

		set_variable = {
			name = segrellion_council_king_total_support
			value = 0
		}

		if = {
			limit = {
				exists = var:segrellion_council_king_support
			}

			change_variable = {
				name = segrellion_council_king_total_support
				add = var:segrellion_council_king_support
			}
		}

		if = {
			limit = {
				exists = var:segrellion_council_king_intent_support
			}

			change_variable = {
				name = segrellion_council_king_total_support
				add = var:segrellion_council_king_intent_support
			}
		}
	}


	# =====================================================
	# CANON
	# Jon always wins the leadership vote.
	# =====================================================

	if = {
		limit = {

			has_game_rule = segrellion_winterfell_council_canon

			scope:host = {

				is_alive = yes

				segrellion_is_current_jon_trigger = yes
			}
		}

		scope:host = {
			save_scope_as = segrellion_council_winning_candidate
		}
	}

	# =====================================================
	# DYNAMIC
	#
	# Actual Council support decides the winner,
	# but ONLY Jon or members of the Stark dynasty
	# are valid candidates for King in the North.
	# =====================================================

	else = {

		scope:activity = {

			ordered_attending_character = {

				limit = {

					is_alive = yes

					NOT = {
						has_trait = incapable
					}


					# =============================================
					# HARD CANDIDATE RULE
					#
					# The Council may vote differently from canon,
					# but it may NEVER invent a non-Stark king.
					# Jon remains explicitly valid regardless of
					# his current dynasty/name.
					# =============================================

					OR = {

						segrellion_is_current_jon_trigger = yes

						dynasty = character:Stark_Dummy_8.dynasty
					}


					# Normal Night's Watch members cannot be crowned.
					# Jon is explicitly exempt.

					OR = {

						NOT = {
							has_trait = nightswatch
						}

						segrellion_is_current_jon_trigger = yes
					}


					exists = var:segrellion_council_king_total_support

					var:segrellion_council_king_total_support > 0
				}


				order_by = {

					value = 0

					add = var:segrellion_council_king_total_support


					# Small tie-break in favor of Jon if tied.

					if = {

						limit = {
							this = scope:host
						}

						add = 0.1
					}
				}


				position = 0

				check_range_bounds = no

				save_scope_as = segrellion_council_winning_candidate
			}
		}
	}


	# =====================================================
	# FALLBACK
	# =====================================================

	if = {
		limit = {

			NOT = {
				exists = scope:segrellion_council_winning_candidate
			}
		}

		scope:host = {
			save_scope_as = segrellion_council_winning_candidate
		}
	}


	scope:activity = {

		set_variable = {
			name = segrellion_council_winning_candidate
			value = scope:segrellion_council_winning_candidate
		}
	}
}


# =========================================================
# SELECT NORTHERN STRATEGY
#
# Current scope = activity.
#
# CANON:
# Red Wedding revenge.
#
# DYNAMIC:
# Highest weighted total.
# =========================================================

segrellion_winterfell_council_select_strategy_effect = {

	set_variable = {
		name = segrellion_council_strategy_revenge_won
		value = 0
	}

	set_variable = {
		name = segrellion_council_strategy_wall_won
		value = 0
	}

	set_variable = {
		name = segrellion_council_strategy_independence_won
		value = 0
	}

	set_variable = {
		name = segrellion_council_strategy_defend_won
		value = 0
	}

	set_variable = {
		name = segrellion_council_strategy_dragon_won
		value = 0
	}


	# =====================================================
	# CANON
	# =====================================================

	if = {
		limit = {
			has_game_rule = segrellion_winterfell_council_canon
		}

		set_variable = {
			name = segrellion_council_strategy_revenge_won
			value = 1
		}
	}


	# =====================================================
	# DYNAMIC
	# =====================================================

	else = {


		# -------------------------------------------------
		# REVENGE
		# Tie priority 1.
		# -------------------------------------------------

		if = {
			limit = {

				var:segrellion_council_revenge_total > 0

				var:segrellion_council_revenge_total >= var:segrellion_council_wall_total
				var:segrellion_council_revenge_total >= var:segrellion_council_independence_total
				var:segrellion_council_revenge_total >= var:segrellion_council_defend_total
				var:segrellion_council_revenge_total >= var:segrellion_council_dragon_total
			}

			set_variable = {
				name = segrellion_council_strategy_revenge_won
				value = 1
			}
		}


		# -------------------------------------------------
		# WALL
		# -------------------------------------------------

		else_if = {
			limit = {

				var:segrellion_council_wall_total > 0

				var:segrellion_council_wall_total >= var:segrellion_council_revenge_total
				var:segrellion_council_wall_total >= var:segrellion_council_independence_total
				var:segrellion_council_wall_total >= var:segrellion_council_defend_total
				var:segrellion_council_wall_total >= var:segrellion_council_dragon_total
			}

			set_variable = {
				name = segrellion_council_strategy_wall_won
				value = 1
			}
		}


		# -------------------------------------------------
		# INDEPENDENCE
		# -------------------------------------------------

		else_if = {
			limit = {

				var:segrellion_council_independence_total > 0

				var:segrellion_council_independence_total >= var:segrellion_council_revenge_total
				var:segrellion_council_independence_total >= var:segrellion_council_wall_total
				var:segrellion_council_independence_total >= var:segrellion_council_defend_total
				var:segrellion_council_independence_total >= var:segrellion_council_dragon_total
			}

			set_variable = {
				name = segrellion_council_strategy_independence_won
				value = 1
			}
		}


		# -------------------------------------------------
		# DEFEND THE NORTH
		# -------------------------------------------------

		else_if = {
			limit = {

				var:segrellion_council_defend_total > 0

				var:segrellion_council_defend_total >= var:segrellion_council_revenge_total
				var:segrellion_council_defend_total >= var:segrellion_council_wall_total
				var:segrellion_council_defend_total >= var:segrellion_council_independence_total
				var:segrellion_council_defend_total >= var:segrellion_council_dragon_total
			}

			set_variable = {
				name = segrellion_council_strategy_defend_won
				value = 1
			}
		}


		# -------------------------------------------------
		# DRAGON QUEEN
		# -------------------------------------------------

		else_if = {
			limit = {

				var:segrellion_council_dragon_total > 0

				var:segrellion_council_dragon_total >= var:segrellion_council_revenge_total
				var:segrellion_council_dragon_total >= var:segrellion_council_wall_total
				var:segrellion_council_dragon_total >= var:segrellion_council_independence_total
				var:segrellion_council_dragon_total >= var:segrellion_council_defend_total
			}

			set_variable = {
				name = segrellion_council_strategy_dragon_won
				value = 1
			}
		}


		# -------------------------------------------------
		# NO CLEAR MANDATE
		# Hold and consolidate.
		# -------------------------------------------------

		else = {

			set_variable = {
				name = segrellion_council_strategy_defend_won
				value = 1
			}
		}
	}
}


# =========================================================
# CROWN ANOTHER CANDIDATE
# =========================================================

segrellion_winterfell_council_crown_other_candidate_effect = {

	scope:activity.var:segrellion_council_winning_candidate ?= {
		save_scope_as = segrellion_council_new_king
	}


	if = {
		limit = {

			exists = scope:segrellion_council_new_king

			scope:segrellion_council_new_king = {

				is_alive = yes

				OR = {

					segrellion_is_current_jon_trigger = yes

					dynasty = character:Stark_Dummy_8.dynasty
				}
			}

			NOT = {
				root = scope:segrellion_council_new_king
			}
		}


		# ---------------------------------------------
		# SAVE NORTHERN VASSALS
		# ---------------------------------------------

		root = {

			every_vassal = {

				limit = {

					segrellion_is_northern_lord_for_council_trigger = yes

					this != scope:segrellion_council_new_king
				}

				add_to_list = segrellion_council_vassals_for_new_king
			}
		}


		# ---------------------------------------------
		# TRANSFER THE NORTH
		# ---------------------------------------------

		create_title_and_vassal_change = {

			type = granted

			save_scope_as = segrellion_council_new_king_title_change

			add_claim_on_loss = no
		}


		title:e_the_north = {

			change_title_holder = {

				holder = scope:segrellion_council_new_king

				change = scope:segrellion_council_new_king_title_change
			}
		}


		resolve_title_and_vassal_change = scope:segrellion_council_new_king_title_change


		# ---------------------------------------------
		# TRANSFER NORTHERN VASSALS
		# ---------------------------------------------

		create_title_and_vassal_change = {

			type = granted

			save_scope_as = segrellion_council_new_king_vassal_change

			add_claim_on_loss = no
		}


		every_in_list = {

			list = segrellion_council_vassals_for_new_king

			change_liege = {

				liege = scope:segrellion_council_new_king

				change = scope:segrellion_council_new_king_vassal_change
			}
		}


		# Jon keeps Winterfell but submits to the chosen ruler.

		root = {

			change_liege = {

				liege = scope:segrellion_council_new_king

				change = scope:segrellion_council_new_king_vassal_change
			}
		}


		resolve_title_and_vassal_change = scope:segrellion_council_new_king_vassal_change


		# ---------------------------------------------
		# NEW KING
		# ---------------------------------------------

		scope:segrellion_council_new_king = {

			set_primary_title_to = title:e_the_north

			if = {
				limit = {
					NOT = {
						has_trait = segrellion_king_in_the_north_trait
					}
				}

				add_trait = segrellion_king_in_the_north_trait
			}

			add_character_flag = segrellion_winterfell_council_chosen_king

			add_prestige = 750
		}


		root = {

			remove_character_flag = segrellion_jon_lord_paramount_north

			add_character_flag = segrellion_jon_accepted_another_northern_king
		}
	}
}
