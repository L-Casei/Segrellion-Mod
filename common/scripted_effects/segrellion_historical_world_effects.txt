segrellion_set_flamelands_realm_capital_effect = {

    if = {
        limit = {
            OR = {
                has_title = title:e_flamelands
                has_title = title:k_flamelands
            }
        }

        # Rhaegar Segrellion II
        if = {
            limit = {
                this = character:Segrellion_25
                has_title = title:c_blackmark
            }

            set_realm_capital = title:c_blackmark
        }

        # Every other ruler of the Flamelands
        else_if = {
            limit = {
                has_title = title:c_farring_cross
            }

            set_realm_capital = title:c_farring_cross
        }
    }
}

        limit = {

        }

        title:k_flamelands.holder = {
            segrellion_set_flamelands_realm_capital_effect = yes
        }
    }
}

#
# Historical renown is considered already spent.
#
# 106 AC  = 1 Segrellion legacy
# 160 AC  = 2 Segrellion legacies
# 165 AC  = 3 Segrellion legacies
# 195 AC  = full Segrellion tree
# 216 AC  = + Dragonlords I
# 283 AC  = + Dragonlords II-III
#
# By 300 AC:
#   Segrellion 1-5
#   Dragonlords 1-3
# =========================================================

segrellion_set_historical_legacies_effect = {

	dynasty:dynn_Segrellion = {


		# =================================================
		# SEGRELLION I
		# =================================================

		if = {
			limit = {
				current_date >= 8106.4.18
				NOT = {
					has_dynasty_perk = segrellion_legacy_1
				}
			}

			add_dynasty_perk = segrellion_legacy_1
		}


		# =================================================
		# SEGRELLION II
		# =================================================

		if = {
			limit = {
				current_date >= 8160.1.1
				NOT = {
					has_dynasty_perk = segrellion_legacy_2
				}
			}

			add_dynasty_perk = segrellion_legacy_2
		}


		# =================================================
		# SEGRELLION III
		# =================================================

		if = {
			limit = {
				current_date >= 8165.1.1
				NOT = {
					has_dynasty_perk = segrellion_legacy_3
				}
			}

			add_dynasty_perk = segrellion_legacy_3
		}


		# =================================================
		# AEGON DRAGONSPEAR ERA
		# THE SEGRELLION TREE IS COMPLETED
		# =================================================

		if = {
			limit = {
				current_date >= 8195.1.1
				NOT = {
					has_dynasty_perk = segrellion_legacy_4
				}
			}

			add_dynasty_perk = segrellion_legacy_4
		}

		if = {
			limit = {
				current_date >= 8195.1.1
				NOT = {
					has_dynasty_perk = segrellion_legacy_5
				}
			}

			add_dynasty_perk = segrellion_legacy_5
		}


		# =================================================
		# DRAGONLORDS I
		# =================================================

		if = {
			limit = {
				current_date >= 8216.1.1
				NOT = {
					has_dynasty_perk = dragonrider_dynasty_legacy_1
				}
			}

			add_dynasty_perk = dragonrider_dynasty_legacy_1
		}


		# =================================================
		# DRAGONLORDS II-III
		# =================================================

		if = {
			limit = {
				current_date >= 8283.1.1
				NOT = {
					has_dynasty_perk = dragonrider_dynasty_legacy_2
				}
			}

			add_dynasty_perk = dragonrider_dynasty_legacy_2
		}

		if = {
			limit = {
				current_date >= 8283.1.1
				NOT = {
					has_dynasty_perk = dragonrider_dynasty_legacy_3
				}
			}

			add_dynasty_perk = dragonrider_dynasty_legacy_3
		}
	}

		limit = {

			set_house_relation = {

			}
		}


		set_global_variable = {
			name = segrellion_black_moon_house_feud_started
			value = yes
		}
	}
