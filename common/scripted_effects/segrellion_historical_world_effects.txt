segrellion_set_flamelands_realm_capital_effect = {

    if = {
        limit = {
            OR = {
                has_title = title:e_flamelands
                has_title = title:k_flamelands
            }
        }

        # Rhaegar Segrellion II
        if = {
            limit = {
                this = character:Segrellion_25
                has_title = title:c_blackmark
            }

            set_realm_capital = title:c_blackmark
        }

        # Every other ruler of the Flamelands
        else_if = {
            limit = {
                has_title = title:c_farring_cross
            }

            set_realm_capital = title:c_farring_cross
        }
    }
}

segrellion_fix_historical_flamelands_capital_effect = {

    if = {
        limit = {
            current_date >= 8144.1.28
            title:e_flamelands = { exists = holder }
        }

        title:e_flamelands.holder = {
            segrellion_set_flamelands_realm_capital_effect = yes
        }
    }

    else_if = {
        limit = {
            current_date >= 8144.1.28
            title:k_flamelands = { exists = holder }
        }

        title:k_flamelands.holder = {
            segrellion_set_flamelands_realm_capital_effect = yes
        }
    }
}
