# =========================================================
# HOUSE VLADWYN - HISTORICAL DYNASTY PROGRESSION
#
# 106 AC: Obscure
# 129 AC: Insignificant - Tristan's Black Sentinel era
# 174 AC: Noteworthy - the landed senior line is established
# 196 AC: Reputable - Arlan the Black / First Blackfyre War
# 250 AC: Well-known - the True Sentinels survive Hugh
#
# Legacy track:
# 129 / 174 / 196 / 217 / 250 AC
#
# Joffrey era:
# Well-known, Vladwyn I-V complete, 700 spendable Renown
# on a fresh historical start.
# =========================================================

vladwyn_historical_dynasty_progression_effect = {

	dynasty:dynn_Vladwyn = {

		# ---------------------------------------------
		# LEVEL OF SPLENDOR
		# ---------------------------------------------

		if = {
			limit = { current_date >= 8250.1.1 }

			while = {
				limit = { dynasty_prestige_level < 5 }
				add_dynasty_prestige_level = 1
			}
		}
		else_if = {
			limit = { current_date >= 8196.8.1 }

			while = {
				limit = { dynasty_prestige_level < 4 }
				add_dynasty_prestige_level = 1
			}
		}
		else_if = {
			limit = { current_date >= 8174.1.2 }

			while = {
				limit = { dynasty_prestige_level < 3 }
				add_dynasty_prestige_level = 1
			}
		}
		else_if = {
			limit = { current_date >= 8129.1.1 }

			while = {
				limit = { dynasty_prestige_level < 2 }
				add_dynasty_prestige_level = 1
			}
		}
		else_if = {
			limit = { current_date >= 8106.1.1 }

			while = {
				limit = { dynasty_prestige_level < 1 }
				add_dynasty_prestige_level = 1
			}
		}


		# ---------------------------------------------
		# VLADWYN LEGACY I - Oaths Written in Blood
		# Tristan's Black Sentinel reputation.
		# ---------------------------------------------

		if = {
			limit = {
				current_date >= 8129.1.1
				NOT = { has_dynasty_perk = vladwyn_legacy_1 }
			}
			add_dynasty_perk = vladwyn_legacy_1
		}


		# ---------------------------------------------
		# VLADWYN LEGACY II - Eyes Behind the Shield
		# Eldric's generation leaves a landed tradition.
		# ---------------------------------------------

		if = {
			limit = {
				current_date >= 8174.1.2
				NOT = { has_dynasty_perk = vladwyn_legacy_2 }
			}
			add_dynasty_perk = vladwyn_legacy_2
		}


		# ---------------------------------------------
		# VLADWYN LEGACY III - The Household Blade
		# Arlan the Black and the martial Proudrack line.
		# ---------------------------------------------

		if = {
			limit = {
				current_date >= 8196.8.1
				NOT = { has_dynasty_perk = vladwyn_legacy_3 }
			}
			add_dynasty_perk = vladwyn_legacy_3
		}


		# ---------------------------------------------
		# VLADWYN LEGACY IV - Hold Until Death
		# Hugh founds the True Sentinels.
		# ---------------------------------------------

		if = {
			limit = {
				current_date >= 8217.1.1
				NOT = { has_dynasty_perk = vladwyn_legacy_4 }
			}
			add_dynasty_perk = vladwyn_legacy_4
		}


		# ---------------------------------------------
		# VLADWYN LEGACY V - The Last Man Standing
		# Hugh dies, but the company and the line survive.
		# ---------------------------------------------

		if = {
			limit = {
				current_date >= 8250.1.1
				NOT = { has_dynasty_perk = vladwyn_legacy_5 }
			}
			add_dynasty_perk = vladwyn_legacy_5
		}


		# ---------------------------------------------
		# JOFFREY ERA RENOWN
		#
		# A fresh late historical start is normalized to
		# exactly 700 spendable Renown. An ongoing campaign
		# is never reduced: it is only topped up to 700.
		# ---------------------------------------------

		if = {
			limit = {
				current_date >= 8295.8.2
				NOT = { has_variable = vladwyn_joffrey_renown_granted }
			}

			if = {
				limit = { game_start_date >= 8295.8.2 }

				if = {
					limit = { dynasty_prestige != 0 }
					add_dynasty_prestige = {
						value = dynasty_prestige
						multiply = -1
					}
				}

				add_dynasty_prestige = 700

				# Fresh Joffrey-era starts must display exactly Well-known.
				while = {
					limit = { dynasty_prestige_level < 5 }
					add_dynasty_prestige_level = 1
				}
				while = {
					limit = { dynasty_prestige_level > 5 }
					add_dynasty_prestige_level = -1
				}
			}
			else_if = {
				limit = { dynasty_prestige < 700 }

				add_dynasty_prestige = {
					value = 700
					subtract = dynasty_prestige
				}
			}

			set_variable = {
				name = vladwyn_joffrey_renown_granted
				value = yes
			}
		}
	}
}

# Historical legacy costs must not leave the starting dynasty in debt.
# Called after legacy setup and on loading existing saves. The dynasty
# marker prevents later gameplay spending from being refunded repeatedly.
segrellion_repair_historical_renown_effect = {
    dynasty:dynn_Segrellion = {
        if = {
            limit = {
                current_date >= 8106.4.18
                has_dynasty_perk = segrellion_legacy_1
                NOT = { has_variable = segrellion_historical_renown_reconciled }
            }

            if = {
                limit = { dynasty_prestige < 0 }

                set_variable = {
                    name = segrellion_splendor_before_renown_repair
                    value = dynasty_prestige_level
                }

                # Refund exactly the deficit, not a fixed 16,000-point reward.
                add_dynasty_prestige = {
                    value = dynasty_prestige
                    multiply = -1
                }

                # Renown compensation must not raise the existing Splendor tier.
                while = {
                    limit = {
                        dynasty_prestige_level > var:segrellion_splendor_before_renown_repair
                    }
                    add_dynasty_prestige_level = -1
                }
                remove_variable = segrellion_splendor_before_renown_repair
            }

            set_variable = {
                name = segrellion_historical_renown_reconciled
                value = yes
            }
        }
    }
}
