vladwyn_upgrade_martial_education_effect = {
	if = {
		limit = {
			has_trait = education_martial
			has_education_rank_5_trigger = no
		}
		change_trait_rank = {
			trait = education_martial
			rank = 1
		}
	}
}

vladwyn_upgrade_fighter_trait_effect = {
	if = {
		limit = {
			NOT = { has_trait = lifestyle_blademaster }
		}

		add_trait = lifestyle_blademaster
	}
	else = {
		add_trait_xp = {
			trait = lifestyle_blademaster
			value = 20
		}
	}
}

vladwyn_followers_gain_knight_opinion_effect = {
	every_courtier = {
		custom = every_follower_custom
		add_opinion = {
			target = root
			modifier = vladwyn_knight_amongst_men_opinion
		}
	}
}

vladwyn_greater_ambition_stress_effect = {
	add_stress = 20
	if = {
		limit = { has_trait = content }
		add_stress = 20
	}
	if = {
		limit = { has_trait = humble }
		add_stress = 15
	}
	if = {
		limit = { has_trait = just }
		add_stress = 15
	}
	if = {
		limit = { has_trait = honest }
		add_stress = 15
	}
}
