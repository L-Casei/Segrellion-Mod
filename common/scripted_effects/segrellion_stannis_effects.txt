# =========================================================
# 300 AC - JON SNOW RESURRECTION STORY
#
# Jon is NO LONGER protected from murder schemes.
#
# Organic AGOT murder schemes are allowed to kill him.
#
# If they fail to do so, a delayed canonical mutiny acts
# as a fallback while Jon remains Lord Commander.
# =========================================================

segrellion_setup_jon_resurrection_story_effect = {

    if = {

        limit = {

            game_start_date = 8300.8.20

            character:Stark_7 ?= {
                is_alive = yes
            }
        }


        # =================================================
        # ENABLE THE STORY
        # =================================================

        set_global_variable = {
            name = segrellion_jon_resurrection_story_enabled
            value = yes
        }


        character:Stark_7 = {

            # ---------------------------------------------
            # REMOVE OLD STORY PROTECTION
            # ---------------------------------------------

            if = {

                limit = {
                    has_character_modifier = segrellion_jon_northern_story_protection_modifier
                }

                remove_character_modifier = segrellion_jon_northern_story_protection_modifier
            }


            remove_character_flag = segrellion_jon_northern_story_protected


            # ---------------------------------------------
            # FALLBACK MUTINY
            #
            # AGOT's normal murder schemes get several
            # months to resolve naturally first.
            #
            # If Jon is still alive and Lord Commander,
            # the canonical mutiny begins.
            # ---------------------------------------------

            if = {

                limit = {

                    NOT = {
                        has_character_flag = segrellion_jon_mutiny_fallback_scheduled
                    }
                }


                add_character_flag = segrellion_jon_mutiny_fallback_scheduled


                trigger_event = {

                    id = segrellion_jon_resurrection.0001

                    days = { 240 360 }
                }
            }
        }
    }
}


# =========================================================
# 300 AC - RAMSAY TAKES THE NORTH
#
# ZERO-TICK SETUP.
#
# This is deliberately separated from Stannis and Jon.
# Ramsay's takeover MUST happen regardless of whether any
# later story event can fire.
#
# We use the current Lord Paramount's primary heir as Ramsay,
# then explicitly ensure the heir receives:
#
# - Winterfell county
# - Winterfell duchy
# - The North
#
# This makes the Northern Campaign deterministic.
# =========================================================

segrellion_setup_ramsay_takeover_effect = {

	if = {

		limit = {

			game_start_date = 8300.8.20

			NOT = {
				has_global_variable = segrellion_ramsay_takeover_completed
			}

			title:e_the_north = {
				exists = holder
			}

			title:e_the_north.holder = {

				is_alive = yes

				primary_heir ?= {
					is_alive = yes
				}
			}
		}


		# =================================================
		# SAVE ROOSE AND RAMSAY BEFORE THE DEATH
		# =================================================

		title:e_the_north.holder = {

			save_scope_as = segrellion_roose_bolton


			primary_heir = {

				save_scope_as = segrellion_ramsay_bolton

				add_character_flag = segrellion_ramsay_northern_takeover
			}
		}


		# =================================================
		# RAMSAY MURDERS ROOSE
		#
		# This happens NOW, during game-start setup.
		# No event. No delayed pulse. No day advancement.
		# =================================================

		scope:segrellion_roose_bolton = {

			death = {

				death_reason = death_murder

				killer = scope:segrellion_ramsay_bolton

				secret = no
			}
		}


		# =================================================
		# FORCE THE POLITICAL RESULT
		#
		# Normal inheritance should already do most of this,
		# but the storyline must not depend on AGOT succession
		# quirks.
		# =================================================

		if = {

			limit = {

				title:c_winterfell.holder != scope:segrellion_ramsay_bolton
			}

			title:c_winterfell = {

				grant_title = scope:segrellion_ramsay_bolton
			}
		}


		if = {

			limit = {

				title:d_winterfell.holder != scope:segrellion_ramsay_bolton
			}

			title:d_winterfell = {

				grant_title = scope:segrellion_ramsay_bolton
			}
		}


		if = {

			limit = {

				title:e_the_north.holder != scope:segrellion_ramsay_bolton
			}

			title:e_the_north = {

				grant_title = scope:segrellion_ramsay_bolton
			}
		}


		scope:segrellion_ramsay_bolton = {

			add_character_flag = segrellion_ramsay_lord_of_winterfell
			add_character_flag = segrellion_ramsay_lord_paramount_north
		}


		set_global_variable = {

			name = segrellion_ramsay_takeover_completed

			value = yes
		}
	}
}

# =========================================================
# 300 AC - STANNIS NORTHERN CAMPAIGN
#
# Ramsay's takeover has already been resolved separately.
#
# Stannis remains landed in Dragonstone but is personally
# located at Castle Black.
# =========================================================

segrellion_setup_stannis_northern_campaign_effect = {

	if = {

		limit = {

			game_start_date = 8300.8.20

			has_global_variable = segrellion_ramsay_takeover_completed


			character:Baratheon_3 ?= {

				is_alive = yes

				NOT = {
					has_character_flag = segrellion_stannis_north_story_initialized
				}
			}

			title:e_the_north = {
				exists = holder
			}
		}


		# =================================================
		# STANNIS AT CASTLE BLACK
		# =================================================

		character:Baratheon_3 = {

			add_character_flag = segrellion_stannis_north_story_initialized
			add_character_flag = segrellion_stannis_preparing_northern_campaign


			set_location = title:c_castle_black.title_province

			# =================================================
			# STANNIS' ROYAL HOUSEHOLD
			#
			# Davos is Lord of the Rainwood and must remain
			# directly sworn to Stannis rather than being
			# absorbed into Young Griff's Stormlands.
			#
			# A duke cannot be the vassal of another duke.
			# Therefore, if Stannis has somehow been reduced
			# to duchy-tier, restore his kingdom-tier
			# Dragonstone title first.
			# =================================================

			if = {

				limit = {

					highest_held_title_tier <= tier_duchy
				}

				title:k_dragonstone = {

					grant_title = character:Baratheon_3
				}

				set_primary_title_to = title:k_dragonstone
			}


			# =================================================
			# DAVOS SEAWORTH
			# LORD OF THE DRENCH
			#
			# Davos holds the newly-created Duchy of
			# The Drench, centred on Rainshore.
			#
			# He must remain directly sworn to Stannis,
			# regardless of the wider Stormlands setup.
			# =================================================

			title:d_the_drench.holder ?= {

				if = {

					limit = {

						is_alive = yes

						this != character:Baratheon_3

						NOT = {
							liege ?= character:Baratheon_3
						}
					}

					save_scope_as = segrellion_stannis_davos
				}
			}


			if = {

				limit = {

					exists = scope:segrellion_stannis_davos
				}


				create_title_and_vassal_change = {

					type = swear_fealty

					save_scope_as = segrellion_davos_vassal_change

					add_claim_on_loss = no
				}


				scope:segrellion_stannis_davos = {

					change_liege = {

						liege = character:Baratheon_3

						change = scope:segrellion_davos_vassal_change
					}
				}


				resolve_title_and_vassal_change = scope:segrellion_davos_vassal_change
			}


			if = {

				limit = {

					is_ai = yes

					NOT = {
						has_character_modifier = segrellion_stannis_northern_story_ai_modifier
					}
				}


				add_character_modifier = {

					modifier = segrellion_stannis_northern_story_ai_modifier
				}
			}


			# =================================================
			# STANNIS STARTUP ROUTER
			#
			# Preserve the original, proven startup order: first let CK3
			# finish the zero-tick title and court recalculations, then let
			# the hidden router launch the visible event chain.
			# =================================================

			trigger_event = {

				id = segrellion_stannis.0090

				days = 1
			}
		}
	}
}

# =========================================================
# JON ACCEPTS STANNIS' OFFER
#
# FINAL ORDER:
#
# 1. Find the next Lord Commander.
# 2. Transfer the ENTIRE Night's Watch away from Jon.
# 3. Verify Jon is completely unlanded.
# 4. Only then release Jon from his vows.
# 5. Legitimize Jon.
# 6. Save the Northern vassals.
# 7. Transfer Winterfell and the North to Jon.
# 8. Transfer ONLY Northern vassals to Jon.
# 9. Make Jon independent.
# 10. Ally Jon and Stannis.
#
# IMPORTANT:
# Davos must NOT follow Jon into the North.
# =========================================================

segrellion_jon_accepts_north_effect = {


	# =====================================================
	# FIND THE NEXT LORD COMMANDER
	# =====================================================

	root = {

		remove_character_flag = segrellion_jon_north_transfer_failed


		# -------------------------------------------------
		# FIRST CHOICE:
		# A landed Black Brother under Jon.
		# -------------------------------------------------

		ordered_vassal_or_below = {

			limit = {

				is_alive = yes
				is_adult = yes

				has_trait = nightswatch

				this != root
			}

			order_by = prowess

			position = 0

			check_range_bounds = no

			save_scope_as = segrellion_new_lord_commander
		}


		# -------------------------------------------------
		# FALLBACK:
		# An unlanded Black Brother in Jon's court.
		# -------------------------------------------------

		if = {

			limit = {

				NOT = {
					exists = scope:segrellion_new_lord_commander
				}
			}


			ordered_courtier = {

				limit = {

					is_alive = yes
					is_adult = yes

					has_trait = nightswatch

					this != root
				}

				order_by = prowess

				position = 0

				check_range_bounds = no

				save_scope_as = segrellion_new_lord_commander
			}
		}


		# -------------------------------------------------
		# SAVE EVERY TITLE JON CURRENTLY HOLDS.
		#
		# At this point Jon has NOT received Winterfell,
		# therefore everything here belongs to the Watch.
		# -------------------------------------------------

		if = {

			limit = {

				exists = scope:segrellion_new_lord_commander
			}


			every_held_title = {

				add_to_list = segrellion_jon_watch_titles
			}
		}
	}


	# =====================================================
	# ONLY CONTINUE IF WE FOUND A NEW LORD COMMANDER
	# =====================================================

	if = {

		limit = {

			exists = scope:segrellion_new_lord_commander
		}


		# =================================================
		# TRANSFER THE NIGHT'S WATCH
		#
		# k_the_wall goes first and transfers its vassals.
		#
		# government_base = Jon is deliberate:
		# Jon is STILL Lord Commander and STILL Night's
		# Watch at this exact moment, so the successor
		# inherits the proper Watch government.
		# =================================================

		create_title_and_vassal_change = {

			type = granted

			save_scope_as = segrellion_nights_watch_transfer

			add_claim_on_loss = no
		}


		title:k_the_wall = {

			change_title_holder_include_vassals = {

				holder = scope:segrellion_new_lord_commander

				change = scope:segrellion_nights_watch_transfer

				government_base = root
			}
		}


		# =================================================
		# TRANSFER EVERY OTHER WATCH TITLE JON HELD
		# =================================================

		every_in_list = {

			list = segrellion_jon_watch_titles


			limit = {

				this != title:k_the_wall
			}


			change_title_holder = {

				holder = scope:segrellion_new_lord_commander

				change = scope:segrellion_nights_watch_transfer
			}
		}


		resolve_title_and_vassal_change = scope:segrellion_nights_watch_transfer


		# =================================================
		# CASTLE BLACK REMAINS THE WATCH CAPITAL
		# =================================================

		scope:segrellion_new_lord_commander = {

			set_realm_capital = title:c_castle_black
		}
	}


	# =====================================================
	# ABSOLUTE SAFETY GATE
	#
	# DO NOT REMOVE NIGHTSWATCH UNLESS JON OWNS NOTHING.
	#
	# This is the bit that prevents the Wall becoming
	# fucking feudal again.
	# =====================================================

	if = {

		limit = {

			OR = {

				exists = scope:segrellion_new_lord_commander


				# Resurrected Jon is already free of the Watch:
				# his death ended the vows and succession has
				# already installed a new Lord Commander.

				AND = {

					NOT = {
						has_trait = nightswatch
					}

					is_landed = no

					NOT = {
						has_title = title:k_the_wall
					}

					NOT = {
						has_title = title:d_the_wall
					}

					NOT = {
						has_title = title:c_castle_black
					}
				}
			}


			root = {

				is_landed = no

				NOT = {
					has_title = title:k_the_wall
				}

				NOT = {
					has_title = title:d_the_wall
				}

				NOT = {
					has_title = title:c_castle_black
				}
			}
		}


		# =================================================
		# JON LEAVES THE NIGHT'S WATCH
		# =================================================

		root = {


			if = {

				limit = {

					has_character_modifier = segrellion_jon_northern_story_protection_modifier
				}

				remove_character_modifier = segrellion_jon_northern_story_protection_modifier
			}


			remove_character_flag = segrellion_jon_northern_story_protected


			if = {

				limit = {

					has_trait = nightswatch
				}

				remove_trait = nightswatch
			}


			# =============================================
			# LEGITIMIZE JON
			# =============================================

			if = {

				limit = {

					has_trait = bastard
				}

				remove_trait = bastard

				add_trait = legitimized_bastard
			}


			add_character_flag = segrellion_jon_left_nights_watch
		}


		# =================================================
		# SAVE WHO CURRENTLY HOLDS THE NORTH
		#
		# Normally this should now be Stannis because he
		# took temporary custody after defeating Ramsay.
		#
		# Saving the current holder makes this resistant
		# to weird save-state situations.
		# =================================================

		title:e_the_north.holder ?= {

			save_scope_as = segrellion_north_holder_before_jon
		}


		# =================================================
		# SAVE ONLY NORTHERN VASSALS
		#
		# This is CRITICAL.
		#
		# Stannis may also have Davos as his direct vassal.
		# Davos holds d_the_drench, which is NOT part of
		# e_the_north, so he is deliberately excluded.
		# =================================================

		scope:segrellion_north_holder_before_jon ?= {

			every_vassal = {

				limit = {

					primary_title = {

						empire ?= title:e_the_north
					}
				}

				add_to_list = segrellion_northern_vassals_for_jon
			}
		}


		# =================================================
		# GIVE THE NORTH TO JON
		#
		# DO NOT use include_vassals here.
		#
		# We transfer the titles explicitly and move only
		# the Northern vassals afterwards.
		# =================================================

		create_title_and_vassal_change = {

			type = granted

			save_scope_as = segrellion_jon_north_restoration_change

			add_claim_on_loss = no
		}


		# -------------------------------------------------
		# THE NORTH
		# -------------------------------------------------

		title:e_the_north = {

			change_title_holder = {

				holder = root

				change = scope:segrellion_jon_north_restoration_change

				# Jon has just left the Night's Watch.
				# His new landed government is based on
				# Stannis, NOT on his old Watch government.

				government_base = character:Baratheon_3
			}
		}


		# -------------------------------------------------
		# WINTERFELL KINGDOM-TIER TITLE
		# -------------------------------------------------

		title:k_winterfell = {

			change_title_holder = {

				holder = root

				change = scope:segrellion_jon_north_restoration_change
			}
		}


		# -------------------------------------------------
		# WINTERFELL DUCHY
		# -------------------------------------------------

		title:d_winterfell = {

			change_title_holder = {

				holder = root

				change = scope:segrellion_jon_north_restoration_change
			}
		}


		# -------------------------------------------------
		# WINTERFELL COUNTY
		# -------------------------------------------------

		title:c_winterfell = {

			change_title_holder = {

				holder = root

				change = scope:segrellion_jon_north_restoration_change
			}
		}


		resolve_title_and_vassal_change = scope:segrellion_jon_north_restoration_change


		# =================================================
		# MOVE THE NORTHERN LORDS TO JON
		#
		# Manderly, Glover, Umber, Reed, etc.
		#
		# DAVOS IS NOT IN THIS LIST.
		# =================================================

		if = {

			limit = {

				any_in_list = {

					list = segrellion_northern_vassals_for_jon

					always = yes
				}
			}


			create_title_and_vassal_change = {

				type = granted

				save_scope_as = segrellion_northern_vassals_transfer

				add_claim_on_loss = no
			}


			every_in_list = {

				list = segrellion_northern_vassals_for_jon


				change_liege = {

					liege = root

					change = scope:segrellion_northern_vassals_transfer
				}
			}


			resolve_title_and_vassal_change = scope:segrellion_northern_vassals_transfer
		}


		# =================================================
		# JON MUST BE INDEPENDENT
		#
		# No Tommen.
		# No Young Griff.
		# No Iron Throne.
		# =================================================

		if = {

			limit = {

				root = {

					NOT = {
						is_independent_ruler = yes
					}
				}
			}


			root = {


				create_title_and_vassal_change = {

					type = independency

					save_scope_as = segrellion_jon_independence_change

					add_claim_on_loss = no
				}


				becomes_independent = {

					change = scope:segrellion_jon_independence_change
				}


				resolve_title_and_vassal_change = scope:segrellion_jon_independence_change
			}
		}


		# =================================================
		# JON STARK
		# LORD OF WINTERFELL
		# LORD PARAMOUNT / FUTURE KING IN THE NORTH
		# =================================================

		root = {

			set_primary_title_to = title:e_the_north

			set_realm_capital = title:c_winterfell

			add_character_flag = segrellion_jon_restored_to_winterfell

			add_character_flag = segrellion_jon_lord_paramount_north


			# Requires:
			# nick_segrellion_the_white_wolf = { }
			#
			# in common/nicknames/segrellion_nicknames.txt

			give_nickname = nick_segrellion_the_white_wolf
		}


		# =================================================
		# STANNIS NO LONGER HOLDS THE NORTH IN TRUST
		# =================================================

		character:Baratheon_3 = {

			remove_character_flag = segrellion_stannis_holds_north_in_trust

			remove_character_flag = segrellion_stannis_searching_northern_heir


			add_character_flag = segrellion_stannis_restored_stark_north

			add_character_flag = segrellion_stannis_north_secured


			# =============================================
			# STANNIS + JON ALLIANCE
			# =============================================

			create_alliance = {

				target = root

				allied_through_owner = character:Baratheon_3

				allied_through_target = root
			}


			add_opinion = {

				target = root

				modifier = segrellion_stannis_restored_jon_opinion
			}
		}


		root = {

			add_opinion = {

				target = character:Baratheon_3

				modifier = segrellion_jon_grateful_to_stannis_opinion
			}
			add_character_flag = segrellion_jon_pledged_to_stannis_iron_throne
		}
	}

	segrellion_transfer_stannis_prisoners_to_jon_effect = yes


	# Ramsay's reckoning.
	title:e_the_north.holder ?= {

		if = {

			limit = {

				this = root


				any_prisoner = {

					has_character_flag = segrellion_ramsay_northern_takeover
				}
			}


			trigger_event = {

				id = segrellion_stannis.1030

				days = { 5 10 }
			}
		}
	}


	# =====================================================
	# FAILURE
	#
	# If the Watch transfer failed, STOP.
	#
	# Jon remains:
	# - Lord Commander
	# - Night's Watch
	# - owner of whatever failed to transfer
	#
	# He does NOT receive Winterfell or the North.
	# =====================================================

	else = {

		root = {

			add_character_flag = segrellion_jon_north_transfer_failed
		}
	}

	# =========================================================
	# SNAPSHOT STANNIS' HOUSEHOLD
	#
	# Called immediately before Stannis may lose Dragonstone.
	#
	# We preserve:
	#
	# - his courtiers;
	# - his wife;
	# - his living children;
	#
	# Melisandre requires no hardcoded ID if she is already
	# part of Stannis' court.
	# =========================================================

	segrellion_snapshot_stannis_retinue_effect = {


		character:Baratheon_3 ?= {


			# =================================================
			# COURTIERS
			# =================================================

			every_courtier = {

				limit = {

					is_alive = yes
				}


				add_to_list = segrellion_stannis_retinue
			}


			# =================================================
			# WIFE
			# =================================================

			every_spouse = {

				limit = {

					is_alive = yes
				}


				add_to_list = segrellion_stannis_family
			}


			# =================================================
			# CHILDREN
			# =================================================

			every_child = {

				limit = {

					is_alive = yes
				}


				add_to_list = segrellion_stannis_family
			}
		}
	}
}

# =========================================================
# STANNIS SECURES THE NORTH
#
# Executed immediately after victory over Ramsay.
#
# Stannis does NOT intend to keep the North permanently.
# He takes custody of it so Bolton rule ends immediately
# and so the North cannot return to Tommen / Young Griff.
# =========================================================

segrellion_stannis_secure_north_after_victory_effect = {


	# =====================================================
	# SAVE THE DEFEATED BOLTON
	# =====================================================

	title:e_the_north.holder ?= {

		save_scope_as = segrellion_defeated_bolton_lord
	}


	# =====================================================
	# TRANSFER THE NORTHERN REALM TO STANNIS
	#
	# Ramsay only rules the Northern realm, so transferring
	# his vassals here is safe.
	# =====================================================

	create_title_and_vassal_change = {

			change = scope:segrellion_stannis_north_conquest_change
		}
	}

	# =====================================================
	# THE NORTH LEAVES THE IRON THRONE

			}
		}

	}

		remove_character_flag = segrellion_ramsay_lord_of_winterfell
		remove_character_flag = segrellion_ramsay_lord_paramount_north

# =========================================================
# STANNIS BECOMES LANDLESS
#
# This effect ONLY executes when Stannis genuinely has
# no landed titles left.
#
# CASE A:
#
# Jon accepted the North
# +
# Daenerys takes Dragonstone
#
# -> Stannis becomes landless immediately.
#
#
# CASE B:
#
# Jon refused
# +
# Stannis still controls the North
# +
# Daenerys takes Dragonstone
#
# -> Nothing happens yet because Stannis remains landed.
#
# Once Stannis eventually gives the North to Sansa,
# Rickon or another Stark, call this effect again.
# =========================================================

segrellion_stannis_try_become_landless_effect = {


	if = {

		limit = {


			character:Baratheon_3 ?= {

				is_alive = yes

				is_landed = no


				NOT = {

					has_government = landless_adventurer_government
				}
			}
		}


		# =================================================
		# CREATE STANNIS' LANDLESS HOST
		# =================================================

		character:Baratheon_3 = {

			create_landless_adventurer_title_effect = {

				REASON = flag:voluntary

				FLAVOR_CHAR = character:Baratheon_3
			}


			add_character_flag = segrellion_stannis_became_landless

			add_character_flag = segrellion_stannis_claimant_host_active
		}


		# =================================================
		# HIS COURTIERS FOLLOW HIM
		# =================================================

		every_in_list = {

			list = segrellion_stannis_retinue


			limit = {

				is_alive = yes

				is_landed = no
			}


			set_employer = character:Baratheon_3
		}


		# =================================================
		# HIS FAMILY FOLLOWS HIM
		# =================================================

		every_in_list = {

			list = segrellion_stannis_family


			limit = {

				is_alive = yes

				is_landed = no
			}


			set_employer = character:Baratheon_3
		}


		# =================================================
		# DAVOS SEAWORTH
		#
		# Davos relinquishes:
		#
		# d_the_drench
		# c_rainshore
		#
		# to his primary heir.
		#
		# He then joins Stannis personally.
		# =================================================

		title:d_the_drench.holder ?= {

			save_scope_as = segrellion_stannis_davos


			primary_heir ?= {

				save_scope_as = segrellion_davos_heir
			}
		}


		if = {

			limit = {

				exists = scope:segrellion_stannis_davos

				exists = scope:segrellion_davos_heir


				scope:segrellion_stannis_davos = {

					is_alive = yes
				}


				scope:segrellion_davos_heir = {

					is_alive = yes
				}
			}


			create_title_and_vassal_change = {

				type = granted

				save_scope_as = segrellion_davos_abdication_change

				add_claim_on_loss = no
			}


			# =============================================
			# THE DRENCH
			# =============================================

			title:d_the_drench = {

				change_title_holder_include_vassals = {

					holder = scope:segrellion_davos_heir

					change = scope:segrellion_davos_abdication_change
				}
			}


			# =============================================
			# RAINSHORE
			# =============================================

			title:c_rainshore = {

				change_title_holder = {

					holder = scope:segrellion_davos_heir

					change = scope:segrellion_davos_abdication_change
				}
			}


			resolve_title_and_vassal_change = scope:segrellion_davos_abdication_change


			# =============================================
			# DAVOS JOINS STANNIS
			# =============================================

			scope:segrellion_stannis_davos = {

				set_employer = character:Baratheon_3

				add_character_flag = segrellion_davos_joined_stannis_last_campaign
			}
		}
	}
}

# =========================================================
# STANNIS -> JON
# TRANSFER THE NORTHERN CAMPAIGN PRISONERS
#
# User-facing rule:
# All current prisoners of Stannis pass into Jon's custody
# when Jon assumes rule of the North.
# =========================================================

segrellion_transfer_stannis_prisoners_to_jon_effect = {

	

	character:Baratheon_3 = {

		every_prisoner = {

			limit = {
				is_alive = yes
			}

			add_to_list = segrellion_stannis_prisoners_for_jon
		}
	}


	every_in_list = {

		list = segrellion_stannis_prisoners_for_jon


		add_character_flag = {
			flag = block_imprisonment_event
			days = 2
		}


		release_from_prison = yes


		root = {

			imprison = {

				target = prev

				type = dungeon
			}
		}
	}
}

# =========================================================
# STANNIS - MARCH FOR THE IRON THRONE
# =========================================================

segrellion_start_stannis_iron_throne_war_effect = {

	if = {

		limit = {

			this = character:Baratheon_3

			is_alive = yes


			NOT = {
				has_character_flag = segrellion_stannis_iron_throne_war_started
			}


			title:h_the_iron_throne = {

				exists = holder

				holder = {

					this != character:Baratheon_3
				}
			}
		}


		# =================================================
		# CLAIM
		# =================================================

		if = {

			limit = {

				NOT = {
					has_claim_on = title:h_the_iron_throne
				}
			}


			add_pressed_claim = title:h_the_iron_throne
		}


		# =================================================
		# SAVE DEFENDER
		# =================================================

		title:h_the_iron_throne.holder = {

			save_scope_as = segrellion_stannis_throne_defender
		}


		add_character_flag = segrellion_stannis_iron_throne_war_started


		# =================================================
		# WAR
		# =================================================

		start_war = {

			cb = claim_cb

			target = scope:segrellion_stannis_throne_defender

			claimant = character:Baratheon_3

			target_title = title:h_the_iron_throne
		}
	}
}
