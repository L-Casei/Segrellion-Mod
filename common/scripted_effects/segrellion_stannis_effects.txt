# =========================================================
# 300 AC - JON SNOW STORY PROTECTION
#
# Jon is essential to the scripted Northern Campaign.
#
# The modifier:
# - makes murder schemes effectively impossible to complete;
# - makes hostile schemes against him extremely difficult;
# - protects him from an unlucky natural death during the
#   opening chapter.
#
# It is TEMPORARY and is removed when Jon is restored to
# Winterfell.
# =========================================================

segrellion_setup_jon_story_protection_effect = {

	if = {

		limit = {

			game_start_date = 8300.8.20

			character:Stark_7 ?= {

				is_alive = yes

				NOT = {
					has_character_modifier = segrellion_jon_northern_story_protection_modifier
				}
			}
		}


		character:Stark_7 = {

			add_character_flag = segrellion_jon_northern_story_protected

			add_character_modifier = {

				modifier = segrellion_jon_northern_story_protection_modifier

				years = 5
			}
		}
	}
}


# =========================================================
# 300 AC - RAMSAY TAKES THE NORTH
#
# ZERO-TICK SETUP.
#
# This is deliberately separated from Stannis and Jon.
# Ramsay's takeover MUST happen regardless of whether any
# later story event can fire.
#
# We use the current Lord Paramount's primary heir as Ramsay,
# then explicitly ensure the heir receives:
#
# - Winterfell county
# - Winterfell duchy
# - The North
#
# This makes the Northern Campaign deterministic.
# =========================================================

segrellion_setup_ramsay_takeover_effect = {

	if = {

		limit = {

			game_start_date = 8300.8.20

			NOT = {
				has_global_variable = segrellion_ramsay_takeover_completed
			}

			title:e_the_north = {
				exists = holder
			}

			title:e_the_north.holder = {

				is_alive = yes

				primary_heir ?= {
					is_alive = yes
				}
			}
		}


		# =================================================
		# SAVE ROOSE AND RAMSAY BEFORE THE DEATH
		# =================================================

		title:e_the_north.holder = {

			save_scope_as = segrellion_roose_bolton


			primary_heir = {

				save_scope_as = segrellion_ramsay_bolton

				add_character_flag = segrellion_ramsay_northern_takeover
			}
		}


		# =================================================
		# RAMSAY MURDERS ROOSE
		#
		# This happens NOW, during game-start setup.
		# No event. No delayed pulse. No day advancement.
		# =================================================

		scope:segrellion_roose_bolton = {

			death = {

				death_reason = death_murder

				killer = scope:segrellion_ramsay_bolton

				secret = no
			}
		}


		# =================================================
		# FORCE THE POLITICAL RESULT
		#
		# Normal inheritance should already do most of this,
		# but the storyline must not depend on AGOT succession
		# quirks.
		# =================================================

		if = {

			limit = {

				title:c_winterfell.holder != scope:segrellion_ramsay_bolton
			}

			title:c_winterfell = {

				grant_title = scope:segrellion_ramsay_bolton
			}
		}


		if = {

			limit = {

				title:d_winterfell.holder != scope:segrellion_ramsay_bolton
			}

			title:d_winterfell = {

				grant_title = scope:segrellion_ramsay_bolton
			}
		}


		if = {

			limit = {

				title:e_the_north.holder != scope:segrellion_ramsay_bolton
			}

			title:e_the_north = {

				grant_title = scope:segrellion_ramsay_bolton
			}
		}


		scope:segrellion_ramsay_bolton = {

			add_character_flag = segrellion_ramsay_lord_of_winterfell
			add_character_flag = segrellion_ramsay_lord_paramount_north
		}


		set_global_variable = {

			name = segrellion_ramsay_takeover_completed

			value = yes
		}
	}
}


# =========================================================
# 300 AC - STANNIS NORTHERN CAMPAIGN
#
# Ramsay's takeover has already been resolved separately.
#
# Stannis remains landed in Dragonstone but is personally
# located at Castle Black.
# =========================================================

segrellion_setup_stannis_northern_campaign_effect = {

	if = {

		limit = {

			game_start_date = 8300.8.20

			has_global_variable = segrellion_ramsay_takeover_completed


			character:Baratheon_3 ?= {

				is_alive = yes

				NOT = {
					has_character_flag = segrellion_stannis_north_story_initialized
				}
			}


			character:Stark_7 ?= {
				is_alive = yes
			}


			title:e_the_north = {
				exists = holder
			}
		}


		# =================================================
		# STANNIS AT CASTLE BLACK
		# =================================================

		character:Baratheon_3 = {

			add_character_flag = segrellion_stannis_north_story_initialized
			add_character_flag = segrellion_stannis_preparing_northern_campaign


			set_location = title:c_castle_black.title_province

			# =================================================
			# STANNIS' ROYAL HOUSEHOLD
			#
			# Davos is Lord of the Rainwood and must remain
			# directly sworn to Stannis rather than being
			# absorbed into Young Griff's Stormlands.
			#
			# A duke cannot be the vassal of another duke.
			# Therefore, if Stannis has somehow been reduced
			# to duchy-tier, restore his kingdom-tier
			# Dragonstone title first.
			# =================================================

			if = {

				limit = {

					highest_held_title_tier <= tier_duchy
				}

				title:k_dragonstone = {

					grant_title = character:Baratheon_3
				}

				set_primary_title_to = title:k_dragonstone
			}


			# =================================================
			# DAVOS SEAWORTH
			# LORD OF THE DRENCH
			#
			# Davos holds the newly-created Duchy of
			# The Drench, centred on Rainshore.
			#
			# He must remain directly sworn to Stannis,
			# regardless of the wider Stormlands setup.
			# =================================================

			title:d_the_drench.holder ?= {

				if = {

					limit = {

						is_alive = yes

						this != character:Baratheon_3

						NOT = {
							liege ?= character:Baratheon_3
						}
					}

					save_scope_as = segrellion_stannis_davos
				}
			}


			if = {

				limit = {

					exists = scope:segrellion_stannis_davos
				}


				create_title_and_vassal_change = {

					type = swear_fealty

					save_scope_as = segrellion_davos_vassal_change

					add_claim_on_loss = no
				}


				scope:segrellion_stannis_davos = {

					change_liege = {

						liege = character:Baratheon_3

						change = scope:segrellion_davos_vassal_change
					}
				}


				resolve_title_and_vassal_change = scope:segrellion_davos_vassal_change
			}


			if = {

				limit = {

					is_ai = yes

					NOT = {
						has_character_modifier = segrellion_stannis_northern_story_ai_modifier
					}
				}


				add_character_modifier = {

					modifier = segrellion_stannis_northern_story_ai_modifier
				}
			}


			# One day is deliberately left here.
			#
			# Roose is ALREADY dead and Ramsay is ALREADY
			# Lord of the North. This merely lets CK3 finish
			# any zero-tick title / court recalculations before
			# the visible Stannis storyline begins.

			trigger_event = {

				id = segrellion_stannis.0090

				days = 1
			}
		}
	}
}

# =========================================================
# JON ACCEPTS STANNIS' OFFER
#
# CRITICAL ORDER:
#
# 1. Find a valid new Lord Commander.
# 2. Transfer EVERY title Jon currently holds.
# 3. Confirm Jon is completely unlanded.
# 4. Only then remove Night's Watch status.
# 5. Legitimize Jon.
# 6. Strip the North from Ramsay.
# 7. Give Winterfell and the North to Jon.
#
# Under no circumstances do we remove the Night's Watch
# trait while Jon still owns Wall titles.
# =========================================================

segrellion_jon_accepts_north_effect = {


	# =====================================================
	# FIND THE NEXT LORD COMMANDER
	#
	# First search Jon's landed Night's Watch vassals.
	# The old version only searched COURTIERS, which could
	# leave the scope empty and destroy the Wall government.
	# =====================================================

	character:Stark_7 = {

		remove_character_flag = segrellion_jon_north_transfer_failed


		ordered_vassal_or_below = {

			limit = {

				is_alive = yes
				is_adult = yes

				has_trait = nightswatch

				this != character:Stark_7
			}

			order_by = prowess

			position = 0

			check_range_bounds = no

			save_scope_as = segrellion_new_lord_commander
		}


		# =================================================
		# FALLBACK: UNLANDED BLACK BROTHER
		# =================================================

		if = {

			limit = {

				NOT = {
					exists = scope:segrellion_new_lord_commander
				}
			}


			ordered_courtier = {

				limit = {

					is_alive = yes
					is_adult = yes

					has_trait = nightswatch

					this != character:Stark_7
				}

				order_by = prowess

				position = 0

				check_range_bounds = no

				save_scope_as = segrellion_new_lord_commander
			}
		}


		# =================================================
		# SNAPSHOT ALL TITLES BEFORE TRANSFER
		#
		# At this point Jon has not received Winterfell.
		# Therefore everything he owns belongs to the Watch.
		# =================================================

		if = {

			limit = {

				exists = scope:segrellion_new_lord_commander
			}


			every_held_title = {

				add_to_list = segrellion_jon_watch_titles
			}
		}
	}


	# =====================================================
	# TRANSFER THE ENTIRE NIGHT'S WATCH
	#
	# Do it as ONE title/vassal transaction.
	# =====================================================

	if = {

		limit = {

			exists = scope:segrellion_new_lord_commander
		}


		create_title_and_vassal_change = {

			type = granted

			save_scope_as = segrellion_nights_watch_transfer

			add_claim_on_loss = no
		}


		every_in_list = {

			list = segrellion_jon_watch_titles


			change_title_holder = {

				holder = scope:segrellion_new_lord_commander

				change = scope:segrellion_nights_watch_transfer
			}
		}


		resolve_title_and_vassal_change = scope:segrellion_nights_watch_transfer
	}


	# =====================================================
	# ABSOLUTE SAFETY GATE
	#
	# We DO NOT continue unless Jon has become completely
	# unlanded.
	#
	# If a single Wall holding failed to transfer, Jon keeps
	# his vows and nothing below executes.
	# =====================================================

	if = {

		limit = {

			character:Stark_7 = {

				is_landed = no
			}
		}


		# =================================================
		# JON IS RELEASED FROM THE NIGHT'S WATCH
		# =================================================

		character:Stark_7 = {


			if = {

				limit = {

					has_character_modifier = segrellion_jon_northern_story_protection_modifier
				}

				remove_character_modifier = segrellion_jon_northern_story_protection_modifier
			}


			remove_character_flag = segrellion_jon_northern_story_protected


			if = {

				limit = {

					has_trait = nightswatch
				}

				remove_trait = nightswatch
			}


			# =============================================
			# JON SNOW BECOMES LEGITIMATE
			# =============================================

			if = {

				limit = {

					has_trait = bastard
				}

				remove_trait = bastard

				add_trait = legitimized_bastard
			}


			add_character_flag = segrellion_jon_left_nights_watch
		}


		# =================================================
		# SAVE RAMSAY BEFORE DISPOSSESSING HIM
		# =================================================

		title:e_the_north.holder ?= {

			save_scope_as = segrellion_defeated_bolton_lord
		}


		# =================================================
		# THE STARK RESTORATION
		#
		# This is a REAL title-and-vassal transaction.
		#
		# e_the_north changes holder WITH its vassals.
		# Ramsay therefore cannot remain Lord Paramount.
		# =================================================

		create_title_and_vassal_change = {

			type = granted

			save_scope_as = segrellion_stark_restoration_change

			add_claim_on_loss = no
		}


		# =================================================
		# THE NORTH
		#
		# Transfer the paramountcy and its vassal structure.
		# =================================================

		title:e_the_north = {

			change_title_holder_include_vassals = {

				holder = character:Stark_7

				change = scope:segrellion_stark_restoration_change
			}
		}


		# =================================================
		# WINTERFELL
		# =================================================

		title:k_winterfell = {

			change_title_holder = {

				holder = character:Stark_7

				change = scope:segrellion_stark_restoration_change
			}
		}

		title:d_winterfell = {

			change_title_holder = {

				holder = character:Stark_7

				change = scope:segrellion_stark_restoration_change
			}
		}


		title:c_winterfell = {

			change_title_holder = {

				holder = character:Stark_7

				change = scope:segrellion_stark_restoration_change
			}
		}


		resolve_title_and_vassal_change = scope:segrellion_stark_restoration_change


		# =================================================
		# JON STARK, LORD PARAMOUNT OF THE NORTH
		# =================================================

		character:Stark_7 = {

			set_primary_title_to = title:e_the_north

			set_realm_capital = title:c_winterfell

			add_character_flag = segrellion_jon_restored_to_winterfell
			add_character_flag = segrellion_jon_lord_paramount_north
		}


		# =================================================
		# RAMSAY IS NO LONGER LORD OF THE NORTH
		# =================================================

		scope:segrellion_defeated_bolton_lord ?= {

			remove_character_flag = segrellion_ramsay_lord_of_winterfell
			remove_character_flag = segrellion_ramsay_lord_paramount_north
		}


		# =================================================
		# STANNIS + JON
		# =================================================

		character:Baratheon_3 = {

			add_character_flag = segrellion_stannis_restored_stark_north
			add_character_flag = segrellion_stannis_north_secured


			create_alliance = {

				target = character:Stark_7

				allied_through_owner = character:Baratheon_3
				allied_through_target = character:Stark_7
			}


			add_opinion = {

				target = character:Stark_7

				modifier = segrellion_stannis_restored_jon_opinion
			}
		}


		character:Stark_7 = {

			add_opinion = {

				target = character:Baratheon_3

				modifier = segrellion_jon_grateful_to_stannis_opinion
			}
		}
	}


	# =====================================================
	# FAILSAFE
	#
	# If Jon still owns land, DO NOT touch his government.
	# This flag makes the failure immediately visible in
	# debug instead of silently feudalizing the Wall.
	# =====================================================

	else = {

		character:Stark_7 = {

			add_character_flag = segrellion_jon_north_transfer_failed
		}
	}
}
