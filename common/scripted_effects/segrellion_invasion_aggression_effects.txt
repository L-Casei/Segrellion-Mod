# =========================================================
# YOUNG GRIFF - SCRIPTED WARS ONLY
#
# The AI must not spend Aegon's claims on ordinary CK3/AGOT
# wars. His Flamelands involvement and every Iron Throne
# campaign are created by this mod's events.
#
# The modifier suppresses AI willingness. sworn_peace blocks
# the normal declare-war interaction outright. A separate
# ownership flag ensures we never remove sworn_peace if some
# other system applied it.
# =========================================================

segrellion_young_griff_apply_scripted_war_lock_effect = {

	if = {

		limit = {

			this = character:IAF_Young_Griff_1

			is_alive = yes
			is_ai = yes

			NOT = {
				has_title = title:h_the_iron_throne
			}

			NOT = {
				has_character_flag = segrellion_young_griff_crowned_after_varys_plan
			}

			NOT = {
				has_character_flag = segrellion_young_griff_daenerys_pact_fulfilled
			}
		}


		if = {

			limit = {

				NOT = {
					has_character_modifier = segrellion_young_griff_scripted_wars_only_ai_modifier
				}
			}

			add_character_modifier = {
				modifier = segrellion_young_griff_scripted_wars_only_ai_modifier
			}
		}


		if = {

			limit = {

				NOT = {
					has_character_flag = sworn_peace
				}
			}

			add_character_flag = sworn_peace
			add_character_flag = segrellion_young_griff_scripted_war_lock_owns_sworn_peace
		}


		# Save repair: discard an autonomous offensive war that
		# was already declared before this lock existed. Scripted
		# throne wars are explicitly preserved.

		every_character_war = {

			limit = {

				primary_attacker = root

				NOT = {

					casus_belli = {

						any_target_title = {
							this = title:h_the_iron_throne
						}
					}
				}
			}

			end_war = invalidated
		}
	}


	else = {

		segrellion_young_griff_release_scripted_war_lock_effect = yes
	}
}


# Temporarily opens only the hard interaction lock. The AI
# suppression modifier remains active during the event tick.

segrellion_young_griff_open_scripted_war_window_effect = {

	if = {

		limit = {

			this = character:IAF_Young_Griff_1

			is_ai = yes

			has_character_flag = segrellion_young_griff_scripted_war_lock_owns_sworn_peace
		}

		remove_character_flag = sworn_peace
		remove_character_flag = segrellion_young_griff_scripted_war_lock_owns_sworn_peace
	}
}


segrellion_young_griff_release_scripted_war_lock_effect = {

	if = {

		limit = {
			has_character_modifier = segrellion_young_griff_scripted_wars_only_ai_modifier
		}

		remove_character_modifier = segrellion_young_griff_scripted_wars_only_ai_modifier
	}


	if = {

		limit = {
			has_character_flag = segrellion_young_griff_scripted_war_lock_owns_sworn_peace
		}

		remove_character_flag = sworn_peace
		remove_character_flag = segrellion_young_griff_scripted_war_lock_owns_sworn_peace
	}
}


# =========================================================
# YOUNG GRIFF - INITIALIZATION
#
# IMPORTANT:
# His normal Iron Throne claim is temporarily removed from
# AI control so CK3 cannot bypass our custom decision and
# declare an ordinary claim war early.
# =========================================================

segrellion_initialize_young_griff_aggression_effect = {

	if = {

		limit = {

			NOT = {
				has_character_flag = segrellion_young_griff_aggression_initialized
			}
		}

		add_character_flag = segrellion_young_griff_aggression_initialized
		add_character_flag = segrellion_young_griff_waiting_for_daemon


		# Young Griff cannot bypass his scripted path
		# with a normal claim war.

		if = {

			limit = {
				has_claim_on = title:h_the_iron_throne
			}

			remove_claim = title:h_the_iron_throne
		}
	}


	segrellion_young_griff_apply_scripted_war_lock_effect = yes
}


# =========================================================
# DAENERYS STORY INITIALIZATION
#
# ROOT = Daenerys
#
# The aggression rule selects an AI campaign profile here,
# but it must never delay the beginning of her storyline.
# The actual strategic delay is applied at the fleet decision.
# =========================================================

segrellion_initialize_daenerys_aggression_effect = {

	if = {

		limit = {

			NOT = {
				has_character_flag = segrellion_daenerys_invasion_event_scheduled
			}

			NOT = {
				has_character_flag = segrellion_daenerys_invasion_abandoned
			}
		}


		# =================================================
		# PLAYER
		#
		# Game Rules control AI behavior only.
		# The player always chooses personally.
		# =================================================

		if = {

			limit = {
				is_ai = no
			}

			add_character_flag = segrellion_daenerys_invasion_event_scheduled
			add_character_flag = segrellion_daenerys_story_opening_scheduled

			trigger_event = {
				id = segrellion_daenerys_invasion.0001
				days = { 3 7 }
			}
		}


		# =================================================
		# AI
		# =================================================

		else = {

			add_character_flag = segrellion_daenerys_invasion_event_scheduled
			add_character_flag = segrellion_daenerys_story_opening_scheduled


			# =================================================
			# HARD WAR LOCK
			# =================================================

			if = {

				limit = {

					NOT = {
						has_character_flag = sworn_peace
					}
				}

				add_character_flag = sworn_peace
			}


			if = {

				limit = {

					NOT = {
						has_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
					}
				}

				add_character_modifier = {
					modifier = segrellion_daenerys_preparing_invasion_ai_modifier
				}
			}


			# =================================================
			# REMOVE BYPASS CLAIMS
			# =================================================

			if = {

				limit = {
					has_claim_on = title:h_the_iron_throne
				}

				remove_claim = title:h_the_iron_throne
			}


			if = {

				limit = {
					has_claim_on = title:c_dragonstone
				}

				remove_claim = title:c_dragonstone
			}


			# =================================================
			# DAENERYS' DESTINY
			# =================================================


			# =================================================
			# FREE CITIES
			# =================================================

			if = {

				limit = {
					has_game_rule = segrellion_daenerys_destiny_free_cities
				}

				add_character_flag = segrellion_daenerys_ai_free_cities_route
				add_character_flag = segrellion_daenerys_destiny_free_cities
			}


			# =================================================
			# DYNAMIC
			#
			# Story-weighted:
			# 75% Westeros
			# 25% Free Cities
			# =================================================

			else_if = {

				limit = {
					has_game_rule = segrellion_daenerys_destiny_dynamic
				}

				random_list = {

					75 = {

						add_character_flag = segrellion_daenerys_destiny_westeros
					}

					25 = {

						add_character_flag = segrellion_daenerys_ai_free_cities_route
						add_character_flag = segrellion_daenerys_destiny_free_cities
					}
				}
			}


			# =================================================
			# RANDOM
			#
			# Pure 50 / 50.
			# =================================================

			else_if = {

				limit = {
					has_game_rule = segrellion_daenerys_destiny_random
				}

				random_list = {

					50 = {

						add_character_flag = segrellion_daenerys_destiny_westeros
					}

					50 = {

						add_character_flag = segrellion_daenerys_ai_free_cities_route
						add_character_flag = segrellion_daenerys_destiny_free_cities
					}
				}
			}


			# =================================================
			# WESTEROS / FALLBACK
			# =================================================

			else = {

				add_character_flag = segrellion_daenerys_destiny_westeros
			}


			# =================================================
			# IRON THRONE INVASION TIMING
			#
			# Only relevant if her Destiny is Westeros.
			# =================================================

			if = {

				limit = {

					NOT = {
						has_character_flag = segrellion_daenerys_ai_free_cities_route
					}
				}


				# =============================================
				# PASSIVE
				# 7-12 years before committing to the fleet.
				# =============================================

				if = {

					limit = {

						OR = {

							has_game_rule = segrellion_daenerys_aggression_passive

							has_character_flag = segrellion_daenerys_random_passive
						}
					}

					add_character_flag = segrellion_daenerys_ai_slow_fleet
				}


				# =============================================
				# DEFAULT
				# 2-5 years.
				# =============================================

				else_if = {

					limit = {

						OR = {

							has_game_rule = segrellion_daenerys_aggression_default

							has_character_flag = segrellion_daenerys_random_default
						}
					}

					add_character_flag = segrellion_daenerys_ai_standard_fleet
				}


				# =============================================
				# AGGRESSIVE
				# Immediate crossing.
				# =============================================

				else_if = {

					limit = {

						OR = {

							has_game_rule = segrellion_daenerys_aggression_aggressive

							has_character_flag = segrellion_daenerys_random_aggressive
						}
					}

					add_character_flag = segrellion_daenerys_ai_rush_crossing
				}


				# =============================================
				# FALLBACK = DEFAULT
				# =============================================

				else = {

					add_character_flag = segrellion_daenerys_ai_standard_fleet
				}
			}


			# =================================================
			# OPEN THE STORY
			#
			# Destiny decides which branch AI Daenerys chooses
			# inside .0001.
			# =================================================

			trigger_event = {

				id = segrellion_daenerys_invasion.0001

				days = { 3 7 }
			}
		}
	}
}

# =========================================================
# DAENERYS - SCRIPTED INVASION PEACE GUARD
#
# Prevent AI Daenerys from declaring ordinary wars while
# waiting for the scripted Westerosi invasion.
# =========================================================

segrellion_daenerys_invasion_peace_guard_effect = {

	character:Targaryen_9 ?= {

		# Save-compatible repair for campaigns created while .0001
		# inherited the old 2-5 / 7-12 year aggression delay. The old
		# delayed event may remain in the queue, but it will harmlessly
		# fail its trigger after this accelerated opening is resolved.

		if = {

			limit = {

				is_alive = yes
				is_ai = yes

				has_character_flag = segrellion_daenerys_invasion_event_scheduled

				NOT = {
					has_character_flag = segrellion_daenerys_story_opening_scheduled
				}

				NOR = {
					has_character_flag = segrellion_daenerys_invasion_committed
					has_character_flag = segrellion_daenerys_invasion_delayed_once
					has_character_flag = segrellion_daenerys_invasion_abandoned
					has_character_flag = segrellion_daenerys_special_story_abandoned
					has_character_flag = segrellion_daenerys_free_cities_path
				}
			}


			add_character_flag = segrellion_daenerys_story_opening_scheduled


			trigger_event = {
				id = segrellion_daenerys_invasion.0001
				days = { 3 7 }
			}
		}

		if = {

			limit = {

				is_alive = yes
				is_ai = yes

				has_character_flag = segrellion_daenerys_invasion_event_scheduled

				NOT = {
					has_character_flag = segrellion_daenerys_landed_in_westeros
				}

				NOT = {
					has_character_flag = segrellion_daenerys_invasion_abandoned
				}

				NOT = {
					has_character_flag = segrellion_daenerys_special_story_abandoned
				}

				NOT = {
					has_character_flag = segrellion_daenerys_young_griff_throne_pact
				}
			}


			# =================================================
			# HARD WAR LOCK
			# =================================================

			if = {

				limit = {

					NOT = {
						has_character_flag = sworn_peace
					}
				}

				add_character_flag = sworn_peace
			}


			# =================================================
			# AI WAR DISCOURAGEMENT
			# Secondary protection.
			# =================================================

			if = {

				limit = {

					NOT = {
						has_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
					}
				}

				add_character_modifier = {
					modifier = segrellion_daenerys_preparing_invasion_ai_modifier
				}

				# =================================================
				# KEEP DAENERYS' CAMP AT PENTOS
				#
				# While the AI is preparing the invasion, Pentos is
				# her fixed staging area. If vanilla/AGOT moves the
				# adventurer camp elsewhere, return it to Pentos.
				#
				# Once the scripted crossing begins, stop interfering.
				# =================================================

				if = {

					limit = {

						has_character_flag = segrellion_daenerys_camped_at_pentos

						NOT = {
							has_character_flag = segrellion_daenerys_landing_chosen
						}

						NOT = {
							exists = current_travel_plan
						}

						has_domicile = yes

						domicile = {

							NOT = {
								domicile_location = title:c_pentos.title_province
							}
						}
					}

					domicile = {
						move_domicile = title:c_pentos.title_province
					}
				}
			}


			# =================================================
			# REMOVE BYPASS CLAIMS
			# =================================================

			if = {

				limit = {
					has_claim_on = title:h_the_iron_throne
				}

				remove_claim = title:h_the_iron_throne
			}


			if = {

				limit = {
					has_claim_on = title:c_dragonstone
				}

				remove_claim = title:c_dragonstone
			}
		}


		else = {

			# Story lock is no longer required.

			if = {

				limit = {
					has_character_flag = sworn_peace
				}

				remove_character_flag = sworn_peace
			}


			if = {

				limit = {
					has_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
				}

				remove_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
			}
		}
	}
}

# =========================================================
# RANDOM STORY GAME RULE INITIALIZATION
# =========================================================

segrellion_initialize_random_story_rules_effect = {

	# =====================================================
	# YOUNG GRIFF AGGRESSION
	# =====================================================

	if = {

		limit = {

			has_game_rule = segrellion_young_griff_aggression_random


			character:IAF_Young_Griff_1 ?= {

				NOT = {
					has_character_flag = segrellion_young_griff_random_aggression_resolved
				}
			}
		}


		character:IAF_Young_Griff_1 = {

			add_character_flag = segrellion_young_griff_random_aggression_resolved


			random_list = {

				25 = {
					add_character_flag = segrellion_young_griff_random_peaceful
				}

				25 = {
					add_character_flag = segrellion_young_griff_random_passive
				}

				25 = {
					add_character_flag = segrellion_young_griff_random_default
				}

				25 = {
					add_character_flag = segrellion_young_griff_random_aggressive
				}
			}
		}
	}

	# =====================================================
	# DAENERYS IRON THRONE INVASION TIMING
	# =====================================================

	if = {

		limit = {

			has_game_rule = segrellion_daenerys_aggression_random

			character:Targaryen_9 ?= {

				NOT = {
					has_character_flag = segrellion_daenerys_random_aggression_resolved
				}
			}
		}


		character:Targaryen_9 = {

			add_character_flag = segrellion_daenerys_random_aggression_resolved


			random_list = {

				33 = {
					add_character_flag = segrellion_daenerys_random_passive
				}

				34 = {
					add_character_flag = segrellion_daenerys_random_default
				}

				33 = {
					add_character_flag = segrellion_daenerys_random_aggressive
				}
			}
		}
	}


	# =====================================================
	# DAEMON GENERAL STRATEGY
	# =====================================================

	if = {

		limit = {

			has_game_rule = segrellion_daemon_ai_strategy_random

			character:Segrellion_60 ?= {

				NOT = {
					has_character_flag = segrellion_daemon_random_strategy_resolved
				}
			}
		}


		character:Segrellion_60 = {

			add_character_flag = segrellion_daemon_random_strategy_resolved


			random_list = {

				50 = {
					add_character_flag = segrellion_daemon_random_flamelands_focused
				}

				50 = {
					add_character_flag = segrellion_daemon_random_wandering_flame
				}
			}
		}
	}
}
