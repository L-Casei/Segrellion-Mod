# INVASION AGGRESSION GAME RULES
# =========================================================


# =========================================================
# YOUNG GRIFF
#
# ROOT = Young Griff
#
# IMPORTANT:
# His normal Iron Throne claim is temporarily removed from
# AI control so CK3 cannot bypass our custom decision and
# declare an ordinary claim war early.
# =========================================================

segrellion_initialize_young_griff_aggression_effect = {

	if = {

		limit = {

			NOT = {
				has_character_flag = segrellion_young_griff_aggression_initialized
			}
		}

		add_character_flag = segrellion_young_griff_aggression_initialized
		add_character_flag = segrellion_young_griff_waiting_for_daemon


		# Young Griff cannot bypass his scripted path
		# with a normal claim war.

		if = {

			limit = {
				has_claim_on = title:h_the_iron_throne
			}

			remove_claim = title:h_the_iron_throne
		}
	}
}


# =========================================================
# DAENERYS
#
# ROOT = Daenerys
# =========================================================

segrellion_initialize_daenerys_aggression_effect = {

	if = {

		limit = {

			NOT = {
				has_character_flag = segrellion_daenerys_invasion_event_scheduled
			}

			NOT = {
				has_character_flag = segrellion_daenerys_invasion_abandoned
			}
		}


		# =================================================
		# PLAYER DAENERYS
		#
		# Game Rule controls AI only.
		# Player receives the storyline normally.
		# =================================================

		if = {

			limit = {
				is_ai = no
			}

			add_character_flag = segrellion_daenerys_invasion_event_scheduled

			trigger_event = {
				id = segrellion_daenerys_invasion.0001
				days = { 15 30 }
			}
		}


		# =================================================
		# AI DAENERYS
		# =================================================

		else = {

			add_character_flag = segrellion_daenerys_invasion_event_scheduled


			# ---------------------------------------------
			# REMOVE MECHANICAL WESTEROSI CLAIMS
			#
			# Daenerys must not bypass the scripted
			# invasion by declaring ordinary claim wars
			# from Essos.
			#
			# The Iron Throne claim was already removed,
			# but her Dragonstone claim was still allowing
			# vanilla / AGOT AI to attack Stannis directly
			# from Pentos.
			# ---------------------------------------------

			if = {

				limit = {
					has_claim_on = title:h_the_iron_throne
				}

				remove_claim = title:h_the_iron_throne
			}


			if = {

				limit = {
					has_claim_on = title:c_dragonstone
				}

				remove_claim = title:c_dragonstone
			}



			# =================================================
			# PEACEFUL
			# =================================================

			if = {

				limit = {
					has_game_rule = segrellion_daenerys_aggression_peaceful
				}

				add_character_flag = segrellion_daenerys_ai_invasion_suppressed
			}


			# =================================================
			# PASSIVE
			# 7-12 years.
			# =================================================

			else_if = {

				limit = {
					has_game_rule = segrellion_daenerys_aggression_passive
				}

				trigger_event = {
					id = segrellion_daenerys_invasion.0001
					days = { 2555 4380 }
				}
			}


			# =================================================
			# DEFAULT
			# 2-5 years.
			# =================================================

			else_if = {

				limit = {
					has_game_rule = segrellion_daenerys_aggression_default
				}

				trigger_event = {
					id = segrellion_daenerys_invasion.0001
					days = { 3 7 }
				}
			}


			# =================================================
			# AGGRESSIVE
			# =================================================

			else_if = {

				limit = {
					has_game_rule = segrellion_daenerys_aggression_aggressive
				}

				trigger_event = {
					id = segrellion_daenerys_invasion.0001
					days = { 1 7 }
				}
			}


			# =================================================
			# SAFETY FALLBACK
			# =================================================

			else = {

				trigger_event = {
					id = segrellion_daenerys_invasion.0001
					days = { 730 1825 }
				}
			}
		}
	}
}

# =========================================================
# DAENERYS - SCRIPTED INVASION PEACE GUARD
#
# Save-compatible repair.
#
# While AI Daenerys has committed to Westeros but has not
# physically landed there yet, she must not initiate normal
# AI wars.
# =========================================================

segrellion_daenerys_invasion_peace_guard_effect = {

	character:Targaryen_9 ?= {

		if = {

			limit = {

				is_alive = yes
				is_ai = yes

				has_character_flag = segrellion_daenerys_invasion_committed

				NOT = {
					has_character_flag = segrellion_daenerys_landed_in_westeros
				}

				NOT = {
					has_character_flag = segrellion_daenerys_invasion_abandoned
				}

				NOT = {
					has_character_flag = segrellion_daenerys_special_story_abandoned
				}
			}


			if = {

				limit = {

					NOT = {
						has_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
					}
				}

				add_character_modifier = {
					modifier = segrellion_daenerys_preparing_invasion_ai_modifier
				}
			}


			# ---------------------------------------------
			# Claims that can bypass the scripted route.
			# ---------------------------------------------

			if = {

				limit = {
					has_claim_on = title:h_the_iron_throne
				}

				remove_claim = title:h_the_iron_throne
			}


			if = {

				limit = {
					has_claim_on = title:c_dragonstone
				}

				remove_claim = title:c_dragonstone
			}
		}


		else_if = {

			limit = {

				has_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
			}

			remove_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
		}
	}
}
