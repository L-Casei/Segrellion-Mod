# INVASION AGGRESSION GAME RULES
# =========================================================


# =========================================================
# YOUNG GRIFF
#
# ROOT = Young Griff
#
# IMPORTANT:
# His normal Iron Throne claim is temporarily removed from
# AI control so CK3 cannot bypass our custom decision and
# declare an ordinary claim war early.
# =========================================================

segrellion_initialize_young_griff_aggression_effect = {

	if = {

		limit = {

			NOT = {
				has_character_flag = segrellion_young_griff_aggression_initialized
			}
		}

		add_character_flag = segrellion_young_griff_aggression_initialized
		add_character_flag = segrellion_young_griff_waiting_for_daemon


		# Young Griff cannot bypass his scripted path
		# with a normal claim war.

		if = {

			limit = {
				has_claim_on = title:h_the_iron_throne
			}

			remove_claim = title:h_the_iron_throne
		}
	}
}


# =========================================================
# DAENERYS
#
# ROOT = Daenerys
# =========================================================

segrellion_initialize_daenerys_aggression_effect = {

	if = {

		limit = {

			NOT = {
				has_character_flag = segrellion_daenerys_invasion_event_scheduled
			}

			NOT = {
				has_character_flag = segrellion_daenerys_invasion_abandoned
			}
		}


		# =================================================
		# PLAYER
		#
		# Game Rule controls AI behavior only.
		# =================================================

		if = {

			limit = {
				is_ai = no
			}

			add_character_flag = segrellion_daenerys_invasion_event_scheduled

			trigger_event = {
				id = segrellion_daenerys_invasion.0001
				days = { 15 30 }
			}
		}


		# =================================================
		# AI
		# =================================================

		else = {

			add_character_flag = segrellion_daenerys_invasion_event_scheduled

			# =================================================
			# HARD WAR LOCK
			#
			# ai_war_chance only influences AI willingness.
			# sworn_peace actually blocks the standard
			# declare-war interaction.
			# =================================================

			if = {

				limit = {

					NOT = {
						has_character_flag = sworn_peace
					}
				}

				add_character_flag = sworn_peace
			}


			# Apply the war lock immediately. Waiting for the first monthly
			# pulse leaves enough time for AGOT's AI to attack Dragonstone.
			if = {

				limit = {

					NOT = {
						has_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
					}
				}

				add_character_modifier = {
					modifier = segrellion_daenerys_preparing_invasion_ai_modifier
				}
			}


			# Prevent vanilla / AGOT claim wars from
			# bypassing the scripted invasion route.

			if = {

				limit = {
					has_claim_on = title:h_the_iron_throne
				}

				remove_claim = title:h_the_iron_throne
			}


			if = {

				limit = {
					has_claim_on = title:c_dragonstone
				}

				remove_claim = title:c_dragonstone
			}


			# =============================================
			# PEACEFUL
			#
			# Daenerys consciously rejects Westeros.
			# =============================================

			if = {

				limit = {
					OR = {

						has_game_rule = segrellion_daenerys_aggression_peaceful

						has_character_flag = segrellion_daenerys_random_peaceful
					}
				}

				add_character_flag = segrellion_daenerys_ai_free_cities_route

				trigger_event = {
					id = segrellion_daenerys_invasion.0001
					days = { 15 30 }
				}
			}


			# =============================================
			# PASSIVE
			#
			# 7-12 years, matching the game-rule description.
			# Do not replace this with the short internal event delays.
			# =============================================

			else_if = {

				limit = {
					OR = {

						has_game_rule = segrellion_daenerys_aggression_passive

						has_character_flag = segrellion_daenerys_random_passive
					}
				}

				add_character_flag = segrellion_daenerys_ai_slow_fleet

				trigger_event = {
					id = segrellion_daenerys_invasion.0001
					days = { 2555 4380 }
				}
			}


			# =============================================
			# DEFAULT
			#
			# 2-5 years, matching the game-rule description.
			# This is the strategic invasion delay, not flavor pacing.
			# =============================================

			else_if = {

				limit = {
					OR = {

						has_game_rule = segrellion_daenerys_aggression_default

						has_character_flag = segrellion_daenerys_random_default
					}
				}

				add_character_flag = segrellion_daenerys_ai_standard_fleet

				trigger_event = {
					id = segrellion_daenerys_invasion.0001
					days = { 730 1825 }
				}
			}


			# =============================================
			# AGGRESSIVE
			# =============================================

			else_if = {

				limit = {
					OR = {

						has_game_rule = segrellion_daenerys_aggression_aggressive

						has_character_flag = segrellion_daenerys_random_aggressive
					}
				}

				add_character_flag = segrellion_daenerys_ai_rush_crossing

				trigger_event = {
					id = segrellion_daenerys_invasion.0001
					days = { 1 7 }
				}
			}


			# =============================================
			# FALLBACK = DEFAULT (2-5 years)
			# =============================================

			else = {

				trigger_event = {
					id = segrellion_daenerys_invasion.0001
					days = { 730 1825 }
				}
			}
		}
	}
}

# =========================================================
# DAENERYS - SCRIPTED INVASION PEACE GUARD
#
# Prevent AI Daenerys from declaring ordinary wars while
# waiting for the scripted Westerosi invasion.
# =========================================================

segrellion_daenerys_invasion_peace_guard_effect = {

	character:Targaryen_9 ?= {

		if = {

			limit = {

				is_alive = yes
				is_ai = yes

				has_character_flag = segrellion_daenerys_invasion_event_scheduled

				NOT = {
					has_character_flag = segrellion_daenerys_landed_in_westeros
				}

				NOT = {
					has_character_flag = segrellion_daenerys_invasion_abandoned
				}

				NOT = {
					has_character_flag = segrellion_daenerys_special_story_abandoned
				}

				NOT = {
					has_character_flag = segrellion_daenerys_young_griff_throne_pact
				}
			}


			# =================================================
			# HARD WAR LOCK
			# =================================================

			if = {

				limit = {

					NOT = {
						has_character_flag = sworn_peace
					}
				}

				add_character_flag = sworn_peace
			}


			# =================================================
			# AI WAR DISCOURAGEMENT
			# Secondary protection.
			# =================================================

			if = {

				limit = {

					NOT = {
						has_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
					}
				}

				add_character_modifier = {
					modifier = segrellion_daenerys_preparing_invasion_ai_modifier
				}
			}


			# =================================================
			# REMOVE BYPASS CLAIMS
			# =================================================

			if = {

				limit = {
					has_claim_on = title:h_the_iron_throne
				}

				remove_claim = title:h_the_iron_throne
			}


			if = {

				limit = {
					has_claim_on = title:c_dragonstone
				}

				remove_claim = title:c_dragonstone
			}
		}


		else = {

			# Story lock is no longer required.

			if = {

				limit = {
					has_character_flag = sworn_peace
				}

				remove_character_flag = sworn_peace
			}


			if = {

				limit = {
					has_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
				}

				remove_character_modifier = segrellion_daenerys_preparing_invasion_ai_modifier
			}
		}
	}
}

# =========================================================
# RANDOM STORY GAME RULE INITIALIZATION
# =========================================================

segrellion_initialize_random_story_rules_effect = {

	# =====================================================
	# DAENERYS AGGRESSION
	# =====================================================

	if = {

		limit = {

			has_game_rule = segrellion_daenerys_aggression_random

			character:Targaryen_9 ?= {

				NOT = {
					has_character_flag = segrellion_daenerys_random_aggression_resolved
				}
			}
		}


		character:Targaryen_9 = {

			add_character_flag = segrellion_daenerys_random_aggression_resolved


			random_list = {

				25 = {
					add_character_flag = segrellion_daenerys_random_peaceful
				}

				25 = {
					add_character_flag = segrellion_daenerys_random_passive
				}

				25 = {
					add_character_flag = segrellion_daenerys_random_default
				}

				25 = {
					add_character_flag = segrellion_daenerys_random_aggressive
				}
			}
		}
	}


	# =====================================================
	# DAEMON GENERAL STRATEGY
	# =====================================================

	if = {

		limit = {

			has_game_rule = segrellion_daemon_ai_strategy_random

			character:Segrellion_60 ?= {

				NOT = {
					has_character_flag = segrellion_daemon_random_strategy_resolved
				}
			}
		}


		character:Segrellion_60 = {

			add_character_flag = segrellion_daemon_random_strategy_resolved


			random_list = {

				50 = {
					add_character_flag = segrellion_daemon_random_flamelands_focused
				}

				50 = {
					add_character_flag = segrellion_daemon_random_wandering_flame
				}
			}
		}
	}
}
