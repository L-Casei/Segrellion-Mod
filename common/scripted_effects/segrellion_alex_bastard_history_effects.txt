# =========================================================
# ALEX SEGRELLION - CANON CHILDREN OUTSIDE HIS MARRIAGE
#
# Character history only exists when a bookmark is loaded. It does
# not create future historical characters in a campaign that began
# earlier, so a 106 AC run needs runtime versions of Alison and Benjen.
#
# Historical bookmarks still use the fixed character IDs. Runtime
# versions are stored in AGOT's born-canon-child scope list with their
# own identity flags so both paths can share the same repair logic.
# =========================================================


# =========================================================
# RUNTIME BIRTHS FOR A 106 AC CAMPAIGN
# =========================================================

segrellion_ensure_runtime_alex_children_effect = {

	# -----------------------------------------------------
	# Alison Hightower
	# Biological parents: Alex Segrellion + Lynett Hightower.
	# Public father remains hidden until 141 AC.
	# -----------------------------------------------------
	if = {
		limit = {
			has_game_rule = segrellion_canon_children_enabled
			current_date >= 8114.1.1
			current_date < 8158.3.1

			character:Segrellion_1 ?= { is_alive = yes }
			character:Hightower_95 ?= { is_alive = yes }

			NOT = { character:Segrellion_Alison_Hightower ?= { exists = yes } }
			NOT = {
				any_in_global_list = {
					variable = agot_born_canon_children_scopes
					has_character_flag = is_Segrellion_Alison_Hightower
				}
			}
		}

		character:Segrellion_1 = { save_scope_as = segrellion_alison_real_father }
		character:Hightower_95 = { save_scope_as = segrellion_alison_mother }

		create_character = {
			name = Alison
			age = 0
			random_traits = no
			mother = scope:segrellion_alison_mother
			real_father = scope:segrellion_alison_real_father
			location = scope:segrellion_alison_mother.location
			gender_female_chance = 100
			faith = scope:segrellion_alison_mother.faith
			culture = scope:segrellion_alison_mother.culture
			dynasty = none
			save_scope_as = segrellion_runtime_alison
		}

		set_global_variable = {
			name = segrellion_runtime_alison
			value = scope:segrellion_runtime_alison
		}

		scope:segrellion_runtime_alison = {
			set_house = scope:segrellion_alison_mother.house
			set_real_father = scope:segrellion_alison_real_father

			add_trait = bastard
			add_trait = beauty_good_3
			add_trait = intellect_good_3

			make_trait_inactive = surname_flowers
			make_trait_inactive = cool_silver_hair
			make_trait_inactive = purple_eyes_13

			add_character_flag = canon_status_canon
			add_character_flag = is_Segrellion_Alison_Hightower
			add_to_global_variable_list = {
				name = agot_born_canon_children_scopes
				target = this
			}
			make_unprunable = yes
		}
	}

	# -----------------------------------------------------
	# Benjen Snow
	# Public father until 147 AC: Cregan Stark.
	# Biological parents: Alex Segrellion + Sara Stark/Snow.
	# -----------------------------------------------------
	if = {
		limit = {
			has_game_rule = segrellion_canon_children_enabled
			current_date >= 8131.1.1
			current_date < 8170.1.1

			character:Segrellion_1 ?= { is_alive = yes }
			character:Stark_86 ?= { exists = yes }
			character:Stark_88 ?= { exists = yes }

			NOT = { character:Segrellion_Benjen_Snow ?= { exists = yes } }
			NOT = {
				any_in_global_list = {
					variable = agot_born_canon_children_scopes
					has_character_flag = is_Segrellion_Benjen_Snow
				}
			}
		}

		character:Segrellion_1 = { save_scope_as = segrellion_benjen_real_father }
		character:Stark_86 = { save_scope_as = segrellion_benjen_public_father }
		character:Stark_88 = { save_scope_as = segrellion_benjen_real_mother }

		create_character = {
			name = Benjen
			age = 0
			random_traits = no
			real_father = scope:segrellion_benjen_real_father
			location = scope:segrellion_benjen_real_mother.location
			gender_female_chance = 0
			faith = scope:segrellion_benjen_real_mother.faith
			culture = scope:segrellion_benjen_real_mother.culture
			dynasty = none
			save_scope_as = segrellion_runtime_benjen
		}

		set_global_variable = {
			name = segrellion_runtime_benjen
			value = scope:segrellion_runtime_benjen
		}

		scope:segrellion_runtime_benjen = {
			# Set the legal/public father first, then restore the hidden
			# biological parentage so Cregan never becomes the real father.
			set_father = scope:segrellion_benjen_public_father
			set_real_father = scope:segrellion_benjen_real_father
			set_real_mother = scope:segrellion_benjen_real_mother
			set_house = scope:segrellion_benjen_public_father.house

			add_trait = bastard
			make_trait_inactive = surname_snow

			add_character_flag = canon_status_canon
			add_character_flag = is_Segrellion_Benjen_Snow
			add_to_global_variable_list = {
				name = agot_born_canon_children_scopes
				target = this
			}
			make_unprunable = yes
		}
	}

	# -----------------------------------------------------
	# Maegor the Usurper
	# Historical birth: 133 AC.
	# Father: Daeryn Segrellion. Mother: Alison Hightower.
	# -----------------------------------------------------
	if = {
		limit = {
			has_game_rule = segrellion_canon_children_enabled
			current_date >= 8133.1.1
			current_date < 8158.3.1

			character:Segrellion_Daeryn_Younger_Braavosi ?= { is_alive = yes }

			NOT = { character:Segrellion_Maegor_Usurper ?= { exists = yes } }
			NOT = { exists = global_var:segrellion_runtime_maegor_usurper }

			OR = {
				character:Segrellion_Alison_Hightower ?= { is_alive = yes }
				exists = global_var:segrellion_runtime_alison
			}
		}

		if = {
			limit = {
				character:Segrellion_Alison_Hightower ?= { is_alive = yes }
			}
			character:Segrellion_Alison_Hightower = {
				save_scope_as = segrellion_maegor_runtime_mother
			}
		}
		else_if = {
			limit = { exists = global_var:segrellion_runtime_alison }
			global_var:segrellion_runtime_alison = {
				save_scope_as = segrellion_maegor_runtime_mother
			}
		}

		if = {
			limit = { exists = scope:segrellion_maegor_runtime_mother }

			create_character = {
				name = Maegor
				age = 0
				random_traits = no
				father = character:Segrellion_Daeryn_Younger_Braavosi
				mother = scope:segrellion_maegor_runtime_mother
				location = scope:segrellion_maegor_runtime_mother.location
				gender_female_chance = 0
				faith = character:Segrellion_Daeryn_Younger_Braavosi.faith
				culture = character:Segrellion_Daeryn_Younger_Braavosi.culture
				dynasty = dynasty:dynn_Segrellion
				save_scope_as = segrellion_runtime_maegor_usurper
			}

			set_global_variable = {
				name = segrellion_runtime_maegor_usurper
				value = scope:segrellion_runtime_maegor_usurper
			}

			scope:segrellion_runtime_maegor_usurper = {
				set_house = house:house_Segrellion
				add_trait = ambitious
				add_trait = vengeful
				add_trait = arrogant
				add_trait = brave
				add_trait = physique_good_2
				add_character_flag = canon_status_canon
				add_character_flag = is_Segrellion_Maegor_Usurper
				make_unprunable = yes
			}
		}
	}

}


# =========================================================
# SHARED HISTORY / PARENTAGE APPLICATION
# =========================================================

segrellion_apply_alison_canon_history_effect = {
	set_real_father = character:Segrellion_1
	set_house = character:Hightower_95.house

	if = {
		limit = {
			current_date >= 8123.1.1
			NOT = { has_trait = curious }
			current_date < 8131.1.1
		}
		add_trait = curious
	}

	if = {
		limit = { current_date >= 8127.1.1 }
		if = { limit = { NOT = { has_trait = ambitious } } add_trait = ambitious }
		if = { limit = { NOT = { has_trait = deceitful } } add_trait = deceitful }
		if = { limit = { NOT = { has_trait = patient } } add_trait = patient }
		if = { limit = { NOT = { has_trait = gregarious } } add_trait = gregarious }
	}

	# Alison is Alex's bastard from birth. Ormund Hightower
	# legitimizes her in 128 AC, so every earlier start must
	# visibly retain the Bastard trait.
	if = {
		limit = { current_date < 8128.1.1 }
		if = { limit = { has_trait = legitimized_bastard } remove_trait = legitimized_bastard }
		if = { limit = { NOT = { has_trait = bastard } } add_trait = bastard }
	}
	else = {
		if = { limit = { has_trait = bastard } remove_trait = bastard }
		if = { limit = { NOT = { has_trait = legitimized_bastard } } add_trait = legitimized_bastard }
		agot_remove_bastard_nickname_effect = yes
	}

	if = {
		limit = { current_date >= 8131.1.1 }
		if = { limit = { has_trait = curious } remove_trait = curious }
		if = { limit = { NOT = { has_trait = education_intrigue_4 } } add_trait = education_intrigue_4 }
	}

	if = {
		limit = {
			current_date >= 8133.1.1
			NOT = { has_character_flag = segrellion_alison_133_progression_done }
		}
		add_character_flag = segrellion_alison_133_progression_done
		give_nickname = nick_segrellion_the_silvertower
		agot_safe_buy_seducer_perk_history_effect = yes
		if = { limit = { NOT = { has_trait = seducer } } add_trait = seducer }
		if = { limit = { NOT = { has_trait = witch } } add_trait = witch }
		if = { limit = { NOT = { has_trait = lifestyle_mystic } } add_trait = lifestyle_mystic }
		add_trait_xp = {
			trait = lifestyle_mystic
			value = trait_second_level
		}
	}

	if = {
		limit = { current_date >= 8141.1.1 }
		set_father = character:Segrellion_1
		set_house = character:Hightower_95.house
	}

	if = {
		limit = {
			current_date >= 8132.1.1
			current_date < 8141.9.1
			character:Segrellion_Daeryn_Younger_Braavosi ?= { is_alive = yes }
			NOT = { is_spouse_of = character:Segrellion_Daeryn_Younger_Braavosi }
		}
		add_spouse = character:Segrellion_Daeryn_Younger_Braavosi
	}

	if = {
		limit = {
			current_date >= 8157.1.1
			NOT = { has_character_flag = segrellion_alison_claimed_sheepstealer }
		}
		add_character_flag = segrellion_alison_claimed_sheepstealer
	}

}


segrellion_apply_benjen_canon_history_effect = {

	# Until the reveal, Cregan is the legal/public father while Alex and
	# Sara remain the biological parents.
	if = {
		limit = { current_date < 8158.4.1 }
		set_father = character:Stark_86
		set_real_father = character:Segrellion_1
		set_real_mother = character:Stark_88
		set_house = character:Stark_86.house
	}
	else = {
		set_father = character:Segrellion_1
		set_mother = character:Stark_88
		set_real_father = character:Segrellion_1
		set_real_mother = character:Stark_88
		set_house = character:Stark_86.house
		if = { limit = { has_trait = bastard } remove_trait = bastard }
		if = { limit = { NOT = { has_trait = legitimized_bastard } } add_trait = legitimized_bastard }
		agot_remove_bastard_nickname_effect = yes

		if = { limit = { NOT = { has_claim_on = title:k_flamelands } } add_unpressed_claim = title:k_flamelands }
		if = { limit = { NOT = { has_claim_on = title:d_the_baywater_woods } } add_unpressed_claim = title:d_the_baywater_woods }
		if = { limit = { NOT = { has_claim_on = title:d_vhalerion } } add_unpressed_claim = title:d_vhalerion }
		if = { limit = { NOT = { has_claim_on = title:c_blackmark } } add_unpressed_claim = title:c_blackmark }
		if = { limit = { NOT = { has_claim_on = title:c_winded_glade } } add_unpressed_claim = title:c_winded_glade }
		if = { limit = { NOT = { has_claim_on = title:b_berryport } } add_unpressed_claim = title:b_berryport }
		if = { limit = { NOT = { has_claim_on = title:b_castle_farring } } add_unpressed_claim = title:b_castle_farring }
		if = { limit = { NOT = { has_claim_on = title:b_benthill_dale } } add_unpressed_claim = title:b_benthill_dale }
	}

	if = {
		limit = {
			current_date >= 8139.1.1
			NOT = { has_trait = humble }
		}
		add_trait = humble
	}

	if = {
		limit = { current_date >= 8143.1.1 }
		if = { limit = { NOT = { has_trait = calm } } add_trait = calm }
		if = { limit = { NOT = { has_trait = brave } } add_trait = brave }
	}

	if = {
		limit = {
			current_date >= 8145.1.1
			NOT = { has_character_flag = segrellion_benjen_145_progression_done }
		}
		add_character_flag = segrellion_benjen_145_progression_done
		if = { limit = { NOT = { has_trait = knight } } add_trait = knight }
		save_scope_as = segrellion_benjen_runtime_scope

		agot_canon_children_get_character_scope_effect = {
			TARGET = character:Segrellion_12
			ID = Segrellion_12
			SCOPE = segrellion_benjen_aeryn
		}

		scope:segrellion_benjen_aeryn ?= {
			add_courtier = scope:segrellion_benjen_runtime_scope
			set_relation_friend = scope:segrellion_benjen_runtime_scope
		}
	}

	if = {
		limit = { current_date >= 8147.1.1 }
		if = { limit = { NOT = { has_trait = education_martial_4 } } add_trait = education_martial_4 }
		if = { limit = { NOT = { has_trait = education_martial_prowess_4 } } add_trait = education_martial_prowess_4 }
		if = { limit = { NOT = { has_trait = chaste } } add_trait = chaste }
		if = { limit = { NOT = { has_trait = logistician } } add_trait = logistician }
		if = { limit = { NOT = { has_trait = unyielding_defender } } add_trait = unyielding_defender }
	}

	if = {
		limit = {
			current_date >= 8158.1.1
			NOT = { has_character_flag = segrellion_benjen_highwind }
		}
		add_character_flag = segrellion_benjen_highwind
		add_character_flag = segrellion_benjen_awakened_icebringer
	}

}


# =========================================================
# POST-HISTORY / SAVE-COMPATIBILITY REPAIR
# =========================================================

segrellion_apply_maegor_usurper_canon_history_effect = {

	if = {
		limit = {
			current_date >= 8141.9.2
			NOT = { has_character_flag = segrellion_maegor_exiled_pentos }
		}
		add_character_flag = segrellion_maegor_exiled_pentos
	}

	if = {
		limit = { current_date >= 8149.1.1 }
		if = { limit = { NOT = { has_trait = education_martial_4 } } add_trait = education_martial_4 }
		if = { limit = { NOT = { has_trait = education_martial_prowess_4 } } add_trait = education_martial_prowess_4 }
		if = { limit = { NOT = { has_trait = lifestyle_blademaster } } add_trait = lifestyle_blademaster }
	}

	if = {
		limit = {
			current_date >= 8157.1.1
			NOT = { has_character_flag = segrellion_maegor_war_started }
		}
		add_character_flag = segrellion_maegor_war_started
		add_character_flag = segrellion_maegor_wields_lightbringer
		add_character_flag = segrellion_maegor_valyrian_armor
		add_character_flag = segrellion_maegor_warriors_of_flame
		give_nickname = nick_segrellion_the_usurper
	}

	if = {
		limit = {
			current_date >= 8158.3.1
			is_alive = yes
		}
		death = {
			death_reason = death_dragon_fire
			killer = character:dragon_maenor
		}
	}
}


segrellion_repair_alex_children_parentage_effect = {

	if = {
		limit = {
			current_date >= 8114.1.1
			character:Segrellion_1 ?= { }
			character:Hightower_95 ?= { }
			character:Segrellion_Alison_Hightower ?= { }
		}
		character:Segrellion_Alison_Hightower = {
			segrellion_apply_alison_canon_history_effect = yes
		}
	}

	random_in_global_list = {
		variable = agot_born_canon_children_scopes
		limit = {
			has_character_flag = is_Segrellion_Alison_Hightower
			is_alive = yes
		}
		segrellion_apply_alison_canon_history_effect = yes
	}

	if = {
		limit = {
			current_date >= 8131.1.1
			character:Segrellion_1 ?= { }
			character:Stark_86 ?= { }
			character:Stark_88 ?= { }
			character:Segrellion_Benjen_Snow ?= { }
		}
		character:Segrellion_Benjen_Snow = {
			segrellion_apply_benjen_canon_history_effect = yes
		}
	}

	random_in_global_list = {
		variable = agot_born_canon_children_scopes
		limit = {
			has_character_flag = is_Segrellion_Benjen_Snow
			is_alive = yes
		}
		segrellion_apply_benjen_canon_history_effect = yes
	}

	if = {
		limit = {
			current_date >= 8133.1.1
			character:Segrellion_Maegor_Usurper ?= { }
		}
		character:Segrellion_Maegor_Usurper = {
			segrellion_apply_maegor_usurper_canon_history_effect = yes
		}
	}

	if = {
		limit = { exists = global_var:segrellion_runtime_maegor_usurper }
		global_var:segrellion_runtime_maegor_usurper = {
			if = {
				limit = { is_alive = yes }
				segrellion_apply_maegor_usurper_canon_history_effect = yes
			}
		}
	}

}
