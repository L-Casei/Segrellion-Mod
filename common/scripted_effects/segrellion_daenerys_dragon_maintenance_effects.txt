# =========================================================
# DAENERYS TARGARYEN
# DRAGON MAINTENANCE
#
# TARGET:
#
# Around age 2:
#   Drogon        >= roughly 30-40 size
#   Rhaegal       >= roughly 25-35 size
#   Viserion      >= roughly 25-35 size
#
# Drogon remains clearly the largest.
#
# Existing saves receive the correction automatically
# because these use new milestone flags.
# =========================================================


segrellion_daenerys_dragon_maintenance_effect = {

	character:Targaryen_9 ?= {

		if = {

			limit = {
				is_alive = yes
			}


			save_scope_as = segrellion_daenerys_dragon_anchor


			# =================================================
			# DROGON
			# =================================================

			character:dragon_drogon ?= {

				if = {

					limit = {
						is_alive = yes
					}


					# -----------------------------------------
					# STAY WITH DAENERYS
					# -----------------------------------------

					if = {

						limit = {

							NOT = {
								has_character_flag = in_dragonpit
							}
						}


						set_location = scope:segrellion_daenerys_dragon_anchor.location
					}


					# -----------------------------------------
					# AGE 2 CANON GROWTH BOOST
					#
					# Existing size is normally around 0-10.
					# +30 puts Drogon around 30-40.
					# -----------------------------------------

					if = {

						limit = {

							age >= 2

							NOT = {
								has_character_flag = segrellion_drogon_age_two_growth_boost
							}
						}


						change_dragon_size = {
							VALUE = 30
						}

						add_prowess_skill = 8


						add_character_flag = segrellion_drogon_age_two_growth_boost

						add_character_flag = {
							flag = segrellion_drogon_bonus_growth_cooldown
							years = 1
						}
					}


					# -----------------------------------------
					# AGE 3 EXTRA GROWTH
					# -----------------------------------------

					if = {

						limit = {

							age >= 3

							NOT = {
								has_character_flag = segrellion_drogon_age_three_growth_boost
							}
						}


						change_dragon_size = {
							VALUE = 6
						}

						add_prowess_skill = 2


						add_character_flag = segrellion_drogon_age_three_growth_boost
					}


					# -----------------------------------------
					# CONTINUED ACCELERATED GROWTH
					# -----------------------------------------

					if = {

						limit = {

							age >= 2

							NOT = {
								has_character_flag = segrellion_drogon_bonus_growth_cooldown
							}
						}


						change_dragon_size = {
							VALUE = 5
						}

						add_prowess_skill = 2


						add_character_flag = {
							flag = segrellion_drogon_bonus_growth_cooldown

						}
					}
				}
			}
