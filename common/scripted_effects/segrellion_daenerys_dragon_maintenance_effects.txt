# =========================================================
# DAENERYS TARGARYEN
# DRAGON MAINTENANCE
#
# Drogon, Rhaegal and Viserion need to match their
# unusually rapid canon growth rather than AGOT's normal
# hatchling growth curve.
# =========================================================


# =========================================================
# FORCE CANON SIZE
#
# CURRENT SCOPE = story_dragon_variable_storage
#
# AGOT stores dragon growth on the story storage.
# =========================================================

segrellion_daenerys_force_dragon_size_effect = {

	set_variable = {
		name = dragon_size_base
		value = $BASE_SIZE$
	}

	set_variable = {
		name = dragon_size
		value = $DISPLAY_SIZE$
	}
}


# =========================================================
# BOOKMARK INITIALIZATION MARKER
# =========================================================

segrellion_setup_daenerys_dragons_300_effect = {

	character:Targaryen_9 ?= {

		if = {

			limit = {
				is_alive = yes
			}

			add_character_flag = segrellion_daenerys_dragons_300_initialized
		}
	}
}


# =========================================================
# DRAGON MAINTENANCE
# =========================================================

segrellion_daenerys_dragon_maintenance_effect = {

	if = {

		limit = {

			current_date >= 8300.1.1

			agot_has_dragon_storage_system_global_list = yes
		}


		every_in_global_list = {

			variable = gl_dragon_variable_storage


			# =================================================
			# ONLY DAENERYS' THREE CANON DRAGONS
			# =================================================

			limit = {

				story_type = story_dragon_variable_storage

				exists = var:dragon_id

				var:dragon_id = {

					is_alive = yes

					OR = {

						AND = {
							exists = character:dragon_drogon
							this = character:dragon_drogon
						}

						AND = {
							exists = character:dragon_rhaegal
							this = character:dragon_rhaegal
						}

						AND = {
							exists = character:dragon_viserion
							this = character:dragon_viserion
						}

						has_inactive_trait = is_dragon_drogon

						has_inactive_trait = is_dragon_rhaegal

						has_inactive_trait = is_dragon_viserion
					}
				}
			}


			# =================================================
			# KEEP THEM WITH DAENERYS
			# =================================================

			var:dragon_id = {

				if = {

					limit = {

						NOT = {
							has_character_flag = in_dragonpit
						}

						exists = character:Targaryen_9

						character:Targaryen_9 = {
							is_alive = yes
						}
					}


					character:Targaryen_9 ?= {

						save_scope_as = segrellion_daenerys_dragon_anchor
					}


					set_location = scope:segrellion_daenerys_dragon_anchor.location
				}
			}


			# =================================================
			# DROGON
			#
			# Physique III = larger than his brothers.
			# =================================================

			if = {

				limit = {

					var:dragon_id = {

						OR = {

							AND = {
								exists = character:dragon_drogon
								this = character:dragon_drogon
							}

							has_inactive_trait = is_dragon_drogon
						}
					}
				}


				segrellion_daenerys_force_dragon_size_effect = {

					BASE_SIZE = 31

					DISPLAY_SIZE = 40
				}
			}


			# =================================================
			# RHAEGAL
			# =================================================

			else_if = {

				limit = {

					var:dragon_id = {

						OR = {

							AND = {
								exists = character:dragon_rhaegal
								this = character:dragon_rhaegal
							}

							has_inactive_trait = is_dragon_rhaegal
						}
					}
				}


				segrellion_daenerys_force_dragon_size_effect = {

					BASE_SIZE = 30

					DISPLAY_SIZE = 36
				}
			}


			# =================================================
			# VISERION
			# =================================================

			else_if = {

				limit = {

					var:dragon_id = {

						OR = {

							AND = {
								exists = character:dragon_viserion
								this = character:dragon_viserion
							}

							has_inactive_trait = is_dragon_viserion
						}
					}
				}


				segrellion_daenerys_force_dragon_size_effect = {

					BASE_SIZE = 30

					DISPLAY_SIZE = 36
				}
			}
		}
	}
}
