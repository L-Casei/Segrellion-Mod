# =========================================================
# DAENERYS TARGARYEN
# DRAGON MAINTENANCE
#
# Drogon, Rhaegal and Viserion grow substantially faster
# than ordinary young dragons.
#
# DESIGN:
#
# - Immediate canon-size correction in the 300 AC story.
# - Drogon remains clearly the largest.
# - Additional accelerated growth every 6 months.
# - Extra scripted growth ends at age 15.
# - Normal AGOT dragon aging continues underneath.
#
# Existing saves that already received the old age-two
# correction will NOT receive the initial correction twice.
# They will still enter the new accelerated growth system.
# =========================================================


segrellion_daenerys_dragon_maintenance_effect = {

	character:Targaryen_9 ?= {

		if = {

			limit = {
				is_alive = yes
			}


			save_scope_as = segrellion_daenerys_dragon_anchor


			# =================================================
			# DROGON
			# =================================================

			character:dragon_drogon ?= {

				if = {

					limit = {
						is_alive = yes
					}


					# =========================================
					# KEEP DROGON WITH DAENERYS
					#
					# Do not pull him out of a dragonpit.
					# =========================================

					if = {

						limit = {

							NOT = {
								has_character_flag = in_dragonpit
							}
						}


						set_location = scope:segrellion_daenerys_dragon_anchor.location
					}


					# =========================================
					# INITIAL CANON SIZE CORRECTION
					#
					# Old saves which already received the
					# previous +30 boost do not receive this.
					# =========================================

					if = {

						limit = {

							age < 15

							NOT = {
								has_character_flag = segrellion_drogon_rapid_growth_initialized
							}
						}


						if = {

							limit = {

								NOT = {
									has_character_flag = segrellion_drogon_age_two_growth_boost
								}
							}


							change_dragon_size = {
								VALUE = 25
							}


							add_prowess_skill = 6
						}


						add_character_flag = segrellion_drogon_rapid_growth_initialized


						add_character_flag = {

							flag = segrellion_drogon_rapid_growth_cooldown

							days = 180
						}
					}


					# =========================================
					# ACCELERATED YOUNG GROWTH
					#
					# +3 extra size every ~6 months.
					#
					# This is ADDITIONAL to AGOT's normal
					# yearly dragon growth.
					# =========================================

					if = {

						limit = {

							age < 15

							has_character_flag = segrellion_drogon_rapid_growth_initialized


							NOT = {
								has_character_flag = segrellion_drogon_rapid_growth_cooldown
							}
						}


						change_dragon_size = {
							VALUE = 3
						}


						add_prowess_skill = 1


						add_character_flag = {

							flag = segrellion_drogon_rapid_growth_cooldown

							days = 180
						}
					}
				}
			}


			# =================================================
			# RHAEGAL
			# =================================================

			character:dragon_rhaegal ?= {

				if = {

					limit = {
						is_alive = yes
					}


					# =========================================
					# KEEP RHAEGAL WITH DAENERYS
					# =========================================

					if = {

						limit = {

							NOT = {
								has_character_flag = in_dragonpit
							}
						}


						set_location = scope:segrellion_daenerys_dragon_anchor.location
					}


					# =========================================
					# INITIAL CANON SIZE CORRECTION
					# =========================================

					if = {

						limit = {

							age < 15

							NOT = {
								has_character_flag = segrellion_rhaegal_rapid_growth_initialized
							}
						}


						if = {

							limit = {

								NOT = {
									has_character_flag = segrellion_rhaegal_age_two_growth_boost
								}
							}


							change_dragon_size = {
								VALUE = 20
							}


							add_prowess_skill = 5
						}


						add_character_flag = segrellion_rhaegal_rapid_growth_initialized


						add_character_flag = {

							flag = segrellion_rhaegal_rapid_growth_cooldown

							days = 180
						}
					}


					# =========================================
					# ACCELERATED YOUNG GROWTH
					#
					# +2 extra size every ~6 months.
					# =========================================

					if = {

						limit = {

							age < 15

							has_character_flag = segrellion_rhaegal_rapid_growth_initialized


							NOT = {
								has_character_flag = segrellion_rhaegal_rapid_growth_cooldown
							}
						}


						change_dragon_size = {
							VALUE = 2
						}


						add_prowess_skill = 1


						add_character_flag = {

							flag = segrellion_rhaegal_rapid_growth_cooldown

							days = 180
						}
					}
				}
			}


			# =================================================
			# VISERION
			# =================================================

			character:dragon_viserion ?= {

				if = {

					limit = {
						is_alive = yes
					}


					# =========================================
					# KEEP VISERION WITH DAENERYS
					# =========================================

					if = {

						limit = {

							NOT = {
								has_character_flag = in_dragonpit
							}
						}


						set_location = scope:segrellion_daenerys_dragon_anchor.location
					}


					# =========================================
					# INITIAL CANON SIZE CORRECTION
					# =========================================

					if = {

						limit = {

							age < 15

							NOT = {
								has_character_flag = segrellion_viserion_rapid_growth_initialized
							}
						}


						if = {

							limit = {

								NOT = {
									has_character_flag = segrellion_viserion_age_two_growth_boost
								}
							}


							change_dragon_size = {
								VALUE = 20
							}


							add_prowess_skill = 5
						}


						add_character_flag = segrellion_viserion_rapid_growth_initialized


						add_character_flag = {

							flag = segrellion_viserion_rapid_growth_cooldown

							days = 180
						}
					}


					# =========================================
					# ACCELERATED YOUNG GROWTH
					#
					# +2 extra size every ~6 months.
					# =========================================

					if = {

						limit = {

							age < 15

							has_character_flag = segrellion_viserion_rapid_growth_initialized


							NOT = {
								has_character_flag = segrellion_viserion_rapid_growth_cooldown
							}
						}


						change_dragon_size = {
							VALUE = 2
						}


						add_prowess_skill = 1


						add_character_flag = {

							flag = segrellion_viserion_rapid_growth_cooldown

							days = 180
						}
					}
				}
			}
		}
	}
}
