# Daenerys's canon dragons: repair the real AGOT storage, including spawned
# versions identified by their inactive canon identity traits.
# Called after lobby setup and by the existing monthly global maintenance.
# ROOT for the helper is story_dragon_variable_storage.
segrellion_daenerys_ensure_dragon_size_effect = {
	# Check the displayed size, which includes physique modifiers, rather than
	# assuming dragon_size_base is the size shown in the dragon window.
	# The count bounds the repair even if another mod changes size calculation.
	while = {
		count = 200
		limit = {
			var:dragon_id = { dragon_size < $MIN_SIZE$ }
		}
		change_dragon_size_direct = { VALUE = 1 }
	}
	set_variable = {
		name = dragon_size
		value = var:dragon_id.dragon_size
	}
}

# Retain the entry point used by existing setup scripts.
segrellion_setup_daenerys_dragons_300_effect = {
	character:Targaryen_9 ?= {
		if = {
			limit = { is_alive = yes }
			add_character_flag = segrellion_daenerys_dragons_300_initialized
		}
	}
}

segrellion_daenerys_dragon_maintenance_effect = {
	if = {
		limit = {
			current_date >= 8300.1.1
			agot_has_dragon_storage_system_global_list = yes
		}
		every_in_global_list = {
			variable = gl_dragon_variable_storage
			limit = {
				story_type = story_dragon_variable_storage
				exists = var:dragon_id
				var:dragon_id = {
					is_alive = yes
					OR = {
						AND = { exists = character:dragon_drogon this = character:dragon_drogon }
						AND = { exists = character:dragon_rhaegal this = character:dragon_rhaegal }
						AND = { exists = character:dragon_viserion this = character:dragon_viserion }
						has_inactive_trait = is_dragon_drogon
						has_inactive_trait = is_dragon_rhaegal
						has_inactive_trait = is_dragon_viserion
					}
				}
			}
			var:dragon_id = {
				if = {
					limit = {
						NOT = { has_character_flag = in_dragonpit }
						exists = character:Targaryen_9
						character:Targaryen_9 = { is_alive = yes }
					}
					character:Targaryen_9 ?= {
						save_scope_as = segrellion_daenerys_dragon_anchor
					}
					set_location = scope:segrellion_daenerys_dragon_anchor.location
				}
			}
			if = {
				limit = {
					var:dragon_id = {
						OR = {
							AND = { exists = character:dragon_drogon this = character:dragon_drogon }
							has_inactive_trait = is_dragon_drogon
						}
					}
				}
				segrellion_daenerys_ensure_dragon_size_effect = { MIN_SIZE = 40 }
			}
			else = {
				segrellion_daenerys_ensure_dragon_size_effect = { MIN_SIZE = 35 }
			}
		}
	}
}
