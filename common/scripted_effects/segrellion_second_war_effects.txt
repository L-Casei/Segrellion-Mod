# =========================================================
# SECOND WAR OF THE FIVE KINGS
# =========================================================


# =========================================================
# REGISTER A NEW IRON THRONE CONTENDER
#
# Called from on_war_started.
#
# Any claimant whose war targets h_the_iron_throne becomes
# an active contender.
#
# The newly-entered claimant immediately declares separate
# elimination wars against every previous active contender.
# =========================================================

segrellion_second_war_register_contender_effect = {


	# =====================================================
	# IDENTIFY THE CLAIMANT
	# =====================================================

	scope:war = {

		if = {

			limit = {
				exists = claimant
			}

			claimant = {

				save_scope_as = segrellion_new_throne_contender
			}
		}


		# Fallback for unusual scripted wars without
		# an explicit claimant.

		else = {

			primary_attacker = {

				save_scope_as = segrellion_new_throne_contender
			}
		}
	}


	# =====================================================
	# REGISTER HIM / HER
	# =====================================================

	scope:segrellion_new_throne_contender ?= {

		if = {

			limit = {
				is_alive = yes
			}

			add_character_flag = segrellion_second_war_active_contender


			# =================================================
			# DECLARE AGAINST EVERY PREVIOUS CONTENDER
			# =================================================

			every_living_character = {

				limit = {

					has_character_flag = segrellion_second_war_active_contender

					this != scope:segrellion_new_throne_contender


					# If this contender has ALREADY obtained
					# the Iron Throne, the normal throne claim
					# war already handles the conflict.

					this != title:h_the_iron_throne.holder


					# Never create duplicate wars.

					NOT = {

						is_at_war_with = scope:segrellion_new_throne_contender
					}
				}


				save_scope_as = segrellion_existing_throne_contender


				scope:segrellion_new_throne_contender = {

					start_war = {

						cb = segrellion_second_war_claimant_elimination_cb

						target = scope:segrellion_existing_throne_contender
					}
				}
			}
		}
	}
}

# =========================================================
# SECOND WAR OF THE FIVE KINGS
# THRONE SETTLEMENT
#
# Runs after title gains.
# =========================================================

segrellion_second_war_throne_settlement_effect = {


	# =====================================================
	# STANNIS WINS
	# =====================================================

	if = {

		limit = {

			title:h_the_iron_throne.holder ?= {

				this = character:Baratheon_3
			}


			character:Baratheon_3 = {

				has_character_flag = segrellion_stannis_iron_throne_war_started

				NOT = {
					has_character_flag = segrellion_stannis_iron_throne_victory_settled
				}
			}
		}


		character:Baratheon_3 = {

			add_character_flag = segrellion_stannis_iron_throne_victory_settled
		}


		# =================================================
		# THE NORTH SWEARS FEALTY
		# =================================================

		title:e_the_north.holder ?= {

			if = {

				limit = {

					is_alive = yes

					this != character:Baratheon_3

					NOT = {
						liege ?= character:Baratheon_3
					}
				}


				save_scope_as = segrellion_stannis_north_vassal
			}
		}


		if = {

			limit = {

				exists = scope:segrellion_stannis_north_vassal
			}


			create_title_and_vassal_change = {

				type = swear_fealty

				save_scope_as = segrellion_stannis_north_fealty_change

				add_claim_on_loss = no
			}


			scope:segrellion_stannis_north_vassal = {

				change_liege = {

					liege = character:Baratheon_3

					change = scope:segrellion_stannis_north_fealty_change
				}
			}


			resolve_title_and_vassal_change = scope:segrellion_stannis_north_fealty_change
		}


		# =================================================
		# THE VALE SWEARS FEALTY
		#
		# Only if the Vale actually answered Stannis' call.
		# =================================================

		title:e_the_vale.holder ?= {

			if = {

				limit = {

					is_alive = yes

					this != character:Baratheon_3

					has_character_flag = segrellion_vale_supported_stannis_iron_throne

					NOT = {
						liege ?= character:Baratheon_3
					}
				}


				save_scope_as = segrellion_stannis_vale_vassal
			}
		}


		if = {

			limit = {

				exists = scope:segrellion_stannis_vale_vassal
			}


			create_title_and_vassal_change = {

				type = swear_fealty

				save_scope_as = segrellion_stannis_vale_fealty_change

				add_claim_on_loss = no
			}


			scope:segrellion_stannis_vale_vassal = {

				change_liege = {

					liege = character:Baratheon_3

					change = scope:segrellion_stannis_vale_fealty_change
				}
			}


			resolve_title_and_vassal_change = scope:segrellion_stannis_vale_fealty_change
		}
	}


	# =====================================================
	# DAENERYS WINS
	#
	# Intentionally NO automatic North/Vale settlement.
	# Future negotiation events will handle them.
	# =====================================================

	else_if = {

		limit = {

			title:h_the_iron_throne.holder ?= {

				this = character:Targaryen_9
			}


			character:Targaryen_9 = {

				has_character_flag = segrellion_daenerys_iron_throne_war_started

				NOT = {
					has_character_flag = segrellion_daenerys_iron_throne_victory_settled
				}
			}
		}


		character:Targaryen_9 = {

			add_character_flag = segrellion_daenerys_iron_throne_victory_settled
		}
	}


	# =====================================================
	# YOUNG GRIFF WINS
	#
	# North and Vale stay independent.
	# Dorne swears fealty.
	# =====================================================

	else_if = {

		limit = {

			title:h_the_iron_throne.holder ?= {

				this = character:IAF_Young_Griff_1
			}


			character:IAF_Young_Griff_1 = {

				has_character_flag = segrellion_young_griff_iron_throne_war_started

				NOT = {
					has_character_flag = segrellion_young_griff_iron_throne_victory_settled
				}
			}
		}


		character:IAF_Young_Griff_1 = {

			add_character_flag = segrellion_young_griff_iron_throne_victory_settled
		}


		# =================================================
		# DORNE
		#
		# Martell_9 is already the canonical Dorne actor
		# used by the 300 AC storyline.
		# =================================================

		character:Martell_9 ?= {

			if = {

				limit = {

					is_alive = yes

					this != character:IAF_Young_Griff_1

					NOT = {
						liege ?= character:IAF_Young_Griff_1
					}
				}


				save_scope_as = segrellion_young_griff_dorne_vassal
			}
		}


		if = {

			limit = {

				exists = scope:segrellion_young_griff_dorne_vassal
			}


			create_title_and_vassal_change = {

				type = swear_fealty

				save_scope_as = segrellion_young_griff_dorne_fealty_change

				add_claim_on_loss = no
			}


			scope:segrellion_young_griff_dorne_vassal = {

				change_liege = {

					liege = character:IAF_Young_Griff_1

					change = scope:segrellion_young_griff_dorne_fealty_change
				}
			}


			resolve_title_and_vassal_change = scope:segrellion_young_griff_dorne_fealty_change
		}
	}
}
