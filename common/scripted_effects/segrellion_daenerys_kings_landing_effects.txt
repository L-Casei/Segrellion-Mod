# =========================================================
# SAVE THE CURRENT DEFENDER
# =========================================================

segrellion_daenerys_save_kings_landing_defender_effect = {

	if = {

		limit = {

			NOT = {
				has_variable = segrellion_daenerys_defeated_throne_holder
			}


			title:h_the_iron_throne ?= {
				exists = holder
			}


			title:h_the_iron_throne.holder != root
		}


		set_variable = {

			name = segrellion_daenerys_defeated_throne_holder

			value = title:h_the_iron_throne.holder
		}


		title:h_the_iron_throne.holder = {

			if = {

				limit = {

					NOT = {

						location ?= {
							county = title:c_kings_landing
						}
					}
				}


				add_character_flag = segrellion_defeated_away_from_kings_landing
			}
		}
	}
}


# =========================================================
# END DAENERYS' IRON THRONE WAR
#
# ALSO RECORD WHO ACTUALLY FOUGHT FOR EACH SIDE.
# =========================================================

segrellion_daenerys_force_iron_throne_victory_effect = {

	segrellion_daenerys_save_kings_landing_defender_effect = yes


	# Invalidate AGOT's Mega War aftermath BEFORE end_war.
	# If we wait until after resolution, AGOT may already have
	# queued agot_mega_wars.0600 in the same tick.
	segrellion_daenerys_cancel_agot_wars_aftermath_effect = yes


	random_character_war = {

		limit = {

			is_attacker = root


			casus_belli = {

				any_target_title = {

					this = title:h_the_iron_throne
				}
			}
		}


		# =================================================
		# ACTUAL SUPPORTERS
		# =================================================

		every_war_attacker = {

			limit = {

				this != character:Targaryen_9
			}


			add_character_flag = segrellion_supported_daenerys_invasion

			remove_character_flag = segrellion_opposed_daenerys_invasion
		}


		# =================================================
		# ACTUAL ENEMIES
		# =================================================

		every_war_defender = {

			limit = {

				this != character:Targaryen_9
			}


			add_character_flag = segrellion_opposed_daenerys_invasion

			remove_character_flag = segrellion_supported_daenerys_invasion
		}


		end_war = attacker
	}


	# AGOT can prepare its Mega War punishment state during
	# the war-resolution tick. Clear it immediately, then let
	# the post-throne handoff verify the title on the next day.
	segrellion_daenerys_cancel_agot_wars_aftermath_effect = yes


	# The title transfer caused by war resolution is not
	# guaranteed to finish in the same script tick.
	add_character_flag = {
		flag = segrellion_daenerys_kings_landing_victory_handoff_active
		days = 30
	}


	trigger_event = {

		id = segrellion_daenerys_kings_landing.0030
		days = 1
	}
}


# =========================================================
# THE QUEEN OF ASHES
# =========================================================

segrellion_daenerys_burn_kings_landing_effect = {

	# Save the king BEFORE the massacre can alter succession.

	segrellion_daenerys_save_kings_landing_defender_effect = yes


	add_character_flag = segrellion_daenerys_burned_kings_landing


	# =====================================================
	# CUSTOM POST-THRONE TYRANNY
	#
	# Burning King's Landing guarantees a minimum of 50/100.
	# =====================================================

	segrellion_daenerys_set_pacification_tyranny_minimum_50_effect = yes


	# Vanilla tyranny remains deliberately catastrophic.

	add_tyranny = 100

	add_dread = 50


	add_character_modifier = {

		modifier = segrellion_daenerys_queen_of_ashes_modifier

		years = 20
	}


	# =====================================================
	# RESOLVE THE THRONE WAR BEFORE THE MASSACRE
	#
	# The old order killed 90% of King's Landing first.
	# If the defending king died there, CK3/AGOT could
	# invalidate or redirect the war before Daenerys gained
	# the Iron Throne, killing the entire post-throne chain.
	# =====================================================

	segrellion_daenerys_force_iron_throne_victory_effect = yes


	# =====================================================
	# EACH PERSON PHYSICALLY IN KING'S LANDING HAS A 90% CHANCE TO DIE
	#
	# Daenerys herself and Drogon are excluded.
	# =====================================================

	every_living_character = {

		limit = {

			this != root

			this != character:dragon_drogon


			location ?= {

				county = title:c_kings_landing
			}
		}


		random = {
			chance = 90
			death = {
				death_reason = death_segrellion_burned_in_kings_landing
				killer = root
			}
		}
	}


	title:c_kings_landing = {

		add_county_modifier = {

			modifier = segrellion_kings_landing_burned_modifier

			years = 25
		}
	}
}


# =========================================================
# PEACEFUL SURRENDER
# =========================================================

segrellion_daenerys_accept_kings_landing_surrender_effect = {

	add_character_flag = segrellion_daenerys_goldcloaks_accepted_surrender


	title:c_kings_landing = {

		add_county_modifier = {

			modifier = segrellion_kings_landing_peaceful_surrender_modifier

			years = 5
		}
	}


	segrellion_daenerys_force_iron_throne_victory_effect = yes
}


# =========================================================
# FINALIZE THE DEFEATED IRON THRONE HOLDER
#
# ROOT = DAENERYS
#
# - Invalidates every other war involving the defeated king
# - Transfers EVERY remaining title personally held by them
#   to Daenerys
# - Leaves their fate to The Fallen Crown
# =========================================================

segrellion_daenerys_finalize_defeated_throne_holder_effect = {

	if = {

		limit = {

			this = character:Targaryen_9

			has_variable = segrellion_daenerys_defeated_throne_holder


			var:segrellion_daenerys_defeated_throne_holder = {

				is_alive = yes


				NOT = {
					has_character_flag = segrellion_defeated_throne_holder_titles_seized
				}
			}
		}


		var:segrellion_daenerys_defeated_throne_holder = {

			save_scope_as = segrellion_daenerys_fallen_king


			# =================================================
			# THEIR OTHER WARS END INCONCLUSIVELY
			# =================================================

			every_character_war = {

				end_war = invalidated
			}


			# =================================================
			# DAENERYS TAKES EVERYTHING
			#
			# No counties, duchies, kingdoms or secondary
			# personal titles remain with the defeated monarch.
			# =================================================

			every_held_title = {

				grant_title = character:Targaryen_9
			}


			remove_character_flag = segrellion_defeated_away_from_kings_landing

			add_character_flag = segrellion_defeated_throne_holder_titles_seized
		}
	}
}
