# =========================================================
# DREAMER VISUAL LEVEL
#
# Keeps the dynamic DDS independent from direct XP checks
# inside the trait icon evaluation.
# =========================================================

segrellion_sync_dreamer_visual_level_effect = {


	remove_character_flag = segrellion_dreamer_visual_level_1
	remove_character_flag = segrellion_dreamer_visual_level_2
	remove_character_flag = segrellion_dreamer_visual_level_3


	if = {

		limit = {

			has_trait = segrellion_dreamer_trait

			has_trait_xp = {
				trait = segrellion_dreamer_trait
				value >= 50
			}
		}

		add_character_flag = segrellion_dreamer_visual_level_3
	}


	else_if = {

		limit = {

			has_trait = segrellion_dreamer_trait

			has_trait_xp = {
				trait = segrellion_dreamer_trait
				value >= 25
			}
		}

		add_character_flag = segrellion_dreamer_visual_level_2
	}


	else_if = {

		limit = {

			has_trait = segrellion_dreamer_trait

			has_trait_xp = {
				trait = segrellion_dreamer_trait
				value >= 10
			}
		}

		add_character_flag = segrellion_dreamer_visual_level_1
	}
}

# =========================================================
# DREAMER EXPERIENCE
# =========================================================

segrellion_gain_dreamer_xp_effect = {


	if = {

		limit = {

			NOT = {
				has_trait = segrellion_dreamer_trait
			}
		}

		add_trait = segrellion_dreamer_trait
	}


	add_trait_xp = {

		trait = segrellion_dreamer_trait
		value = 1
	}


	segrellion_sync_dreamer_visual_level_effect = yes
}

# =========================================================
# DREAMER FLAVOR DISPATCHER
#
# Level 0 -> only small dreams.
# Level 1+ -> medium dreams.
#
# Later:
# Level 2 can mix Medium + Major.
# Level 3 can mix Medium + Major + Prophetic.
# =========================================================

segrellion_dreamer_flavor_pool_effect = {

	if = {
		limit = {

			has_trait_xp = {
				trait = segrellion_dreamer_trait
				value >= 10
			}
		}

		segrellion_dreamer_level_1_plus_flavor_pool_effect = yes
	}

	else = {

		segrellion_dreamer_level_0_flavor_pool_effect = yes
	}
}
