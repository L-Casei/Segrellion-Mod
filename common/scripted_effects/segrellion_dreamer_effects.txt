# =========================================================
# DREAMER EXPERIENCE
# =========================================================

segrellion_gain_dreamer_xp_effect = {

	if = {
		limit = {
			NOT = {
				has_trait = segrellion_dreamer_trait
			}
		}

		add_trait = segrellion_dreamer_trait
	}


	add_trait_xp = {
		trait = segrellion_dreamer_trait
		value = 1
	}
}

# =========================================================
# DREAMER FLAVOR DISPATCHER
#
# Level 0 -> only small dreams.
# Level 1+ -> medium dreams.
#
# Later:
# Level 2 can mix Medium + Major.
# Level 3 can mix Medium + Major + Prophetic.
# =========================================================

segrellion_dreamer_flavor_pool_effect = {

	if = {
		limit = {

			has_trait_xp = {
				trait = segrellion_dreamer_trait
				value >= 10
			}
		}

		segrellion_dreamer_level_1_plus_flavor_pool_effect = yes
	}

	else = {

		segrellion_dreamer_level_0_flavor_pool_effect = yes
	}
}
