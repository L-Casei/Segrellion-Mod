# =========================================================
# DREAMER EXPERIENCE
#
# Every dream / vision grants exactly 1 XP.
# If the character somehow reaches a Dreamer event without
# the trait, the trait is granted automatically.
# =========================================================

segrellion_gain_dreamer_xp_effect = {

	if = {
		limit = {
			NOT = {
				has_trait = segrellion_dreamer_trait
			}
		}

		add_trait = segrellion_dreamer_trait
	}


	add_trait_xp = {
		trait = segrellion_dreamer_trait
		value = 1
	}
}
