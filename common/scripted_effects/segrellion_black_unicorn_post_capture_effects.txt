# =========================================================
# BLACK UNICORN - POST CAPTURE EFFECTS
# =========================================================


# =========================================================
# RELEASE VISENYA AFTER THE BEACON DUNGEON ARC
#
# ROOT = DAEMON
# =========================================================

segrellion_black_unicorn_release_visenya_from_dungeon_effect = {

	character:Black_Unicorn_1 ?= {

		if = {
			limit = {
				is_alive = yes
				is_imprisoned_by = root
			}

			release_from_prison = yes
		}

		remove_character_flag = segrellion_black_unicorn_imprisoned_by_daemon
		remove_character_flag = segrellion_black_unicorn_imprisoned_visenya_story_pending
	}

	remove_character_flag = segrellion_black_unicorn_visenya_imprisoned
	remove_character_flag = segrellion_black_unicorn_imprisoned_visenya_story_pending
}


# =========================================================
# DAEMON AND VISENYA RECONCILE AFTER THE DUNGEON
#
# This does not marry them directly.
# It only clears the prison state and feeds the existing
# marriage / double-wife machinery.
# =========================================================

segrellion_black_unicorn_dungeon_reconciliation_effect = {

	segrellion_black_unicorn_release_visenya_from_dungeon_effect = yes

	remove_character_modifier = segrellion_black_unicorn_treacherous_host_modifier
	remove_character_flag = segrellion_black_unicorn_daemon_betrayed_visenya

	character:Black_Unicorn_1 = {
		remove_character_flag = segrellion_black_unicorn_betrayed_by_daemon
	}

	add_character_flag = segrellion_black_unicorn_dungeon_reconciliation
	add_character_flag = segrellion_black_unicorn_visenya_forgave_beacon_betrayal

	character:Black_Unicorn_1 = {

		add_character_flag = segrellion_black_unicorn_daemon_forgiven_after_dungeon

		add_opinion = {
			target = root
			modifier = segrellion_visenya_forgave_dungeon_betrayal_opinion
		}
	}
}


# =========================================================
# DAEMON RELEASES VISENYA AND RECOGNIZES HER AS FAMILY
#
# Visenya becomes his acknowledged niece and temporary
# designated heir. She receives no extra claim here.
# =========================================================

segrellion_black_unicorn_recognize_visenya_as_niece_effect = {

	segrellion_black_unicorn_release_visenya_from_dungeon_effect = yes

	segrellion_reveal_black_unicorn_identity_effect = yes

	add_character_flag = segrellion_black_unicorn_recognized_visenya_as_niece

	# Visenya only becomes the temporary designated heir if
	# Daemon does not already have a living son.
	if = {

		limit = {

			NOT = {
				any_child = {
					is_alive = yes
					is_female = no
				}
			}
		}

		add_character_flag = segrellion_black_unicorn_visenya_temporary_heir
		set_designated_heir_unsafe = character:Black_Unicorn_1

		character:Black_Unicorn_1 = {
			add_character_flag = segrellion_black_unicorn_temporary_heir_to_daemon
		}
	}

	character:Black_Unicorn_1 = {

		add_character_flag = segrellion_black_unicorn_recognized_niece_of_daemon

		add_opinion = {
			target = root
			modifier = segrellion_visenya_recognized_as_family_opinion
		}
	}

	remove_character_modifier = segrellion_black_unicorn_treacherous_host_modifier
	remove_character_flag = segrellion_black_unicorn_daemon_betrayed_visenya

	character:Black_Unicorn_1 = {
		remove_character_flag = segrellion_black_unicorn_betrayed_by_daemon
	}
}


# =========================================================
# EXECUTE VISENYA AND HER SKAGOSI FOLLOWERS
#
# RICKON AND OSHA ARE EXPLICITLY SPARED.
# =========================================================

segrellion_black_unicorn_execute_visenya_and_followers_effect = {

	# -----------------------------------------------------
	# HER FOUR NAMED SKAGOSI FOLLOWERS
	# -----------------------------------------------------

	global_var:segrellion_visenya_skagosi_warrior_1 ?= {
		if = {
			limit = { is_alive = yes }
			death = {
				death_reason = death_execution
				killer = root
			}
		}
	}

	global_var:segrellion_visenya_skagosi_warrior_2 ?= {
		if = {
			limit = { is_alive = yes }
			death = {
				death_reason = death_execution
				killer = root
			}
		}
	}

	global_var:segrellion_visenya_skagosi_warrior_3 ?= {
		if = {
			limit = { is_alive = yes }
			death = {
				death_reason = death_execution
				killer = root
			}
		}
	}

	global_var:segrellion_visenya_skagosi_warrior_4 ?= {
		if = {
			limit = { is_alive = yes }
			death = {
				death_reason = death_execution
				killer = root
			}
		}
	}

	# -----------------------------------------------------
	# RICKON AND OSHA SURVIVE
	# -----------------------------------------------------

	character:Stark_Dummy_11 ?= {

		if = {
			limit = { is_alive = yes }

			root = {
				add_courtier = character:Stark_Dummy_11
			}

			character:Stark_Dummy_11 = {
				set_location = title:b_castle_farring.title_province
				add_character_flag = segrellion_rickon_spared_after_visenya_execution
				add_character_flag = segrellion_rickon_under_daemon_protection
			}
		}
	}

	character:Stark_hh_13 ?= {

		if = {
			limit = { is_alive = yes }

			root = {
				add_courtier = character:Stark_hh_13
			}

			character:Stark_hh_13 = {
				set_location = title:b_castle_farring.title_province
				add_character_flag = segrellion_osha_spared_after_visenya_execution
				add_character_flag = segrellion_osha_protects_rickon
			}
		}
	}

	# -----------------------------------------------------
	# RHAEGON SURVIVES VISENYA
	# -----------------------------------------------------

	global_var:segrellion_historical_rhaegon ?= {

		if = {
			limit = { is_alive = yes }

			set_location = title:b_castle_farring.title_province
			add_character_flag = segrellion_rhaegon_survived_visenya_execution
		}
	}

	# -----------------------------------------------------
	# VISENYA
	# -----------------------------------------------------

	character:Black_Unicorn_1 ?= {

		if = {
			limit = { is_alive = yes }

			if = {
				limit = { is_imprisoned_by = root }
				release_from_prison = yes
			}

			death = {
				death_reason = death_execution
				killer = root
			}
		}
	}

	add_character_flag = segrellion_black_unicorn_daemon_executed_visenya
	add_character_flag = segrellion_black_unicorn_dungeon_arc_resolved

	remove_character_flag = segrellion_black_unicorn_visenya_imprisoned
	remove_character_flag = segrellion_black_unicorn_imprisoned_visenya_story_pending

	remove_character_modifier = segrellion_black_unicorn_treacherous_host_modifier
}
