segrellion_spawn_canon_maenor_effect = {

	$OWNER$ = {
		save_scope_as = owner
	}

	create_character = {

		name = "Maenor"
		gender = female

		culture = culture:dragon_culture
		faith = faith:valyrian_pan_freehold

		location = scope:owner.location

		age = 0
		dynasty = none

		trait = dragon

		random_traits = no

		save_scope_as = dragon

		after_creation = {

			add_character_flag = segrellion_maenor_dragon
			add_character_flag = agot_no_silly_name

			set_variable = {
				name = dragon_gender_hidden
				value = 1
			}

			create_story = story_dragon_alive

			set_variable = {
				name = dragon_age
				value = age
			}

			set_variable = {
				name = dragon_size
				value = 0
			}

			set_variable = {
				name = dragon_size_base
				value = 0
			}

			agot_back_apply_dragon_aging_effect = yes


			# ==================================================
			# PERSONALITY
			# ==================================================

			add_trait = dragon_majestic
			add_trait = dragon_cooperative
			add_trait = dragon_accepting
			add_trait = dragon_calculating
			add_trait = dragon_restrained

			add_trait = dragon_physique_good_2
			add_trait = education_dragon_4


			# ==================================================
			# APPEARANCE
			#
			# Aquí metemos luego todos sus genes visuales.
			# ==================================================


			agot_dragon_transfer_vars_to_story_cycle_effect = {
				DRAGON = scope:dragon
			}

			agot_dragon_birth_magic_effect = yes
		}
	}


	# Make it an owned dragon

	agot_set_as_owned_dragon = {
		OWNER = scope:owner
		DRAGON = scope:dragon
	}


	# Add to dragon family tree

	agot_dragon_tree_creation_effect = {
		DRAGON = scope:dragon
	}


	# Bond to canon rider

	agot_bond_dragon_relation_effect = {
		ACTOR = scope:owner
		DRAGON = scope:dragon
	}


	# Remember this specific dragon

	set_global_variable = {
		name = segrellion_canon_maenor
		value = scope:dragon
	}
}
