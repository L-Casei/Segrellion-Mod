# =========================================================
# SEGRELLION RELIC SEARCH
# =========================================================


# =========================================================
# CLEAR PENDING DESTINATION
# =========================================================

segrellion_clear_relic_pending_destination_effect = {

	remove_character_flag = segrellion_relic_pending_tyrosh
	remove_character_flag = segrellion_relic_pending_oldtown
	remove_character_flag = segrellion_relic_pending_winterfell

	remove_character_flag = segrellion_relic_pending_castle_black
	remove_character_flag = segrellion_relic_pending_shadow_tower
	remove_character_flag = segrellion_relic_pending_skagos
	remove_character_flag = segrellion_relic_pending_nightfort
}


# =========================================================
# FINISH SEARCH
# =========================================================

segrellion_finish_relic_search_effect = {

	remove_character_flag = segrellion_legendary_relic_search_active
	remove_character_flag = segrellion_legendary_relic_search_paused

	remove_character_flag = segrellion_relic_route_tyrosh
	remove_character_flag = segrellion_relic_route_oldtown
	remove_character_flag = segrellion_relic_route_winterfell

	remove_character_flag = segrellion_relic_winterfell_full_access

	segrellion_clear_relic_pending_destination_effect = yes

	return_home = yes
}


# =========================================================
# RESUME SEARCH TRAVEL
# =========================================================

segrellion_resume_relic_travel_effect = {

	remove_character_flag = segrellion_legendary_relic_search_paused


	# -----------------------------------------------------
	# TYROSH
	# -----------------------------------------------------

	if = {
		limit = {
			has_character_flag = segrellion_relic_pending_tyrosh
		}

		start_travel_plan = {
			destination = title:c_tyrosh.title_province
			return_trip = no
			on_arrival_event = segrellion_relic.1000
			on_arrival_destinations = last
			on_travel_planner_cancel_event = segrellion_relic.0090
		}
	}


	# -----------------------------------------------------
	# OLDTOWN
	# -----------------------------------------------------

	else_if = {
		limit = {
			has_character_flag = segrellion_relic_pending_oldtown
		}

		start_travel_plan = {
			destination = title:c_oldtown.title_province
			return_trip = no
			on_arrival_event = segrellion_relic.2000
			on_arrival_destinations = last
			on_travel_planner_cancel_event = segrellion_relic.0090
		}
	}


	# -----------------------------------------------------
	# WINTERFELL
	# -----------------------------------------------------

	else_if = {
		limit = {
			has_character_flag = segrellion_relic_pending_winterfell
		}

		start_travel_plan = {
			destination = title:c_winterfell.title_province
			return_trip = no
			on_arrival_event = segrellion_relic.3000
			on_arrival_destinations = last
			on_travel_planner_cancel_event = segrellion_relic.0090
		}
	}


	# -----------------------------------------------------
	# CASTLE BLACK
	# -----------------------------------------------------

	else_if = {
		limit = {
			has_character_flag = segrellion_relic_pending_castle_black
		}

		start_travel_plan = {
			destination = title:c_castle_black.title_province
			return_trip = no
			on_arrival_event = segrellion_relic.4000
			on_arrival_destinations = last
			on_travel_planner_cancel_event = segrellion_relic.0090
		}
	}


	# -----------------------------------------------------
	# SHADOW TOWER
	# -----------------------------------------------------

	else_if = {
		limit = {
			has_character_flag = segrellion_relic_pending_shadow_tower
		}

		start_travel_plan = {
			destination = title:c_the_shadow_tower.title_province
			return_trip = no
			on_arrival_event = segrellion_relic.6000
			on_arrival_destinations = last
			on_travel_planner_cancel_event = segrellion_relic.0090
		}
	}


	# -----------------------------------------------------
	# SKAGOS - SKANE
	# -----------------------------------------------------

	else_if = {
		limit = {
			has_character_flag = segrellion_relic_pending_skagos
		}

		start_travel_plan = {
			destination = title:c_skane.title_province

			return_trip = no

			on_arrival_event = segrellion_relic.7000
			on_arrival_destinations = last

			on_travel_planner_cancel_event = segrellion_relic.0090
		}
	}


	# -----------------------------------------------------
	# NIGHTFORT
	# -----------------------------------------------------

	else_if = {
		limit = {
			has_character_flag = segrellion_relic_pending_nightfort
		}

		start_travel_plan = {
			destination = title:b_the_nightfort.title_province
			return_trip = no
			on_arrival_event = segrellion_relic.5000
			on_arrival_destinations = last
			on_travel_planner_cancel_event = segrellion_relic.0090
		}
	}
}
