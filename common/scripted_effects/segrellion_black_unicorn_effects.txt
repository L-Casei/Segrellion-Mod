# =========================================================
# SELECT DAEMON'S INVESTIGATOR
#
# The investigator is permanently stored as a CHARACTER
# VARIABLE on Daemon.
# =========================================================

segrellion_black_unicorn_begin_courtier_investigation_effect = {


	if = {

		limit = {
			has_variable = segrellion_black_unicorn_investigator
		}

		remove_variable = segrellion_black_unicorn_investigator
	}


	random_courtier_or_guest = {

		limit = {

			is_alive = yes
			is_adult = yes

			intrigue >= 8

			NOT = {
				has_trait = incapable
			}

			NOT = {
				this = root
			}
		}


		alternative_limit = {

			is_alive = yes
			is_adult = yes

			NOT = {
				has_trait = incapable
			}

			NOT = {
				this = root
			}
		}


		save_scope_as = black_unicorn_investigator
	}


	if = {

		limit = {
			exists = scope:black_unicorn_investigator
		}


		set_variable = {

			name = segrellion_black_unicorn_investigator

			value = scope:black_unicorn_investigator
		}


		scope:black_unicorn_investigator = {

			add_character_flag = segrellion_black_unicorn_investigator
		}
	}


	add_character_flag = segrellion_black_unicorn_courtier_investigation


	trigger_event = {

		id = segrellion_black_unicorn.0020

		days = 90
	}
}

# =========================================================
# DUEL SAFETY
#
# The duel itself is REAL single combat.
# These effects only prevent its random injury system from
# permanently mutilating either story character.
# =========================================================

segrellion_black_unicorn_snapshot_duel_injuries_effect = {

	}

	if = {
		limit = { has_trait = maimed }

	}

	if = {
		limit = { has_trait = one_legged }

	}

}

# =========================================================
# REVEAL THE BLACK UNICORN
#
# DO NOT CALL THIS EFFECT YET.
#
# The same character becomes Visenya Segrellion.
# Her genealogy only appears after the reveal.
# =========================================================

segrellion_reveal_black_unicorn_identity_effect = {


	character:Black_Unicorn_1 = {


		change_first_name = segrellion_black_unicorn_true_name


		set_father = character:Segrellion_55
		set_mother = character:Segrellion_59


		set_house = house:house_Segrellion


		remove_character_flag = segrellion_black_unicorn_mysterious_woman

		add_character_flag = segrellion_black_unicorn_identity_revealed
		add_character_flag = segrellion_black_unicorn_visenya_revealed
	}
}

# =========================================================
# RETURN TO DAEMON'S ORIGINAL CAMP
#
# Uses the province stored before the Skagos expedition.
# =========================================================

segrellion_black_unicorn_return_to_daemon_camp_effect = {


	# Save wherever Daemon physically is when the return begins.

	location = {

		save_scope_as = black_unicorn_return_start
	}


	if = {

		limit = {

			exists = scope:black_unicorn_return_start
		}


		set_variable = {

			name = segrellion_black_unicorn_return_start

			value = scope:black_unicorn_return_start
		}
	}


	if = {

		limit = {

			exists = current_travel_plan
		}


		current_travel_plan = {

			complete_travel_plan = yes
		}
	}


	trigger_event = {

		id = segrellion_black_unicorn.0195

		days = 1
	}
}

# =========================================================
# BLACK UNICORN
# CHECK THE THREE SKAGOS EXPEDITIONS
#
# If all three have resolved and nobody has returned with
# She-Unicorn's letter, the trail simply goes cold.
# =========================================================

segrellion_black_unicorn_check_three_scouts_effect = {

	if = {

		limit = {

			has_character_flag = segrellion_black_unicorn_shipmaster_resolved
			has_character_flag = segrellion_black_unicorn_swordsman_resolved
			has_character_flag = segrellion_black_unicorn_guide_resolved


			NOT = {
				has_character_flag = segrellion_black_unicorn_survivor_pending
			}

			NOT = {
				has_character_flag = segrellion_black_unicorn_letter_received
			}

			NOT = {
				has_character_flag = segrellion_black_unicorn_trail_gone_cold
			}
		}


		add_character_flag = segrellion_black_unicorn_three_scouts_failed
		add_character_flag = segrellion_black_unicorn_trail_gone_cold

		# Reserved for the future storyline where Visenya
		# eventually comes south after the Stormflame war.
		add_character_flag = segrellion_black_unicorn_future_post_war_return
	}
}
