# =========================================================
# SELECT DAEMON'S INVESTIGATOR
#
# The investigator is permanently stored as a CHARACTER
# VARIABLE on Daemon.
# =========================================================

segrellion_black_unicorn_begin_courtier_investigation_effect = {


	if = {

		limit = {
			has_variable = segrellion_black_unicorn_investigator
		}

		remove_variable = segrellion_black_unicorn_investigator
	}


	random_courtier_or_guest = {

		limit = {

			is_alive = yes
			is_adult = yes

			intrigue >= 8

			NOT = {
				has_trait = incapable
			}

			NOT = {
				this = root
			}
		}


		alternative_limit = {

			is_alive = yes
			is_adult = yes

			NOT = {
				has_trait = incapable
			}

			NOT = {
				this = root
			}
		}


		save_scope_as = black_unicorn_investigator
	}


	if = {

		limit = {
			exists = scope:black_unicorn_investigator
		}


		set_variable = {

			name = segrellion_black_unicorn_investigator

			value = scope:black_unicorn_investigator
		}


		scope:black_unicorn_investigator = {

			add_character_flag = segrellion_black_unicorn_investigator
		}
	}


	add_character_flag = segrellion_black_unicorn_courtier_investigation


	trigger_event = {

		id = segrellion_black_unicorn.0020

		days = 90
	}
}

# =========================================================
# VISENYA - RESOLVE SECRET OF SKAGOS
#
# ROOT = Black_Unicorn_1
#
# The result is stored directly on She-Unicorn.
# This effect is idempotent.
# =========================================================

segrellion_black_unicorn_resolve_secret_effect = {

	if = {

		limit = {

			NOT = {
				has_character_flag = segrellion_black_unicorn_secret_icebringer
			}

			NOT = {
				has_character_flag = segrellion_black_unicorn_secret_rhaegon
			}
		}


		# =================================================
		# ICEBRINGER
		# =================================================

		if = {

			limit = {
				has_game_rule = segrellion_visenya_secret_icebringer
			}


			add_character_flag = segrellion_black_unicorn_secret_icebringer
		}


		# =================================================
		# RHAEGON
		# =================================================

		else_if = {

			limit = {
				has_game_rule = segrellion_visenya_secret_rhaegon
			}


			add_character_flag = segrellion_black_unicorn_secret_rhaegon
		}


		# =================================================
		# RANDOM
		# =================================================

		else_if = {

			limit = {
				has_game_rule = segrellion_visenya_secret_random
			}


			random_list = {

				50 = {
					add_character_flag = segrellion_black_unicorn_secret_icebringer
				}

				50 = {
					add_character_flag = segrellion_black_unicorn_secret_rhaegon
				}
			}
		}


		# =================================================
		# DEFENSIVE FALLBACK
		#
		# Preserve the current mod behavior.
		# =================================================

		else = {

			add_character_flag = segrellion_black_unicorn_secret_icebringer
		}
	}
}


# =========================================================
# CREATE SECRET RHAEGON
#
# ROOT = Black_Unicorn_1
#
# Rhaegon already exists when Daemon begins investigating.
#
# This deliberately does NOT use
# agot_spawn_owned_hatchling_from_egg_effect because that
# effect sends the owner a hatching interface message.
#
# There is no physical egg here because its discovery and
# hatching happened before the playable storyline.
# =========================================================

segrellion_black_unicorn_create_rhaegon_effect = {

	if = {

		limit = {

			NOT = {
				exists = global_var:segrellion_historical_rhaegon
			}
		}


		save_scope_as = rhaegon_owner


		# =================================================
		# CREATE RHAEGON
		#
		# Direct creation.
		# NO hatch effect.
		# NO interface notification.
		# =================================================

		create_character = {

			name = Rhaegon

			location = root.location

			template = agot_dragon_hatchling_template

			age = 0

			random_traits = no

			save_scope_as = rhaegon
		}


		# =================================================
		# OWNERSHIP
		# =================================================

		hidden_effect = {

			agot_set_as_owned_dragon = {

				OWNER = scope:rhaegon_owner
				DRAGON = scope:rhaegon
			}
		}


		# =================================================
		# RHAEGON TRAITS
		# =================================================

		scope:rhaegon = {

			change_first_name = Rhaegon


			add_character_flag = segrellion_rhaegon_dragon
			add_character_flag = segrellion_rhaegon_secret
			add_character_flag = agot_no_silly_name


			# =============================================
			# PHYSIQUE
			# =============================================

			remove_trait = dragon_physique_bad
			remove_trait = dragon_physique_good_1
			remove_trait = dragon_physique_good_2
			remove_trait = dragon_physique_good_3

			# Tyrannic Leviathan
			add_trait = dragon_physique_good_3


			# =============================================
			# SPECIAL
			# =============================================

			add_trait = dragon_majestic


			# =============================================
			# PERSONALITY
			# =============================================

			add_trait = dragon_bloodthirsty
			add_trait = dragon_voracious
			add_trait = dragon_calculating
			add_trait = dragon_imperious


			# =============================================
			# GROWTH-APPROPRIATE PROWESS
			#
			# Our historical dragon growth system gives
			# +3 prowess alongside +3 size during early
			# growth.
			#
			# Size 3 therefore corresponds to 3 prowess.
			# =============================================

			add_prowess_skill = 3


			# =============================================
			# RHAEGON DNA
			#
			# Still intentionally absent.
			#
			# DO NOT add has_scripted_appearance until
			# Rhaegon's actual dragon DNA is supplied.
			# =============================================
		}


		# =================================================
		# AGOT DRAGON RUNTIME STORAGE
		#
		# This is the part currently missing.
		# =================================================

		segrellion_initialize_runtime_dragon_stats_effect = {

			DRAGON = scope:rhaegon

			DRAGON_AGE = scope:rhaegon.age

			DRAGON_SIZE = 3
		}


		scope:rhaegon = {

			add_character_flag = segrellion_rhaegon_early_growth_active
		}


		# Bloodthirsty receives +4 Draconic Dread when AGOT
		# normally assigns the personality through its
		# personality event. We reproduce that initialization
		# because the trait is being assigned directly.

		scope:rhaegon = {

			change_draconic_dread = {
				VALUE = 4
			}
		}


		# =================================================
		# AGOT DRAGON SYSTEM
		# =================================================

		agot_dragon_tree_creation_effect = {

			DRAGON = scope:rhaegon
		}


		agot_bond_dragon_relation_effect = {

			ACTOR = scope:rhaegon_owner
			DRAGON = scope:rhaegon
		}


		# =================================================
		# PERMANENT REFERENCE
		# =================================================

		set_global_variable = {

			name = segrellion_historical_rhaegon

			value = scope:rhaegon
		}


		add_character_flag = segrellion_black_unicorn_rhaegon_created
	}
}

# =========================================================
# RHAEGON - EARLY EXPLOSIVE GROWTH
#
# Birth : 3
# Age 1 : 18
# Age 2 : 35
# Age 3 : 55
#
# After age 3 the custom growth stops completely and AGOT's
# normal dragon growth system takes over.
# =========================================================

segrellion_rhaegon_early_growth_effect = {

	if = {

		limit = {

			exists = global_var:segrellion_historical_rhaegon
		}


		global_var:segrellion_historical_rhaegon = {

			save_scope_as = segrellion_rhaegon_growth_dragon
		}


		if = {

			limit = {

				scope:segrellion_rhaegon_growth_dragon = {

					is_alive = yes

					has_character_flag = segrellion_rhaegon_early_growth_active
				}
			}


			# =================================================
			# FIND RHAEGON'S AGOT RUNTIME STORAGE
			# =================================================

			random_in_global_list = {

				variable = gl_dragon_variable_storage


				limit = {

					var:dragon_id ?= scope:segrellion_rhaegon_growth_dragon
				}


				save_scope_as = segrellion_rhaegon_growth_storage
			}


			# =================================================
			# AGE 0
			# SIZE 3
			# =================================================

			if = {

				limit = {

					scope:segrellion_rhaegon_growth_dragon = {

						age < 1
					}
				}


				scope:segrellion_rhaegon_growth_dragon = {

					set_variable = {

						name = dragon_size_base

						value = 3
					}

					set_variable = {

						name = dragon_size

						value = 3
					}
				}


				scope:segrellion_rhaegon_growth_storage ?= {

					set_variable = {

						name = dragon_size_base

						value = 3
					}

					set_variable = {

						name = dragon_size

						value = 3
					}
				}
			}


			# =================================================
			# AGE 1
			# SIZE 18
			# =================================================

			else_if = {

				limit = {

					scope:segrellion_rhaegon_growth_dragon = {

						age >= 1
						age < 2
					}
				}


				scope:segrellion_rhaegon_growth_dragon = {

					set_variable = {

						name = dragon_size_base

						value = 18
					}

					set_variable = {

						name = dragon_size

						value = 18
					}
				}


				scope:segrellion_rhaegon_growth_storage ?= {

					set_variable = {

						name = dragon_size_base

						value = 18
					}

					set_variable = {

						name = dragon_size

						value = 18
					}
				}
			}


			# =================================================
			# AGE 2
			# SIZE 35
			# =================================================

			else_if = {

				limit = {

					scope:segrellion_rhaegon_growth_dragon = {

						age >= 2
						age < 3
					}
				}


				scope:segrellion_rhaegon_growth_dragon = {

					set_variable = {

						name = dragon_size_base

						value = 35
					}

					set_variable = {

						name = dragon_size

						value = 35
					}
				}


				scope:segrellion_rhaegon_growth_storage ?= {

					set_variable = {

						name = dragon_size_base

						value = 35
					}

					set_variable = {

						name = dragon_size

						value = 35
					}
				}
			}


			# =================================================
			# AGE 3
			# SIZE 55
			#
			# Release him into AGOT's normal growth system.
			# =================================================

			else = {


				scope:segrellion_rhaegon_growth_dragon = {

					set_variable = {

						name = dragon_size_base

						value = 55
					}

					set_variable = {

						name = dragon_size

						value = 55
					}


					remove_character_flag = segrellion_rhaegon_early_growth_active

					add_character_flag = segrellion_rhaegon_early_growth_complete
				}


				scope:segrellion_rhaegon_growth_storage ?= {

					set_variable = {

						name = dragon_size_base

						value = 55
					}

					set_variable = {

						name = dragon_size

						value = 55
					}
				}
			}
		}
	}
}

# =========================================================
# BLACK UNICORN - POST BEACON STORY
#
# Starts the one-year clock after Daemon gains the
# Flamelands.
#
# This effect is deliberately idempotent because it is
# called both from on_title_gain and from the monthly
# failsafe.
# =========================================================

segrellion_black_unicorn_start_post_beacon_clock_effect = {

	character:Segrellion_60 ?= {

		if = {

			limit = {

				is_alive = yes

				title:e_flamelands.holder = this


				character:Black_Unicorn_1 ?= {
					is_alive = yes
				}


				NOT = {
					has_character_flag = segrellion_black_unicorn_post_beacon_clock_started
				}

				NOT = {
					has_character_flag = segrellion_black_unicorn_post_beacon_resolved
				}
			}


			# =================================================
			# THE LAST UNICORN
			# =================================================

			give_nickname = nick_segrellion_the_last_unicorn

			add_character_flag = segrellion_daemon_last_unicorn_nickname_granted


			# =================================================
			# ENSURE VISENYA'S SECRET HAS ALREADY BEEN CHOSEN
			#
			# This does NOT choose a route based on this story.
			# It simply uses the existing game-rule/random
			# resolver if the secret was never initialized.
			# =================================================

			character:Black_Unicorn_1 = {

				segrellion_black_unicorn_resolve_secret_effect = yes
			}


			# =================================================
			# ONE-YEAR CLOCK
			# =================================================

			add_character_flag = segrellion_black_unicorn_post_beacon_clock_started


			add_character_flag = {

				flag = segrellion_black_unicorn_post_beacon_waiting_year

				years = 1
			}
		}
	}
}


# =========================================================
# MONTHLY POST-BEACON CHECK
#
# Once the one-year flag expires, this does NOT start a new
# timer.
#
# It simply waits indefinitely until:
#
# - Daemon is alive
# - Visenya is alive
# - Daemon still rules the Flamelands
# - Visenya has not already joined him
# - Daemon is not at war
# - Daemon is physically at the Beacon
#
# Then the event starts immediately.
# =========================================================

segrellion_black_unicorn_post_beacon_progression_effect = {

	# =====================================================
	# SAVE-COMPATIBLE CLOCK START
	# =====================================================

	segrellion_black_unicorn_start_post_beacon_clock_effect = yes


	character:Segrellion_60 ?= {

		if = {

			limit = {

				is_alive = yes

				is_imprisoned = no


				title:e_flamelands.holder = this


				has_character_flag = segrellion_black_unicorn_post_beacon_clock_started


				NOT = {
					has_character_flag = segrellion_black_unicorn_post_beacon_waiting_year
				}


				NOT = {
					has_character_flag = segrellion_black_unicorn_post_beacon_started
				}


				NOT = {
					has_character_flag = segrellion_black_unicorn_post_beacon_resolved
				}


				NOT = {
					has_character_flag = segrellion_black_unicorn_visenya_returned_with_daemon
				}


				is_at_war = no


				location = title:b_castle_farring.title_province


				character:Black_Unicorn_1 ?= {

					is_alive = yes

					is_imprisoned = no


					NOT = {
						has_character_flag = segrellion_black_unicorn_visenya_joined_daemon
					}
				}
			}


			# Mark it BEFORE firing so another pulse can never
			# duplicate the opening.

			add_character_flag = segrellion_black_unicorn_post_beacon_started


			trigger_event = {

				id = segrellion_black_unicorn.0600

				days = 0
			}
		}
	}
}

# =========================================================
# PREPARE DAEMON FOR THE BEACON DUEL
#
# -25 prowess because he is still partially bound.
#
# EVERY currently equipped artifact is temporarily
# unequipped:
#
# - Trueflame
# - Icebringer
# - armor
# - accessory bonuses
# - anything else the player may have equipped
# =========================================================

segrellion_black_unicorn_prepare_beacon_duel_effect = {

	add_character_modifier = {

		modifier = segrellion_black_unicorn_bound_duel_modifier
	}

		set_variable = {

			name = segrellion_black_unicorn_beacon_duel_unequipped

			value = yes
		}


		unequip_artifact_from_owner = yes
	}


	add_character_flag = segrellion_black_unicorn_beacon_duel_loadout_removed
}


# =========================================================
# RESTORE DAEMON AFTER THE DUEL

segrellion_black_unicorn_restore_beacon_duel_effect = {

}

# =========================================================
# VISENYA TAKES THE FLAMELANDS
#
# ROOT = DAEMON
#
# IMPORTANT:
# Titles are transferred BEFORE Daemon dies.
# Otherwise normal inheritance could steal the realm from
# Visenya immediately after the assassination.
# =========================================================

segrellion_black_unicorn_visenya_take_flamelands_effect = {

	# =====================================================
	# REVEAL HER REAL IDENTITY
	# =====================================================

	segrellion_reveal_black_unicorn_identity_effect = yes


	character:Black_Unicorn_1 = {

		remove_character_flag = segrellion_black_unicorn_lock_camp_skagos


		set_location = title:b_castle_farring.title_province


		add_character_flag = segrellion_black_unicorn_seized_flamelands

		add_character_flag = segrellion_black_unicorn_illegitimate_lady_of_flamelands


		# =================================================
		# MURDERER + DYNASTIC KINSLAYER
		# =================================================

		if = {

			limit = {
				NOT = { has_trait = murderer }
			}

			add_trait = murderer
		}


		if = {

			limit = {
				NOT = { has_trait = kinslayer_1 }
			}

			add_trait = kinslayer_1
		}


		# =================================================
		# ILLEGITIMATE RULER
		# =================================================

		add_character_modifier = {

			modifier = segrellion_black_unicorn_illegitimate_ruler_modifier
		}
	}


	# =====================================================
	# TITLE TRANSFER
	# =====================================================

	create_title_and_vassal_change = {

		type = usurped

		save_scope_as = segrellion_visenya_flamelands_takeover
	}


	# -----------------------------------------------------
	# THE BEACON / CAPITAL COUNTY
	# -----------------------------------------------------

	if = {

		limit = {

			title:c_farring_cross.holder = root
		}


		title:c_farring_cross = {

			change_title_holder = {

				holder = character:Black_Unicorn_1

				change = scope:segrellion_visenya_flamelands_takeover

				government_base = root
			}
		}
	}


	# -----------------------------------------------------
	# KINGDOM TITLE, IF DAEMON HOLDS IT
	# -----------------------------------------------------

	if = {

		limit = {

			title:k_flamelands.holder ?= {
				this = root
			}
		}


		title:k_flamelands = {

			change_title_holder = {

				holder = character:Black_Unicorn_1

				change = scope:segrellion_visenya_flamelands_takeover

				government_base = root
			}
		}
	}


	# -----------------------------------------------------
	# FLAMELANDS + ITS VASSALS
	# -----------------------------------------------------

	title:e_flamelands = {

		change_title_holder_include_vassals = {

			holder = character:Black_Unicorn_1

			change = scope:segrellion_visenya_flamelands_takeover

			government_base = root
		}
	}


	resolve_title_and_vassal_change = scope:segrellion_visenya_flamelands_takeover


	# =====================================================
	# DESTROY HER OLD LANDLESS WARBAND TITLE
	#
	# She is no longer the hidden ruler of a Skagosi camp.
	# She is now Lady of the Flamelands.
	# =====================================================

	if = {

		limit = {

			title:d_laamp_red_unicorn.holder ?= {
				this = character:Black_Unicorn_1
			}
		}


		character:Black_Unicorn_1 = {

			destroy_title = title:d_laamp_red_unicorn
		}
	}


	# =====================================================
	# STORY STATE
	# =====================================================

	add_character_flag = segrellion_black_unicorn_post_beacon_resolved

	add_character_flag = segrellion_black_unicorn_daemon_killed_by_visenya


	character:Black_Unicorn_1 = {

		add_character_flag = segrellion_black_unicorn_post_beacon_visenya_victory
	}


	# =====================================================
	# DAEMON DIES
	# =====================================================

	death = {

		death_reason = death_execution

		killer = character:Black_Unicorn_1
	}
}
