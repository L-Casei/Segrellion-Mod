# =========================================================
# SEGRELLION LORE
# =========================================================

segrellion_add_lore_effect = {

	if = {
		limit = {
			exists = var:segrellion_lore
		}

		change_variable = {
			name = segrellion_lore
			add = $AMOUNT$
		}
	}
	else = {
		set_variable = {
			name = segrellion_lore
			value = $AMOUNT$
		}
	}

	set_variable = {
		name = $VISITED_VARIABLE$
		value = yes
	}
}
