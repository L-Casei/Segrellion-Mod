# =========================================================
# SEGRELLION LORE
# =========================================================

segrellion_add_lore_effect = {

	if = {
		limit = {
			exists = var:segrellion_lore
		}

		change_variable = {
			name = segrellion_lore
			add = $AMOUNT$
		}
	}
	else = {
		set_variable = {
			name = segrellion_lore
			value = $AMOUNT$
		}
	}

	set_variable = {
		name = $VISITED_VARIABLE$
		value = yes
	}
}

# =========================================================
# SEGRELLION LORE - TRAVEL TOAST
#
# Non-blocking parchment notification shown while travelling.
# =========================================================

segrellion_lore_discovery_toast_effect = {

	send_interface_toast = {

		type = event_toast_effect_neutral

		title = $TITLE$
		desc = $DESC$

		left_icon = root

		custom_tooltip = $GAIN_TT$
	}


	segrellion_add_lore_effect = {
		AMOUNT = $AMOUNT$
		VISITED_VARIABLE = $VISITED_VARIABLE$
	}
}
