# =========================================================
# JON POST-RESURRECTION - TRIGGERS
# =========================================================

# La cadena del asesinato marca a todos los participantes con
# segrellion_jon_mutineer antes de matar a Jon.

segrellion_jon_is_murder_mutineer_trigger = {
	OR = {
		has_character_flag = segrellion_jon_mutineer
	}
}


# The historical Jon stays dead when the runtime Jon is created.
segrellion_is_current_jon_trigger = {
	is_alive = yes
	OR = {
		AND = { exists = global_var:segrellion_resurrected_jon this = global_var:segrellion_resurrected_jon }
		AND = {
			this = character:Stark_7
			NOT = { exists = global_var:segrellion_resurrected_jon global_var:segrellion_resurrected_jon ?= { is_alive = yes } }
		}
		has_character_flag = segrellion_canonical_current_jon
	}
}

# =========================================================
# JON - TRUE PARENTAGE PUBLICLY REVEALED
#
# Works with:
# - Segrellion's own reveal chain
# - AGOT / Howland Reed revealing Rhaegar as Jon's father
# =========================================================

segrellion_jon_true_parentage_revealed_trigger = {

	segrellion_is_current_jon_trigger = yes

	OR = {

		has_character_flag = segrellion_jon_targaryen_revealed

		father = character:Targaryen_3
		dynasty ?= dynasty:dynn_Targaryen

		# AGOT may reveal the historical Jon rather than the resurrected character.
		AND = {
			exists = global_var:segrellion_resurrected_jon
			this = global_var:segrellion_resurrected_jon
			character:Stark_7 = {
				OR = {
					father = character:Targaryen_3
					dynasty ?= dynasty:dynn_Targaryen
				}
			}
		}
	}
}


# =========================================================
# JON - MAY PUBLICLY TAKE THE STARK NAME
# =========================================================

segrellion_jon_can_choose_stark_identity_trigger = {

	has_trait = legitimized_bastard
}
