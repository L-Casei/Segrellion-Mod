# =========================================================
# SEGRELLION - SECOND WIFE / EXPANDED EXCEPTIONALISM
# =========================================================

# The special mechanic is reserved for dragonriders,
# characters of Valyrian cultural heritage, or adherents
# of the Fourteen Flames.

segrellion_can_seek_second_wife_trigger = {
	OR = {
		is_current_dragonrider = yes

		culture = {
			has_cultural_pillar = heritage_valyrian
		}

		faith = faith:valyrian_pan_freehold
	}
}


# The custom mechanic has two supported religious routes:
# - Faith of the Seven: personal permission from the High Septon.
# - Fourteen Flames: already polygamous, so no permission is needed.

segrellion_has_second_wife_religious_path_trigger = {
	OR = {
		religion = religion:the_seven_religion
		faith = faith:valyrian_pan_freehold
	}
}


# Permission to actually activate the second-wife search.

segrellion_has_second_wife_permission_trigger = {
	OR = {
		faith = faith:valyrian_pan_freehold
		has_character_flag = segrellion_two_wives_authorized
	}
}
