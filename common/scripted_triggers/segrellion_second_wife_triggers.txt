# =========================================================
# SEGRELLION - SECOND WIFE / EXPANDED EXCEPTIONALISM
# =========================================================

# This custom mechanic exists ONLY for adherents of a faith
# belonging to the Faith of the Seven religion.
#
# Other religions use their own native marriage doctrines.
#
# Within the Seven, the petitioner must either:
# - be a dragonrider, or
# - have Valyrian cultural heritage.

segrellion_can_seek_second_wife_trigger = {

	religion = religion:the_seven_religion

	OR = {

		is_current_dragonrider = yes

		culture = {
			has_cultural_pillar = heritage_valyrian
		}
	}
}


# Kept as a separate trigger because the decision references it.

segrellion_has_second_wife_religious_path_trigger = {

	religion = religion:the_seven_religion
}


# The custom second-wife interaction requires explicit approval
# from the High Septon.

segrellion_has_second_wife_permission_trigger = {

	religion = religion:the_seven_religion

	has_character_flag = segrellion_two_wives_authorized
}
