# =========================================================
# SEGRELLION - SECOND WIFE / EXPANDED EXCEPTIONALISM
# =========================================================

# This custom mechanic exists ONLY inside the Faith of the Seven.
# Other religions use their own native marriage doctrines.

segrellion_can_seek_second_wife_trigger = {

	religion = religion:the_seven_religion

	faith = {
		has_doctrine = agot_exceptionalism_doctrine
	}

	OR = {

		is_current_dragonrider = yes

		culture = {
			has_cultural_pillar = heritage_valyrian
		}
	}
}


# The High Septon has personally recognised the expansion.

segrellion_has_second_wife_permission_trigger = {

	religion = religion:the_seven_religion

	faith = {
		has_doctrine = agot_exceptionalism_doctrine
	}

	has_character_flag = segrellion_two_wives_authorized
}


# The player has converted the permission into one usable
# exceptional second-wife slot through the decision.

segrellion_has_second_wife_slot_trigger = {

	religion = religion:the_seven_religion

	faith = {
		has_doctrine = agot_exceptionalism_doctrine
	}

	has_character_flag = segrellion_two_wives_authorized
	has_character_flag = segrellion_second_wife_slot_unlocked
}

# Candidate scope; scope:actor is the prospective husband.
# Shared by the picker and the direct interaction's validation.
segrellion_second_wife_candidate_trigger = {
	this != scope:actor
	is_alive = yes
	is_female = yes
	is_adult = yes
	is_married = no
	is_betrothed = no
	is_imprisoned = no

	# Do not call can_marry_trigger here. AGOT story locks expose raw
	# NOT_story_type_trigger text and incorrectly exclude Visenya.
	custom_description = {
		text = segrellion_second_wife_refusing_marriage_requirement
		OR = {
			NOT = { has_trait = refusing_marriage }
			AND = {
				this = character:Black_Unicorn_1
				scope:actor = character:Segrellion_60
			}
		}
	}


	custom_description = {

		text = segrellion_second_wife_incest_requirement

		OR = {

			NOT = {
				is_close_family_of = scope:actor
			}

			scope:actor = {

				faith = {
					has_doctrine = doctrine_consanguinity_siblings
				}
			}

			AND = {

				scope:actor = {
					agot_exceptionalism_incest_allowed = yes
				}

				agot_exceptionalism_incest_allowed = yes
			}
		}
	}
}
