# WESTEROS LANDING ROUTES

# =========================================================
# DRAGONSTONE LANDING
# =========================================================

segrellion_daenerys_dragonstone_landing_available_trigger = {

	character:Baratheon_3 = {
		is_alive = no
	}
}


# =========================================================
# DAEMON IS FIRMLY COMMITTED TO AEGON
#
# ROOT = Daemon
# =========================================================

segrellion_daemon_committed_to_aegon_trigger = {

	character:IAF_Young_Griff_1 ?= {
		is_alive = yes
	}

	OR = {

		has_character_flag = segrellion_pledged_to_young_griff

		has_character_flag = segrellion_deadly_triarchy_formed

		has_character_flag = segrellion_daemon_aegon_fealty_accepted

		has_character_flag = segrellion_daemon_aegon_hand_and_regent

		has_character_flag = segrellion_daemon_aegon_confidant

		has_relation_best_friend = character:IAF_Young_Griff_1


		AND = {

			has_relation_friend = character:IAF_Young_Griff_1

			is_allied_to = character:IAF_Young_Griff_1
		}


		character:IAF_Young_Griff_1 = {

			OR = {

				has_character_flag = segrellion_young_griff_daemon_oath_secured

				has_character_flag = segrellion_young_griff_supports_daemon

				has_character_flag = segrellion_young_griff_daemon_confidant
			}
		}
	}
}


# =========================================================
# DAEMON IS PERSONALLY INCLINED TOWARD DAENERYS
#
# ROOT = Daemon
# =========================================================

segrellion_daemon_inclined_to_daenerys_trigger = {

	character:Targaryen_9 ?= {
		is_alive = yes
	}

	OR = {

		has_character_flag = segrellion_pledged_to_daenerys

		has_relation_friend = character:Targaryen_9

		has_relation_best_friend = character:Targaryen_9


		character:Targaryen_9 = {

			OR = {

				has_character_flag = segrellion_daenerys_supports_daemon

				has_character_flag = segrellion_marriage_promised_to_daemon

				has_character_flag = segrellion_daenerys_promised_beacon_marriage
			}
		}


		AND = {

			opinion = {
				target = character:Targaryen_9
				value >= 40
			}

			NOT = {
				segrellion_daemon_committed_to_aegon_trigger = yes
			}
		}
	}
}


# =========================================================
# MASSEY MUST BETRAY DAEMON TO ALLOW THE LANDING
#
# ROOT = Daenerys
# =========================================================

segrellion_daenerys_massey_betrayal_required_trigger = {

	segrellion_daemon_controls_flamelands_trigger = yes


	character:Segrellion_60 = {

		segrellion_daemon_committed_to_aegon_trigger = yes

		NOT = {
			segrellion_daemon_inclined_to_daenerys_trigger = yes
		}
	}


	title:d_masseys_hook.holder ?= {

		is_alive = yes

		this != character:Segrellion_60


		liege ?= {
			this = character:Segrellion_60
		}
	}
}


# =========================================================
# MASSEY'S HOOK LANDING
# =========================================================

segrellion_daenerys_masseys_hook_landing_available_trigger = {

	NOT = {
		has_character_flag = segrellion_daenerys_massey_landing_refused
	}


	OR = {

		title:d_masseys_hook.holder ?= {
			is_alive = yes
		}

		title:c_stonedance.holder ?= {
			is_alive = yes
		}
	}
}


# =========================================================
# BEACON DIRECT LANDING
# =========================================================

segrellion_daenerys_beacon_direct_landing_available_trigger = {

	segrellion_daemon_controls_flamelands_trigger = yes


	OR = {

		has_character_flag = segrellion_daenerys_beacon_invitation_ready


		character:Segrellion_60 = {

			has_character_flag = segrellion_pledged_to_daenerys
		}
	}
}


# =========================================================
# BEACON NEGOTIATION
#
# Daemon may be neutral or aligned with Aegon.
# His political situation affects his ANSWER, not whether
# Daenerys is allowed to ask.
# =========================================================

segrellion_daenerys_beacon_negotiation_available_trigger = {

	segrellion_daemon_controls_flamelands_trigger = yes


	NOT = {
		has_character_flag = segrellion_daenerys_beacon_negotiation_started
	}


	NOT = {
		has_character_flag = segrellion_daenerys_beacon_negotiation_refused
	}


	NOT = {
		has_character_flag = segrellion_daenerys_beacon_invitation_ready
	}


	character:Segrellion_60 = {

		is_alive = yes

		is_landed = yes
	}
}


# =========================================================
# BEACON ROUTE
# =========================================================

segrellion_daenerys_beacon_route_available_trigger = {

	segrellion_daemon_controls_flamelands_trigger = yes


	OR = {

		segrellion_daenerys_beacon_direct_landing_available_trigger = yes

		segrellion_daenerys_beacon_negotiation_available_trigger = yes
	}
}
