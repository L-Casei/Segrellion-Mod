# =========================================================
# SEGRELLION - THE LONG NIGHT COMPATIBILITY
# =========================================================
#
# The Long Night & Azor Ahai stamps every true member of
# the dead with the character flag "ln_one_of_them".
#
# The Night King also carries "is_nightking" and the older
# "agot_immersion_bannermen_blocker" marker.
#
# Character flags do not require a database definition, so
# these checks remain safe when the Long Night mod is absent.
# =========================================================

segrellion_is_long_night_dead_trigger = {

	OR = {

		has_character_flag = ln_one_of_them

		has_character_flag = is_nightking

		has_character_flag = agot_immersion_bannermen_blocker
	}
}
