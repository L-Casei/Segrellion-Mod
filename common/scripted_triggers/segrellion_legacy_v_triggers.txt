# =========================================================
# VALID GREAT HOUSE
#
# CALLER = character making the request
# =========================================================

segrellion_legacy_v_valid_great_house_target_trigger = {

	is_alive = yes
	is_adult = yes
	is_ruler = yes
	is_imprisoned = no

	highest_held_title_tier >= tier_kingdom

	NOT = {
		this = $CALLER$
	}

	# Never ask someone who is currently fighting
	# directly against the caller.
	NOT = {
		is_at_war_with = $CALLER$
	}

	NOR = {
		government_has_flag = government_is_theocracy
		government_has_flag = government_is_holy_order
		is_theocratic_lessee = yes

		faith.religious_head ?= this
	}
}


# =========================================================
# CAN THIS LORD JOIN THIS WAR?
#
# Scope = WAR
#
# CALLER = Segrellion ruler
# TARGET = Lord Paramount being asked
#
# A war is NOT eligible if:
# - TARGET is already in it
# - TARGET is fighting CALLER
# - the opposing primary ruler is TARGET
# - the opposing primary ruler is TARGET's vassal
# =========================================================

# =========================================================
# CALLER LEADS AT LEAST ONE WAR
# =========================================================

segrellion_legacy_v_caller_leads_war_trigger = {

	any_character_war = {

		OR = {

			primary_attacker = root

			primary_defender = root
		}
	}
}


# =========================================================
# CAN THIS LORD JOIN THIS WAR?
#
# Scope = WAR
#
# IMPORTANT:
# The caller must be the PRIMARY war leader.
# Merely participating in somebody else's war is not enough.
# =========================================================

segrellion_legacy_v_callable_war_trigger = {

	OR = {

		# -------------------------------------------------
		# CALLER LEADS THE ATTACKING SIDE
		# -------------------------------------------------

		AND = {

			primary_attacker = {
				this = $CALLER$
			}

			NOT = {
				is_attacker = $TARGET$
			}

			NOT = {
				is_defender = $TARGET$
			}

			primary_defender = {

				NOR = {

					this = $TARGET$

					is_vassal_or_below_of = $TARGET$
				}
			}
		}


		# -------------------------------------------------
		# CALLER LEADS THE DEFENDING SIDE
		# -------------------------------------------------

		AND = {

			primary_defender = {
				this = $CALLER$
			}

			NOT = {
				is_attacker = $TARGET$
			}

			NOT = {
				is_defender = $TARGET$
			}

			primary_attacker = {

				NOR = {

					this = $TARGET$

					is_vassal_or_below_of = $TARGET$
				}
			}
		}
	}
}
