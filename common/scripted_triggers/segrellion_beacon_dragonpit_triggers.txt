# =========================================================
# THE GREAT DRAGONPIT - NATIVE GREAT PROJECT CONTRIBUTORS
# =========================================================
#
# ROOT = potential contributor
# scope:owner = current owner of the Great Project
#
# The engine first applies the native allowed_contributor_filter.
# This trigger then narrows the candidate to the groups intended for
# the Great Dragonpit.
# =========================================================

segrellion_beacon_dragonpit_contributor_valid_trigger = {

	OR = {

		# Current project owner / Lord of the Beacon.
		scope:owner ?= {
			this = root
		}

		# Any direct or lower vassal of the Lord of the Beacon, whenever
		# the native Great Project contributor category exposes them.
		is_vassal_or_below_of = title:c_farring_cross.holder

		# Any Segrellion dynasty member exposed to the native contributor
		# system through vassalage, alliance, liege or top-liege relations.
		dynasty = dynasty:dynn_Segrellion

		# Close or extended family of the project owner.
		scope:owner ?= {
			is_close_or_extended_family_of = root
		}
	}
}
