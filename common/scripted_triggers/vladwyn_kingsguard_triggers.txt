# VLADWYN KINGSGUARD
# =========================================================


# =========================================================
# KINGSGUARD VACANCY
#
# Expected scope:
# The ruler who possesses the Kingsguard.
# =========================================================

vladwyn_kingsguard_has_vacancy_trigger = {

	ruler_has_kingsguard_trigger = yes


	AND = {

		# At least one of the six ordinary Kingsguard
		# council positions is physically vacant.

		OR = {

			NOT = {
				exists = cp:kingsguard_1
			}

			NOT = {
				exists = cp:kingsguard_2
			}

			NOT = {
				exists = cp:kingsguard_3
			}

			NOT = {
				exists = cp:kingsguard_4
			}

			NOT = {
				exists = cp:kingsguard_5
			}

			NOT = {
				exists = cp:kingsguard_6
			}
		}


		# AGOT counts the six normal members plus
		# the Lord Commander = seven total.
		#
		# IMPORTANT:
		# PREV is the ruler from whose scope this
		# any_living_kingsguard iterator was entered.
		#
		# ROOT would still be the original decision
		# character when called from a decision.

		any_living_kingsguard = {

			top_liege = prev

			count < 7
		}
	}
}
