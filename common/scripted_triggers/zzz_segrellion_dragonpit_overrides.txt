# ============================================================
# THE CAVERNS OF VHAL
# Makes Caverns II+ count as a native AGOT dragonpit
# ============================================================

# Explicit level checks are deliberate. They avoid relying only on
# has_building_or_higher for a custom duchy-capital upgrade chain.
segrellion_vhal_dragonpit_building_active_trigger = {
	OR = {
		has_building = segrellion_vhal_dragonpit_02
		has_building = segrellion_vhal_dragonpit_03
		has_building_or_higher = segrellion_vhal_dragonpit_02
	}
}

# Character-scope fallback used by our own GUI/decisions even if another
# submod later overwrites one of AGOT's generic dragonpit triggers.
segrellion_has_active_vhal_dragonpit = {
	any_held_county = {
		this = title:c_blackmark
		any_county_province = {
			segrellion_vhal_dragonpit_building_active_trigger = yes
		}
		has_variable = has_dragonkeeper_order
	}
}


agot_can_use_dragonpit_amenities = {
	any_held_county = {
		OR = {
			any_county_province = {
				has_holding_type = castle_holding
				has_building_or_higher = generic_dragon_pit_01
			}

			any_county_province = {
				OR = {
					has_building_or_higher = dragonpit_01
					has_building = dragonpit_ruins_03
					has_building = agot_dragonmont_01

					# Caverns of Vhal
					segrellion_vhal_dragonpit_building_active_trigger = yes
				}
			}
		}

		has_variable = has_dragonkeeper_order
	}
}
