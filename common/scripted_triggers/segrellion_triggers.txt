# =========================================================
# ICEBRINGER
# =========================================================


# ---------------------------------------------------------
# CANON BEARERS
#
# These characters are known in canon to be capable of
# carrying Icebringer without being rejected.
# ---------------------------------------------------------

segrellion_icebringer_canon_bearer_trigger = {
	OR = {
		this = character:Segrellion_1       # Alex
		this = character:Vladwyn_1          # Tristan Vladwyn
		this = character:Segrellion_19      # Valerion
		this = character:Segrellion_21      # Aegon Dragonspear
		this = character:Segrellion_60      # Daemon Segrellion

		has_trait = segrellion_canon_alex_modifier
		has_trait = vladwyn_canon_tristan_modifier
		has_trait = segrellion_canon_valerion_modifier
		has_trait = segrellion_canon_aegon_dragonspear_modifier
		has_trait = segrellion_canon_daemon_bastard_unicorn_modifier

		# Aegon Reborn may be generated through Canon Children,
		# so his canon identity trait is safer than a fixed ID.
		has_trait = segrellion_canon_aegon_reborn_modifier
	}
}


# ---------------------------------------------------------
# CANON REJECTIONS
#
# Aeryn canonically cannot wield Icebringer without being
# burned by the blade.
# ---------------------------------------------------------

segrellion_icebringer_canon_rejected_trigger = {
	OR = {
		this = character:Segrellion_12
		has_trait = segrellion_canon_aeryn_modifier
	}
}


# ---------------------------------------------------------
# FULL ACCEPTANCE
# ---------------------------------------------------------

segrellion_icebringer_accepted_trigger = {
	NOR = {
		segrellion_icebringer_canon_rejected_trigger = yes
	}

	OR = {
		segrellion_icebringer_canon_bearer_trigger = yes

		segrellion_icebringer_affinity_value >= 8
	}
}


# ---------------------------------------------------------
# PARTIAL ACCEPTANCE
#
# The bearer receives Icebringer's full artifact bonuses,
# but the blade remains uncomfortable and unpredictable.
# ---------------------------------------------------------

segrellion_icebringer_partial_trigger = {
	NOR = {
		segrellion_icebringer_canon_bearer_trigger = yes
		segrellion_icebringer_canon_rejected_trigger = yes
	}

	segrellion_icebringer_affinity_value >= 4
	segrellion_icebringer_affinity_value < 8
}


# ---------------------------------------------------------
# REJECTION
# ---------------------------------------------------------

segrellion_icebringer_rejected_trigger = {
	OR = {
		segrellion_icebringer_canon_rejected_trigger = yes

		AND = {
			NOT = {
				segrellion_icebringer_canon_bearer_trigger = yes
			}

			segrellion_icebringer_affinity_value < 4
		}
	}
}


# ---------------------------------------------------------
# GAMEPLAY "CHOSEN"
#
# Both partial and complete acceptance unlock the full
# artifact modifier.
# ---------------------------------------------------------

segrellion_icebringer_chosen_trigger = {
	OR = {
		segrellion_icebringer_partial_trigger = yes
		segrellion_icebringer_accepted_trigger = yes
	}
}

segrellion_has_flamelands_hook_power_trigger = {
	OR = {
		title:k_the_hook = {
			any_de_jure_county = {
				count >= 8
				holder.top_liege = root
			}
		}
		any_vassal = {
			count >= 4
			any_realm_county = {
				de_jure_liege.de_jure_liege = title:k_the_hook
			}
		}
	}
}
