segrellion_rickon_return_jon_king_trigger = {
	segrellion_is_current_jon_trigger = yes
	is_alive = yes
	has_title = title:e_the_north
	has_character_flag = segrellion_jon_king_in_the_north
}

segrellion_rickon_return_flamelands_protector_trigger = {
	OR = {
		this = character:Segrellion_60
		this = character:Black_Unicorn_1
	}
	is_alive = yes
	is_ruler = yes
	has_title = title:e_flamelands
	is_imprisoned = no
	is_incapable = no
	is_at_war = no
	is_travelling = no
}
