# =========================================================
# NORTH
#
# The option remains visible while Jon rules the North,
# but cannot be selected until the Winterfell Council
# has resolved.
#
# The separate Dragonstone / Shireen storyline may already
# have opened negotiations with Jon. In that case we do not
# send a duplicate raven.
# =========================================================

segrellion_daenerys_north_support_shown_trigger = {

	NOT = {
		has_character_flag = segrellion_daenerys_support_north_approached
	}

	NOT = {
		has_character_flag = segrellion_daenerys_invited_davos_and_jon
	}

	NOT = {
		has_character_flag = segrellion_daenerys_jon_parley_invitation_sent
	}

	title:e_the_north.holder ?= {
		is_alive = yes
	}
}


segrellion_daenerys_north_support_available_trigger = {

	segrellion_daenerys_north_support_shown_trigger = yes


	title:e_the_north.holder ?= {

		is_alive = yes

		segrellion_is_current_jon_trigger = yes

		has_character_flag = segrellion_jon_winterfell_council_completed
	}
}


# =========================================================
# REACH
# =========================================================

segrellion_daenerys_reach_support_shown_trigger = {

	NOT = {
		has_character_flag = segrellion_daenerys_support_reach_approached
	}


	OR = {

		# Normal case: the Reach still has a living ruler.

		title:e_the_reach.holder ?= {
			is_alive = yes
		}


		# Even if the Reach title is in an unusual state,
		# keep the option visible when Aegon has married Margaery.
		# It will then be shown as blocked rather than disappearing.

		character:IAF_Young_Griff_1 ?= {

			is_alive = yes

			any_spouse = {
				this = character:Tyrell_14
			}
		}
	}
}


segrellion_daenerys_reach_support_available_trigger = {

	segrellion_daenerys_reach_support_shown_trigger = yes


	# Margaery is Aegon's queen.
	# The option remains visible, but cannot be selected.

	NOT = {

		character:IAF_Young_Griff_1 ?= {

			any_spouse = {
				this = character:Tyrell_14
			}
		}
	}
}

# =========================================================
# RIVERLANDS HELPERS
# =========================================================

segrellion_daenerys_riverlands_holder_is_jon_trigger = {

	OR = {

		title:e_the_riverlands.holder = character:Stark_7

		AND = {

			has_global_variable = segrellion_resurrected_jon

			title:e_the_riverlands.holder = global_var:segrellion_resurrected_jon
		}
	}
}


segrellion_daenerys_riverlands_holder_is_sansa_trigger = {

	title:e_the_riverlands.holder = {

		OR = {

			this = character:Stark_Dummy_8

			AND = {
				is_female = yes
				dynasty = dynasty:dynn_Stark
			}
		}
	}
}


segrellion_daenerys_riverlands_holder_is_tully_trigger = {

	title:e_the_riverlands.holder = {
		dynasty = dynasty:dynn_Tully
	}
}

	}

	NOT = {
		has_character_flag = segrellion_daenerys_promised_beacon_marriage
	}

	NOT = {
		has_character_flag = segrellion_daenerys_promised_beacon_council
	}

	NOT = {
		has_character_flag = segrellion_daenerys_promised_beacon_independence
	}

	NOT = {
		has_character_flag = segrellion_marriage_promised_to_daemon
	}

	is_married = no

	character:Segrellion_60 ?= {

		is_alive = yes
		is_landed = yes

		segrellion_daemon_controls_flamelands_trigger = yes

		NOT = {
			has_character_flag = segrellion_pledged_to_daenerys
		}
	}
}


# =========================================================
# STORMLANDS HELPERS
# =========================================================

segrellion_daenerys_stormlands_holder_is_roberts_bastard_trigger = {

	title:e_the_stormlands.holder = {

		OR = {
			has_trait = bastard
			has_trait = legitimized_bastard
		}

		father ?= {
			dynasty = dynasty:dynn_Baratheon
		}
	}
}


segrellion_daenerys_stormlands_holder_negotiable_trigger = {

	OR = {

		title:e_the_stormlands.holder = character:Vladwyn_5

		segrellion_daenerys_stormlands_holder_is_roberts_bastard_trigger = yes
	}
}

# =========================================================
# DAENERYS MARRIAGE PROMISE AVAILABILITY
#
# A lord may still ASK for marriage without knowing that
# Daenerys has promised herself elsewhere.
#
# This trigger only determines whether SHE may accept.
# =========================================================

segrellion_daenerys_can_accept_support_marriage_trigger = {

	is_married = no

	NOT = {
		has_trait = refusing_marriage
	}

	NOT = {
		has_character_flag = segrellion_marriage_promised_to_daemon
	}

	NOT = {
		has_character_flag = segrellion_daenerys_promised_beacon_marriage
	}

	NOT = {
		has_character_flag = segrellion_daenerys_support_marriage_already_promised
	}
}


# =========================================================
# ANY ROUTE SHOWN
# =========================================================

segrellion_daenerys_has_westeros_support_route_trigger = {

	OR = {

		segrellion_daenerys_north_support_shown_trigger = yes

		segrellion_daenerys_reach_support_shown_trigger = yes

		segrellion_daenerys_westerlands_support_shown_trigger = yes

		segrellion_daenerys_dorne_support_shown_trigger = yes

		segrellion_daenerys_riverlands_support_shown_trigger = yes

		segrellion_daenerys_flamelands_support_available_trigger = yes

		segrellion_daenerys_stormlands_support_shown_trigger = yes
	}
}
