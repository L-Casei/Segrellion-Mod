# =========================================================
# VISENYA - PART III
# =========================================================
#
# Territorial truth:
# Daemon must actually rule the Flamelands.
#
# Story truth:
# The Unicorn Returns was completed OR an older Visenya
# Part III route flag already exists.
#
# This deliberately avoids depending on fragile pairs of
# flags from older versions.
# =========================================================

segrellion_black_unicorn_part_iii_eligible_trigger = {

	this = character:Segrellion_60
	is_alive = yes

	has_title = title:e_flamelands


	# =====================================================
	# STORY HAS REACHED THE POINT WHERE PART III CAN EXIST
	# =====================================================

	OR = {

		# Canonical case.
		has_character_flag = segrellion_unicorn_returns_completed

		# Unresolved earlier routes now deliberately feed Part III.
		has_character_flag = segrellion_black_unicorn_future_post_war_return
		has_character_flag = segrellion_black_unicorn_future_beacon_ambush
		has_character_flag = segrellion_black_unicorn_left_without_truth
		has_character_flag = segrellion_black_unicorn_left_skagos_without_visenya
		has_character_flag = segrellion_black_unicorn_returned_to_camp_without_visenya
		has_character_flag = segrellion_black_unicorn_visenya_declined_first_return

		# Save compatibility with older Visenya implementations.
		has_character_flag = segrellion_black_unicorn_part_iii_pending
		has_character_flag = segrellion_black_unicorn_part_iii_after_flamelands
		has_character_flag = segrellion_black_unicorn_route_d_resolved
		has_character_flag = segrellion_black_unicorn_route_c_destiny
		has_character_flag = segrellion_black_unicorn_destiny_route_pending
		has_character_flag = segrellion_black_unicorn_part_iii_available
	}


	# =====================================================
	# PART III HAS NOT ALREADY HAPPENED
	# =====================================================

	NOT = {
		has_character_flag = segrellion_black_unicorn_post_beacon_resolved
	}

	NOT = {
		has_character_flag = segrellion_black_unicorn_daemon_abducted_in_beacon
	}

	NOT = {
		has_character_flag = segrellion_black_unicorn_visenya_returned_with_daemon
	}


	# =====================================================
	# VISENYA STILL EXISTS AS THE FUTURE THREAT
	# =====================================================

	exists = character:Black_Unicorn_1

	character:Black_Unicorn_1 = {

		is_alive = yes


		NOT = {
			has_character_flag = segrellion_black_unicorn_visenya_joined_daemon
		}


		# =================================================
		# HUGH CUSTOM ROUTES
		#
		# Normal Part III must not fire while Hugh has
		# Visenya hidden at Vhal, is preparing a coup,
		# or is running Daemon's counter-trap.
		# =================================================

		NOR = {

			has_character_flag = segrellion_hugh_route_no_ice_vhal_pending

			has_character_flag = segrellion_hugh_route_peaceful_vhal

			has_character_flag = segrellion_hugh_route_coup_vhal

			has_character_flag = segrellion_hugh_route_daemon_trap

			has_character_flag = segrellion_hugh_route_daemon_trap_active

			has_character_flag = segrellion_hugh_route_vhal_guest
		}
	}
}


# =========================================================
# SOFT / NEGOTIATED PATH III ENTRY
#
# These outcomes mean Daemon and Visenya still have
# unfinished business, but not a reason to begin the Beacon
# reunion with murder, binding or a forced duel.
# =========================================================

segrellion_black_unicorn_part_iii_soft_entry_trigger = {

	OR = {

		has_character_flag = segrellion_black_unicorn_future_post_war_return
		has_character_flag = segrellion_black_unicorn_future_beacon_ambush
		has_character_flag = segrellion_black_unicorn_left_without_truth
		has_character_flag = segrellion_black_unicorn_left_skagos_without_visenya
		has_character_flag = segrellion_black_unicorn_returned_to_camp_without_visenya
		has_character_flag = segrellion_black_unicorn_visenya_declined_first_return
		has_character_flag = segrellion_black_unicorn_daemon_abandoned_after_warning
		has_character_flag = segrellion_black_unicorn_trail_gone_cold
		has_character_flag = segrellion_black_unicorn_route_a_refused_condition
		has_character_flag = segrellion_black_unicorn_route_a_failed
		has_character_flag = segrellion_black_unicorn_route_b_abandoned
		has_character_flag = segrellion_black_unicorn_shared_night
		has_character_flag = segrellion_black_unicorn_visenya_trust_earned
		has_character_flag = segrellion_black_unicorn_daemon_knows_visenya_identity
	}
}


# =========================================================
# READY TO ACTUALLY OPEN PATH III
# =========================================================

segrellion_black_unicorn_part_iii_ready_trigger = {

	segrellion_black_unicorn_part_iii_eligible_trigger = yes

	has_character_flag = segrellion_black_unicorn_post_beacon_timer_v4

	NOT = {
		has_character_flag = segrellion_black_unicorn_post_beacon_started
	}

	NOT = {
		has_character_flag = segrellion_black_unicorn_path_iii_variant_started
	}

	is_imprisoned = no

	character:Black_Unicorn_1 = {
		is_imprisoned = no
	}
}

# =========================================================
# VISENYA - PRE-STORY MARRIAGE LOCK
#
# Until the Black Unicorn storyline has reached a real
# resolution, Visenya is reserved for the Daemon storyline.
# Once Daemon is dead or the story is resolved, the lock
# disappears automatically.
# =========================================================

segrellion_visenya_marriage_story_locked_trigger = {

	this = character:Black_Unicorn_1

	is_alive = yes


	character:Segrellion_60 ?= {

		is_alive = yes


		NOT = {
			has_character_flag = segrellion_black_unicorn_post_beacon_resolved
		}


		NOT = {
			has_character_flag = segrellion_black_unicorn_dungeon_arc_resolved
		}
	}
}
