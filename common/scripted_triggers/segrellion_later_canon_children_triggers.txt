iaf_later_canon_identity_exists_trigger = {
	OR = {
		# Historical character loaded directly from history.
		character:$ID$ ?= {
			exists = yes
			is_alive = yes
		}

		# Character born dynamically through AGOT Canon Children.
		any_in_global_list = {
			variable = agot_born_canon_children_scopes
			has_character_flag = is_$ID$
			is_alive = yes
		}
	}
}
