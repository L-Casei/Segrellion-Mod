	construction_time = very_slow_construction_time
	cost_gold = 500
	type_icon = "icon_structure_firstflame.dds"

	asset = {
		type = pdxmesh
		name = "building_western_castle_01_mesh"
		illustration = @segrellion_vhalerion_holding_illustration
		provinces = { 4427 }
	}

	can_construct_potential = {
		building_requirement_tribal = no
		barony = title:b_berryport
	}

	can_construct = {
		trigger_if = {
			limit = {
				NOT = { scope:holder = county.holder }
			}
			county = { holder.dynasty = dynasty:dynn_Segrellion }
			scope:holder = { dynasty = dynasty:dynn_Segrellion }
		}
		trigger_else = {
			scope:holder = { dynasty = dynasty:dynn_Segrellion }
		}
	}

	is_enabled = {
		agot_generic_is_special_building_enabled = yes
	}

	province_modifier = {
		fort_level = 2
		monthly_income = 1.5
		garrison_size = 0.25
		defender_holding_advantage = 3
	}

	county_modifier = {
		development_growth_factor = 0.05
	}

	character_modifier = {
		monthly_prestige = 0.10
		monthly_dread = 0.15
	}

	effect_desc = segrellion_firstflame_dynasty_bonus_01_effect_desc

	next_building = segrellion_aeryns_den_02
