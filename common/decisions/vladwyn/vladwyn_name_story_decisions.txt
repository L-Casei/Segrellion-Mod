# =========================================================
# REQUEST ACCESS TO STORM'S END
# =========================================================

vladwyn_request_storms_end_records_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/tristan_vladwyn_landless_decision_background.dds"
	}
	decision_group_type = ice_and_fire

	desc = vladwyn_request_storms_end_records_decision_desc
	selection_tooltip = vladwyn_request_storms_end_records_decision_tooltip
	confirm_text = vladwyn_request_storms_end_records_decision_confirm


	is_shown = {

		has_character_flag = is_Vladwyn_1

		has_character_flag = vladwyn_storms_end_investigation_unlocked

		NOT = {
			has_character_flag = vladwyn_boremund_discovered
		}

		exists = title:b_storms_end.holder

		title:b_storms_end.holder = {

			NOT = {
				has_character_flag = vladwyn_tristan_storms_end_permission
			}

			NOT = {
				has_character_flag = vladwyn_tristan_storms_end_barred
			}
		}
	}


	is_valid = {

		custom_description = {

			text = vladwyn_storms_end_not_on_cooldown_requirement

			title:b_storms_end.holder = {

				NOT = {
					has_character_flag = vladwyn_tristan_storms_end_request_cooldown
				}
			}
		}
	}


	effect = {

		save_scope_as = vladwyn_tristan

		title:b_storms_end.holder = {
			save_scope_as = vladwyn_storm_lord
		}

		trigger_event = vladwyn_name.4000
	}


	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
