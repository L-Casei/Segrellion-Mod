vladwyn_take_to_open_road_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/tristan_vladwyn_landless_decision_background.dds"
	}
	decision_group_type = ice_and_fire

	desc = vladwyn_take_to_open_road_decision_desc
	selection_tooltip = vladwyn_take_to_open_road_decision_tooltip

	is_shown = {
		this = character:Vladwyn_1
		has_dlc_feature = roads_to_power
		OR = {
			has_character_flag = vladwyn_landless_route_unlocked
			is_landless_adventurer = yes
			has_government = landless_adventurer_government
		}
		NOT = { has_character_flag = vladwyn_open_road_taken }
	}

	is_valid_showing_failures_only = {
		is_adult = yes
		is_imprisoned = no
	}

	effect = {
		custom_tooltip = vladwyn_take_to_open_road_effect_tt
		trigger_event = vladwyn_adventurer.0001
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}

	picture = {
		reference = "gfx/interface/illustrations/decisions/tristan_vladwyn_landless_decision_background.dds"
	}
	decision_group_type = ice_and_fire

	desc = vladwyn_gather_men_to_banner_decision_desc
	selection_tooltip = vladwyn_gather_men_to_banner_decision_tooltip

	is_shown = {
		this = character:Vladwyn_1
		has_dlc_feature = roads_to_power
		has_character_flag = vladwyn_open_road_taken
		NOT = { has_character_flag = vladwyn_gathered_banner_men }
	}

	is_valid = {
		prestige_level >= 2
		num_character_followers >= 15
		maa_regiments_count >= 2
		OR = {
			custom_tooltip = {
				text = vladwyn_has_2500_soldiers_tt
				current_military_strength >= 2500
			}
			AND = {
				custom_description = {
					text = vladwyn_completed_justicar_contracts_tt
					var:justicar_contracts_succeeded_counter >= 10
				}
				custom_tooltip = {
					text = vladwyn_has_500_soldiers_tt
					current_military_strength >= 500
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_adult = yes
		is_imprisoned = no
	}

	effect = {
		custom_tooltip = vladwyn_gather_men_to_banner_effect_tt
		trigger_event = vladwyn_adventurer.0002
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
