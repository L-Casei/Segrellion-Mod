# =========================================================
# DAEMON SEGRELLION
# THE UNICORN RETURNS
# =========================================================

segrellion_the_unicorn_returns_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/proclaim_paramountcy_of_the_flamelands_decision_background.dds"
	}

	decision_group_type = major

	desc = segrellion_the_unicorn_returns_decision_desc
	selection_tooltip = segrellion_the_unicorn_returns_decision_tooltip
	confirm_text = segrellion_the_unicorn_returns_decision_confirm


	# =====================================================
	# COST
	# 500 Prestige normally.
	# Free in debug for testing.
	# =====================================================

	cost = {
		prestige = {
			value = 0

			if = {
				limit = {
					NOT = {
						debug_only = yes
					}
				}

				add = 500
			}
		}
	}


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {

		# This storyline decision belongs specifically to Daemon.
		this = character:Segrellion_60

		# In debug, ignore all storyline visibility restrictions.
		trigger_if = {
			limit = {
				NOT = {
					debug_only = yes
				}
			}

			current_date >= 8300.1.1

			NOT = {
				has_character_flag = segrellion_unicorn_returns_completed
			}

			NOT = {
				has_character_flag = segrellion_unicorn_returns_event_pending
			}

			NOT = {
				has_character_flag = segrellion_unicorn_returns_war_active
			}
		}
	}


	# =====================================================
	# REQUIREMENTS
	# All bypassed while running in debug mode.
	# =====================================================

	is_valid = {

		trigger_if = {
			limit = {
				NOT = {
					debug_only = yes
				}
			}


			is_adult = yes
			is_imprisoned = no

			# =================================================
			# AI STORY ORDER
			#
			# Player Daemon remains free to play differently.
			# AI Daemon follows his intended historical route.
			# =================================================

			trigger_if = {

				limit = {
					is_ai = yes
				}

				has_character_flag = segrellion_foreign_support_secured
			}


			# =================================================
			# FLAMELANDS MUST NOT ALREADY BE SEGRELLION
			# =================================================

			custom_description = {
				text = segrellion_unicorn_returns_not_segrellion_ruled_tt

				title:e_flamelands = {

					exists = holder

					holder = {
						NOT = {
							house = house:house_Segrellion
						}
					}
				}
			}


			# =================================================
			# DAEMON MUST NOT ALREADY BE FIGHTING FOR THEM
			# =================================================

			custom_description = {
				text = segrellion_unicorn_returns_not_at_war_tt

				NOT = {
					any_character_war = {

						casus_belli = {

							any_target_title = {

								OR = {
									this = title:e_flamelands
									this = title:k_flamelands
								}
							}
						}
					}
				}
			}


			# =================================================
			# RESTORATION REQUIREMENT
			#
			# AT LEAST ONE:
			#   - Trueflame + Plate of Segrell
			#   - Icebringer
			#   - 4,000 soldiers
			# =================================================

			OR = {

				# ---------------------------------------------
				# TRUEFLAME + PLATE OF SEGRELL
				# ---------------------------------------------

				AND = {

					custom_description = {
						text = segrellion_unicorn_returns_has_trueflame_tt

						any_character_artifact = {
							has_variable = segrellion_trueflame_artifact
						}
					}

					custom_description = {
						text = segrellion_unicorn_returns_has_plate_tt

						any_character_artifact = {
							has_variable = segrellion_plate_of_segrell_artifact
						}
					}
				}


				# ---------------------------------------------
				# ICEBRINGER
				# ---------------------------------------------

				custom_description = {
					text = segrellion_unicorn_returns_has_icebringer_tt

					any_character_artifact = {
						has_variable = segrellion_icebringer_artifact
					}
				}


				# ---------------------------------------------
				# 3,000 SOLDIERS
				# ---------------------------------------------

				custom_description = {
					text = segrellion_unicorn_returns_host_ready_tt

					current_military_strength >= 3000
				}
			}


		}
	}


	# =====================================================
	# EFFECT
	# =====================================================

	effect = {

		custom_tooltip = segrellion_unicorn_returns_effect_war_tt
		custom_tooltip = segrellion_unicorn_returns_effect_allies_tt
		custom_tooltip = segrellion_unicorn_returns_effect_modifier_tt
		custom_tooltip = segrellion_unicorn_returns_effect_riders_tt


		hidden_effect = {

			remove_character_modifier = segrellion_daemon_preparing_restoration_ai_modifier
			
			# Permanently records that Daemon has committed
			# to The Unicorn Returns.
			add_character_flag = segrellion_unicorn_returns_decision_taken

			add_character_flag = segrellion_unicorn_returns_event_pending

			trigger_event = {
				id = segrellion_winds.0001
				days = 1
			}
		}
	}


	# =====================================================
	# AI
	# =====================================================

	ai_check_interval = 1

	ai_potential = {

		this = character:Segrellion_60
		is_ai = yes

		has_character_flag = segrellion_foreign_support_secured


		# =====================================================
		# PLATE OF SEGRELL
		#
		# Daemon only waits for the Plate negotiation if Hugh
		# ALREADY possesses the Plate, meaning the demand can
		# actually be made immediately.
		#
		# If Hugh has not yet recovered the Plate, the artifact
		# storyline no longer blocks The Unicorn Returns.
		# =====================================================

		trigger_if = {

			limit = {

				character:Segrellion_43 ?= {

					is_alive = yes

					any_character_artifact = {

						has_variable = segrellion_plate_of_segrell_artifact
					}
				}
			}

			has_character_flag = segrellion_plate_demand_resolved
		}
	}

	ai_will_do = {
		base = 10000
	}
}

# =========================================================
# HUGH SEGRELLION
# RECOVER THE PLATE OF SEGRELL
# =========================================================

segrellion_recover_plate_of_segrell_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/proclaim_paramountcy_of_the_flamelands_decision_background.dds"
	}

	decision_group_type = major

	desc = segrellion_recover_plate_of_segrell_decision_desc
	selection_tooltip = segrellion_recover_plate_of_segrell_decision_tooltip
	confirm_text = segrellion_recover_plate_of_segrell_decision_confirm


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {

		this = character:Segrellion_43

		has_character_flag = segrellion_plate_recovery_ready

		NOT = {
			has_character_flag = segrellion_plate_recovery_in_progress
		}

		NOT = {
			has_character_flag = segrellion_plate_recovery_complete
		}

		NOT = {
			segrellion_plate_recovery_blocked_trigger = yes
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		is_adult = yes
		is_imprisoned = no

		gold >= 200


		custom_description = {

			text = segrellion_recover_plate_not_already_recovered_tt

			NOT = {
				segrellion_plate_recovery_blocked_trigger = yes
			}
		}
	}


	# =====================================================
	# COST
	#
	# This is NOT a purchase price.
	# It covers couriers, guards, ships, wagons,
	# harbour fees and secure transport.
	# =====================================================

	cost = {
		gold = 50
	}


	# =====================================================
	# EFFECT
	# =====================================================

	effect = {

		custom_tooltip = segrellion_recover_plate_transport_tt


		hidden_effect = {

			remove_character_flag = segrellion_plate_recovery_ready

			add_character_flag = segrellion_plate_recovery_in_progress


			trigger_event = {
				id = segrellion_winds.0130
				days = { 20 60 }
			}
		}
	}


	# =====================================================
	# AI
	# =====================================================

	ai_check_interval = 7

	ai_potential = {

		this = character:Segrellion_43

		has_character_flag = segrellion_plate_recovery_ready

		NOT = {
			has_character_flag = segrellion_plate_recovery_in_progress
		}

		NOT = {
			has_character_flag = segrellion_plate_recovery_complete
		}

		NOT = {
			segrellion_plate_recovery_blocked_trigger = yes
		}

		gold >= 200
	}

	ai_will_do = {
		base = 1000
	}
}
