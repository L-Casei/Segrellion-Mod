segrellion_attempt_to_awaken_icebringer_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/awaken_icebringer_decision_background.dds"
	}

	desc = segrellion_attempt_to_awaken_icebringer_decision_desc
	selection_tooltip = segrellion_attempt_to_awaken_icebringer_decision_tooltip


		# =========================================================
		# VISIBILITY
		#
		# If you do not personally possess Icebringer,
		# the decision does not exist for you.
		# =========================================================

		is_shown = {
			OR = {
				debug_only = yes

				AND = {

					any_character_artifact = {
						has_variable = segrellion_icebringer_artifact
					}

					NOT = {
						has_global_variable = segrellion_icebringer_has_awakened
					}

					NOT = {
						has_character_flag = segrellion_icebringer_awakening_chain_active
					}

					NOT = {
						has_character_flag = segrellion_icebringer_awakening_succeeded
					}
					NOT = {
						has_character_flag = segrellion_icebringer_awaken_path_closed
					}
					NOT = {
						has_character_flag = segrellion_icebringer_stage5_entry_reached
					}
				}
			}
		}


		# =========================================================
		# ONLY REQUIREMENTS
		# =========================================================

		is_valid = {

			OR = {
				debug_only = yes

				AND = {
					is_adult = yes

					custom_description = {
						text = segrellion_icebringer_equipped_tt

						hidden_trigger = {
							any_equipped_character_artifact = {
								has_variable = segrellion_icebringer_artifact
							}
						}
					}

					NOT = {
						has_character_flag = segrellion_icebringer_decision_cooldown
					}
				}
			}

			# ---------------------------------------------------------
			# YOU PERSONALLY HOLD ICEBRINGER
			# ---------------------------------------------------------

			custom_description = {
				text = segrellion_icebringer_holder_requirement

				global_var:segrellion_icebringer = {
					artifact_owner = root
				}
			}


			# ---------------------------------------------------------
			# ICEBRINGER HAS CHOSEN YOU
			# ---------------------------------------------------------

			custom_description = {
				text = segrellion_icebringer_chosen_desc
				segrellion_icebringer_chosen_trigger = yes
			}
		}


		effect = {

			add_character_flag = {
				flag = segrellion_icebringer_decision_cooldown
				years = 1
			}

			# =========================================================
			# WARNING
			# =========================================================

			custom_tooltip = segrellion_awaken_icebringer_danger_tt


			# =========================================================
			# CROSS THE WAKING VEIL
			# =========================================================

			add_character_modifier = {
				modifier = segrellion_beyond_waking_veil_modifier
				days = 30
			}


			# =========================================================
			# BEGIN / RESUME AWAKENING V2
			# =========================================================

			add_character_flag = segrellion_icebringer_awakening_chain_active


			# =========================================================
			# RESUME AWAKENING
			#
			# Route locks always take priority.
			# =========================================================


			# ---------------------------------------------------------
			# STAGE IV - GUARDIAN
			# ---------------------------------------------------------

			if = {
				limit = {
					has_character_flag = segrellion_icebringer_stage4_guardian_locked
				}

				trigger_event = {
					id = segrellion_icebringer.7000
					days = 1
				}
			}


			# ---------------------------------------------------------
			# STAGE IV - AZOR AHAI
			# ---------------------------------------------------------

			else_if = {
				limit = {
					has_character_flag = segrellion_icebringer_stage4_azor_locked
				}

				trigger_event = {
					id = segrellion_icebringer.7100
					days = 1
				}
			}


			# ---------------------------------------------------------
			# STAGE IV - DARKNESS
			# ---------------------------------------------------------

			else_if = {
				limit = {
					has_character_flag = segrellion_icebringer_stage4_darkness_locked
				}

				trigger_event = {
					id = segrellion_icebringer.7200
					days = 1
				}
			}


			# ---------------------------------------------------------
			# LEGACY / SAFETY REPAIR
			#
			# A save from the old system might have Stage III passed
			# without possessing one of the new route locks.
			#
			# Send that character back to Stage III.
			# ---------------------------------------------------------

			else_if = {
				limit = {
					has_character_flag = segrellion_icebringer_v2_stage_three_passed
				}

				remove_character_flag = segrellion_icebringer_v2_stage_three_passed

				trigger_event = {
					id = segrellion_icebringer.6500
					days = 1
				}
			}


			# ---------------------------------------------------------
			# STAGE II COMPLETED
			# Resume Stage III
			# ---------------------------------------------------------

			else_if = {
				limit = {
					has_character_flag = segrellion_icebringer_v2_stage_two_passed
				}

				trigger_event = {
					id = segrellion_icebringer.6500
					days = 1
				}
			}


			# ---------------------------------------------------------
			# STAGE I COMPLETED
			# Resume Stage II
			# ---------------------------------------------------------

			else_if = {
				limit = {
					has_character_flag = segrellion_icebringer_v2_stage_one_passed
				}

				trigger_event = {
					id = segrellion_icebringer.6400
					days = 1
				}
			}


			# ---------------------------------------------------------
			# NO PROGRESS
			# Begin Stage I
			# ---------------------------------------------------------

			else = {

				trigger_event = {
					id = segrellion_icebringer.6000
					days = 1
				}
			}
		}


		ai_check_interval = 0

		ai_potential = {
			always = no
		}

		ai_will_do = {
			base = 0
		
		}
}

	# VISIBILITY
	# =====================================================

	is_shown = {

		OR = {

			# -------------------------------------------------
			# DEBUG MODE
			#
			# Always visible. Ignores date, Icebringer,
			# knowledge and active-search state.
			# -------------------------------------------------

			debug_only = yes


			# -------------------------------------------------
			# NORMAL GAME
			# -------------------------------------------------

			AND = {

				is_adult = yes

				segrellion_can_search_legendary_relic_trigger = yes

				# Not before 196 AC.
				current_date >= 8196.1.1

				# The actual artifact must be lost.
				NOT = {
					segrellion_icebringer_exists_anywhere_trigger = yes
				}

				# Cannot start another search while one is active.
				NOT = {
					has_character_flag = segrellion_legendary_relic_search_active
				}
			}
		}
	}


		# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		OR = {

			# DEBUG MODE:
			# usable regardless of normal requirements.
			debug_only = yes


			# NORMAL GAME
			AND = {

				is_imprisoned = no

				NOT = {
					has_trait = incapable
				}

				custom_description = {
					text = segrellion_search_relic_knowledge_requirement

					segrellion_can_search_legendary_relic_trigger = yes
				}

				custom_description = {
					text = segrellion_search_relic_icebringer_lost_requirement

					NOT = {
						segrellion_icebringer_exists_anywhere_trigger = yes
					}
				}

				custom_description = {
					text = segrellion_search_relic_cooldown_requirement

					NOT = {
						has_character_flag = segrellion_legendary_relic_search_cooldown
					}
				}
			}
		}
	}


	# =====================================================
	# CHOOSE THE FIRST TRAIL
	# =====================================================

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button = "segrellion_search_for_legendary_relic_decision_next_step"


		# =====================================================
		# TYROSH
		# =====================================================

		item = {
			value = segrellion_relic_choice_tyrosh

			current_description = segrellion_relic_choice_tyrosh_desc
			localization = segrellion_relic_choice_tyrosh

			icon = "gfx/interface/icons/message_feed/intrigue.dds"

			is_valid = {
				always = yes
			}

			ai_chance = {
				value = 30
			}
		}


		# =====================================================
		# OLDTOWN
		# =====================================================

		item = {
			value = segrellion_relic_choice_oldtown

			current_description = segrellion_relic_choice_oldtown_desc
			localization = segrellion_relic_choice_oldtown

			icon = "gfx/interface/icons/message_feed/learning.dds"

			is_valid = {
				always = yes
			}

			ai_chance = {
				value = 30
			}
		}

		# WINTERFELL
		# =====================================================

			value = segrellion_relic_choice_winterfell

			current_description = segrellion_relic_choice_winterfell_desc
			localization = segrellion_relic_choice_winterfell

			icon = "gfx/interface/icons/message_feed/martial.dds"

			is_valid = {
				always = yes
			}

			ai_chance = {
				value = 40
			}
		}
	}


	# =====================================================
	# BEGIN SEARCH
	# =====================================================

	effect = {

		# Displayed as an actual effect before the cooldown notice.
		custom_tooltip = segrellion_search_relic_multiple_journeys_effect_tt

		# =====================================================
		# DEBUG CLEANUP
		#
		# Allows restarting a broken search from scratch.
		# =====================================================

		if = {

			limit = {
				debug_only = yes
			}

			remove_character_flag = segrellion_legendary_relic_search_active
			remove_character_flag = segrellion_legendary_relic_search_paused

			remove_character_flag = segrellion_relic_route_tyrosh
			remove_character_flag = segrellion_relic_route_oldtown
			remove_character_flag = segrellion_relic_route_winterfell

		# BEGIN SEARCH
		# =====================================================

		add_character_flag = segrellion_legendary_relic_search_active

		# Debug ignores this because the Debug branch bypasses

				on_arrival_destinations = first

				on_travel_planner_cancel_event = segrellion_relic.0090
			}
		}


		# =====================================================

		# =====================================================

		else_if = {
			limit = {
				scope:segrellion_relic_choice_oldtown = yes
			}

				on_arrival_destinations = first

				on_travel_planner_cancel_event = segrellion_relic.0090
			}
		}

		# =====================================================

		else_if = {

			limit = {
				scope:segrellion_relic_choice_winterfell = yes
			}

			segrellion_clear_relic_pending_destination_effect = yes

				destination = title:c_winterfell.title_province

				# The northern search continues from Winterfell.
				# Do NOT automatically travel back home.
				return_trip = no

				# This is a single outbound journey.
				on_arrival_destinations = first

				on_travel_planner_cancel_event = segrellion_relic.0090
			}
		}
	}


	ai_check_interval = 0

	ai_potential = {
		always = no
	}

	ai_will_do = {

	}

# ESTABLISH THE GREATER FLAMELANDS

segrellion_establish_greater_flamelands_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/proclaim_paramountcy_of_the_flamelands_decision_background.dds"
	}

	decision_group_type = major

	desc = segrellion_establish_greater_flamelands_decision_desc
	selection_tooltip = segrellion_establish_greater_flamelands_decision_tooltip
	confirm_text = segrellion_establish_greater_flamelands_decision_confirm


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {

		# Available to anyone who holds the Paramountcy.
		has_title = title:e_flamelands

		# Can only ever be established once.
		title:e_flamelands = {
			NOT = {
				has_variable = segrellion_greater_flamelands_established
			}
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		is_adult = yes
		is_imprisoned = no


		# -------------------------------------------------
		# TERRITORIAL POWER
		#
		# EITHER:
		#   Dragonstone + The Claw + Flamelands
		#
		# OR:
		#   Paramountcy of the Flamelands
		#   + Dragonstone OR The Claw
		# -------------------------------------------------

		OR = {

			AND = {

				custom_description = {
					text = segrellion_greater_flamelands_dragonstone_requirement_tt
					has_title = title:k_dragonstone
				}

				custom_description = {
					text = segrellion_greater_flamelands_claw_requirement_tt
					has_title = title:k_the_hook
				}

				custom_description = {
					text = segrellion_greater_flamelands_flamelands_requirement_tt
					has_title = title:k_flamelands
				}
			}


			AND = {

				custom_description = {
					text = segrellion_greater_flamelands_paramountcy_requirement_tt
					has_title = title:e_flamelands
				}

				OR = {

					custom_description = {
						text = segrellion_greater_flamelands_dragonstone_requirement_tt
						has_title = title:k_dragonstone
					}

					custom_description = {
						text = segrellion_greater_flamelands_claw_requirement_tt
						has_title = title:k_the_hook
					}
				}
			}
		}


		# -------------------------------------------------
		# EXALTED AMONGST MEN
		# -------------------------------------------------

		prestige_level >= 4


		# -------------------------------------------------
		# IRON THRONE APPROVAL OR INDEPENDENCE
		# -------------------------------------------------

		OR = {

			is_independent_ruler = yes

			custom_description = {
				text = segrellion_greater_flamelands_iron_throne_opinion_tt

				AND = {

					title:h_the_iron_throne = {
						exists = holder
					}

					title:h_the_iron_throne.holder = {
						opinion = {
							target = root
							value >= 100
						}
					}
				}
			}
		}


		# -------------------------------------------------
		# PRESTIGE COST
		# -------------------------------------------------

		prestige >= 2500
	}


	# =====================================================
	# EFFECTS
	# =====================================================

	effect = {

		custom_tooltip = segrellion_establish_greater_flamelands_effect_tt

		hidden_effect = {

			# ---------------------------------------------
			# GREAT FOUNDER
			# Permanent character modifier.
			# ---------------------------------------------

			add_character_modifier = {
				modifier = segrellion_great_founder_modifier
			}


			# ---------------------------------------------
			# MAXIMUM LEGITIMACY
			# Only attempt this on characters with a valid
			# legitimacy type.
			# ---------------------------------------------

			if = {
				limit = {
					has_legitimacy = yes
				}

				add_legitimacy = 1500
			}


			# ---------------------------------------------
			# EXALTED AMONGST MEN -> LIVING LEGEND
			#
			# If already Living Legend, do nothing.
			# ---------------------------------------------

			if = {
				limit = {
					prestige_level = 4
				}

				add_prestige_level = 1
			}


			# ---------------------------------------------
			# CROWN OF THE FLAMEKING
			# ---------------------------------------------

			segrellion_create_artifact_crown_of_the_flameking_effect = {
				OWNER = this
			}


			# ---------------------------------------------
			# THE GREATER FLAMELANDS NOW EXIST
			# ---------------------------------------------

			title:e_flamelands = {
				set_variable = {
					name = segrellion_greater_flamelands_established
					value = yes
				}
			}
		}
	}

	trigger_event = {
		id = segrellion_decision.0007
	}


	# =====================================================
	# COST
	# =====================================================

	cost = {
		prestige = 2500
	}


	# =====================================================
	# AI
	# =====================================================

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
