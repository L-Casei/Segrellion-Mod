segrellion_search_for_trueflame_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/segrellion_search_legendary_relic_decision_background.dds"
	}

	decision_group_type = ice_and_fire

	desc = segrellion_search_for_trueflame_decision_desc
	selection_tooltip = segrellion_search_for_trueflame_decision_tooltip
	confirm_text = segrellion_search_for_trueflame_decision_confirm

	cost = {
		gold = 200
	}

	is_shown = {
		is_adult = yes
		house = house:house_Segrellion
		is_house_head = yes

		has_global_variable = segrellion_trueflame_story_known

		NOT = {
			exists = global_var:segrellion_trueflame
		}

		NOT = {
			has_character_flag = segrellion_trueflame_search_active
		}
	}

	is_valid = {
		is_landed = yes
		primary_title.tier >= tier_county
		is_imprisoned = no
		is_travelling = no

		NOT = {
			has_trait = incapable
		}

		custom_description = {
			text = segrellion_trueflame_prestige_requirement
			prestige >= 1500
		}

		custom_description = {
			text = segrellion_trueflame_skill_requirement

			OR = {
				martial >= 10
				intrigue >= 10
			}
		}
	}

	effect = {
		custom_tooltip = segrellion_search_trueflame_effect_start_tt
		custom_tooltip = segrellion_search_trueflame_effect_agents_tt
		custom_tooltip = segrellion_search_trueflame_effect_reward_tt

		hidden_effect = {
			add_character_flag = segrellion_trueflame_search_active
			trigger_event = {
				id = segrellion_trueflame.0100
			}
		}
	}

	ai_check_interval = 24

	ai_potential = {
		house = house:house_Segrellion
		is_house_head = yes
		is_landed = yes
	}

	ai_will_do = {
		base = 35
	}
}
