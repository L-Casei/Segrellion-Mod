# =========================================================
# TAKE DRAGONSTONE
#
# Daenerys has already landed elsewhere in Westeros.
# Once Stannis is dead, she may return to the island of
# her birth and claim Dragonstone for herself.
# =========================================================

segrellion_daenerys_take_dragonstone_decision = {

	picture = {
		reference = "gfx/interface/illustrations/event_scenes/segrellion_dragonstone_arrival.dds"
	}

	decision_group_type = ice_and_fire

	desc = segrellion_daenerys_take_dragonstone_decision_desc
	selection_tooltip = segrellion_daenerys_take_dragonstone_decision_tooltip
	confirm_text = segrellion_daenerys_take_dragonstone_decision_confirm


	# =====================================================
	# SHOWN
	# =====================================================

	is_shown = {

		this = character:Targaryen_9

		is_alive = yes

		has_character_flag = segrellion_daenerys_landed_in_westeros


		NOT = {
			has_character_flag = segrellion_daenerys_seized_dragonstone
		}


		NOT = {
			has_character_flag = segrellion_daenerys_taking_dragonstone
		}


		# Old-save / recovery protection.

		NOT = {
			has_title = title:k_dragonstone
		}

		NOT = {
			has_title = title:d_dragonstone
		}

		NOT = {
			has_title = title:c_dragonstone
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		# =================================================
		# STANNIS MUST BE DEAD
		# =================================================

		custom_description = {

			text = segrellion_daenerys_take_dragonstone_stannis_dead_requirement


			character:Baratheon_3 = {

				is_alive = no
			}
		}


		# =================================================
		# AEGON MUST NOT CONTROL / BE TAKING DRAGONSTONE
		# =================================================

		custom_description = {

			text = segrellion_daenerys_take_dragonstone_aegon_requirement


			AND = {

				NOT = {

					title:c_dragonstone.holder = character:IAF_Young_Griff_1
				}


				NOT = {

					character:IAF_Young_Griff_1 ?= {

						has_character_flag = segrellion_young_griff_taking_dragonstone
					}
				}
			}
		}


		# -------------------------------------------------
		# DAENERYS CANNOT BE IMPRISONED
		# -------------------------------------------------

		custom_description = {

			text = segrellion_daenerys_take_dragonstone_not_imprisoned_requirement

			is_imprisoned = no
		}


		# -------------------------------------------------
		# DAENERYS CANNOT ALREADY BE TRAVELLING
		# -------------------------------------------------

		custom_description = {

			text = segrellion_daenerys_take_dragonstone_not_travelling_requirement

			NOT = {
				exists = current_travel_plan
			}
		}
	}


	# =====================================================
	# EFFECT
	# =====================================================

	effect = {

		custom_tooltip = segrellion_daenerys_take_dragonstone_travel_tt
		custom_tooltip = segrellion_daenerys_take_dragonstone_claim_tt


		hidden_effect = {

			add_character_flag = segrellion_daenerys_taking_dragonstone


			start_travel_plan = {

				destination = title:c_dragonstone.title_province

				players_use_planner = no

				return_trip = no

				travel_with_domicile = yes

				on_arrival_event = segrellion_daenerys_dragonstone.0090

				on_arrival_destinations = last

				on_travel_planner_cancel_event = segrellion_daenerys_dragonstone.0091
			}
		}
	}


	# =====================================================
	# AI
	#
	# Daenerys should take Dragonstone almost immediately
	# once Stannis is dead and the decision becomes valid.
	# =====================================================

	ai_check_interval = 1


	ai_potential = {

		this = character:Targaryen_9

		is_ai = yes
	}


	ai_will_do = {

		base = 1000
	}
}


# =========================================================
# DECLARE THE WAR FOR THE IRON THRONE
# =========================================================

segrellion_daenerys_claim_iron_throne_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/segrellion_take_the_iron_throne_decision.dds"
	}

	decision_group_type = ice_and_fire

	desc = segrellion_daenerys_claim_iron_throne_decision_desc
	selection_tooltip = segrellion_daenerys_claim_iron_throne_decision_tooltip
	confirm_text = segrellion_daenerys_claim_iron_throne_decision_confirm


	is_shown = {

		this = character:Targaryen_9

		is_alive = yes

		has_character_flag = segrellion_daenerys_landed_in_westeros

		has_character_flag = segrellion_daenerys_westeros_strategy_unlocked


		NOT = {
			has_character_flag = segrellion_daenerys_iron_throne_war_started
		}


		NOT = {
			has_title = title:h_the_iron_throne
		}


		title:h_the_iron_throne = {
			exists = holder
		}
	}


	is_valid = {

		# =================================================
		# DAENERYS MUST FIRST TEST THE POLITICAL LANDSCAPE
		# =================================================

		custom_description = {

			text = segrellion_daenerys_claim_iron_throne_gather_support_requirement

			AND = {

				has_variable = segrellion_daenerys_gather_support_attempts

				var:segrellion_daenerys_gather_support_attempts >= 2
			}
		}


		# =================================================
		# NO ACTIVE SUPPORT NEGOTIATION
		# =================================================

		custom_description = {

			text = segrellion_daenerys_claim_iron_throne_support_pending_requirement

			NOT = {
				has_character_flag = segrellion_daenerys_support_negotiation_pending
			}
		}


		# =================================================
		# NOT TRAVELLING
		# =================================================

		custom_description = {

			text = segrellion_daenerys_claim_iron_throne_not_travelling_requirement

			NOT = {
				exists = current_travel_plan
			}
		}


		# =================================================
		# NOT ALREADY AT WAR
		# =================================================

		custom_description = {

			text = segrellion_daenerys_claim_iron_throne_not_at_war_requirement

			is_at_war = no
		}
	}


	effect = {

		custom_tooltip = segrellion_daenerys_claim_iron_throne_effect_war_tt

		custom_tooltip = segrellion_daenerys_claim_iron_throne_effect_supporters_tt

		hidden_effect = {

			if = {

				limit = {

					NOT = {
						has_claim_on = title:h_the_iron_throne
					}
				}

				add_pressed_claim = title:h_the_iron_throne
			}


			title:h_the_iron_throne.holder = {
				save_scope_as = segrellion_daenerys_throne_defender
			}


			# =================================================
			# THE CURRENT KING WILL FACE DAENERYS AFTER VICTORY
			# =================================================

			add_character_flag = segrellion_daenerys_defeated_throne_holder_fate_pending

			remove_character_flag = segrellion_daenerys_defeated_throne_holder_fate_resolved

			remove_character_flag = segrellion_daenerys_defeated_throne_holder_event_scheduled


			# =================================================
			# SAVE THE KING DAENERYS IS ACTUALLY FIGHTING
			#
			# Do this BEFORE the war starts so the post-throne
			# chain never depends solely on title-gain or
			# war-resolution on_action ordering.
			# =================================================

			set_variable = {

				name = segrellion_daenerys_defeated_throne_holder

				value = scope:segrellion_daenerys_throne_defender
			}


			start_war = {

				cb = claim_cb

				target = scope:segrellion_daenerys_throne_defender

				claimant = root

				target_title = title:h_the_iron_throne
			}


			add_character_flag = segrellion_daenerys_iron_throne_war_started


			# =================================================
			# GREAT HOUSES ALREADY WON THROUGH GATHER SUPPORT
			# =================================================

			segrellion_daenerys_call_promised_supporters_effect = yes
		}
	}


	ai_check_interval = 14


	ai_potential = {

		this = character:Targaryen_9

		is_ai = yes
	}


	ai_will_do = {

		base = 100


		modifier = {

			factor = 0

			has_character_flag = segrellion_daenerys_recently_landed_in_westeros
		}


		modifier = {

			add = 75

			OR = {

				has_character_flag = segrellion_daenerys_north_support_secured

				has_character_flag = segrellion_daenerys_reach_support_secured

				has_character_flag = segrellion_daenerys_flamelands_support_secured
			}
		}
	}
}
