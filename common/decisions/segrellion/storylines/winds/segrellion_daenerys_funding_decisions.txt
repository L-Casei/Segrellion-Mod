segrellion_settle_invasion_debts_decision = {

	decision_group_type = ice_and_fire

	picture = {
		reference = "gfx/interface/illustrations/decisions/segrellion_repay_debts.dds"
	}

	desc = segrellion_settle_invasion_debts_decision_desc
	selection_tooltip = segrellion_settle_invasion_debts_decision_tooltip
	confirm_text = segrellion_settle_invasion_debts_decision_confirm


	is_shown = {

		this = character:Targaryen_9

		is_alive = yes

		has_character_flag = segrellion_daenerys_landed_in_westeros

		OR = {
			has_character_flag = segrellion_daenerys_pentoshi_syndicate_debt
			has_character_flag = segrellion_daenerys_iron_bank_debt
			has_character_flag = segrellion_daenerys_captains_credit_debt
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		custom_description = {

			text = segrellion_settle_invasion_debts_affordable_requirement

			OR = {

				AND = {
					has_character_flag = segrellion_daenerys_pentoshi_syndicate_debt
					gold >= 1200
				}

				AND = {

					has_character_flag = segrellion_daenerys_iron_bank_debt

					exists = var:segrellion_daenerys_iron_bank_debt_value

					gold >= var:segrellion_daenerys_iron_bank_debt_value
				}

				AND = {
					has_character_flag = segrellion_daenerys_captains_credit_debt
					gold >= 650
				}
			}
		}
	}


	widget = {

		gui = "decision_view_widget_option_list_generic"

		controller = decision_option_list_controller

		decision_to_second_step_button = "segrellion_settle_invasion_debts_next_step"


		item = {

			value = segrellion_repay_choice_syndicate

			localization = segrellion_repay_choice_syndicate

			current_description = segrellion_repay_choice_syndicate_desc

			icon = "gfx/interface/icons/icon_gold.dds"


			is_shown = {

				root = {
					has_character_flag = segrellion_daenerys_pentoshi_syndicate_debt
				}
			}


			is_valid = {

				root = {

					custom_description = {

						text = segrellion_repay_choice_syndicate_gold_requirement

						gold >= 1200
					}
				}
			}
		}


		item = {

			value = segrellion_repay_choice_iron_bank

			value = segrellion_repay_choice_iron_bank

			localization = segrellion_repay_choice_iron_bank

			current_description = segrellion_repay_choice_iron_bank_desc

			icon = "gfx/interface/icons/icon_imperial_treasury.dds"


			is_shown = {

				root = {
					has_character_flag = segrellion_daenerys_iron_bank_debt
				}
			}


			is_valid = {

				root = {

					custom_description = {

						text = segrellion_repay_choice_iron_bank_gold_requirement

						exists = var:segrellion_daenerys_iron_bank_debt_value

						gold >= var:segrellion_daenerys_iron_bank_debt_value
					}
				}
			}
		}


		item = {

			value = segrellion_repay_choice_captains

			localization = segrellion_repay_choice_captains

			current_description = segrellion_repay_choice_captains_desc

			icon = "gfx/interface/icons/icon_group_of_characters.dds"


			is_shown = {

				root = {
					has_character_flag = segrellion_daenerys_captains_credit_debt
				}
			}


			is_valid = {

				root = {

					custom_description = {

						text = segrellion_repay_choice_captains_gold_requirement

						gold >= 650
					}
				}
			}
		}
	}


	effect = {

		custom_tooltip = segrellion_settle_invasion_debts_effect_one_debt_tt
		custom_tooltip = segrellion_settle_invasion_debts_effect_remove_penalty_tt

		hidden_effect = {

			if = {

				limit = {
					scope:segrellion_repay_choice_syndicate = yes
				}

				add_gold = -1200

				remove_character_flag = segrellion_daenerys_pentoshi_syndicate_debt
				remove_character_flag = segrellion_daenerys_pentoshi_syndicate_debt_applied

				remove_character_modifier = segrellion_pentoshi_syndicate_debt_modifier

				add_character_flag = segrellion_daenerys_pentoshi_syndicate_debt_paid
			}


			else_if = {

				limit = {
					scope:segrellion_repay_choice_iron_bank = yes
				}


				add_gold = {
					value = var:segrellion_daenerys_iron_bank_debt_value
					multiply = -1
				}


				segrellion_daenerys_clear_iron_bank_debt_state_effect = yes

				add_character_flag = segrellion_daenerys_iron_bank_debt_paid
			}


			else_if = {

				limit = {
					scope:segrellion_repay_choice_captains = yes
				}

				add_gold = -650

				remove_character_flag = segrellion_daenerys_captains_credit_debt
				remove_character_flag = segrellion_daenerys_captains_credit_applied

				remove_character_modifier = segrellion_sellsail_credit_modifier

				add_character_flag = segrellion_daenerys_captains_credit_paid
			}
		}
	}
