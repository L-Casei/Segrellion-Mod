# =========================================================
# REPAY THE VALE DEBT
# =========================================================

segrellion_repay_vale_debt_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/segrellion_negotiate_for_reinforcements_decision_background.dds"
	}

	decision_group_type = major

	sort_order = 180

	desc = segrellion_repay_vale_debt_decision_desc
	selection_tooltip = segrellion_repay_vale_debt_decision_tooltip
	confirm_text = segrellion_repay_vale_debt_decision_confirm


	is_shown = {

		this = character:Segrellion_60

		has_character_flag = segrellion_vale_debt_active

		has_variable = segrellion_vale_debt

		var:segrellion_vale_debt > 0
	}


	is_valid = {

		is_adult = yes

		is_imprisoned = no


		custom_description = {

			text = segrellion_vale_debt_outstanding_requirement

			has_variable = segrellion_vale_debt

			var:segrellion_vale_debt > 0
		}


		custom_description = {

			text = segrellion_vale_debt_can_make_payment_requirement

			OR = {

				# At least the minimum fixed payment is possible.
				AND = {
					var:segrellion_vale_debt >= 100
					gold >= 100
				}

				# Less than 100 remains, so the full-balance
				# option is the only necessary payment.
				AND = {
					var:segrellion_vale_debt < 100
					gold >= var:segrellion_vale_debt
				}
			}
		}
	}


	widget = {

		gui = "decision_view_widget_option_list_generic"

		controller = decision_option_list_controller

		decision_to_second_step_button = "segrellion_repay_vale_debt_next_step"


		# -------------------------------------------------
		# 100 GOLD
		# -------------------------------------------------

		item = {

			value = segrellion_vale_repayment_100

			current_description = {
				desc = segrellion_vale_repayment_100_desc
			}

			localization = segrellion_vale_repayment_100

			icon = "gfx/interface/icons/message_feed/stewardship.dds"

			is_valid = {

				root = {
					gold >= 100
					var:segrellion_vale_debt >= 100
				}
			}

			ai_chance = {
				value = 0
			}
		}


		# -------------------------------------------------
		# 250 GOLD
		# -------------------------------------------------

		item = {

			value = segrellion_vale_repayment_250

			current_description = {
				desc = segrellion_vale_repayment_250_desc
			}

			localization = segrellion_vale_repayment_250

			icon = "gfx/interface/icons/message_feed/stewardship.dds"

			is_valid = {

				root = {
					gold >= 250
					var:segrellion_vale_debt >= 250
				}
			}

			ai_chance = {
				value = 0
			}
		}


		# -------------------------------------------------
		# 500 GOLD
		# -------------------------------------------------

		item = {

			value = segrellion_vale_repayment_500

			current_description = {
				desc = segrellion_vale_repayment_500_desc
			}

			localization = segrellion_vale_repayment_500

			icon = "gfx/interface/icons/message_feed/stewardship.dds"

			is_valid = {

				root = {
					gold >= 500
					var:segrellion_vale_debt >= 500
				}
			}

			ai_chance = {
				value = 0
			}
		}


		# -------------------------------------------------
		# FULL REMAINING BALANCE
		# -------------------------------------------------

		item = {

			value = segrellion_vale_repayment_full

			current_description = {
				desc = segrellion_vale_repayment_full_desc
			}

			localization = segrellion_vale_repayment_full

			icon = "gfx/interface/icons/message_feed/stewardship.dds"

			is_valid = {

				root = {
					gold >= var:segrellion_vale_debt
				}
			}

			ai_chance = {
				value = 0
			}
		}
	}


	effect = {

		custom_tooltip = segrellion_vale_debt_payment_effect_tt


		if = {
			limit = {
				scope:segrellion_vale_repayment_100 = yes
			}

			custom_tooltip = segrellion_vale_debt_payment_100_effect_tt
		}


		else_if = {
			limit = {
				scope:segrellion_vale_repayment_250 = yes
			}

			custom_tooltip = segrellion_vale_debt_payment_250_effect_tt
		}


		else_if = {
			limit = {
				scope:segrellion_vale_repayment_500 = yes
			}

			custom_tooltip = segrellion_vale_debt_payment_500_effect_tt
		}


		else_if = {
			limit = {
				scope:segrellion_vale_repayment_full = yes
			}

			custom_tooltip = segrellion_vale_debt_payment_full_effect_tt
		}


		custom_tooltip = segrellion_vale_debt_completion_effect_tt

		hidden_effect = {

			# =================================================
			# SAVE THE PAYEE
			# =================================================

			var:segrellion_vale_creditor ?= {

				if = {

					limit = {
						is_alive = yes
					}


					save_scope_as = segrellion_vale_payment_recipient
				}
			}


			# Creditor died: the debt remains owed to the Vale.
			if = {

				limit = {

					NOT = {
						exists = scope:segrellion_vale_payment_recipient
					}


					title:e_the_vale ?= {
						exists = holder
					}
				}


				title:e_the_vale.holder = {

					save_scope_as = segrellion_vale_payment_recipient
				}
			}


			# =================================================
			# 100 GOLD
			# =================================================

			if = {

				limit = {
					scope:segrellion_vale_repayment_100 = yes
				}


				if = {

					limit = {
						exists = scope:segrellion_vale_payment_recipient
					}


					pay_short_term_gold = {

						target = scope:segrellion_vale_payment_recipient

						gold = 100
					}
				}


				else = {

					remove_short_term_gold = 100
				}


				change_variable = {

					name = segrellion_vale_debt

					add = -100
				}
			}


			# =================================================
			# 250 GOLD
			# =================================================

			else_if = {

				limit = {
					scope:segrellion_vale_repayment_250 = yes
				}


				if = {

					limit = {
						exists = scope:segrellion_vale_payment_recipient
					}


					pay_short_term_gold = {

						target = scope:segrellion_vale_payment_recipient

						gold = 250
					}
				}


				else = {

					remove_short_term_gold = 250
				}


				change_variable = {

					name = segrellion_vale_debt

					add = -250
				}
			}


			# =================================================
			# 500 GOLD
			# =================================================

			else_if = {

				limit = {
					scope:segrellion_vale_repayment_500 = yes
				}


				if = {

					limit = {
						exists = scope:segrellion_vale_payment_recipient
					}


					pay_short_term_gold = {

						target = scope:segrellion_vale_payment_recipient

						gold = 500
					}
				}


				else = {

					remove_short_term_gold = 500
				}


				change_variable = {

					name = segrellion_vale_debt

					add = -500
				}
			}


			# =================================================
			# FULL REMAINING BALANCE
			# =================================================

			else_if = {

				limit = {
					scope:segrellion_vale_repayment_full = yes
				}


				save_scope_value_as = {

					name = segrellion_vale_payment_amount

					value = var:segrellion_vale_debt
				}


				if = {

					limit = {
						exists = scope:segrellion_vale_payment_recipient
					}


					pay_short_term_gold = {

						target = scope:segrellion_vale_payment_recipient

						gold = scope:segrellion_vale_payment_amount
					}
				}


				else = {

					remove_short_term_gold = scope:segrellion_vale_payment_amount
				}


				set_variable = {

					name = segrellion_vale_debt

					value = 0
				}
			}


			# =================================================
			# FULLY REPAID
			# =================================================

			if = {

				limit = {

					var:segrellion_vale_debt <= 0
				}


				if = {

					limit = {
						exists = scope:segrellion_vale_payment_recipient
					}


					set_variable = {

						name = segrellion_vale_debt_repaid_creditor

						value = scope:segrellion_vale_payment_recipient
					}
				}


				add_character_flag = segrellion_vale_debt_paid_in_full


				segrellion_finish_vale_debt_effect = yes


				trigger_event = {

					id = segrellion_support.6210

					days = 1
				}
			}
		}
	}


	ai_check_interval = 0

	ai_will_do = {
		base = 0
	}
}
