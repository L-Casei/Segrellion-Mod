segrellion_investigate_black_unicorn_decision = {


	picture = {

		reference = "gfx/interface/illustrations/decisions/segrellion_search_legendary_relic_decision_background.dds"
	}


	decision_group_type = major

	sort_order = 80


	desc = segrellion_investigate_black_unicorn_decision_desc

	selection_tooltip = segrellion_investigate_black_unicorn_decision_tooltip

	confirm_text = segrellion_investigate_black_unicorn_decision_confirm


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {


		# This storyline belongs specifically to Daemon.

		this = character:Segrellion_60


		# The White Harbor investigation has revealed
		# enough information to make the decision visible.

		has_character_flag = segrellion_black_unicorn_decision_unlocked


		# Once Daemon actually commits to the expedition,
		# the decision disappears.

		NOT = {
			has_character_flag = segrellion_black_unicorn_expedition_committed
		}


		# No reason to show it after the encounter has
		# already reached its current conclusion.

		NOT = {
			has_character_flag = segrellion_black_unicorn_duel_resolved
		}
	}

	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {


		is_alive = yes
		is_adult = yes

		is_imprisoned = no


		custom_description = {

			text = segrellion_black_unicorn_not_travelling_requirement

			NOT = {
				exists = current_travel_plan
			}
		}


		custom_description = {

			text = segrellion_black_unicorn_woman_alive_requirement

			character:Black_Unicorn_1 ?= {
				is_alive = yes
			}
		}


		trigger_if = {

			limit = {

				NOT = {
					debug_only = yes
				}
			}


			custom_description = {

				text = segrellion_black_unicorn_unicorn_returns_requirement

				has_character_flag = segrellion_unicorn_returns_completed
			}


			custom_description = {

				text = segrellion_black_unicorn_beacon_requirement

				title:c_farring_cross = {
					holder = root
				}
			}
		}
	}


	# =====================================================
	# COST
	# =====================================================

	cost = {

		gold = 150
	}


	# =====================================================
	# EFFECT
	# =====================================================

	effect = {


		custom_tooltip = segrellion_investigate_black_unicorn_effect_trail_tt

		custom_tooltip = segrellion_investigate_black_unicorn_effect_skagos_tt

		custom_tooltip = segrellion_investigate_black_unicorn_effect_danger_tt


		hidden_effect = {

			trigger_event = {

				id = segrellion_black_unicorn.0030
			}
		}
	}


	# =====================================================
	# AI
	# =====================================================

	ai_check_interval = 30


	ai_potential = {

		this = character:Segrellion_60

		is_ai = yes

		has_character_flag = segrellion_unicorn_returns_completed

		has_character_flag = segrellion_black_unicorn_decision_unlocked
	}


	ai_will_do = {

		base = 100
	}
}
