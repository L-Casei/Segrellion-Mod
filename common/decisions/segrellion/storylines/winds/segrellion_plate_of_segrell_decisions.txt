# =========================================================
# HUGH SEGRELLION
# RECOVER THE PLATE OF SEGRELL
# =========================================================

segrellion_recover_plate_of_segrell_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/proclaim_paramountcy_of_the_flamelands_decision_background.dds"
	}

	decision_group_type = major

	desc = segrellion_recover_plate_of_segrell_decision_desc
	selection_tooltip = segrellion_recover_plate_of_segrell_decision_tooltip
	confirm_text = segrellion_recover_plate_of_segrell_decision_confirm


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {

		this = character:Segrellion_43

		has_character_flag = segrellion_plate_recovery_ready

		NOT = {
			has_character_flag = segrellion_plate_recovery_in_progress
		}

		NOT = {
			has_character_flag = segrellion_plate_recovery_complete
		}

		NOT = {
			segrellion_plate_recovery_blocked_trigger = yes
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		is_adult = yes
		is_imprisoned = no

		gold >= 200


		custom_description = {

			text = segrellion_recover_plate_not_already_recovered_tt

			NOT = {
				segrellion_plate_recovery_blocked_trigger = yes
			}
		}
	}


	# =====================================================
	# COST
	#
	# This is NOT a purchase price.
	# It covers couriers, guards, ships, wagons,
	# harbour fees and secure transport.
	# =====================================================

	cost = {
		gold = 50
	}


	# =====================================================
	# EFFECT
	# =====================================================

	effect = {

		custom_tooltip = segrellion_recover_plate_transport_tt


		hidden_effect = {

			remove_character_flag = segrellion_plate_recovery_ready

			add_character_flag = segrellion_plate_recovery_in_progress


			trigger_event = {
				id = segrellion_winds.0130
				days = { 20 60 }
			}
		}
	}


	# =====================================================
	# AI
	# =====================================================

	ai_check_interval = 7

	ai_potential = {

		this = character:Segrellion_43

		has_character_flag = segrellion_plate_recovery_ready

		NOT = {
			has_character_flag = segrellion_plate_recovery_in_progress
		}

		NOT = {
			has_character_flag = segrellion_plate_recovery_complete
		}

		NOT = {
			segrellion_plate_recovery_blocked_trigger = yes
		}

		gold >= 200
	}

	ai_will_do = {
		base = 1000
	}
}

# =========================================================
# DAEMON SEGRELLION
# DEMAND THE PLATE OF SEGRELL
# =========================================================

segrellion_demand_plate_of_segrell_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/proclaim_paramountcy_of_the_flamelands_decision_background.dds"
	}

	decision_group_type = major

	cooldown = {
		years = 1
	}

	desc = segrellion_demand_plate_of_segrell_decision_desc
	selection_tooltip = segrellion_demand_plate_of_segrell_decision_tooltip
	confirm_text = segrellion_demand_plate_of_segrell_decision_confirm


	# =====================================================
	# COST
	# 250 Prestige normally.
	# Free in debug.
	# =====================================================

	cost = {
		prestige = {
			value = 0

			if = {
				limit = {
					NOT = {
						debug_only = yes
					}
				}

				add = 250
			}
		}
	}


	# =====================================================
	# VISIBILITY
	# Normally hidden until Hugh possesses the Plate.
	# Debug bypasses these restrictions.
	# =====================================================

	is_shown = {

		this = character:Segrellion_60

		trigger_if = {
			limit = {
				NOT = {
					debug_only = yes
				}
			}


			current_date >= 8300.8.20


			character:Segrellion_43 ?= {

				is_alive = yes

				any_character_artifact = {
					has_variable = segrellion_plate_of_segrell_artifact
				}
			}


			NOT = {
				has_character_flag = segrellion_plate_demand_pending
			}
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		trigger_if = {
			limit = {
				NOT = {
					debug_only = yes
				}
			}


			is_adult = yes
			is_imprisoned = no


			custom_description = {

				text = segrellion_demand_plate_hugh_has_plate_tt

				character:Segrellion_43 = {

					any_character_artifact = {
						has_variable = segrellion_plate_of_segrell_artifact
					}
				}
			}
		}
	}


	# =====================================================
	# EFFECT
	# =====================================================

	effect = {

		# Visible in the Effects section of the decision.
		custom_tooltip = segrellion_demand_plate_effect_send_tt
		custom_tooltip = segrellion_demand_plate_effect_response_tt


		# Internal execution should not clutter the UI.
		hidden_effect = {

			add_character_flag = segrellion_plate_demand_pending

			character:Segrellion_43 = {

				trigger_event = {
					id = segrellion_winds.0102
					days = 1
				}
			}
		}
	}


	# =====================================================
	# AI
	# =====================================================

	ai_check_interval = 1

	ai_potential = {

		this = character:Segrellion_60
		is_ai = yes

		has_character_flag = segrellion_foreign_support_secured
	}

	ai_will_do = {
		base = 10000
	}
}
