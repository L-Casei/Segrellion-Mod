# =========================================================
# I NEED A DRAGON
# =========================================================

segrellion_young_griff_need_dragon_decision = {
	decision_group_type = ice_and_fire

	picture = {
		reference = "gfx/interface/illustrations/decisions/young_griff_need_dragon_background.dds"
	}

	sort_order = 175

	desc = segrellion_young_griff_need_dragon_decision_desc

	selection_tooltip = segrellion_young_griff_need_dragon_decision_tooltip

	confirm_text = segrellion_young_griff_need_dragon_decision_confirm


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {

		this = character:IAF_Young_Griff_1

		is_alive = yes

		has_character_flag = segrellion_young_griff_varys_contacted


		NOT = {
			has_character_flag = segrellion_young_griff_dragon_path_chosen
		}

		NOT = {
			has_title = title:h_the_iron_throne
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		is_adult = yes

		is_imprisoned = no

		NOT = {
			exists = current_travel_plan
		}


		# -------------------------------------------------
		# NO DRAGON
		# -------------------------------------------------

		custom_description = {

			text = segrellion_young_griff_need_dragon_no_dragon_requirement

			NOR = {

				has_trait = dragonrider

				any_relation = {
					type = agot_dragon
				}
			}
		}


		# -------------------------------------------------
		# NO DRAGON EGG
		# -------------------------------------------------

		custom_description = {

			text = segrellion_young_griff_need_dragon_no_egg_requirement

			NOT = {

				any_character_artifact = {
					has_variable = dragon_egg
				}
			}
		}
	}


	# =====================================================
	# ROUTE SELECTOR
	# =====================================================

	widget = {

		gui = "decision_view_widget_option_list_generic"

		controller = decision_option_list_controller

		decision_to_second_step_button = "segrellion_young_griff_need_dragon_next_step"

		show_from_start = yes


		# =================================================
		# SEARCH FOR AN EGG
		# =================================================

		item = {

			value = segrellion_young_griff_dragon_choice_egg

			localization = segrellion_young_griff_dragon_choice_egg

			current_description = segrellion_young_griff_dragon_choice_egg_tt

			icon = "gfx/interface/icons/message_feed/intrigue.dds"


			is_shown = {
				always = yes
			}


			is_valid = {

				gold >= 100

				prestige >= 150
				
			}


			ai_chance = {
				value = 97
			}
		}


		# =================================================
		# RHAEGAL / VISERION
		# =================================================

		item = {

			value = segrellion_young_griff_dragon_choice_daenerys

			localization = segrellion_young_griff_dragon_choice_daenerys

			current_description = segrellion_young_griff_dragon_choice_daenerys_tt

			icon = "gfx/interface/icons/message_feed/martial.dds"


			is_shown = {

				OR = {

					character:dragon_rhaegal ?= {

						is_alive = yes

						NOT = {
							exists = var:current_rider
						}
					}

					character:dragon_viserion ?= {

						is_alive = yes

						NOT = {
							exists = var:current_rider
						}
					}
				}
			}


			is_valid = {

				OR = {

					character:dragon_rhaegal ?= {

						is_alive = yes

						NOT = {
							exists = var:current_rider
						}
					}

					character:dragon_viserion ?= {

						is_alive = yes

						NOT = {
							exists = var:current_rider
						}
					}
				}
			}


			ai_chance = {
				value = 3
			}
		}
	}


	# =====================================================
	# EFFECTS
	# =====================================================

	effect = {

		custom_tooltip = segrellion_young_griff_need_dragon_danger_warning_tt


		# =====================================================
		# AI
		# =====================================================

		if = {

			limit = {
				is_ai = yes
			}


			hidden_effect = {

				add_character_flag = segrellion_young_griff_dragon_path_chosen


				# =================================================
				# BOTH ROUTES AVAILABLE
				# =================================================

				if = {

					limit = {

						any_courtier = {

							is_alive = yes
							is_adult = yes
							is_imprisoned = no

							learning >= 8
							stewardship >= 8
						}


						OR = {

							character:dragon_rhaegal ?= {

								is_alive = yes

								NOT = {
									exists = var:current_rider
								}
							}

							character:dragon_viserion ?= {

								is_alive = yes

								NOT = {
									exists = var:current_rider
								}
							}
						}
					}


					random_list = {

						97 = {

							segrellion_young_griff_choose_dragon_egg_route_effect = yes
						}

						3 = {

							segrellion_young_griff_choose_daenerys_dragon_route_effect = yes
						}
					}
				}


				# =================================================
				# ONLY EGG ROUTE
				# =================================================

				else_if = {

					limit = {

						any_courtier = {

							is_alive = yes
							is_adult = yes
							is_imprisoned = no

							learning >= 8
							stewardship >= 8
						}
					}


					segrellion_young_griff_choose_dragon_egg_route_effect = yes
				}


				# =================================================
				# ONLY DRAGON ROUTE
				# =================================================

				else_if = {

					limit = {

						OR = {

							character:dragon_rhaegal ?= {

								is_alive = yes

								NOT = {
									exists = var:current_rider
								}
							}

							character:dragon_viserion ?= {

								is_alive = yes

								NOT = {
									exists = var:current_rider
								}
							}
						}
					}


					segrellion_young_griff_choose_daenerys_dragon_route_effect = yes
				}


				else = {

					remove_character_flag = segrellion_young_griff_dragon_path_chosen
				}
			}
		}


		# =====================================================
		# PLAYER
		#
		# decision_option_list_controller passes the selected
		# option as a scope. It must be handled with switch.
		# =====================================================

		else = {

			switch = {

				trigger = yes


				# =================================================
				# SEARCH FOR AN EGG
				# =================================================

				scope:segrellion_young_griff_dragon_choice_egg = {

					custom_tooltip = segrellion_young_griff_dragon_choice_egg_tt

					custom_tooltip = segrellion_young_griff_dragon_choice_egg_effect_tt


					hidden_effect = {

						add_character_flag = segrellion_young_griff_dragon_path_chosen

						segrellion_young_griff_choose_dragon_egg_route_effect = yes
					}
				}


				# =================================================
				# TAKE ONE OF DAENERYS' DRAGONS
				# =================================================

				scope:segrellion_young_griff_dragon_choice_daenerys = {

					custom_tooltip = segrellion_young_griff_dragon_choice_daenerys_tt

					custom_tooltip = segrellion_young_griff_dragon_choice_daenerys_effect_tt


					hidden_effect = {

						add_character_flag = segrellion_young_griff_dragon_path_chosen

						segrellion_young_griff_choose_daenerys_dragon_route_effect = yes
					}
				}
			}
		}
	}


	ai_check_interval = 30


	ai_potential = {

		this = character:IAF_Young_Griff_1

		is_ai = yes
	}


	ai_will_do = {
		base = 33
	}
}

# =========================================================
# TAKE DRAGONSTONE
#
# Aegon demands Dragonstone from Shireen after Stannis dies.
#
# Uses the exact same decision artwork as Daenerys'
# Take Dragonstone decision.
# =========================================================

segrellion_young_griff_take_dragonstone_decision = {

	picture = {

		reference = "gfx/interface/illustrations/event_scenes/segrellion_dragonstone_arrival.dds"
	}


	decision_group_type = ice_and_fire

	sort_order = 176


	desc = segrellion_young_griff_take_dragonstone_decision_desc

	selection_tooltip = segrellion_young_griff_take_dragonstone_decision_tooltip

	confirm_text = segrellion_young_griff_take_dragonstone_decision_confirm


	# =====================================================
	# SHOWN
	# =====================================================

	is_shown = {

		this = character:IAF_Young_Griff_1

		is_alive = yes

		has_character_flag = segrellion_young_griff_varys_contacted


		# Intended primarily for Aegon's pre-Iron-Throne phase.

		NOT = {
			has_title = title:h_the_iron_throne
		}


		NOT = {
			has_character_flag = segrellion_young_griff_took_dragonstone
		}

		NOT = {
			has_character_flag = segrellion_young_griff_taking_dragonstone
		}

		NOT = {
			has_character_flag = segrellion_young_griff_dragonstone_request_pending
		}


		NOT = {
			has_title = title:k_dragonstone
		}

		NOT = {
			has_title = title:d_dragonstone
		}

		NOT = {
			has_title = title:c_dragonstone
		}


		character:Baratheon_13 ?= {

			is_alive = yes
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		# =================================================
		# STANNIS MUST BE DEAD
		# =================================================

		custom_description = {

			text = segrellion_young_griff_take_dragonstone_stannis_dead_requirement


			character:Baratheon_3 = {

				is_alive = no
			}
		}


		# =================================================
		# SHIREEN MUST CONTROL DRAGONSTONE
		# =================================================

		custom_description = {

			text = segrellion_young_griff_take_dragonstone_shireen_requirement


			title:c_dragonstone.holder = character:Baratheon_13
		}


		# =================================================
		# DAENERYS MUST NOT ALREADY HAVE DRAGONSTONE
		# =================================================

		custom_description = {

			text = segrellion_young_griff_take_dragonstone_daenerys_requirement


			OR = {

				character:Targaryen_9 = {

					is_alive = no
				}


				AND = {

					NOT = {
						title:c_dragonstone.holder = character:Targaryen_9
					}


					character:Targaryen_9 = {

						NOT = {
							location = title:c_dragonstone.title_province
						}


						NOR = {

							has_character_flag = segrellion_daenerys_landing_dragonstone

							has_character_flag = segrellion_daenerys_taking_dragonstone
						}
					}
				}
			}
		}


		# =================================================
		# AEGON AVAILABLE
		# =================================================

		custom_description = {

			text = segrellion_young_griff_take_dragonstone_not_imprisoned_requirement

			is_imprisoned = no
		}


		custom_description = {

			text = segrellion_young_griff_take_dragonstone_not_travelling_requirement


			NOT = {
				exists = current_travel_plan
			}
		}


		custom_description = {

			text = segrellion_young_griff_take_dragonstone_not_at_war_requirement

			is_at_war = no
		}
	}


	# =====================================================
	# EFFECT
	# =====================================================

	effect = {

		custom_tooltip = segrellion_young_griff_take_dragonstone_letter_tt


		hidden_effect = {

			add_character_flag = segrellion_young_griff_dragonstone_request_pending


			character:Baratheon_13 = {

				trigger_event = {

					id = segrellion_young_griff_dragonstone.0001

					days = { 3 7 }
				}
			}
		}
	}


	# =====================================================
	# AI
	# =====================================================

	ai_check_interval = 30


	ai_potential = {

		this = character:IAF_Young_Griff_1

		is_ai = yes
	}


	ai_will_do = {

		base = 300


		# Especially attractive if Aegon still needs an egg.

		modifier = {

			add = 700


			NOT = {

				any_character_artifact = {

					has_variable = dragon_egg
				}
			}
		}
	}
}
