# =========================================================
# CHANGE FOCUS
# =========================================================

segrellion_change_knightly_ideal_focus_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/knightly_ideal_decision.dds"
	}

	desc = segrellion_change_knightly_ideal_focus_decision_desc

	selection_tooltip = segrellion_change_knightly_ideal_focus_decision_tooltip

	confirm_text = segrellion_change_knightly_ideal_focus_decision_confirm


	is_shown = {

		has_trait = segrellion_knightly_ideal_trait

		NOT = {
			has_character_flag = segrellion_knightly_ideal_completed
		}
	}


	is_valid = {

		is_alive = yes

		is_imprisoned = no


		custom_description = {

			text = segrellion_knightly_focus_cooldown_requirement


			OR = {


				NOT = {
					has_character_flag = segrellion_knightly_focus_change_cooldown
				}


				AND = {

					has_character_flag = segrellion_knightly_focus_blade

					has_trait_xp = {
						trait = segrellion_knightly_ideal_trait
						track = knightly_blade
						value >= 100
					}
				}


				AND = {

					has_character_flag = segrellion_knightly_focus_shield

					has_trait_xp = {
						trait = segrellion_knightly_ideal_trait
						track = knightly_shield
						value >= 100
					}
				}


				AND = {

					has_character_flag = segrellion_knightly_focus_oath

					has_trait_xp = {
						trait = segrellion_knightly_ideal_trait
						track = knightly_oath
						value >= 100
					}
				}


				AND = {

					has_character_flag = segrellion_knightly_focus_laurel

					has_trait_xp = {
						trait = segrellion_knightly_ideal_trait
						track = knightly_laurel
						value >= 100
					}
				}
			}
		}
	}


	effect = {

		custom_tooltip = segrellion_knightly_change_focus_tt

		trigger_event = {
			id = segrellion_knightly_ideal.0010
		}
	}


	ai_potential = {
		always = no
	}


	ai_will_do = {
		base = 0
	}
}
