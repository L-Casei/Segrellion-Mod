# =========================================================
# SEGRELLION
# DEBUG / CAMPAIGN RECOVERY DECISIONS
#
# PLAYER:
#   Can use them whenever their normal requirements are met.
#
# AI:
#   Cannot use them during the first 5 years of the campaign.
#   After 5 years they may use them only as stuck-story
#   recovery tools.
# =========================================================


# =========================================================
# REMOVE REFUSING MARRIAGE
# =========================================================

segrellion_debug_remove_refusing_marriage_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_knight_kneeling.dds"
	}

	decision_group_type = ice_and_fire

	desc = segrellion_debug_remove_refusing_marriage_decision_desc

	selection_tooltip = segrellion_debug_remove_refusing_marriage_decision_tooltip

	confirm_text = segrellion_debug_remove_refusing_marriage_decision_confirm


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {

		has_trait = refusing_marriage


		OR = {

			this = character:Segrellion_60

			this = character:IAF_Young_Griff_1
		}


		OR = {

			is_ai = no

			years_from_game_start >= 5
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		is_alive = yes

		is_adult = yes

		is_imprisoned = no

		has_trait = refusing_marriage


		trigger_if = {

			limit = {
				is_ai = yes
			}

			years_from_game_start >= 5
		}
	}


	# =====================================================
	# EFFECT
	# =====================================================

	effect = {

		custom_tooltip = segrellion_debug_remove_refusing_marriage_effect_tt


		hidden_effect = {

			remove_trait = refusing_marriage

			add_character_flag = segrellion_debug_refusing_marriage_removed
		}
	}


	# =====================================================
	# AI
	#
	# First possible use: 5 years after campaign start.
	# =====================================================

	ai_check_interval = 12


	ai_potential = {

		years_from_game_start >= 5

		has_trait = refusing_marriage


		OR = {

			this = character:Segrellion_60

			this = character:IAF_Young_Griff_1
		}
	}


	ai_will_do = {

		base = 100
	}
}


# =========================================================
# DAENERYS
# RESTORE FERTILITY
# =========================================================

segrellion_debug_daenerys_restore_fertility_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}

	decision_group_type = ice_and_fire

	desc = segrellion_debug_daenerys_restore_fertility_decision_desc

	selection_tooltip = segrellion_debug_daenerys_restore_fertility_decision_tooltip

	confirm_text = segrellion_debug_daenerys_restore_fertility_decision_confirm


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {

		this = character:Targaryen_9

		current_date >= 8303.1.1

		has_character_modifier = no_more_children_modifier


		OR = {

			is_ai = no

			years_from_game_start >= 5
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		is_alive = yes

		is_adult = yes

		is_imprisoned = no

		current_date >= 8303.1.1

		has_character_modifier = no_more_children_modifier


		trigger_if = {

			limit = {
				is_ai = yes
			}

			years_from_game_start >= 5
		}
	}


	# =====================================================
	# EFFECT
	# =====================================================

	effect = {

		custom_tooltip = segrellion_debug_daenerys_restore_fertility_effect_tt


		hidden_effect = {

			remove_character_modifier = no_more_children_modifier

			add_character_flag = segrellion_debug_daenerys_fertility_restored
		}
	}


	# =====================================================
	# AI
	#
	# Even though the decision appears from 303 AC for
	# players, AI cannot use it until 5 full campaign years
	# have elapsed.
	# =====================================================

	ai_check_interval = 12


	ai_potential = {

		years_from_game_start >= 5

		this = character:Targaryen_9

		current_date >= 8303.1.1

		has_character_modifier = no_more_children_modifier
	}


	ai_will_do = {

		base = 100
	}
}
