# =========================================================
# REMOVE REFUSING MARRIAGE
#
# Only available to Daemon Segrellion and Young Griff.
# Intended solely to recover a campaign where the scripted
# marriage storyline failed to remove the trait.
# =========================================================

segrellion_debug_remove_refusing_marriage_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_knight_kneeling.dds"
	}

	decision_group_type = ice_and_fire

	desc = segrellion_debug_remove_refusing_marriage_decision_desc

	selection_tooltip = segrellion_debug_remove_refusing_marriage_decision_tooltip

	confirm_text = segrellion_debug_remove_refusing_marriage_decision_confirm


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {

		is_ai = no

		has_trait = refusing_marriage


		OR = {

			this = character:Segrellion_60

			this = character:IAF_Young_Griff_1
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		is_alive = yes

		is_adult = yes

		is_imprisoned = no

		has_trait = refusing_marriage
	}


	# =====================================================
	# EFFECT
	# =====================================================

	effect = {

		custom_tooltip = segrellion_debug_remove_refusing_marriage_effect_tt


		hidden_effect = {

			remove_trait = refusing_marriage

			add_character_flag = segrellion_debug_refusing_marriage_removed
		}
	}


	# =====================================================
	# NEVER AI
	# =====================================================

	ai_check_interval = 0

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
