

segrellion_raise_unicorn_riders_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/reinstitute_unicorn_riders_decision_background.dds"
	}

	decision_group_type = major

	desc = segrellion_raise_unicorn_riders_decision_desc
	selection_tooltip = segrellion_raise_unicorn_riders_decision_tooltip
	confirm_text = segrellion_raise_unicorn_riders_decision_confirm


	# =========================================================
	# VISIBILITY
	# =========================================================

	is_shown = {

		# The Unicorn Riders historically exist from roughly 140 AC onward.
		current_date >= 8140.1.1

		# Only rulers of the historical seats of the Flamelands
		# may restore the Unicorn Riders.
		OR = {
			has_title = title:c_farring_cross
			has_title = title:k_flamelands
			has_title = title:e_flamelands
		}

		# This character has not already restored the order.
		NOT = {
			has_character_flag = segrellion_unicorn_riders_recruitment_unlocked
		}
	}


	# =========================================================
	# REQUIREMENTS
	# =========================================================

	is_valid = {

		is_adult = yes
		is_imprisoned = no
		is_incapable = no

		NOT = {
			government_has_flag = government_is_nw
		}

		custom_description = {
			text = segrellion_unicorn_riders_title_requirement_tt

			OR = {
				has_title = title:c_farring_cross
				has_title = title:k_flamelands
				has_title = title:e_flamelands
			}
		}

		gold >= 1000
		prestige >= 500
	}


	# =========================================================
	# COST
	# =========================================================

	cost = {
		gold = 1000
		prestige = 500
	}


	# =========================================================
	# EFFECT
	# =========================================================

	effect = {

		add_character_flag = segrellion_unicorn_riders_recruitment_unlocked

		custom_tooltip = segrellion_reinstitute_unicorn_riders_effect_tt

		trigger_event = segrellion_unicorn_riders.0001
	}


	# =========================================================
	# AI
	# =========================================================

	ai_potential = {
		always = yes
	}

	ai_will_do = {

		base = 0

		modifier = {
			add = 100
			gold >= 2000
			prestige >= 1000
		}
	}
}

segrellion_call_the_banners_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/segrellion_call_the_banners_decision.dds"
	}

	decision_group_type = major

	desc = segrellion_call_the_banners_decision_desc
	selection_tooltip = segrellion_call_the_banners_decision_tooltip
	confirm_text = segrellion_call_the_banners_decision_confirm


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {

		has_game_rule = segrellion_call_the_banners_enabled

		is_landed = yes
		is_ruler = yes

		highest_held_title_tier >= tier_duchy

		any_vassal = {

			is_alive = yes
			is_landed = yes

			highest_held_title_tier > tier_barony
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		is_adult = yes
		is_imprisoned = no
		is_incapable = no

		is_at_war = yes


		custom_description = {

			text = segrellion_call_the_banners_war_requirement

			any_character_war = {

				is_civil_war = no

				is_war_leader = root
			}
		}


		custom_description = {

			text = segrellion_call_the_banners_vassal_requirement

			any_vassal = {

				is_alive = yes
				is_landed = yes
				is_imprisoned = no
				is_incapable = no

				highest_held_title_tier > tier_barony
			}
		}
	}


	# =====================================================
	# COOLDOWN
	# =====================================================

	cooldown = {
		months = 12
	}


	# =====================================================
	# EFFECT
	# =====================================================

	effect = {

		custom_tooltip = segrellion_call_the_banners_effect_tt

		hidden_effect = {
			segrellion_call_the_banners_effect = yes
		}
	}


	# =====================================================
	# AI
	# =====================================================

	ai_check_interval = 1

	ai_potential = {

		has_game_rule = segrellion_call_the_banners_enabled

		is_landed = yes
		is_ruler = yes

		is_adult = yes
		is_imprisoned = no
		is_incapable = no

		is_at_war = yes

		highest_held_title_tier >= tier_duchy


		any_character_war = {

			is_civil_war = no

			is_war_leader = root
		}


		any_vassal = {

			is_alive = yes
			is_landed = yes
			is_imprisoned = no
			is_incapable = no

			highest_held_title_tier > tier_barony
		}
	}


	ai_will_do = {
		base = 100
	}


	sort_order = 1001
}
