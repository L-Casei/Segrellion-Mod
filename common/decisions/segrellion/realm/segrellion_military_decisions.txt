segrellion_raise_unicorn_riders_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/reinstitute_unicorn_riders_decision_background.dds"
	}

	decision_group_type = ice_and_fire

	desc = segrellion_raise_unicorn_riders_decision_desc
	selection_tooltip = segrellion_raise_unicorn_riders_decision_tooltip
	confirm_text = segrellion_raise_unicorn_riders_decision_confirm


	# =========================================================
	# VISIBILITY
	#
	# The Unicorn Riders are founded after The Beacon exists.
	# Once founded, their existence is stored globally and access
	# follows the Segrellion lord who personally holds The Beacon.
	# =========================================================

	is_shown = {

		dynasty = dynasty:dynn_Segrellion

		has_title = title:b_castle_farring

		title:b_castle_farring = {
			holder = root
		}

		# The decision only exists after Castle Farring has actually
		# become The Beacon.
		title:b_castle_farring.title_province = {
			has_building_or_higher = segrellion_beacon_01
		}

		# This is a founding decision, not a repeatable restoration.
		NOT = {
			has_global_variable = segrellion_unicorn_riders_created
		}
	}


	# =========================================================
	# REQUIREMENTS
	# =========================================================

	is_valid = {

		is_adult = yes
		is_imprisoned = no
		is_incapable = no

		NOT = {
			government_has_flag = government_is_nw
		}

		custom_description = {
			text = segrellion_unicorn_riders_title_requirement_tt

			AND = {
				has_title = title:b_castle_farring

				title:b_castle_farring = {
					holder = root
				}

				title:b_castle_farring.title_province = {
					has_building_or_higher = segrellion_beacon_01
				}
			}
		}

		gold >= 1000
		prestige >= 500
	}


	# =========================================================
	# COST
	# =========================================================

	cost = {
		gold = 1000
		prestige = 500
	}


	# =========================================================
	# EFFECT
	# =========================================================

	effect = {

		# Permanent institution state. This survives succession.
		set_global_variable = {
			name = segrellion_unicorn_riders_created
			value = yes
		}

		# Legacy compatibility for saves or scripts that still inspect
		# the old personal unlock flag.
		add_character_flag = segrellion_unicorn_riders_recruitment_unlocked

		custom_tooltip = segrellion_create_unicorn_riders_effect_tt

		trigger_event = segrellion_unicorn_riders.0001
	}


	# =========================================================
	# AI
	# =========================================================

	ai_potential = {
		always = yes
	}

	ai_will_do = {

		base = 0

		modifier = {
			add = 100
			gold >= 2000
			prestige >= 1000
		}
	}
}

# CALL THE BANNERS
# =========================================================

segrellion_call_the_banners_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/segrellion_call_the_banners_decision.dds"
	}

	decision_group_type = ice_and_fire

	desc = segrellion_call_the_banners_decision_desc
	selection_tooltip = segrellion_call_the_banners_decision_tooltip
	confirm_text = segrellion_call_the_banners_decision_confirm


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {

		has_game_rule = segrellion_call_the_banners_enabled

		NOT = {
			segrellion_is_long_night_dead_trigger = yes
		}

		is_landed = yes
		is_ruler = yes

		highest_held_title_tier >= tier_duchy

		NOT = {
			has_character_flag = segrellion_disable_call_the_banners
		}

		# The Dance uses AGOT Mega War allegiance. Calling our separate
		# banner system here would falsely brand legitimate opposing
		# vassals as traitors/oathbreakers.
		NOT = {
			any_character_war = {
				using_cb = segrellion_dance_of_the_dragons_war
			}
		}

		OR = {

			any_vassal = {

				is_alive = yes
				is_landed = yes

				highest_held_title_tier > tier_barony
			}


			# Unicorn Returns special call:
			# Gaunt and Pyle may answer Daemon even though
			# they currently stand with Leo Stormflame.

			AND = {

				this = character:Segrellion_60

				has_character_flag = segrellion_unicorn_returns_war_active

				OR = {

					title:c_the_gauntlet.holder ?= {
						is_alive = yes
					}

					title:c_greyhelm.holder ?= {
						is_alive = yes
					}
				}
			}
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		is_adult = yes
		is_imprisoned = no
		is_incapable = no

		custom_description = {
			text = segrellion_call_the_banners_others_blocked_tt

			NOT = {
				segrellion_is_long_night_dead_trigger = yes
			}
		}

		is_at_war = yes


		# =================================================
		# HIDDEN STORY RESTRICTIONS
		# =================================================

		hidden_trigger = {

			NOT = {
				has_character_flag = segrellion_disable_call_the_banners
			}
		}


		hidden_trigger = {

			NOT = {
				has_character_flag = segrellion_sansa_baelish_war_active
			}
		}

		hidden_trigger = {
			NOT = {
				any_character_war = {
					using_cb = segrellion_dance_of_the_dragons_war
				}
			}
		}


		# =================================================
		# ELIGIBLE BANNERS
		# =================================================

		custom_description = {

			text = segrellion_call_the_banners_vassal_requirement

			OR = {

				any_vassal = {

					is_alive = yes
					is_landed = yes
					is_imprisoned = no
					is_incapable = no

					highest_held_title_tier > tier_barony
				}


				AND = {

					this = character:Segrellion_60

					has_character_flag = segrellion_unicorn_returns_war_active

					OR = {

						title:c_the_gauntlet.holder ?= {
							is_alive = yes
						}

						title:c_greyhelm.holder ?= {
							is_alive = yes
						}
					}
				}
			}
		}
	}


	# =====================================================
	# COOLDOWN
	# =====================================================

	cooldown = {
		months = 12
	}


	# =====================================================
	# EFFECT
	# =====================================================

	effect = {

		custom_tooltip = segrellion_call_the_banners_effect_tt

		hidden_effect = {
			segrellion_call_the_banners_effect = yes
		}
	}


	# =====================================================
	# AI
	# =====================================================

	ai_check_interval = 1

	ai_potential = {

		has_game_rule = segrellion_call_the_banners_enabled

		NOT = {
			segrellion_is_long_night_dead_trigger = yes
		}

		NOT = {
			has_character_flag = segrellion_disable_call_the_banners
		}

		NOT = {
			has_character_flag = segrellion_sansa_baelish_war_active
		}

		NOT = {
			any_character_war = {
				using_cb = segrellion_dance_of_the_dragons_war
			}
		}

		is_landed = yes
		is_ruler = yes

		is_adult = yes
		is_imprisoned = no
		is_incapable = no

		is_at_war = yes

		highest_held_title_tier >= tier_duchy


		OR = {

			any_vassal = {

				is_alive = yes
				is_landed = yes
				is_imprisoned = no
				is_incapable = no

				highest_held_title_tier > tier_barony
			}


			AND = {

				this = character:Segrellion_60

				has_character_flag = segrellion_unicorn_returns_war_active

				OR = {

					title:c_the_gauntlet.holder ?= {
						is_alive = yes
					}

					title:c_greyhelm.holder ?= {
						is_alive = yes
					}
				}
			}
		}
	}


	ai_will_do = {

		base = 10


		# =================================================
		# OWN WAR
		#
		# If this ruler actually leads the war, calling the
		# banners becomes substantially more attractive.
		# =================================================

		modifier = {

			factor = 6

			any_character_war = {
				is_war_leader = root
			}
		}


		# =================================================
		# DEFENSIVE WAR
		#
		# Defending your own realm is especially important.
		# =================================================

		modifier = {

			factor = 2

			any_character_war = {

				is_war_leader = root

				NOT = {
					is_attacker = root
				}
			}
		}


		# =================================================
		# LOYAL TO THEIR OWN LIEGE
		#
		# A Duke fighting for a liege they respect is more
		# willing to mobilize their own banners.
		# =================================================

		modifier = {

			factor = 2

			exists = liege

			opinion = {
				target = liege
				value >= 50
			}
		}


		modifier = {

			factor = 1.5

			exists = liege

			opinion = {
				target = liege
				value >= 20
			}
		}


		# =================================================
		# HOSTILE TO THEIR OWN LIEGE
		# =================================================

		modifier = {

			factor = 0.5

			exists = liege

			opinion = {
				target = liege
				value < 0
			}
		}


		modifier = {

			factor = 0.25

			exists = liege

			opinion = {
				target = liege
				value <= -50
			}
		}


		# =================================================
		# PERSONALITY
		# =================================================

		modifier = {

			factor = 1.5

			has_trait = loyal
		}


		modifier = {

			factor = 0.5

			has_trait = disloyal
		}


		# =================================================
		# EXPECTED RESPONSE FROM THEIR BANNERS
		# =================================================

		modifier = {

			factor = 1.5

			any_vassal = {

				is_alive = yes
				is_landed = yes

				highest_held_title_tier > tier_barony

				opinion = {
					target = root
					value >= 50
				}
			}
		}


		modifier = {

			factor = 0.5

			NOT = {

				any_vassal = {

					is_alive = yes
					is_landed = yes

					highest_held_title_tier > tier_barony

					opinion = {
						target = root
						value >= 0
					}
				}
			}
		}


		# =================================================
		# STORY WAR
		# =================================================

		modifier = {

			factor = 10

			has_character_flag = segrellion_unicorn_returns_war_active
		}
	}


	sort_order = 1001
}
