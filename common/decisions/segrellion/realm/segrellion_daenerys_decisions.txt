# =========================================================
# DAENERYS - SAIL FOR WESTEROS
# =========================================================

segrellion_daenerys_sail_for_westeros_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_found_kingdom.dds"
	}

	decision_group_type = major


	desc = segrellion_daenerys_sail_for_westeros_decision_desc

	selection_tooltip = segrellion_daenerys_sail_for_westeros_decision_tooltip

	confirm_text = segrellion_daenerys_sail_for_westeros_decision_confirm


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {

		this = character:Targaryen_9

		has_character_flag = segrellion_daenerys_invasion_committed

		has_character_flag = segrellion_daenerys_camped_at_pentos


		OR = {

			has_character_flag = segrellion_daenerys_fleet_strategy_chosen

			has_character_flag = segrellion_daenerys_fleet_project_active

			has_character_flag = segrellion_daenerys_dragon_fleet_completed

			has_character_flag = segrellion_daenerys_aurane_route_chosen
		}


		NOT = {
			has_character_flag = segrellion_daenerys_landed_in_westeros
		}

		NOT = {
			has_character_flag = segrellion_daenerys_crossing_committed
		}

		NOT = {
			has_character_flag = segrellion_daenerys_aurane_negotiation_active
		}

		NOT = {
			has_character_flag = segrellion_daenerys_aurane_deal_accepted
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		is_alive = yes

		is_imprisoned = no

		is_incapable = no


		NOT = {
			exists = current_travel_plan
		}
	}


	# =====================================================
	# EFFECT
	# =====================================================

	effect = {


		# -------------------------------------------------
		# 100%
		# -------------------------------------------------

		if = {

			limit = {
				has_character_flag = segrellion_daenerys_dragon_fleet_completed
			}

			custom_tooltip = segrellion_daenerys_crossing_capacity_100_tt
		}


		# -------------------------------------------------
		# 90%
		# -------------------------------------------------

		else_if = {

			limit = {

				exists = var:segrellion_daenerys_fleet_readiness

				var:segrellion_daenerys_fleet_readiness >= 90
			}

			custom_tooltip = segrellion_daenerys_crossing_capacity_90_tt
		}


		# -------------------------------------------------
		# 75%
		# -------------------------------------------------

		else_if = {

			limit = {

				exists = var:segrellion_daenerys_fleet_readiness

				var:segrellion_daenerys_fleet_readiness >= 75
			}

			custom_tooltip = segrellion_daenerys_crossing_capacity_75_tt
		}


		# -------------------------------------------------
		# 60%
		# -------------------------------------------------

		else_if = {

			limit = {

				exists = var:segrellion_daenerys_fleet_readiness

				var:segrellion_daenerys_fleet_readiness >= 60
			}

			custom_tooltip = segrellion_daenerys_crossing_capacity_60_tt
		}


		# -------------------------------------------------
		# 50%
		# -------------------------------------------------

		else_if = {

			limit = {

				exists = var:segrellion_daenerys_fleet_readiness

				var:segrellion_daenerys_fleet_readiness >= 50
			}

			custom_tooltip = segrellion_daenerys_crossing_capacity_50_tt
		}


		# -------------------------------------------------
		# 40%
		# -------------------------------------------------

		else_if = {

			limit = {

				exists = var:segrellion_daenerys_fleet_readiness

				var:segrellion_daenerys_fleet_readiness >= 40
			}

			custom_tooltip = segrellion_daenerys_crossing_capacity_40_tt
		}


		# -------------------------------------------------
		# 20%
		# -------------------------------------------------

		else = {

			custom_tooltip = segrellion_daenerys_crossing_capacity_20_tt
		}


		hidden_effect = {

			segrellion_daenerys_apply_crossing_capacity_effect = yes


			# An unfinished fleet remains behind in Pentos.
			remove_character_flag = segrellion_daenerys_fleet_project_active

			remove_character_flag = segrellion_daenerys_waiting_at_pentos


			add_character_flag = segrellion_daenerys_crossing_committed


			trigger_event = {

				id = segrellion_daenerys_invasion.0010

				days = 1
			}
		}
	}


	# =====================================================
	# AI
	#
	# Do NOT disable the decision for AI. The scheduler is a
	# recovery layer, not the only way Daenerys is allowed to sail.
	# This gives the normal decision system a second independent
	# route to launch the invasion when the story state is valid.
	# =====================================================

	ai_check_interval = 14

	ai_potential = {
		this = character:Targaryen_9
		is_ai = yes

		NOT = {
			has_game_rule = segrellion_daenerys_destiny_free_cities
		}

		NOT = {
			has_character_flag = segrellion_daenerys_young_griff_throne_pact
		}
	}

	ai_will_do = {
		base = 0

		# Absolute emergency. If the decision is valid this late,
		# there is no reason for AI Daenerys to keep waiting.
		modifier = {
			add = 20000
			current_date >= 8306.1.1
		}

		# SUPER AGGRESSIVE: may sail as soon as ~40% readiness.
		modifier = {
			add = 1500
			OR = {
				has_game_rule = segrellion_daenerys_aggression_super_aggressive
				has_character_flag = segrellion_daenerys_invasion_profile_super_aggressive
				has_character_flag = segrellion_daenerys_random_super_aggressive
			}
			exists = var:segrellion_daenerys_fleet_readiness
			var:segrellion_daenerys_fleet_readiness >= 40
		}

		# AGGRESSIVE: sail readily from 303 AC at ~50% readiness.
		modifier = {
			add = 1500
			OR = {
				has_game_rule = segrellion_daenerys_aggression_aggressive
				has_character_flag = segrellion_daenerys_invasion_profile_aggressive
				has_character_flag = segrellion_daenerys_random_aggressive
			}
			current_date >= 8303.1.1
			exists = var:segrellion_daenerys_fleet_readiness
			var:segrellion_daenerys_fleet_readiness >= 50
		}

		# DEFAULT: 75% is the intended threshold. Start trying in
		# 303, become very eager through 304, and make 305 decisive.
		modifier = {
			add = 300
			OR = {
				has_game_rule = segrellion_daenerys_aggression_default
				has_character_flag = segrellion_daenerys_invasion_profile_default
				has_character_flag = segrellion_daenerys_random_default
			}
			current_date >= 8303.1.1
			exists = var:segrellion_daenerys_fleet_readiness
			var:segrellion_daenerys_fleet_readiness >= 75
		}

		modifier = {
			add = 1200
			OR = {
				has_game_rule = segrellion_daenerys_aggression_default
				has_character_flag = segrellion_daenerys_invasion_profile_default
				has_character_flag = segrellion_daenerys_random_default
			}
			current_date >= 8304.1.1
			exists = var:segrellion_daenerys_fleet_readiness
			var:segrellion_daenerys_fleet_readiness >= 75
		}

		modifier = {
			add = 15000
			OR = {
				has_game_rule = segrellion_daenerys_aggression_default
				has_character_flag = segrellion_daenerys_invasion_profile_default
				has_character_flag = segrellion_daenerys_random_default
			}
			current_date >= 8305.1.1
		}

		# PASSIVE: waits for 90% in 305, but 306 is absolute.
		modifier = {
			add = 1200
			OR = {
				has_game_rule = segrellion_daenerys_aggression_passive
				has_character_flag = segrellion_daenerys_invasion_profile_passive
				has_character_flag = segrellion_daenerys_random_passive
			}
			current_date >= 8305.1.1
			exists = var:segrellion_daenerys_fleet_readiness
			var:segrellion_daenerys_fleet_readiness >= 90
		}

		modifier = {
			add = 15000
			OR = {
				has_game_rule = segrellion_daenerys_aggression_passive
				has_character_flag = segrellion_daenerys_invasion_profile_passive
				has_character_flag = segrellion_daenerys_random_passive
			}
			current_date >= 8306.1.1
		}
	}


	sort_order = 20
}
