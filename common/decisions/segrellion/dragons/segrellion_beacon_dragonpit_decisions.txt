# =========================================================
# BUILD THE DRAGONPIT OF KNIGHT'S BRIDGE
# =========================================================

segrellion_build_beacon_dragonpit_decision = {

	picture = {
		reference = "gfx/interface/illustrations/great_projects/segrellion_beacon_dragonpit.dds"
	}

	decision_group_type = major
	ai_goal = yes

	desc = segrellion_build_beacon_dragonpit_decision_desc
	selection_tooltip = segrellion_build_beacon_dragonpit_decision_tooltip
	confirm_text = segrellion_build_beacon_dragonpit_decision_confirm

	cost = {
		gold = 250
		prestige = 100
	}

	# Completely hidden from everyone except the personal holder
	# of the Beacon, and completely hidden once the real building exists.
	is_shown = {

		is_landed = yes
		has_title = title:c_farring_cross

		NOT = {
			has_global_variable = segrellion_beacon_dragonpit_project_active
		}

		province:7263 = {
			NOT = {
				has_building_or_higher = segrellion_beacon_dragonpit_01
			}
		}
	}

	# Every player-facing requirement has an explicit tooltip.
	is_valid = {

		custom_description = {
			text = segrellion_beacon_dragonpit_requirement_adult_tt
			is_capable_adult = yes
		}

		custom_description = {
			text = segrellion_beacon_dragonpit_requirement_not_imprisoned_tt
			is_imprisoned = no
		}

		custom_description = {
			text = segrellion_beacon_dragonpit_requirement_county_tt
			has_title = title:c_farring_cross
		}

		custom_description = {
			text = segrellion_beacon_dragonpit_requirement_dragonpit_tt
			agot_has_dragonpit_requirements_trigger = yes
		}

		custom_description = {
			text = segrellion_beacon_dragonpit_requirement_not_built_tt
			province:7263 = {
				NOT = {
					has_building_or_higher = segrellion_beacon_dragonpit_01
				}
			}
		}

		custom_description = {
			text = segrellion_beacon_dragonpit_requirement_site_tt
			province:7263 = {
				OR = {
					has_holding = no
					has_holding_type = castle_holding
				}
			}
		}
	}

	effect = {

		custom_tooltip = segrellion_build_beacon_dragonpit_decision_effect_tt

		hidden_effect = {

			set_global_variable = {
				name = segrellion_beacon_dragonpit_project_active
				value = yes
			}

			province:7263 = {

				plan_great_project = {
					founder = root
					great_project_type = segrellion_construct_beacon_dragonpit
					save_scope_as = segrellion_beacon_dragonpit_project
				}
			}
		}
	}

	ai_check_interval = 12

	ai_potential = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		has_title = title:c_farring_cross
		agot_has_dragonpit_requirements_trigger = yes

		NOT = {
			has_global_variable = segrellion_beacon_dragonpit_project_active
		}

		province:7263 = {
			NOT = {
				has_building_or_higher = segrellion_beacon_dragonpit_01
			}

			OR = {
				has_holding = no
				has_holding_type = castle_holding
			}
		}
	}

	ai_will_do = {

		base = 100

		modifier = {
			add = 500

			any_relation = {
				type = agot_dragon
			}
		}

		modifier = {
			add = 300

			any_courtier_or_guest = {
				has_trait = dragon
			}
		}

		modifier = {
			add = 250

			any_close_or_extended_family_member = {
				any_relation = {
					type = agot_dragon
				}
			}
		}
	}
}


# =========================================================
# FINANCE THE DRAGONPIT
#
# Native Great Projects can directly expose owner + vassals.
# This decision extends financing to the founder's family and
# the entire Segrellion dynasty without inventing unsupported
# allowed_contributor_filter keywords.
# =========================================================

segrellion_finance_beacon_dragonpit_decision = {

	picture = {
		reference = "gfx/interface/illustrations/great_projects/segrellion_beacon_dragonpit.dds"
	}

	decision_group_type = major
	ai_goal = yes

	desc = segrellion_finance_beacon_dragonpit_decision_desc
	selection_tooltip = segrellion_finance_beacon_dragonpit_decision_tooltip
	confirm_text = segrellion_finance_beacon_dragonpit_decision_confirm

	is_shown = {
		segrellion_can_finance_beacon_dragonpit_project_trigger = yes
	}

	is_valid = {

		custom_description = {
			text = segrellion_beacon_dragonpit_finance_requirement_adult_tt
			is_adult = yes
		}

		custom_description = {
			text = segrellion_beacon_dragonpit_finance_requirement_not_imprisoned_tt
			is_imprisoned = no
		}

		custom_description = {
			text = segrellion_beacon_dragonpit_finance_requirement_contribution_tt
			segrellion_beacon_dragonpit_has_unfunded_contribution_trigger = yes
		}
	}

	effect = {
		custom_tooltip = segrellion_finance_beacon_dragonpit_decision_effect_tt

		hidden_effect = {
			trigger_event = {
				id = segrellion_beacon_dragonpit.0100
				days = 1
			}
		}
	}

	ai_check_interval = 12

	ai_potential = {
		is_adult = yes
		is_imprisoned = no
		segrellion_can_finance_beacon_dragonpit_project_trigger = yes
	}

	ai_will_do = {
		base = 50

		modifier = {
			add = 100
			gold >= 1000
		}
	}
}
