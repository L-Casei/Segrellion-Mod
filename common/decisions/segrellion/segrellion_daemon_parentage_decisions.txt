# =========================================================
# DAEMON SEGRELLION - REVEAL MY TRUE PARENTAGE
# =========================================================

segrellion_daemon_reveal_true_parentage_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/proclaim_paramountcy_of_the_flamelands_decision_background.dds"
	}

	decision_group_type = major

	desc = segrellion_daemon_reveal_true_parentage_decision_desc

	selection_tooltip = segrellion_daemon_reveal_true_parentage_decision_tooltip

	confirm_text = segrellion_daemon_reveal_true_parentage_decision_confirm


	# =====================================================
	# VISIBILITY
	#
	# All the ugly mechanical gating goes here rather than
	# being dumped into the visible Requirements list.
	# =====================================================

	is_shown = {

		this = character:Segrellion_60

		has_character_flag = segrellion_daemon_parentage_known

		NOT = {
			has_character_flag = segrellion_daemon_parentage_publicly_revealed
		}
	}


	# Nothing else is required once Daemon knows.
	is_valid = {
	}


	# =====================================================
	# EFFECTS
	# =====================================================

	effect = {

		custom_tooltip = segrellion_daemon_parentage_effect_aerys_tt
		custom_tooltip = segrellion_daemon_parentage_effect_public_tt
		custom_tooltip = segrellion_daemon_parentage_effect_genealogy_tt
		custom_tooltip = segrellion_daemon_parentage_effect_irreversible_tt


		hidden_effect = {
			segrellion_publicly_reveal_daemon_parentage_effect = yes
		}
	}


	# Daemon AI does not randomly expose himself.
	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
