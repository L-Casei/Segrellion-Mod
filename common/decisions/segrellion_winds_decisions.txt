# =========================================================
# NEGOTIATE FOR REINFORCEMENTS
# 300 AC - DAEMON SEGRELLION
# =========================================================

segrellion_negotiate_reinforcements_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/segrellion_negotiate_for_reinforcements_decision_background.dds"
	}

	decision_group_type = major

	desc = segrellion_negotiate_reinforcements_decision_desc
	selection_tooltip = segrellion_negotiate_reinforcements_decision_tooltip
	confirm_text = segrellion_negotiate_reinforcements_decision_confirm


	# =====================================================
	# VISIBILITY
	# =====================================================

	is_shown = {

		this = character:Segrellion_60

		current_date >= 8300.8.20

		NOT = {
			has_character_flag = segrellion_foreign_support_secured
		}

		NOT = {
			has_character_flag = segrellion_troop_negotiation_active
		}

		OR = {

			NOT = {
				has_character_flag = segrellion_stannis_support_route_closed
			}

			NOT = {
				has_character_flag = segrellion_dorne_support_route_closed
			}

			NOT = {
				has_character_flag = segrellion_daenerys_support_route_closed
			}

			NOT = {
				has_character_flag = segrellion_vale_support_route_closed
			}
		}
	}


	# =====================================================
	# REQUIREMENTS
	# =====================================================

	is_valid = {

		is_adult = yes
		is_imprisoned = no


		# =================================================
		# DAEMON HAS NOT YET RECLAIMED THE FLAMELANDS
		# =================================================

		custom_description = {

			text = segrellion_negotiate_reinforcements_no_flamelands_requirement

			NOT = {
				OR = {
					has_title = title:k_flamelands
					has_title = title:e_flamelands
				}
			}
		}


		# =================================================
		# THE UNICORN RETURNS HAS NOT YET BEGUN
		# =================================================

		custom_description = {

			text = segrellion_negotiate_reinforcements_before_unicorn_returns_requirement

			NOT = {
				OR = {
					has_character_flag = segrellion_unicorn_returns_decision_taken
					has_character_flag = segrellion_unicorn_returns_event_pending
					has_character_flag = segrellion_unicorn_returns_war_active
					has_character_flag = segrellion_unicorn_returns_completed
				}
			}
		}


		# =================================================
		# NOT ALREADY TRAVELLING
		# =================================================

		custom_description = {

			text = segrellion_negotiate_reinforcements_not_travelling_requirement

			NOT = {
				exists = current_travel_plan
			}
		}
	}


	# =====================================================
	# ROUTES
	# =====================================================

	widget = {

		gui = "decision_view_widget_option_list_generic"

		controller = decision_option_list_controller

		decision_to_second_step_button = "segrellion_negotiate_reinforcements_next_step"


		# -------------------------------------------------
		# STANNIS
		# -------------------------------------------------

		item = {

			value = segrellion_support_choice_stannis

			current_description = segrellion_support_choice_stannis_desc

			localization = segrellion_support_choice_stannis

			icon = "gfx/interface/icons/message_feed/diplomacy.dds"

			is_shown = {
				root = {
					NOT = {
						has_character_flag = segrellion_stannis_support_route_closed
					}
				}
			}

			is_valid = {

				character:Baratheon_3 ?= {

					is_alive = yes

					any_child = {
						is_alive = yes
						is_female = yes
						is_married = no
					}
				}

				root = {
					is_married = no
				}
			}

			ai_chance = {
				value = 10
			}
		}


		# -------------------------------------------------
		# DORNE
		# -------------------------------------------------

		item = {

			value = segrellion_support_choice_dorne

			current_description = segrellion_support_choice_dorne_desc

			localization = segrellion_support_choice_dorne

			# Different from Stannis.
			icon = "gfx/interface/icons/message_feed/intrigue.dds"

			is_shown = {
				root = {
					NOT = {
						has_character_flag = segrellion_dorne_support_route_closed
					}
				}
			}

			is_valid = {

				character:Martell_9 ?= {
					is_alive = yes
					is_married = no
				}

				root = {
					is_married = no
				}
			}

			ai_chance = {
				value = 40
			}
		}


		# -------------------------------------------------
		# DAENERYS
		# -------------------------------------------------

		item = {

			value = segrellion_support_choice_daenerys

			current_description = segrellion_support_choice_daenerys_desc

			localization = segrellion_support_choice_daenerys

			icon = "gfx/interface/icons/message_feed/martial.dds"

			is_shown = {
				root = {
					NOT = {
						has_character_flag = segrellion_daenerys_support_route_closed
					}
				}
			}

			is_valid = {

				character:Targaryen_9 ?= {
					is_alive = yes
				}
			}

			ai_chance = {
				value = 50
			}
		}

		# -------------------------------------------------
		# THE VALE
		# -------------------------------------------------

		item = {

			value = segrellion_support_choice_vale

			current_description = segrellion_support_choice_vale_desc

			localization = segrellion_support_choice_vale

			icon = "gfx/interface/icons/message_feed/diplomacy.dds"


			is_shown = {

				root = {

					NOT = {
						has_character_flag = segrellion_vale_support_route_closed
					}
				}
			}


			is_valid = {

				title:e_the_vale = {

					exists = holder

					holder = {

						is_alive = yes

						has_active_diarchy = yes

						diarch ?= {
							is_alive = yes
						}
					}
				}
			}

			# Weight for choosing this ROUTE among the four.
			# The 95/5 negotiation chance happens later.
			ai_chance = {
				value = 40

			}
		}
	}


	# =====================================================
	# EFFECTS
	# =====================================================

	effect = {

		# Visible Effects section.
		custom_tooltip = segrellion_negotiate_reinforcements_effect_journey_tt
		custom_tooltip = segrellion_negotiate_reinforcements_effect_bargain_tt
		custom_tooltip = segrellion_negotiate_reinforcements_effect_one_supporter_tt


		# Everything below is mechanical and should not
		# clutter the Effects section.
		hidden_effect = {

			add_character_flag = segrellion_troop_negotiation_active


			# -------------------------------------------------
			# STANNIS
			# -------------------------------------------------

			if = {

				limit = {
					scope:segrellion_support_choice_stannis = yes
				}

				add_character_flag = segrellion_negotiation_pending_stannis

				start_travel_plan = {

					destination = title:c_dragonstone.title_province

					return_trip = no

					on_arrival_event = segrellion_support.1000
					on_arrival_destinations = first

					on_travel_planner_cancel_event = segrellion_support.0090
				}
			}


			# -------------------------------------------------
			# DORNE
			# -------------------------------------------------

			else_if = {

				limit = {
					scope:segrellion_support_choice_dorne = yes
				}

				add_character_flag = segrellion_negotiation_pending_dorne

				start_travel_plan = {

					destination = title:c_sunspear.title_province

					return_trip = yes

					on_arrival_event = segrellion_support.1990
					on_arrival_destinations = first

					on_travel_planner_cancel_event = segrellion_support.0090
				}
			}


			# -------------------------------------------------
			# DAENERYS
			# -------------------------------------------------

			else_if = {

				limit = {
					scope:segrellion_support_choice_daenerys = yes
				}

				add_character_flag = segrellion_negotiation_pending_daenerys

				start_travel_plan = {

					destination = title:c_ghoyan_drohe.title_province

					return_trip = no

					on_arrival_event = segrellion_support.3000
					on_arrival_destinations = first

					on_travel_planner_cancel_event = segrellion_support.0090
					
				}
			}

			# -------------------------------------------------
			# THE VALE
			# -------------------------------------------------

			else_if = {

				limit = {
					scope:segrellion_support_choice_vale = yes
				}

				add_character_flag = segrellion_negotiation_pending_vale

				start_travel_plan = {

					# We use Robin's actual capital instead of hardcoding
					# a possibly changing Eyrie province key.
					destination = title:e_the_vale.holder.capital_province

					# Daemon returns to his camp after negotiations.
					return_trip = yes

					on_arrival_event = segrellion_support.6000
					on_arrival_destinations = first

					on_travel_planner_cancel_event = segrellion_support.0090
				}
			}
		}
	}


	ai_check_interval = 7

	ai_potential = {
		this = character:Segrellion_60
	}

	ai_will_do = {
		base = 100
	}
	
	trigger_if = {

		limit = {

			is_ai = yes

			character:Segrellion_43 ?= {
				is_alive = yes
			}
		}

		has_character_flag = segrellion_plate_demand_resolved
	}
}

# =========================================================
# EXPOSE MY TRUE PARENTAGE
# DAEMON SEGRELLION
#
# Visibility is handled by us.
# Public revelation is handled by AGOT's native
# royal bastard system.
# =========================================================

segrellion_expose_my_true_parentage_decision = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}

	decision_group_type = major

	sort_order = 151

	desc = agot_expose_my_true_parentage_decision_desc

	selection_tooltip = agot_expose_my_true_parentage_decision_tooltip

	confirm_text = agot_expose_my_true_parentage_decision_confirm


	is_shown = {

		this = character:Segrellion_60

		is_adult = yes
		is_ruler = yes

		NOT = {
			has_character_flag = royal_bastard
		}

		knows_self_royal_bastard_secret = yes

		any_known_secret = {

			secret_type = secret_agot_disputed_heritage

			secret_target = {

				this = root

				real_paternal_held_iron_throne_claim = yes
			}
		}
	}


	is_valid = {

		is_adult = yes
		is_ruler = yes

		real_paternal_held_iron_throne_claim = yes

		knows_self_royal_bastard_secret = yes
	}


	effect = {

		random_known_secret = {

			limit = {

				secret_type = secret_agot_disputed_heritage

				secret_target = {
					this = root
					real_paternal_held_iron_throne_claim = yes
				}

				can_be_exposed_by = root
			}

			save_scope_as = heritage_secret
		}


		# AGOT expects the person publicly revealing
		# the parentage in scope:revealer.
		save_scope_as = revealer


		# Native AGOT handler for making the royal bastard
		# revelation public and creating its story cycle.
		trigger_event = agot_events_bastard.0950
	}


	ai_check_interval = 12

	ai_potential = {
		this = character:Segrellion_60
	}

	ai_will_do = {
		base = 0
	}
}
