segrellion_poi_firstflame_dragonlore = {
	build_province_list = {
		province:4427 = {
			if = {
				limit = {
					OR = {
						has_building = segrellion_fort_01
						has_building = segrellion_aeryns_den_02
						has_building = segrellion_firstflame_03
					}
				}
				add_to_list = provinces
			}
		}
	}

	on_visit = {
		root = { trigger_event = segrellion_dragonlore.0001 }
		custom_tooltip = segrellion_poi_firstflame_dragonlore_tt
	}
}
