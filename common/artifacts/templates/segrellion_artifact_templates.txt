segrellion_icebringer_template = {
	can_equip = {
		always = yes
	}
	can_benefit = {
		custom_tooltip = {
			text = segrellion_icebringer_chosen_desc
			segrellion_icebringer_chosen_trigger = yes
		}
	}
	fallback = {
		# An undecided Icebringer remains cold and grants no power.
	}
	unique = yes
	ai_score = {
		value = 150
		if = {
			limit = { segrellion_icebringer_rejected_trigger = yes }
			add = -10000
		}
	}
}

segrellion_plate_of_segrell_template = {
	can_equip = {
		custom_tooltip = {
			text = segrellion_plate_of_segrell_worthy_desc
			is_adult = yes
			segrellion_valyrian_culture_trigger = yes
		}
	}
	can_benefit = {
		is_capable_adult = yes
		segrellion_valyrian_culture_trigger = yes
	}
	fallback = {
		#always = yes
	}
	unique = yes
	ai_score = {
		value = 150
		if = {
			limit = {
				NOT = {
					culture = {
						has_cultural_pillar = heritage_valyrian
					}
				}
			}
			add = -10000
		}
	}
}
