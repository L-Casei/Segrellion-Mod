# ============================================================
# DRAGONPIT AMENITIES
# Overridden from AGOT 00_agot_dragons_scripted_gui.txt
# ============================================================

agot_dragonpit_amenities_gui = {
	scope = character

	is_shown = {
		any_held_title = {
			tier = tier_county

			OR = {
				# Generic constructed Dragonpit
				any_county_province = {
					has_holding_type = castle_holding
					has_building_or_higher = generic_dragon_pit_01
				}

				# Historical / special Dragonpits
				any_county_province = {
					OR = {
						has_building_or_higher = dragonpit_01
						has_building = dragonpit_ruins_03
						has_building = agot_dragonmont_01

						# Caverns of Vhal
						has_building_or_higher = segrellion_vhal_dragonpit_02
					}
				}
			}

			has_variable = has_dragonkeeper_order
		}
	}
}


# ============================================================
# VIEW DRAGONS IN DRAGONPIT
# Overridden from AGOT 00_agot_dragons_scripted_gui.txt
# ============================================================

agot_view_dragons_in_dragonpit = {
	scope = title

	is_shown = {
		OR = {
			has_building_or_higher = generic_dragon_pit_01
			has_building_or_higher = dragonpit_01
			has_building = dragonpit_ruins_03
			has_building = agot_dragonmont_01

			# Caverns of Vhal
			has_building_or_higher = segrellion_vhal_dragonpit_02
		}
	}

	effect = {
		trigger_event = agot_dragon_pits.0999
	}
}
