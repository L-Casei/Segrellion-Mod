# =========================================================
# SEGRELLION - SPECIAL TWO-WIVES UI
#
# Any character using the native Segrellion second-wife
# system can display the secondary spouse row.
#
# The row is only shown once a real second spouse exists.
# =========================================================

segrellion_two_wives_secondary_spouses_shown = {

	scope = character


	is_shown = {

		has_character_flag = segrellion_two_wives_active

		number_of_spouses >= 2
	}
}
