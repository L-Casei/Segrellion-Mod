# =========================================================
# BOND WITH ICEBRINGER
#
# XP TRAIT
#
# TIER 0:   0 - 59 XP
# TIER I:  60 - 149 XP
# TIER II: 150 - 299 XP
# TIER III: 300+ XP
# =========================================================

segrellion_icebringer_bearer_trait = {

	category = lifestyle

	physical = no
	good = yes

	icon = segrellion_icebringer_bearer_trait.dds

	shown_in_ruler_designer = no
	shown_in_encyclopedia = yes


	# =====================================================
	# TIER 0
	# =====================================================

	prowess_no_portrait = 1
	advantage = 1


	# =====================================================
	# EXPERIENCE TRACK
	#
	# Bonuses are cumulative.
	# =====================================================

	track = {

		# -------------------------------------------------
		# TIER I - 60 XP
		# -------------------------------------------------

		60 = {
			martial = 1
			prowess_no_portrait = 2
			advantage = 1

			monthly_prestige = 0.10
		}


		# -------------------------------------------------
		# TIER II - 150 XP
		# -------------------------------------------------

		150 = {
			martial = 1
			prowess_no_portrait = 3
			advantage = 2

			monthly_prestige = 0.15

			knight_effectiveness_mult = 0.05
		}


		# -------------------------------------------------
		# TIER III - 300 XP
		# -------------------------------------------------

		300 = {
			martial = 2
			prowess_no_portrait = 4
			advantage = 2

			monthly_prestige = 0.25

			knight_effectiveness_mult = 0.10

			stress_gain_mult = -0.10
		}
	}


	# =====================================================
	# DYNAMIC NAME
	# =====================================================

	name = {
		first_valid = {

			triggered_desc = {
				trigger = {
					exists = this

					has_trait_xp = {
						trait = segrellion_icebringer_bearer_trait
						value >= 300
					}
				}

				desc = segrellion_icebringer_bond_trait_3
			}


			triggered_desc = {
				trigger = {
					exists = this

					has_trait_xp = {
						trait = segrellion_icebringer_bearer_trait
						value >= 150
					}
				}

				desc = segrellion_icebringer_bond_trait_2
			}


			triggered_desc = {
				trigger = {
					exists = this

					has_trait_xp = {
						trait = segrellion_icebringer_bearer_trait
						value >= 60
					}
				}

				desc = segrellion_icebringer_bond_trait_1
			}


			desc = segrellion_icebringer_bond_trait_0
		}
	}


	desc = desc_icebringer_bond
}
