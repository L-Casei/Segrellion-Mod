# =========================================================
# ICEBRINGER BEARER
#
# XP TRAIT
#
# 0-49   Bearer
# 50-99  Adept
# 100    Master
# =========================================================

segrellion_icebringer_bearer_trait = {

	category = lifestyle

	physical = no
	good = yes

	icon = segrellion_icebringer_bearer_trait.dds

	shown_in_ruler_designer = no
	shown_in_encyclopedia = yes


	# -----------------------------------------------------
	# LEVEL I
	# -----------------------------------------------------

	martial = 1
	prowess_no_portrait = 2
	advantage = 1


	# -----------------------------------------------------
	# EXPERIENCE TRACK
	# -----------------------------------------------------

	track = {

		50 = {
			martial = 1
			prowess_no_portrait = 2
			advantage = 1
		}


		100 = {
			martial = 1
			prowess_no_portrait = 2
			advantage = 1

			monthly_prestige = 0.25

			stress_gain_mult = -0.10
		}
	}


	name = {
		first_valid = {

			triggered_desc = {
				trigger = {
					exists = this

					has_trait_xp = {
						trait = segrellion_icebringer_bearer_trait
						value >= 100
					}
				}

				desc = segrellion_icebringer_bearer_trait_3
			}


			triggered_desc = {
				trigger = {
					exists = this

					has_trait_xp = {
						trait = segrellion_icebringer_bearer_trait
						value >= 50
					}
				}

				desc = segrellion_icebringer_bearer_trait_2
			}


			desc = segrellion_icebringer_bearer_trait_1
		}
	}


	desc = segrellion_icebringer_bearer_trait_desc
}
