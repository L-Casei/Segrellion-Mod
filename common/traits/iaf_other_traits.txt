# =========================================================
# UNSULLIED
# =========================================================

segrellion_unsullied = {

	category = health

	icon = segrellion_unsullied_trait.dds

	# A lifetime of military drill and discipline.
	martial = 1
	prowess = 2

	# Brotherhood between Unsullied.
	same_opinion = 10

	# Their reputation alone inspires a measure of fear.
	dread_baseline_add = 5

	# Unsullied cannot marry or father children.
	flag = can_not_marry
	can_have_children = no

	shown_in_ruler_designer = yes
	ruler_designer_cost = 10
}

# =========================================================
# RESURRECTED
#
# A character who has truly died and returned to life.
#
# This trait represents the universal consequences of
# resurrection itself.
#
# Character-specific changes belong in separate modifiers
# or events.
# =========================================================

segrellion_resurrected_trait = {

    category = health

    physical = yes

    icon = segrellion_resurrected_trait.dds

    shown_in_ruler_designer = no
    shown_in_encyclopedia = yes

    name = segrellion_resurrected_trait
    desc = segrellion_resurrected_trait_desc


    # =====================================================
    # DEATH HAS ALREADY CLAIMED THEM ONCE
    # =====================================================

    health = 0.5


    # =====================================================
    # SOMETHING UNNATURAL HAS RETURNED
    # =====================================================

    dread_baseline_add = 50

    dread_decay_mult = -0.50


    # =====================================================
    # THE LIVING ARE UNEASY AROUND THEM
    # =====================================================

    general_opinion = -10

    attraction_opinion = -25


    # =====================================================
    # LESS ATTACHED TO LIFE
    # =====================================================

    fertility = -0.25

    stress_gain_mult = -0.20
}

# =========================================================
# DREAMER
#
# PROGRESSION TRAIT
#
# LEVEL 0:   0 - 9 XP
# LEVEL I:  10 - 24 XP
# LEVEL II: 25 - 49 XP
# LEVEL III: 50 - 99 XP
# LEVEL IV: 100+ XP
# =========================================================

segrellion_dreamer_trait = {

	category = lifestyle
	physical = no
	good = yes

	icon = segrellion_dreamer_trait.dds

	shown_in_ruler_designer = no
	shown_in_encyclopedia = yes


	# =====================================================
	# DREAMER PROGRESSION
	#
	# Track modifiers are cumulative.
	#
	# LEVEL IV TOTAL:
	# +3 Learning
	# +2 Intrigue
	# -10% Stress Gain
	# =====================================================

	track = {

		# =================================================
		# LEVEL I
		# 10 XP
		# =================================================

		10 = {
			learning = 1
		}


		# =================================================
		# LEVEL II
		# 25 XP
		# =================================================

		25 = {
			intrigue = 1
		}


		# =================================================
		# LEVEL III
		# 50 XP
		# =================================================

		50 = {
			learning = 1
			stress_gain_mult = -0.05
		}


		# =================================================
		# LEVEL IV
		# 100 XP
		# =================================================

		100 = {
			learning = 1
			intrigue = 1
			stress_gain_mult = -0.05
		}
	}


	# =====================================================
	# DYNAMIC NAME
	# =====================================================

	name = {
		first_valid = {

			triggered_desc = {
				trigger = {
					exists = this
					has_trait_xp = {
						trait = segrellion_dreamer_trait
						value >= 100
					}
				}

				desc = segrellion_dreamer_trait_4
			}


			triggered_desc = {
				trigger = {
					exists = this
					has_trait_xp = {
						trait = segrellion_dreamer_trait
						value >= 50
					}
				}

				desc = segrellion_dreamer_trait_3
			}


			triggered_desc = {
				trigger = {
					exists = this
					has_trait_xp = {
						trait = segrellion_dreamer_trait
						value >= 25
					}
				}

				desc = segrellion_dreamer_trait_2
			}


			triggered_desc = {
				trigger = {
					exists = this
					has_trait_xp = {
						trait = segrellion_dreamer_trait
						value >= 10
					}
				}

				desc = segrellion_dreamer_trait_1
			}


			desc = segrellion_dreamer_trait_0
		}
	}


	desc = segrellion_dreamer_trait_desc
}
