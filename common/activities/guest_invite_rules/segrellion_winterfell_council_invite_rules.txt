# =========================================================
# GREAT LORDS OF THE NORTH
#
# Duchy tier and above.
# These are automatically selected by default.
# =========================================================

activity_invite_rule_segrellion_northern_great_lords = {

	effect = {

		every_ruler = {

			limit = {

				this != root

				is_alive = yes
				is_adult = yes

				segrellion_is_northern_great_lord_for_council_trigger = yes
			}

			add_to_list = characters
		}
	}
}

# =========================================================
# NIGHT'S WATCH
#
# Lord Commander + notable Black Brothers.
# =========================================================

activity_invite_rule_segrellion_nights_watch = {

	effect = {

		title:k_the_wall.holder ?= {

			if = {

				limit = {

					this != root

					is_alive = yes
					is_adult = yes
				}

				add_to_list = characters
			}


			every_courtier = {

				limit = {

					is_alive = yes
					is_adult = yes

					has_trait = nightswatch

					prowess >= 10
				}

				add_to_list = characters
			}
		}
	}
}
