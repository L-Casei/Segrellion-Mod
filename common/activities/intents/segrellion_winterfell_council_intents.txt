# ---------------------------------------------------------
# NAME A KING
# Target-based intent
# ---------------------------------------------------------

segrellion_winterfell_council_kingmaker_intent = {

	icon = intent_legitimacy

	on_invalidated = {
		trigger_event = activity_system.0090
	}

	ai_targets = {
		ai_recipients = activity_host
	}

	ai_targets = {
		ai_recipients = activity_guests
		max = 20
	}

	is_target_valid = {

		this != scope:target

		scope:target = {

			is_alive = yes
			is_adult = yes

			NOT = {
				has_trait = nightswatch
			}

			involved_activity ?= root.involved_activity
		}
	}

	on_target_invalidated = {
		trigger_event = activity_system.0091
	}

	ai_will_do = {

		value = 80

		if = {
			limit = {
				has_trait = ambitious
			}

			add = 30
		}

		if = {
			limit = {
				has_trait = content
			}

			add = -20
		}
	}

	ai_target_score = {

		value = 1

		if = {
			limit = {
				scope:target = {
					has_title = title:e_the_north
				}
			}

			add = 250
		}

		if = {
			limit = {
				scope:target = {
					has_title = title:d_winterfell
				}
			}

			add = 100
		}

		if = {
			limit = {
				scope:target = {
					is_landed = yes
				}
			}

			add = 25
		}

		if = {
			limit = {
				scope:target = {
					has_trait = nightswatch
				}
			}

			add = -1000
		}
	}

	scripted_animation = {
		animation = thinking
	}
}
