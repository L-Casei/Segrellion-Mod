activity_segrellion_winterfell_council = {

	sort_order = 100

	# =====================================================
	# SHOWN ONLY FOR JON AFTER STANNIS' DEATH
	# =====================================================

	is_shown = {

		segrellion_is_current_jon_trigger = yes

		has_character_flag = segrellion_jon_winterfell_council_called

		NOT = {
			has_character_flag = segrellion_jon_king_in_the_north
		}
	}


	# =====================================================
	# PLANNING / START REQUIREMENTS
	# =====================================================

	can_start_showing_failures_only = {

		is_available_adult = yes

		is_landed = yes

		has_title = title:d_winterfell
	}


	can_plan = {

		is_available_adult = yes

		is_landed = yes

		has_title = title:d_winterfell
	}


	can_start = {

		is_available_adult = yes

		is_landed = yes

		has_title = title:d_winterfell
	}


	# =====================================================
	# VALIDITY
	# =====================================================

	is_valid = {

		NOT = {
			has_variable = activity_invalidated
		}


		scope:host = {

			is_alive = yes

			is_capable_adult = yes

			is_imprisoned = no

			is_landed = yes

			has_title = title:d_winterfell
		}


		trigger_if = {

			limit = {
				is_current_phase_active = yes
			}

			has_attending_activity_guests = yes
		}
	}


	# =====================================================
	# ACTIVITY HEADER BACKGROUND
	# =====================================================

	background = {

		texture = "gfx/interface/illustrations/activity_header_backgrounds/activity_segrellion_winterfell_council.dds"
	}


	# =====================================================
	# COST
	# =====================================================

	cost = {

		gold = {
			value = 25
		}
	}


	# =====================================================
	# LOCATION
	# WINTERFELL ONLY
	# =====================================================

	is_single_location = yes


	province_filter = realm

	ai_province_filter = realm


	max_route_deviation_mult = 1.0


	is_location_valid = {

		this = title:c_winterfell.title_province
	}


	province_description = {

		desc = activity_segrellion_winterfell_council_province_desc
	}


	max_province_icons = 1


	province_score = {

		value = 0


		if = {

			limit = {

				this = title:c_winterfell.title_province
			}

			add = 1000
		}
	}


	# =====================================================
	# PREPARATION TIME
	# LORDS GATHER FOR TWO MONTHS
	# =====================================================

	wait_time_before_start = {

		days = 60
	}


	max_guest_arrival_delay_time = {

		days = 30
	}


	# =====================================================
	# ACTIVITY OPTIONS
	# =====================================================

	options = {

		segrellion_winterfell_council_scope = {

			segrellion_winterfell_council_scope_full = {

				default = yes


				ai_will_do = {

					value = 100
				}
			}
		}
	}


	# =====================================================
	# HOST INTENT
	# USE VANILLA LEGITIMACY INTENT
	# =====================================================

	host_intents = {

		intents = {
			legitimacy_intent
		}

		default = legitimacy_intent

		player_defaults = {
			legitimacy_intent
		}
	}


	# =====================================================
	# GUESTS
	# =====================================================

	max_guests = 100


	# =====================================================
	# INVITATION GROUPS
	#
	# Great Northern Lords are selected automatically.
	# The player may additionally summon Counts, Barons,
	# family, friends and members of the Night's Watch.
	# =====================================================

	guest_invite_rules = {

		rules = {

			2 = activity_invite_rule_segrellion_northern_counts

			3 = activity_invite_rule_segrellion_northern_barons

			2 = activity_invite_rule_close_family

			2 = activity_invite_rule_friends

			3 = activity_invite_rule_segrellion_nights_watch
		}


		defaults = {

			1 = activity_invite_rule_segrellion_northern_great_lords
		}
	}


	# =====================================================
	# WHO MAY ATTEND
	#
	# Do NOT restrict this to Jon's current vassals.
	# The council represents the political North as a whole.
	# =====================================================

	can_be_activity_guest = {

		is_adult = yes

		in_diplomatic_range = scope:host

		is_pool_guest = no
	}


	# =====================================================
	# ATTENDANCE
	#
	# Anyone formally summoned to this council attends.
	# =====================================================

	guest_join_chance = {

		base = 100
	}


	# =====================================================
	# THE COUNCIL PHASE
	# =====================================================

	phases = {

		segrellion_winterfell_council_phase = {

			is_predefined = yes

			order = 1

			location_source = pickable


			can_pick = {

				scope:province = title:c_winterfell.title_province
			}


			is_valid = {

				scope:province = title:c_winterfell.title_province
			}


			on_phase_active = {

				scope:host = {

					trigger_event = {

						id = segrellion_jon_frey.0020

						days = 1
					}
				}
			}
		}
	}


	# =====================================================
	# AI JON
	# =====================================================

	ai_check_interval = 1


	ai_will_do = {

		value = 1000
	}
}
