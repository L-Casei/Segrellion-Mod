activity_segrellion_winterfell_council = {

	sort_order = 100


	# =====================================================
	# STORY AVAILABILITY
	# =====================================================

	is_shown = {

		segrellion_is_current_jon_trigger = yes

		has_character_flag = segrellion_jon_winterfell_council_called


		character:Baratheon_3 ?= {
			is_alive = no
		}


		NOT = {
			has_character_flag = segrellion_jon_winterfell_council_completed
		}
	}


	# =====================================================
	# NO VISIBLE REQUIREMENTS
	# =====================================================

	can_start_showing_failures_only = {
		always = yes
	}

	# Modern activity planner gate.
	# "can_plan" is obsolete in the current activity schema.
	can_always_plan = yes

	can_start = {
		always = yes
	}


	# =====================================================
	# RUNTIME VALIDITY
	# =====================================================

	is_valid = {
		scope:host = {
			is_alive = yes
		}
	}


	# =====================================================
	# PRESENTATION
	# =====================================================

	background = {
		texture = "gfx/interface/illustrations/activity_header_backgrounds/activity_segrellion_winterfell_council.dds"
	}


	# =====================================================
	# COST
	# =====================================================

	cost = {
		prestige = {

			# Story-critical AI Jon must never fail to plan the
			# Council because his prestige is low or negative.
			if = {
				limit = {
					is_ai = yes
					segrellion_is_current_jon_trigger = yes
					has_character_flag = segrellion_jon_winterfell_council_called
				}
				value = 0
			}

			else = {
				value = 300
			}
		}
	}


	# =====================================================
	# LOCATION
	# =====================================================

	is_single_location = yes

	province_filter = realm
	ai_province_filter = realm

	max_route_deviation_mult = 1.0

	is_location_valid = {
		this = title:c_winterfell.title_province
	}

	province_description = {
		desc = activity_segrellion_winterfell_council_province_desc
	}

	max_province_icons = 1

	province_score = {
		value = 0

		if = {
			limit = {
				this = title:c_winterfell.title_province
			}

			add = 1000
		}
	}

	# Required by the current AI planner to actually select a
	# valid location after choosing this activity type.
	ai_will_select_province = {
		value = 0

		if = {
			limit = {
				this = title:c_winterfell.title_province
			}

			add = 10000
		}
	}


	# =====================================================
	# PREPARATION
	# =====================================================

	# Northern lords need actual time to reach Winterfell.
	# One day caused AI-hosted councils to begin before any
	# invited lord could physically arrive.
	wait_time_before_start = {
		days = 60
	}

	max_guest_arrival_delay_time = {
		days = 60
	}


	# =====================================================
	# HOST INTENTS
	# =====================================================

	host_intents = {

		intents = {
			segrellion_winterfell_council_intent
			segrellion_winterfell_council_long_night_intent
			segrellion_winterfell_council_revenge_intent
			segrellion_winterfell_council_independence_intent
			segrellion_winterfell_council_kingmaker_intent
			segrellion_winterfell_council_defend_north_intent
			segrellion_winterfell_council_dragon_alliance_intent
		}

		default = segrellion_winterfell_council_intent

		player_defaults = {
			segrellion_winterfell_council_intent
		}
	}


	# =====================================================
	# GUEST INTENTS
	# =====================================================

	guest_intents = {

		intents = {
			segrellion_winterfell_council_intent
			segrellion_winterfell_council_long_night_intent
			segrellion_winterfell_council_revenge_intent
			segrellion_winterfell_council_independence_intent
			segrellion_winterfell_council_kingmaker_intent
			segrellion_winterfell_council_defend_north_intent
			segrellion_winterfell_council_dragon_alliance_intent
		}

		default = segrellion_winterfell_council_intent

		player_defaults = {
			segrellion_winterfell_council_intent
		}
	}


	# =====================================================
	# GUESTS
	# =====================================================

	max_guests = 100

	# Do not let a temporary lack of valid invitees prevent the
	# story-critical Council from being created.
	allow_zero_guest_invites = yes

	guest_invite_rules = {

		rules = {
			2 = activity_invite_rule_segrellion_northern_counts
			2 = activity_invite_rule_close_family
			2 = activity_invite_rule_friends
			3 = activity_invite_rule_segrellion_nights_watch
		}

		defaults = {
			1 = activity_invite_rule_segrellion_northern_great_lords
			1 = activity_invite_rule_segrellion_northern_counts
			1 = activity_invite_rule_vassals
			1 = activity_invite_rule_close_family
		}
	}

	can_be_activity_guest = {

		OR = {

			is_adult = yes

			# Stark heirs can be brought before the council
			# even if they are still underage.
			dynasty = character:Stark_Dummy_8.dynasty
		}

		in_diplomatic_range = scope:host

		is_pool_guest = no
	}

	guest_join_chance = {
		base = 100
	}


	# =====================================================
	# PHASES
	# =====================================================

	phases = {


		# =================================================
		# I - THE LORDS ARRIVE
		# =================================================

		segrellion_winterfell_council_arrivals_phase = {

			is_predefined = yes
			order = 1
			location_source = pickable

			can_pick = {
				scope:province = title:c_winterfell.title_province
			}

			is_valid = {
				scope:province = title:c_winterfell.title_province
			}

			on_phase_active = {

				if = {
					limit = {
						this = scope:host
					}

					segrellion_winterfell_council_initialize_effect = yes

					scope:activity = {
						progress_activity_phase_after = {
							days = 14
						}
					}
				}

				segrellion_winterfell_council_apply_intent_support_effect = yes

				trigger_event = {
					on_action = segrellion_winterfell_council_arrivals_events
				}
			}

			on_weekly_pulse = {

				segrellion_winterfell_council_apply_intent_support_effect = yes

				trigger_event = {
					on_action = segrellion_winterfell_council_arrivals_events
				}
			}
		}


		# =================================================
		# II - THE DEBATE
		# =================================================

		segrellion_winterfell_council_debate_phase = {

			is_predefined = yes
			order = 2
			location_source = pickable

			can_pick = {
				scope:province = title:c_winterfell.title_province
			}

			is_valid = {
				scope:province = title:c_winterfell.title_province
			}


			# CRITICAL:
			# later predefined phases must transition from
			# "entering phase" to "active phase".

			on_enter_phase = {

				if = {
					limit = {
						this = scope:host
					}

					scope:activity = {
						progress_activity_phase_after = {
							days = 1
						}
					}
				}
			}

			on_phase_active = {

				if = {
					limit = {
						this = scope:host
					}

					scope:activity = {
						progress_activity_phase_after = {
							days = 21
						}
					}
				}

				segrellion_winterfell_council_apply_intent_support_effect = yes

				trigger_event = {
					on_action = segrellion_winterfell_council_debate_events
				}
			}

			on_weekly_pulse = {

				segrellion_winterfell_council_apply_intent_support_effect = yes

				trigger_event = {
					on_action = segrellion_winterfell_council_debate_events
				}
			}
		}


		# =================================================
		# III - THE PROPOSALS
		# =================================================

		segrellion_winterfell_council_proposals_phase = {

			is_predefined = yes
			order = 3
			location_source = pickable

			can_pick = {
				scope:province = title:c_winterfell.title_province
			}

			is_valid = {
				scope:province = title:c_winterfell.title_province
			}

			on_enter_phase = {

				if = {
					limit = {
						this = scope:host
					}

					scope:activity = {
						progress_activity_phase_after = {
							days = 1
						}
					}
				}
			}

			on_phase_active = {

				if = {
					limit = {
						this = scope:host
					}

					scope:activity = {
						progress_activity_phase_after = {
							days = 14
						}
					}
				}

				segrellion_winterfell_council_apply_intent_support_effect = yes

				trigger_event = {
					on_action = segrellion_winterfell_council_proposals_events
				}
			}

			on_weekly_pulse = {

				segrellion_winterfell_council_apply_intent_support_effect = yes

				trigger_event = {
					on_action = segrellion_winterfell_council_proposals_events
				}
			}
		}


		# =================================================
		# IV - THE PROCLAMATION
		# =================================================

		segrellion_winterfell_council_proclamation_phase = {

			is_predefined = yes
			order = 4
			location_source = pickable

			can_pick = {
				scope:province = title:c_winterfell.title_province
			}

			is_valid = {
				scope:province = title:c_winterfell.title_province
			}

			on_enter_phase = {

				if = {
					limit = {
						this = scope:host
					}

					scope:activity = {
						progress_activity_phase_after = {
							days = 1
						}
					}
				}
			}

			on_phase_active = {

				if = {
					limit = {
						this = scope:host
					}

					scope:activity = {
						progress_activity_phase_after = {
							days = 8
						}
					}

					trigger_event = {
						id = segrellion_winterfell_council.4900
						days = 4
					}
				}

				segrellion_winterfell_council_apply_intent_support_effect = yes

				trigger_event = {
					on_action = segrellion_winterfell_council_proclamation_events
				}
			}

			on_weekly_pulse = {

				segrellion_winterfell_council_apply_intent_support_effect = yes

				trigger_event = {
					on_action = segrellion_winterfell_council_proclamation_events
				}
			}
		}
	}


	# =====================================================
	# COMPLETION
	# =====================================================

	on_complete = {

		scope:activity = {

			every_attending_character = {

				remove_variable = segrellion_council_king_support
				remove_variable = segrellion_council_king_intent_support
				remove_variable = segrellion_council_king_total_support
				remove_variable = segrellion_council_influence
			}
		}


		scope:host = {

			remove_variable = segrellion_council_king_support
			remove_variable = segrellion_council_king_intent_support
			remove_variable = segrellion_council_king_total_support
			remove_variable = segrellion_council_influence


			# =================================================
			# THE COUNCIL IS NOW COMPLETE
			# =================================================

			add_character_flag = segrellion_jon_winterfell_council_completed


			# =================================================
			# CANON HARD FAILSAFE
			#
			# Normally Jon has already been proclaimed during
			# the proclamation phase.
			#
			# This now exists ONLY to repair a failed final
			# council event.
			# =================================================

			segrellion_jon_force_canon_north_recovery_effect = yes


			remove_character_flag = segrellion_jon_winterfell_council_called

			remove_character_flag = segrellion_jon_winterfell_council_war_locked

			remove_character_flag = sworn_peace

			remove_character_flag = segrellion_jon_winterfell_council_ai_spawn_cooldown


			# =================================================
			# DAENERYS' UNANSWERED LETTER
			#
			# Jon deliberately postponed the matter until
			# after the Winterfell Council.
			# =================================================

			if = {

				limit = {

					has_character_flag = segrellion_jon_daenerys_reply_deferred

					character:Targaryen_9 ?= {
						is_alive = yes
					}
				}


				if = {

					limit = {

						character:Seaworth_1 ?= {

							is_alive = yes
							is_imprisoned = no
						}
					}


					trigger_event = {

						id = segrellion_daenerys_jon.0002
						days = 1
					}
				}


				else = {

					trigger_event = {

						id = segrellion_daenerys_jon.0003
						days = 1
					}
				}
			}
		}
	}


	# =====================================================
	# AI
	# =====================================================

	# Check every month at every landed tier Jon can hold.
	ai_check_interval_by_tier = {
		barony = 0
		county = 1
		duchy = 1
		kingdom = 1
		empire = 1
		hegemony = 1
	}

	# Far above the normal activity threshold. Once the Council
	# is available, Jon should choose it over ordinary activities.
	ai_will_do = {
		value = 1000
	}
}
