on_birthday = {
	on_actions = {
		segrellion_legacy_childhood_birthday
	}
}

segrellion_legacy_childhood_birthday = {
	trigger = {
		age >= 8
		age <= 15
		dynasty ?= { has_dynasty_perk = segrellion_legacy_1 }
		NOT = { has_character_flag = segrellion_weight_of_truth_event_seen }
		NOR = {
			has_trait = honest
			has_trait = just
		}
	}

	random_events = {
		65 = 0
		35 = segrellion_legacy.0001
	}
}
