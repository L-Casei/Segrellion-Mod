# =========================================================
# DREAMER FLAVOR ON ACTIONS
# =========================================================


yearly_global_pulse = {

	on_actions = {

		segrellion_dreamer_flavor_yearly_check
	}
}


# =========================================================
# DREAM FREQUENCY
#
# RARE
# LEVEL 0: 20%
# LEVEL 1: 25%
# LEVEL 2: 30%
# LEVEL 3: 35%
#
# NORMAL
# LEVEL 0: 45%
# LEVEL 1: 50%
# LEVEL 2: 55%
# LEVEL 3: 60%
#
# FREQUENT
# LEVEL 0: 70%
# LEVEL 1: 75%
# LEVEL 2: 80%
# LEVEL 3: 85%
# =========================================================

segrellion_dreamer_flavor_yearly_check = {

	effect = {

		every_player = {

			limit = {

				is_alive = yes
				is_human = yes
				is_adult = yes

				has_trait = segrellion_dreamer_trait

				NOT = {
					has_trait = incapable
				}

				# Do not throw a random flavor dream into the
				# middle of the Icebringer awakening chain.
				NOT = {
					has_character_flag = segrellion_icebringer_awakening_chain_active
				}
			}

			segrellion_sync_dreamer_visual_level_effect = yes


			# =================================================
			# LEVEL 3
			# =================================================

			if = {

				limit = {

					has_trait_xp = {
						trait = segrellion_dreamer_trait
						value >= 50
					}
				}


				# RARE
				if = {

					limit = {
						has_game_rule = segrellion_dreamer_dream_frequency_rare
					}

					random = {
						chance = 35
						segrellion_dreamer_flavor_pool_effect = yes
					}
				}


				# FREQUENT
				else_if = {

					limit = {
						has_game_rule = segrellion_dreamer_dream_frequency_frequent
					}

					random = {
						chance = 85
						segrellion_dreamer_flavor_pool_effect = yes
					}
				}


				# NORMAL / FALLBACK
				else = {

					random = {
						chance = 60
						segrellion_dreamer_flavor_pool_effect = yes
					}
				}
			}


			# =================================================
			# LEVEL 2
			# =================================================

			else_if = {

				limit = {

					has_trait_xp = {
						trait = segrellion_dreamer_trait
						value >= 25
					}
				}


				# RARE
				if = {

					limit = {
						has_game_rule = segrellion_dreamer_dream_frequency_rare
					}

					random = {
						chance = 30
						segrellion_dreamer_flavor_pool_effect = yes
					}
				}


				# FREQUENT
				else_if = {

					limit = {
						has_game_rule = segrellion_dreamer_dream_frequency_frequent
					}

					random = {
						chance = 80
						segrellion_dreamer_flavor_pool_effect = yes
					}
				}


				# NORMAL / FALLBACK
				else = {

					random = {
						chance = 55
						segrellion_dreamer_flavor_pool_effect = yes
					}
				}
			}


			# =================================================
			# LEVEL 1
			# =================================================

			else_if = {

				limit = {

					has_trait_xp = {
						trait = segrellion_dreamer_trait
						value >= 10
					}
				}


				# RARE
				if = {

					limit = {
						has_game_rule = segrellion_dreamer_dream_frequency_rare
					}

					random = {
						chance = 25
						segrellion_dreamer_flavor_pool_effect = yes
					}
				}


				# FREQUENT
				else_if = {

					limit = {
						has_game_rule = segrellion_dreamer_dream_frequency_frequent
					}

					random = {
						chance = 75
						segrellion_dreamer_flavor_pool_effect = yes
					}
				}


				# NORMAL / FALLBACK
				else = {

					random = {
						chance = 50
						segrellion_dreamer_flavor_pool_effect = yes
					}
				}
			}


			# =================================================
			# LEVEL 0
			# =================================================

			else = {


				# RARE
				if = {

					limit = {
						has_game_rule = segrellion_dreamer_dream_frequency_rare
					}

					random = {
						chance = 20
						segrellion_dreamer_flavor_pool_effect = yes
					}
				}


				# FREQUENT
				else_if = {

					limit = {
						has_game_rule = segrellion_dreamer_dream_frequency_frequent
					}

					random = {
						chance = 70
						segrellion_dreamer_flavor_pool_effect = yes
					}
				}


				# NORMAL / FALLBACK
				else = {

					random = {
						chance = 45
						segrellion_dreamer_flavor_pool_effect = yes
					}
				}
			}
		}
	}
}
