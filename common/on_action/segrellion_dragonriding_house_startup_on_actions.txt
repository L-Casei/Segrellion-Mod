on_game_start_after_lobby = {
	on_actions = {
		segrellion_check_dragonriding_house_on_start
	}
}

segrellion_check_dragonriding_house_on_start = {
	effect = {
		if = {
			limit = {
				current_date >= 8120.1.1
			}

			house:house_Segrellion = {
				if = {
					limit = {
						NOT = { has_house_modifier = dragonrider_house_modifier }
					}
					add_house_modifier = {
						modifier = dragonrider_house_modifier
					}
				}
			}

			# Opcional: también para Stormflame
			house:house_Stormflame = {
				if = {
					limit = {
						NOT = { has_house_modifier = dragonrider_house_modifier }
					}
					add_house_modifier = {
						modifier = dragonrider_house_modifier
					}
				}
			}
		}
	}
}
