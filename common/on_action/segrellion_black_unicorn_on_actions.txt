# =========================================================
# THE BLACK UNICORN
# STARTUP
# =========================================================


on_game_start_after_lobby = {

	on_actions = {

		segrellion_black_unicorn_startup
	}
}


segrellion_black_unicorn_startup = {

	effect = {


		# =================================================
		# MYSTERIOUS WOMAN + ICEBRINGER
		# =================================================

		character:Black_Unicorn_1 ?= {

			if = {

				limit = {

					is_alive = yes

					game_start_date = 8300.8.20
				}


				add_character_flag = segrellion_black_unicorn_mysterious_woman
				add_character_flag = segrellion_icebringer_authorized

				add_character_flag = segrellion_black_unicorn_skagos_outfit

				add_character_flag = segrellion_black_unicorn_lock_camp_skagos


				set_location = title:b_cold_peaks.title_province


				domicile ?= {

					if = {

						limit = {

							domicile_location != title:b_cold_peaks.title_province
						}


						move_domicile = title:b_cold_peaks.title_province
					}
				}


				if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_black_unicorn_army_created
						}
					}


					create_maa_regiment = {

						type = segrellion_skagosi_unicorns

						check_can_recruit = no

						size = 1
					}


					create_maa_regiment = {

						type = segrellion_skagosi_unicorns

						check_can_recruit = no

						size = 1
					}


					create_maa_regiment = {

						type = segrellion_skagosi_raiders

						check_can_recruit = no

						size = 1
					}


					create_maa_regiment = {

						type = segrellion_skagosi_raiders

						check_can_recruit = no

						size = 1
					}


					add_character_flag = segrellion_black_unicorn_army_created
				}


				if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_black_unicorn_camp_lock_started
						}
					}


					add_character_flag = segrellion_black_unicorn_camp_lock_started


					trigger_event = {

						id = segrellion_black_unicorn.0000

						days = 3
					}
				}


				set_global_variable = {
					name = segrellion_black_unicorn_woman
					value = this
				}


				if = {

					limit = {

						NOT = {
							exists = global_var:segrellion_icebringer
						}
					}


					segrellion_create_artifact_icebringer_effect = {
						OWNER = this
					}
				}
			}
		}


		# =================================================
		# DAEMON STORY START
		#
		# Randomly 60-120 days after game start.
		# =================================================

		character:Segrellion_60 ?= {

			if = {

				limit = {

					is_alive = yes
					is_adult = yes

					game_start_date = 8300.8.20

					NOT = {
						has_character_flag = segrellion_black_unicorn_opening_queued
					}

					NOT = {
						has_character_flag = segrellion_black_unicorn_opening_seen
					}
				}


				add_character_flag = segrellion_black_unicorn_opening_queued


				trigger_event = {

					id = segrellion_black_unicorn.0001

					days = { 60 120 }
				}
			}
		}
	}
}
