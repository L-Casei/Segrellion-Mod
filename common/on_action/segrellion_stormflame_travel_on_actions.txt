# Check every province entered, independently of the random travel-event pool.
on_travel_plan_movement = {
	on_actions = {
		segrellion_stormflame_travel_check
	}
}

segrellion_stormflame_travel_check = {
	trigger = {
		exists = current_travel_plan
		current_travel_plan.current_location = {
			is_sea_province = no
			OR = {
				province_owner = {
					is_alive = yes
					house = house:house_Stormflame
				}
				county.holder = {
					is_alive = yes
					house = house:house_Stormflame
				}
			}
		}
		# The personal Black Unicorn expedition must reach Skagos and return
		# without the generic Stormflame border capture ending the story.
		# Both flags are set before departure; storyline_active is cleared
		# on return (.0190) or cancellation (.0090). Protect this party only.
		NOT = {
			character:Segrellion_60 = {
				has_character_flag = segrellion_black_unicorn_storyline_active
				has_character_flag = segrellion_black_unicorn_daemon_personal_route
				exists = current_travel_plan
				current_travel_plan = root.current_travel_plan
			}
		}
		OR = {
			character:Segrellion_60 = {
				is_alive = yes
				is_imprisoned = no
				exists = current_travel_plan
				current_travel_plan = root.current_travel_plan
			}
			character:Segrellion_43 = {
				is_alive = yes
				is_imprisoned = no
				exists = current_travel_plan
				current_travel_plan = root.current_travel_plan
			}
		}
	}
	effect = {
		# Resolve both travelers before imprisonment can end their shared plan.
		character:Segrellion_60 ?= {
			if = {
				limit = {
					is_alive = yes
					is_imprisoned = no
					exists = current_travel_plan
					current_travel_plan = root.current_travel_plan
				}
				save_scope_as = stormflame_travel_daemon
			}
		}
		character:Segrellion_43 ?= {
			if = {
				limit = {
					is_alive = yes
					is_imprisoned = no
					exists = current_travel_plan
					current_travel_plan = root.current_travel_plan
				}
				save_scope_as = stormflame_travel_hugh
			}
		}
		current_travel_plan.current_location = {
			# Includes the Beacon and White Hall while held by Stormflames.
			# Missing owners (e.g. at sea) must fail, not pass via ?=.
			if = {
				limit = {
					exists = province_owner
					province_owner = { is_alive = yes house = house:house_Stormflame }
				}
				province_owner = { save_scope_as = stormflame_travel_captor }
			}
			else_if = {
				limit = {
					exists = county.holder
					county.holder = { is_alive = yes house = house:house_Stormflame }
				}
				county.holder = { save_scope_as = stormflame_travel_captor }
			}
		}
		if = {
			limit = {
				exists = scope:stormflame_travel_captor
				scope:stormflame_travel_captor = {
					is_alive = yes
					house = house:house_Stormflame
				}
				# Reject a captor scope inherited from an earlier event/location.
				current_travel_plan.current_location = {
					OR = {
						province_owner = scope:stormflame_travel_captor
						county.holder = scope:stormflame_travel_captor
					}
				}
			}
			scope:stormflame_travel_daemon ?= { segrellion_stormflame_capture_traveler_effect = yes }
			scope:stormflame_travel_hugh ?= { segrellion_stormflame_capture_traveler_effect = yes }
		}
	}
}
