# =========================================================
# SEGRELLION - RICKON CUSTOM EDUCATION RECOVERY
#
# Keeps existing saves healthy and guarantees that the
# Daemon/Visenya education handoff cannot be stranded.
# =========================================================


on_game_start_after_lobby = {

	on_actions = {

		delay = {
			days = 2
		}

		segrellion_rickon_education_recovery
	}
}


monthly_global_pulse = {

	on_actions = {

		segrellion_rickon_education_recovery
	}
}


segrellion_rickon_education_recovery = {

	effect = {

		# Existing-save failsafe: older Path III resolutions could finish
		# without ever queuing the Rickon-at-the-Beacon introduction.
		character:Segrellion_60 ?= {

			if = {

				limit = {
					is_alive = yes
					is_imprisoned = no
					has_character_flag = segrellion_black_unicorn_post_beacon_resolved

					character:Black_Unicorn_1 ?= {
						is_alive = yes
						has_character_flag = segrellion_black_unicorn_visenya_joined_daemon
					}

					character:Stark_Dummy_11 ?= {
						is_alive = yes
						is_adult = no
					}

					NOT = {
						has_character_flag = segrellion_black_unicorn_rickon_beacon_intro_completed
					}
				}

				segrellion_black_unicorn_try_rickon_beacon_introduction_effect = yes
			}
		}


		character:Stark_Dummy_11 ?= {

			if = {

				limit = {

					is_alive = yes
					is_adult = no

					NOT = {
						has_character_flag = segrellion_rickon_education_chain_ended
					}

					OR = {
						has_character_flag = segrellion_rickon_under_daemon_protection
						has_character_flag = segrellion_rickon_under_visenya_protection
					}
				}


				segrellion_rickon_ensure_education_watch_effect = yes


				if = {

					limit = {
						has_character_flag = segrellion_rickon_education_chain_active
					}


					segrellion_rickon_lock_agot_random_personality_effect = yes
					segrellion_rickon_clean_unapproved_personality_traits_effect = yes
				}
			}
		}
	}
}
