on_birthday = {
	on_actions = {
		vladwyn_legacy_childhood_birthday
	}
}

vladwyn_legacy_childhood_birthday = {
	trigger = {
		age >= 8
		age <= 15
		dynasty ?= { has_dynasty_perk = vladwyn_legacy_1 }
		NOT = { has_character_flag = vladwyn_oaths_written_in_blood_event_seen }
		NOR = {
			has_trait = brave
			has_trait = just
		}
	}

	random_events = {
		65 = 0
		35 = vladwyn_legacy.0001
	}
}
