# =========================================================
# VISERYS I - STORY PROTECTION
#
# Applied only to the 128 AC bookmark. It lasts long enough
# for the prelude and succession chain, and 0120 removes it
# immediately before Viserys' scripted death.
# =========================================================

segrellion_dance_128_viserys_protection = {
	effect = {
		if = {
			limit = {
				game_start_date = 8128.11.12

				character:Targaryen_60 ?= {
					is_alive = yes
					NOT = { has_character_flag = segrellion_dance_viserys_death_processed }
					NOT = { has_character_modifier = segrellion_dance_viserys_story_protection_modifier }
				}
			}

			character:Targaryen_60 = {
				add_character_modifier = {
					modifier = segrellion_dance_viserys_story_protection_modifier
					months = 12
				}
			}
		}
	}
}


# =========================================================
# THE DANCE OF THE DRAGONS - 128 AC
# Opening handoff for Alex Segrellion.
# =========================================================

segrellion_dance_128_startup = {
	effect = {
		if = {
			limit = {
				game_start_date = 8128.11.12
			}

			# Ensure stale dragon storage from older builds cannot leave
			# Vhalyrax at 100-130 size in the Dance bookmark.
			segrellion_force_vhalyrax_128_size_repair_effect = yes

			character:Segrellion_1 ?= {
				if = {
					limit = {
						is_alive = yes
						NOT = { has_character_flag = segrellion_dance_128_intro_seen }
						NOT = { has_character_flag = segrellion_dance_128_intro_queued }
					}

					add_character_flag = segrellion_dance_128_intro_queued

					trigger_event = {
						id = segrellion_dance.0001
						days = 3
					}
				}
			}
		}
	}
}

segrellion_dance_on_player_select_destiny_confirmed = {

# Save-compatible fallback if another startup sequence swallows
# the first scheduled event. Idempotent because of queued/seen flags.

	effect = {
		character:Segrellion_1 ?= {
			if = {
				limit = {
					is_alive = yes
					game_start_date = 8128.11.12
					NOT = { has_character_flag = segrellion_dance_128_intro_seen }
					NOT = { has_character_flag = segrellion_dance_128_intro_queued }
				}

				add_character_flag = segrellion_dance_128_intro_queued

				trigger_event = {
					id = segrellion_dance.0001
					days = 2
				}
			}
		}
	}
}


# If the direct handoff from the Vhalyrax omen is ever swallowed,
# restore the succession chain without duplicating it.
segrellion_dance_128_succession_fallback = {
	effect = {
		character:Segrellion_1 ?= {
			if = {
				limit = {
					is_alive = yes
					game_start_date = 8128.11.12
					has_character_flag = segrellion_dance_prelude_complete
					NOT = { has_character_flag = segrellion_dance_dying_king_seen }
					character:Targaryen_60 ?= { is_alive = yes }
				}

				trigger_event = {
					id = segrellion_dance.0100
					days = 2
				}
			}
		}
	}
}

# =========================================================
# SAVE-COMPATIBLE REPAIR
#
# Older builds allowed Call the Banners to run on top of
# the Dance Mega War, incorrectly branding legitimate
# opposite-side vassals as traitors/oathbreakers.
# =========================================================

segrellion_dance_128_false_banner_traitor_cleanup = {
	effect = {
		if = {
			limit = {
				game_start_date = 8128.11.12

				character:Targaryen_63 ?= {
					any_character_war = {
						using_cb = segrellion_dance_of_the_dragons_war
					}
				}
			}

			every_ruler = {
				limit = {
					any_character_war = {
						using_cb = segrellion_dance_of_the_dragons_war
					}

					OR = {
						has_character_flag = segrellion_banner_betrayed_current_call
						has_character_flag = segrellion_banner_fought_against_liege
					}
				}

				if = {
					limit = {
						has_trait = segrellion_traitor
					}
					remove_trait = segrellion_traitor
				}

				if = {
					limit = {
						has_character_modifier = segrellion_banner_oathbreaker_modifier
					}
					remove_character_modifier = segrellion_banner_oathbreaker_modifier
				}

				if = {
					limit = {
						has_character_modifier = segrellion_banner_oathbreaker_minor_modifier
					}
					remove_character_modifier = segrellion_banner_oathbreaker_minor_modifier
				}

				remove_character_flag = segrellion_banner_betrayed_current_call
				remove_character_flag = segrellion_banner_fought_against_liege
				remove_character_flag = segrellion_banner_oathbreaker_unpunished
				remove_character_flag = segrellion_banner_refused_current_call
			}
		}
	}
}


# One-time monthly recovery for 128 saves created before the size rebalance.
segrellion_dance_128_vhalyrax_size_repair = {
	effect = {
		if = {
			limit = {
				game_start_date = 8128.11.12
				character:dragon_vhalyrax ?= {
					is_alive = yes
					NOT = {
						has_character_flag = segrellion_vhalyrax_128_size_repaired
					}
				}
			}

			segrellion_force_vhalyrax_128_size_repair_effect = yes
		}
	}
}
