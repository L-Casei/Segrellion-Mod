# Covers any marriage route, with either partner as the marriage root.
on_marriage = {
    on_actions = {
        segrellion_daemon_visenya_marriage_traits
        segrellion_visenya_pre_daemon_marriage_guard
    }
}

on_game_start_after_lobby = {
    on_actions = {
        segrellion_visenya_marriage_guard_startup
    }
}

segrellion_daemon_visenya_marriage_traits = {
    effect = {
        if = {
            limit = {
                OR = {
                    AND = {
                        this = character:Segrellion_60
                        scope:spouse = character:Black_Unicorn_1
                    }
                    AND = {
                        this = character:Black_Unicorn_1
                        scope:spouse = character:Segrellion_60
                    }
                }
            }
            character:Segrellion_60 = {
                segrellion_daemon_visenya_marriage_traits_effect = yes
            }
        }
    }
}

segrellion_visenya_pre_daemon_marriage_guard = {
    effect = {
        character:Black_Unicorn_1 ?= {
            segrellion_visenya_pre_daemon_marriage_guard_effect = yes
        }
    }
}

segrellion_visenya_marriage_guard_startup = {
    effect = {
        character:Black_Unicorn_1 ?= {
            segrellion_visenya_pre_daemon_marriage_guard_effect = yes
            segrellion_visenya_schedule_marriage_guard_watch_effect = yes
        }
    }
}
