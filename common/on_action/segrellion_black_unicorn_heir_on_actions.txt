# =========================================================
# VISENYA - TEMPORARY HEIR AFTER DUNGEON RECONCILIATION
# =========================================================

on_birth_mother = {
	on_actions = {
		segrellion_black_unicorn_visenya_temporary_heir_birth_check
	}
}


segrellion_black_unicorn_visenya_temporary_heir_birth_check = {

	trigger = {

		exists = scope:father
		exists = scope:child

		scope:father = {

			this = character:Segrellion_60

			has_character_flag = segrellion_black_unicorn_visenya_temporary_heir
		}

		scope:child = {
			is_alive = yes
			is_female = no
		}
	}

	effect = {

		scope:father = {

			set_designated_heir_unsafe = scope:child

			remove_character_flag = segrellion_black_unicorn_visenya_temporary_heir

			add_character_flag = segrellion_black_unicorn_visenya_heirship_ended_by_son
		}

		character:Black_Unicorn_1 ?= {

			remove_character_flag = segrellion_black_unicorn_temporary_heir_to_daemon

			add_character_flag = segrellion_black_unicorn_no_longer_daemon_heir_after_son
		}
	}
}
