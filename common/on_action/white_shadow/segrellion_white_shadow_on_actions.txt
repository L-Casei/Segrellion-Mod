			}

			}
		}
	}
}

segrellion_first_stepstones_attacker_victory = {
	trigger = {
		exists = scope:war
		scope:war = {
			has_variable = segrellion_tracked_first_stepstones_war
		}
		character:Segrellion_1 ?= {
			is_alive = yes
			has_character_flag = segrellion_stepstones_war_joined
			NOR = {
				has_character_flag = segrellion_stepstones_story_completed
				has_character_flag = segrellion_stepstones_aftermath_pending
			}
		}
	}
	effect = {
		character:Segrellion_1 ?= {
			add_character_flag = segrellion_stepstones_daemon_won
			add_character_flag = segrellion_stepstones_aftermath_pending
			trigger_event = {
				id = segrellion_intro.0006
				days = 3
			}
		}
	}
}

segrellion_restless_for_war_monthly_check = {
	effect = {
		character:Segrellion_1 ?= {
			if = {
				limit = {
					is_alive = yes
					has_character_flag = segrellion_restless_for_war_path
					NOT = { has_character_flag = segrellion_restless_for_war_answered_call }
				}
				if = {
					limit = {
						character:Targaryen_61 ?= {
							any_character_war = {
								primary_attacker ?= prev
							}
						}
					}
					character:Targaryen_61 = {
						random_character_war = {
							limit = {
								primary_attacker ?= prev
							}
							add_attacker = root
							root = {
								add_character_flag = segrellion_restless_for_war_answered_call
							}
						}
					}
				}
				else_if = {
					limit = {
						character:Velaryon_31 ?= {
							any_character_war = {
								primary_attacker ?= prev
							}
						}
					}
					character:Velaryon_31 = {
						random_character_war = {
							limit = {
								primary_attacker ?= prev
							}
							add_attacker = root
							root = {
								add_character_flag = segrellion_restless_for_war_answered_call
							}
						}
					}
				}
			}
		}
	}
}

vladwyn_8106_startup = {
    effect = {
        every_player = {
            limit = {
                this = character:Vladwyn_1
                game_start_date = 8106.4.19
                NOT = { has_character_flag = vladwyn_sentinels_path_resolved }
                NOT = { has_character_flag = vladwyn_sentinels_path_queued }
            }

            # -------------------------------------------------
            # FAILSAFE:
            # Tristan MUST begin as a Landless Adventurer.
            # If AGOT/startup has reduced him to a courtier,
            # remove him from that court and rebuild his camp.
            # -------------------------------------------------
            if = {
                limit = {
                    has_dlc_feature = roads_to_power
                    NOT = { is_landless_adventurer = yes }
                }

                move_to_pool = yes

                create_landless_adventurer_title_effect = {
                    REASON = flag:historical
                    FLAVOR_CHAR = this
                }
            }

            add_character_flag = vladwyn_sentinels_path_queued

            trigger_event = {
                id = segrellion_sentinel.0001
                days = 1
            }
        }
    }
}
