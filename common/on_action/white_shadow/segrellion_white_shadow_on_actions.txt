# =========================================================
# TRISTAN VLADWYN - 106 STARTUP
# =========================================================

vladwyn_8106_startup = {

	effect = {

		if = {

			limit = {
				game_start_date = 8106.4.19
			}

			character:Vladwyn_1 ?= {

				if = {

					limit = {

						is_alive = yes

						NOT = {
							has_character_flag = vladwyn_106_startup_applied
						}
					}

					add_character_flag = vladwyn_106_startup_applied


					# =================================================
					# CANON ROUTE
					# Tristan begins at Alex's side.
					# No Kingsguard preparation yet.
					# =================================================

					if = {

						limit = {
							has_game_rule = segrellion_tristan_106_start_canon
						}


						add_character_flag = vladwyn_106_canon_start
						add_character_flag = vladwyn_sentinels_path_resolved

						remove_character_flag = vladwyn_sentinels_path_queued
						remove_character_flag = vladwyn_landless_route_unlocked
						remove_character_flag = vladwyn_open_road_taken
						remove_character_flag = vladwyn_adventurer_events_unlocked

						remove_character_flag = kingsguard_ambition


						if = {

							limit = {
								has_character_modifier = training_for_kingsguard
							}

							remove_character_modifier = training_for_kingsguard
						}


						if = {

							limit = {
								has_trait = refusing_marriage
							}

							remove_trait = refusing_marriage
						}


						# If Tristan himself was selected in the
						# bookmark, move the player to Alex first.
						if = {

							limit = {
								is_ai = no
								has_multiple_players = no
							}

							add_internal_flag = special
							set_player_character = character:Segrellion_1
						}


						# Save and destroy Tristan's adventurer camp.
						primary_title ?= {
							save_scope_as = vladwyn_106_old_camp
						}


						if = {

							limit = {

								exists = scope:vladwyn_106_old_camp

								OR = {
									is_ai = yes
									has_multiple_players = no
								}
							}

							destroy_title = scope:vladwyn_106_old_camp
						}


						# Tristan joins Alex's court.
						if = {

							limit = {

								OR = {
									is_ai = yes
									has_multiple_players = no
								}
							}

							set_employer = character:Segrellion_1
						}


						# Alex and Tristan are already deeply bonded.
						if = {

							limit = {

								character:Segrellion_1 ?= {
									is_alive = yes
								}

								NOT = {
									has_relation_best_friend = character:Segrellion_1
								}
							}

							set_relation_best_friend = {
								target = character:Segrellion_1
								reason = best_friend_childhood_oath
								copy_reason = friend
							}
						}
					}


					# =================================================
					# NON-CANON ROUTE
					# The Open Road
					# =================================================

					else_if = {

						limit = {
							has_game_rule = segrellion_tristan_106_start_landless
						}


						if = {

							limit = {

								has_dlc_feature = roads_to_power

								NOT = {
									is_landless_adventurer = yes
								}
							}

							move_to_pool = yes

							create_landless_adventurer_title_effect = {
								REASON = flag:historical
								FLAVOR_CHAR = this
							}
						}


						add_character_flag = vladwyn_106_landless_start
						add_character_flag = vladwyn_landless_route_unlocked
						add_character_flag = vladwyn_open_road_taken
						add_character_flag = vladwyn_adventurer_events_unlocked
						add_character_flag = vladwyn_sentinels_path_resolved

						remove_character_flag = vladwyn_sentinels_path_queued


						primary_title ?= {

							save_title_name_change_data = {
								TRANSFER_VAR = yes
							}

							reset_title_prefix = yes
							set_title_name = vladwyn_the_honorables_name
						}


						add_prestige = 200
						add_martial_lifestyle_xp = 250


						add_character_modifier = {
							modifier = vladwyn_wandering_knight_modifier
							years = 8
						}
					}
				}
			}
		}
	}
}

# =========================================================
# TRISTAN VLADWYN CANON STORYLINE
# 106 AC -> AEGON III
# =========================================================

vladwyn_canon_story_progression = {

	effect = {

		character:Vladwyn_1 ?= {

			if = {

				limit = {

					is_alive = yes

					has_character_flag = vladwyn_106_canon_start

					NOT = {
						has_character_flag = vladwyn_canon_story_completed
					}
				}


				# =================================================
				# 114 AC
				# RHAENYRA & LAENOR'S WEDDING
				# TRISTAN MEETS ELYSE SUNGLASS
				# =================================================

				if = {

					limit = {

						current_date >= 8114.1.1

						NOT = {
							has_character_flag = vladwyn_met_elyse_sunglass
						}

						NOT = {
							has_character_flag = vladwyn_elyse_meeting_queued
						}

						character:IAF_Sunglass_2 ?= {
							is_alive = yes
						}

						character:Segrellion_1 ?= {
							is_alive = yes
						}
					}

					add_character_flag = vladwyn_elyse_meeting_queued

					character:Segrellion_1 = {

						trigger_event = {
							id = vladwyn_canon.1000
							days = 1
						}
					}
				}


				# =================================================
				# MARRIAGE FAILSAFE
				# =================================================

				if = {

					limit = {

						current_date >= 8114.6.1

						has_character_flag = vladwyn_met_elyse_sunglass

						NOT = {
							has_character_flag = vladwyn_married_elyse_sunglass
						}

						NOT = {
							has_character_flag = vladwyn_elyse_marriage_queued
						}

						character:IAF_Sunglass_2 ?= {
							is_alive = yes
						}

						character:Segrellion_1 ?= {
							is_alive = yes
						}
					}

					add_character_flag = vladwyn_elyse_marriage_queued

					character:Segrellion_1 = {

						trigger_event = {
							id = vladwyn_canon.1010
							days = 1
						}
					}
				}


				# =================================================
				# 122 AC
				# ELYSE DIES
				# =================================================

				if = {

					limit = {

						current_date >= 8122.1.1

						has_character_flag = vladwyn_married_elyse_sunglass

						NOT = {
							has_character_flag = vladwyn_elyse_died
						}

						NOT = {
							has_character_flag = vladwyn_elyse_death_queued
						}

						character:IAF_Sunglass_2 ?= {
							is_alive = yes
						}

						character:Segrellion_1 ?= {
							is_alive = yes
						}
					}

					add_character_flag = vladwyn_elyse_death_queued

					character:Segrellion_1 = {

						trigger_event = {
							id = vladwyn_canon.1020
							days = 1
						}
					}
				}


				# If Elyse died through some other mechanic,
				# continue the canon storyline anyway.
				else_if = {

					limit = {

						current_date >= 8122.1.1

						has_character_flag = vladwyn_married_elyse_sunglass

						NOT = {
							has_character_flag = vladwyn_elyse_died
						}

						character:IAF_Sunglass_2 ?= {
							is_alive = no
						}
					}

					add_character_flag = vladwyn_elyse_died
					add_character_flag = vladwyn_tristan_widowed


					if = {

						limit = {
							NOT = {
								has_trait = refusing_marriage
							}
						}

						add_trait = refusing_marriage
					}
				}


				# =================================================
				# 123 AC
				# RHAENYRA CALLS TRISTAN TO DRAGONSTONE
				# =================================================

				if = {

					limit = {

						current_date >= 8123.1.1

						has_character_flag = vladwyn_elyse_died

						NOT = {
							has_character_flag = vladwyn_rhaenyra_household_captain
						}

						NOT = {
							has_character_flag = vladwyn_rhaenyra_service_queued
						}

						character:Targaryen_63 ?= {
							is_alive = yes
							is_landed = yes
						}

						character:Segrellion_1 ?= {
							is_alive = yes
						}
					}

					add_character_flag = vladwyn_rhaenyra_service_queued

					character:Segrellion_1 = {

						trigger_event = {
							id = vladwyn_canon.1030
							days = 1
						}
					}
				}


				# =================================================
				# DANCE OF THE DRAGONS
				# BLACK QUEENSGUARD
				# =================================================

				if = {

					limit = {

						current_date >= 8129.1.1

						has_character_flag = vladwyn_rhaenyra_household_captain

						NOT = {
							has_character_flag = vladwyn_black_queensguard_lord_commander
						}

						NOT = {
							has_character_flag = vladwyn_black_queensguard_queued
						}

						character:Targaryen_63 ?= {
							is_alive = yes
							is_at_war = yes
						}

						character:Segrellion_1 ?= {
							is_alive = yes
						}
					}

					add_character_flag = vladwyn_black_queensguard_queued

					character:Segrellion_1 = {

						trigger_event = {
							id = vladwyn_canon.1040
							days = 1
						}
					}
				}


				# =================================================
				# AEGON III TAKES THE IRON THRONE
				# TRISTAN BECOMES REAL LORD COMMANDER
				# =================================================

				if = {

					limit = {

						current_date >= 8131.1.1

						NOT = {
							has_character_flag = vladwyn_tristan_served_kingsguard
						}

						NOT = {
							has_character_flag = vladwyn_aegon_iii_kingsguard_queued
						}

						character:Targaryen_78 ?= {

							is_alive = yes

							has_title = title:e_the_iron_throne
						}

						character:Segrellion_1 ?= {
							is_alive = yes
						}
					}

					add_character_flag = vladwyn_aegon_iii_kingsguard_queued

					character:Segrellion_1 = {

						trigger_event = {
							id = vladwyn_canon.1050
							days = 1
						}
					}
				}
			}
		}
	}
}

# =========================================================
# STEPSTONES - LATER WAR START
# =========================================================

segrellion_stepstones_followup_war_started = {

	trigger = {

		exists = scope:war


		character:Segrellion_1 ?= {

			is_alive = yes

			has_character_flag = segrellion_stepstones_campaign_active

			NOT = {
				has_character_flag = segrellion_stepstones_story_completed
			}
		}


		scope:war = {

			OR = {

				AND = {

					using_cb = daemon_stepstones_second_war

					OR = {
						primary_attacker ?= character:Targaryen_61
						primary_attacker ?= character:Velaryon_31
					}
				}


				AND = {

					using_cb = triarchy_anti_daemon_war

					primary_defender ?= character:Targaryen_61
				}
			}
		}
	}


	effect = {

		scope:war = {


			# =============================================
			# DAEMON'S SECOND STEPSTONES WAR
			# =============================================

			if = {

				limit = {

					using_cb = daemon_stepstones_second_war

					OR = {
						primary_attacker ?= character:Targaryen_61
						primary_attacker ?= character:Velaryon_31
					}
				}


				set_variable = {
					name = segrellion_tracked_second_stepstones_war
					value = yes
				}


				if = {

					limit = {

						NOT = {
							is_participant = character:Segrellion_1
						}
					}

					add_attacker = character:Segrellion_1
				}


				character:Segrellion_1 = {

					add_character_flag = segrellion_stepstones_second_war_joined

					trigger_event = {
						id = segrellion_intro.0101
						days = 1
					}
				}
			}


			# =============================================
			# TRIARCHY COUNTERATTACK
			# =============================================

			else_if = {

				limit = {

					using_cb = triarchy_anti_daemon_war

					primary_defender ?= character:Targaryen_61
				}


				set_variable = {
					name = segrellion_tracked_triarchy_stepstones_war
					value = yes
				}


				if = {

					limit = {

						NOT = {
							is_participant = character:Segrellion_1
						}
					}

					add_defender = character:Segrellion_1
				}


				character:Segrellion_1 = {

					add_character_flag = segrellion_stepstones_triarchy_war_joined

					trigger_event = {
						id = segrellion_intro.0101
						days = 1
					}
				}
			}
		}
	}
}


# =========================================================
# STEPSTONES - ATTACKER VICTORY
# =========================================================

segrellion_stepstones_attacker_victory = {

	trigger = {

		exists = scope:war

		scope:war = {

			OR = {
				has_variable = segrellion_tracked_first_stepstones_war
				has_variable = segrellion_tracked_second_stepstones_war
				has_variable = segrellion_tracked_triarchy_stepstones_war
			}
		}
	}


	effect = {


		# =============================================
		# FIRST WAR WON BY DAEMON
		# =============================================

		if = {

			limit = {

				scope:war = {
					has_variable = segrellion_tracked_first_stepstones_war
				}

				character:Segrellion_1 ?= {

					is_alive = yes

					NOT = {
						has_character_flag = segrellion_stepstones_first_war_completed
					}
				}
			}


			character:Segrellion_1 = {

				add_character_flag = segrellion_stepstones_daemon_won
				add_character_flag = segrellion_stepstones_first_war_completed
				add_character_flag = segrellion_stepstones_aftermath_pending

				trigger_event = {
					id = segrellion_intro.0006
					days = 3
				}
			}
		}


		# =============================================
		# SECOND WAR WON
		# =============================================

		else_if = {

			limit = {

				scope:war = {
					has_variable = segrellion_tracked_second_stepstones_war
				}

				character:Segrellion_1 ?= {
					has_character_flag = segrellion_stepstones_campaign_active
				}
			}


			character:Segrellion_1 = {

				add_character_flag = segrellion_stepstones_second_war_completed

				trigger_event = {
					id = segrellion_intro.0190
					days = 180
				}
			}
		}


		# =============================================
		# TRIARCHY WON ITS COUNTERATTACK
		# DAEMON LOST
		# =============================================

		else_if = {

			limit = {

				scope:war = {
					has_variable = segrellion_tracked_triarchy_stepstones_war
				}
			}


			character:Segrellion_1 ?= {

				add_character_flag = segrellion_stepstones_daemon_failed
				add_character_flag = segrellion_stepstones_aftermath_pending

				trigger_event = {
					id = segrellion_intro.0008
					days = 3
				}
			}
		}
	}
}


# =========================================================
# STEPSTONES - DEFENDER VICTORY
# =========================================================

segrellion_stepstones_defender_victory = {

	trigger = {

		exists = scope:war

		scope:war = {

			OR = {
				has_variable = segrellion_tracked_first_stepstones_war
				has_variable = segrellion_tracked_second_stepstones_war
				has_variable = segrellion_tracked_triarchy_stepstones_war
			}
		}
	}


	effect = {


		# First/second Daemon war:
		# defender victory means Daemon lost.

		if = {

			limit = {

				scope:war = {

					OR = {
						has_variable = segrellion_tracked_first_stepstones_war
						has_variable = segrellion_tracked_second_stepstones_war
					}
				}
			}


			character:Segrellion_1 ?= {

				add_character_flag = segrellion_stepstones_daemon_failed
				add_character_flag = segrellion_stepstones_aftermath_pending

				trigger_event = {
					id = segrellion_intro.0008
					days = 3
				}
			}
		}


		# Triarchy attacks Daemon:
		# defender victory means Daemon held the islands.

		else_if = {

			limit = {

				scope:war = {
					has_variable = segrellion_tracked_triarchy_stepstones_war
				}
			}


			character:Segrellion_1 ?= {

				trigger_event = {
					id = segrellion_intro.0103
					days = 3
				}
			}
		}
	}
}
