# CK3 has quarterly_playable_pulse, not monthly_global_pulse.
# Only the two story recipients seed the checks, including in existing saves.
quarterly_playable_pulse = {
	on_actions = {
		segrellion_story_letters_quarterly
		segrellion_aegon_daemon_quarterly_recovery
	}
}

segrellion_story_letters_quarterly = {
	trigger = {
		OR = {
			this = character:Segrellion_60
			segrellion_is_current_jon_trigger = yes
		}
	}
	on_actions = {
		segrellion_sansa_story_progression
		segrellion_jon_stannis_death_news_progression
		delay = { days = 30 }
		segrellion_sansa_story_progression
		segrellion_jon_stannis_death_news_progression
		delay = { days = 60 }
		segrellion_sansa_story_progression
		segrellion_jon_stannis_death_news_progression
	}
}

on_game_start_after_lobby = {

	on_actions = {

		segrellion_sansa_story_progression

		segrellion_jon_stannis_death_news_progression

		segrellion_aegon_daemon_vassalage_progression

		segrellion_aegon_daemon_appointment_recovery

		segrellion_dynasty_splendor_repair
	}
}

on_birthday = {
	on_actions = { segrellion_sansa_sixteenth_birthday }
}
segrellion_sansa_sixteenth_birthday = {
	trigger = {
		this = character:Stark_Dummy_8
		age >= 16
	}
	on_actions = { segrellion_sansa_story_progression }
}

on_death = {
	on_actions = {
		segrellion_stannis_death_letter
		segrellion_aegon_daemon_death_handoff
	}
}

segrellion_stannis_death_letter = {

	trigger = {

		this = character:Baratheon_3
	}


	effect = {

		segrellion_schedule_stannis_death_news_for_jon_effect = yes
	}
}

segrellion_aegon_daemon_quarterly_recovery = {

	trigger = {

		OR = {

			this = character:IAF_Young_Griff_1

			this = character:Segrellion_60

			dynasty = dynasty:dynn_Segrellion
		}
	}


	on_actions = {

		segrellion_aegon_daemon_vassalage_progression

		segrellion_aegon_daemon_appointment_recovery
	}
}

segrellion_aegon_daemon_appointment_recovery = {

	trigger = {


		character:IAF_Young_Griff_1 ?= {

			is_alive = yes

			has_title = title:h_the_iron_throne

			has_character_flag = segrellion_young_griff_supports_daemon

			has_character_flag = segrellion_young_griff_daemon_hand_and_regent


			NOT = {

				has_character_flag = segrellion_young_griff_daemon_appointment_story_resolved
			}
		}


		character:Segrellion_60 ?= {

			is_alive = yes

			liege ?= character:IAF_Young_Griff_1

			has_character_flag = segrellion_pledged_to_young_griff

			has_character_flag = segrellion_daemon_aegon_fealty_accepted

			has_character_flag = segrellion_daemon_aegon_hand_and_regent


			NOT = {

				has_character_flag = segrellion_daemon_aegon_appointment_story_resolved
			}


			NOT = {

				has_character_flag = segrellion_daemon_aegon_appointment_story_scheduled
			}
		}
	}


	effect = {

		character:Segrellion_60 = {

			add_character_flag = {

				flag = segrellion_daemon_aegon_appointment_story_scheduled

				days = 30
			}


			trigger_event = {

				id = segrellion_aegon_daemon.0006

				days = { 2 4 }
			}
		}
	}
}


segrellion_dynasty_splendor_repair = {

	effect = {

		segrellion_set_historical_splendor_effect = yes
	}
}


segrellion_aegon_daemon_death_handoff = {

	trigger = {

		this = character:IAF_Young_Griff_1


		character:Segrellion_60 ?= {

			is_alive = yes

			has_character_flag = segrellion_daemon_aegon_fealty_accepted

			has_character_flag = segrellion_daemon_aegon_hand_and_regent


			NOT = {
				has_character_flag = segrellion_daemon_aegon_death_claim_pending
			}

			NOT = {
				has_character_flag = segrellion_daemon_aegon_succession_resolved
			}
		}
	}


	effect = {

		character:Segrellion_60 = {

			add_character_flag = segrellion_daemon_aegon_death_claim_pending


			trigger_event = {

				id = segrellion_aegon_daemon.0010

				days = 1
			}
		}
	}
}
