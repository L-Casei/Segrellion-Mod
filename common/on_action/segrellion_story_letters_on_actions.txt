# CK3 has quarterly_playable_pulse, not monthly_global_pulse.
# Only the two story recipients seed the checks, including in existing saves.
# =========================================================
# AEGON / DAEMON
#
# Valid CK3 playable monthly pulse.
# This is the failsafe for lost title-gain events and
# existing saves.
# =========================================================

monthly_playable_pulse = {

	on_actions = {

		segrellion_aegon_daemon_monthly_recovery

		segrellion_jon_winterfell_council_monthly_recovery

		segrellion_daenerys_post_throne_monthly_recovery
	}
}


segrellion_jon_winterfell_council_monthly_recovery = {

	trigger = {

		is_ai = yes

		segrellion_is_current_jon_trigger = yes

		has_title = title:e_the_north

		character:Baratheon_3 ?= {
			is_alive = no
		}

		NOT = {
			has_character_flag = segrellion_jon_winterfell_council_completed
		}

		OR = {
			has_character_flag = segrellion_jon_stannis_death_news_received
			has_character_flag = segrellion_jon_winterfell_council_called
			has_character_flag = segrellion_jon_king_in_the_north
		}
	}


	effect = {

		add_character_flag = segrellion_jon_winterfell_council_called

		remove_character_flag = segrellion_jon_winterfell_council_ai_spawn_cooldown

		segrellion_jon_try_start_winterfell_council_effect = yes


		if = {

			limit = {

				NOT = {
					in_activity_type = activity_segrellion_winterfell_council
				}
			}


			trigger_event = {

				id = segrellion_jon_frey.0011

				days = 3
			}
		}
	}
}


segrellion_daenerys_post_throne_monthly_recovery = {

	trigger = {

		this = character:Targaryen_9

		is_alive = yes

		has_title = title:h_the_iron_throne
	}


	effect = {

		segrellion_daenerys_post_throne_recovery_effect = yes
	}
}

quarterly_playable_pulse = {

	on_actions = {

		segrellion_story_letters_quarterly
	}
}

segrellion_story_letters_quarterly = {
	trigger = {
		OR = {
			this = character:Segrellion_60
			segrellion_is_current_jon_trigger = yes
		}
	}
	on_actions = {
		segrellion_sansa_story_progression
		segrellion_jon_stannis_death_news_progression
		delay = { days = 30 }
		segrellion_sansa_story_progression
		segrellion_jon_stannis_death_news_progression
		delay = { days = 60 }
		segrellion_sansa_story_progression
		segrellion_jon_stannis_death_news_progression
	}
}

on_game_start_after_lobby = {

	on_actions = {

		segrellion_sansa_story_progression

		segrellion_jon_stannis_death_news_progression

		segrellion_aegon_daemon_vassalage_progression

		segrellion_aegon_daemon_appointment_recovery

		segrellion_dynasty_splendor_repair
	}
}

on_birthday = {
	on_actions = { segrellion_sansa_sixteenth_birthday }
}
segrellion_sansa_sixteenth_birthday = {
	trigger = {
		this = character:Stark_Dummy_8
		age >= 16
	}
	on_actions = { segrellion_sansa_story_progression }
}

on_death = {
	on_actions = {
		segrellion_stannis_death_letter
		segrellion_aegon_daemon_death_handoff
	}
}

segrellion_stannis_death_letter = {

	trigger = {

		this = character:Baratheon_3
	}


	effect = {

		segrellion_schedule_stannis_death_news_for_jon_effect = yes


		character:Targaryen_9 ?= {

			if = {

				limit = {

					is_alive = yes

					has_character_flag = segrellion_daenerys_massey_network_established

					has_game_rule = segrellion_story_news_events_enabled

					NOT = {
						has_character_flag = segrellion_daenerys_news_stannis_dead
					}
				}


				add_character_flag = segrellion_daenerys_news_stannis_dead


				trigger_event = {

					id = segrellion_daenerys_massey.0035

					days = { 1 4 }
				}
			}
		}
	}
}

segrellion_aegon_daemon_monthly_recovery = {

	trigger = {

		OR = {

			this = character:IAF_Young_Griff_1

			this = character:Segrellion_60
		}
	}


	on_actions = {

		segrellion_aegon_daemon_vassalage_progression

		segrellion_aegon_daemon_appointment_recovery
	}
}

segrellion_aegon_daemon_appointment_recovery = {

	trigger = {

		character:IAF_Young_Griff_1 ?= {

			is_alive = yes

			has_title = title:h_the_iron_throne

			has_character_flag = segrellion_young_griff_supports_daemon
		}


		character:Segrellion_60 ?= {

			is_alive = yes

			liege ?= character:IAF_Young_Griff_1

			has_character_flag = segrellion_pledged_to_young_griff

			has_character_flag = segrellion_daemon_aegon_fealty_accepted


			NOT = {
				has_character_flag = segrellion_daemon_aegon_appointment_story_resolved
			}


			NOT = {
				has_character_flag = segrellion_daemon_aegon_appointment_story_scheduled
			}
		}
	}


	effect = {

		character:Segrellion_60 = {

			# Repair the actual offices first.

			segrellion_daemon_becomes_aegons_hand_and_regent_effect = yes


			add_character_flag = {

				flag = segrellion_daemon_aegon_appointment_story_scheduled

				days = 30
			}


			# Resume exactly where a lost delayed event stopped.

			if = {

				limit = {

					has_character_flag = segrellion_daemon_aegon_hand_event_seen

					NOT = {
						has_character_flag = segrellion_daemon_aegon_regent_event_seen
					}
				}


				trigger_event = {

					id = segrellion_aegon_daemon.0007

					days = { 1 2 }
				}
			}


			else_if = {

				limit = {

					has_character_flag = segrellion_daemon_aegon_regent_event_seen

					NOT = {
						has_character_flag = segrellion_daemon_aegon_friend
					}
				}


				trigger_event = {

					id = segrellion_aegon_daemon.0008

					days = { 1 2 }
				}
			}


			else_if = {

				limit = {

					has_character_flag = segrellion_daemon_aegon_friend

					NOT = {
						has_character_flag = segrellion_daemon_aegon_appointment_story_resolved
					}
				}


				trigger_event = {

					id = segrellion_aegon_daemon.0009

					days = { 1 2 }
				}
			}


			else = {

				trigger_event = {

					id = segrellion_aegon_daemon.0006

					days = { 1 2 }
				}
			}
		}
	}
}


segrellion_dynasty_splendor_repair = {

	effect = {

		segrellion_set_historical_splendor_effect = yes
	}
}


segrellion_aegon_daemon_death_handoff = {

	trigger = {

		this = character:IAF_Young_Griff_1


		character:Segrellion_60 ?= {

			is_alive = yes

			has_character_flag = segrellion_daemon_aegon_fealty_accepted

			has_character_flag = segrellion_daemon_aegon_hand_and_regent


			NOT = {
				has_character_flag = segrellion_daemon_aegon_death_claim_pending
			}

			NOT = {
				has_character_flag = segrellion_daemon_aegon_succession_resolved
			}
		}
	}


	effect = {

		character:Segrellion_60 = {

			add_character_flag = segrellion_daemon_aegon_death_claim_pending


			trigger_event = {

				id = segrellion_aegon_daemon.0010

				days = 1
			}
		}
	}
}
