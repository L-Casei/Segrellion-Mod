segrellion_on_game_start = {
	effect = {
		dynasty:dynn_Segrellion = {
			while = {
				limit = { dynasty_prestige_level < 2 }
				add_dynasty_prestige_level = 1
			}
			add_dynasty_prestige = 250
		}
		dynasty:dynn_Segrellion = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
		title:b_castle_farring.title_province = {
			if = {
				limit = { has_special_building_slot = no }
				add_special_building_slot = segrellion_beacon_01
			}
		}
		character:Segrellion_1 ?= {
			if = {
				limit = { is_alive = yes }
				if = {
					limit = {
						has_character_flag = segrellion_intro_event_seen
						NOR = {
							has_character_flag = segrellion_opening_route_dragonstone
							has_character_flag = segrellion_opening_route_vhalerion
							has_character_flag = segrellion_chose_stepstones
							has_character_flag = segrellion_returned_to_vhalerion
						}
					}
					remove_character_flag = segrellion_intro_event_seen
				}
				add_character_flag = has_scripted_appearance
				add_character_flag = hair_trait_male_hair_custom_20
				add_character_flag = beard_trait_no_beard
			}
		}
		character:Segrellion_1 ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_character_flag = segrellion_intro_event_seen }
				}
				trigger_event = {
					id = segrellion_intro.0001
					days = 3
				}
			}
		}
		character:Vladwyn_1 ?= {
			if = {
				limit = { is_alive = yes }
				add_character_flag = segrellion_icebringer_authorized
				add_character_flag = has_scripted_appearance
				add_character_flag = hair_trait_m_hair_ep3_byzantine_01
				add_character_flag = beard_trait_no_beard
			}
		}
		character:Segrellion_2 ?= {
			if = {
				limit = { is_alive = yes }
				add_character_flag = has_scripted_appearance
				add_character_flag = hair_trait_female_hair_custom_15
				add_character_flag = beard_trait_no_beard
				add_character_flag = segrellion_alysanne_custom_clothes
			}
		}
		character:Segrellion_3 ?= {
			if = {
				limit = { is_alive = yes }
				add_character_flag = has_scripted_appearance
				add_character_flag = hair_trait_female_hair_custom_14
				add_character_flag = beard_trait_no_beard
				add_character_flag = segrellion_daenerys_custom_clothes
				if = {
					limit = {
						current_date < 8113.1.1
						any_spouse = { this = character:Segrellion_1 }
					}
					add_character_modifier = {
						modifier = segrellion_daenerys_canon_pregnancy_waiting_modifier
						years = 8
					}
				}
			}
		}
		character:Segrellion_12 ?= {
			if = {
				limit = { is_alive = yes }
				add_character_flag = has_scripted_appearance
				add_character_flag = hair_trait_m_hair_ep3_byzantine_01
				add_character_flag = beard_trait_no_beard
				add_character_flag = segrellion_aeryn_custom_clothes
			}
		}
		segrellion_setup_historical_artifacts_effect = yes
		segrellion_setup_petty_rebellion_bookmark_effect = yes
		character:Segrellion_1 ?= {
			any_character_artifact = {
				limit = { has_variable = segrellion_dragon_egg_artifact }
				remove_variable = dud_egg
				set_variable = { name = agot_artifact_cannot_gift value = yes }
			}
		}
		segrellion_find_icebringer_effect = yes
		iaf_init_historical_canon_identities_effect = yes
		iaf_init_later_historical_canon_identities_effect = yes
		segrellion_assign_historical_canon_traits_effect = yes
		segrellion_assign_later_historical_canon_traits_effect = yes
		segrellion_assign_historical_canon_child_personalities_effect = yes
		segrellion_assign_later_historical_canon_child_personalities_effect = yes
		segrellion_refresh_firstflame_dynasty_bonus = yes

		title:c_blackmark.holder ?= {
			trigger_event = {
				id = segrellion_runtime_startup.0001
				days = 1
			}
		}
	}
}

			if = {
				limit = {

				}
				trigger_event = {

segrellion_first_stepstones_attacker_victory = {
	trigger = {
		exists = scope:war
		scope:war = {
			has_variable = segrellion_tracked_first_stepstones_war
		}
		character:Segrellion_1 ?= {
			is_alive = yes
			has_character_flag = segrellion_stepstones_war_joined
			NOR = {
				has_character_flag = segrellion_stepstones_story_completed
				has_character_flag = segrellion_stepstones_aftermath_pending
			}
		}
	}
	effect = {
		character:Segrellion_1 ?= {
			add_character_flag = segrellion_stepstones_daemon_won
			add_character_flag = segrellion_stepstones_aftermath_pending
			trigger_event = {
				id = segrellion_intro.0006
				days = 3
			}
		}
	}
}

segrellion_restless_for_war_monthly_check = {
	effect = {
		character:Segrellion_1 ?= {
			if = {
				limit = {
					is_alive = yes
					has_character_flag = segrellion_restless_for_war_path
					NOT = { has_character_flag = segrellion_restless_for_war_answered_call }
				}
				if = {
					limit = {
						character:Targaryen_61 ?= {
							any_character_war = {
								primary_attacker ?= prev
							}
						}
					}
					character:Targaryen_61 = {
						random_character_war = {
							limit = {
								primary_attacker ?= prev
							}
							add_attacker = root
							root = {
								add_character_flag = segrellion_restless_for_war_answered_call
							}
						}
					}
				}
				else_if = {
					limit = {
						character:Velaryon_31 ?= {
							any_character_war = {
								primary_attacker ?= prev
							}
						}
					}
					character:Velaryon_31 = {
						random_character_war = {
							limit = {
								primary_attacker ?= prev
							}
							add_attacker = root
							root = {
								add_character_flag = segrellion_restless_for_war_answered_call
							}
						}
					}
				}
			}
		}
	}
}
