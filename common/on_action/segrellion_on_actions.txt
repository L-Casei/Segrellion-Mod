			if = {
				limit = { has_special_building_slot = no }
				add_special_building_slot = segrellion_beacon_01
			}
		}

			if = {
				limit = { is_alive = yes }
				if = {
					limit = {
						has_character_flag = segrellion_intro_event_seen

							has_character_flag = segrellion_opening_route_dragonstone
							has_character_flag = segrellion_opening_route_vhalerion
							has_character_flag = segrellion_chose_stepstones
							has_character_flag = segrellion_returned_to_vhalerion

					}
					remove_character_flag = segrellion_intro_event_seen
				}
				add_character_flag = has_scripted_appearance
				add_character_flag = hair_trait_male_hair_custom_20
				add_character_flag = beard_trait_no_beard
			}
		}
		character:Segrellion_1 ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_character_flag = segrellion_intro_event_seen }
				}
				trigger_event = {
					id = segrellion_intro.0001
					days = 3
				}
			}
		}
		character:Vladwyn_1 ?= {
			if = {
				limit = { is_alive = yes }
				add_character_flag = segrellion_icebringer_authorized

			}

				limit = { is_alive = yes }
				add_character_flag = has_scripted_appearance
				add_character_flag = hair_trait_female_hair_custom_14
				add_character_flag = beard_trait_no_beard
				add_character_flag = segrellion_daenerys_custom_clothes
				if = {
					limit = {
						current_date < 8113.1.1
						any_spouse = { this = character:Segrellion_1 }
					}
					add_character_modifier = {
						modifier = segrellion_daenerys_canon_pregnancy_waiting_modifier

					}
				}
			}
		}
		character:Segrellion_12 ?= {

				add_character_flag = has_scripted_appearance
				add_character_flag = hair_trait_m_hair_ep3_byzantine_01
				add_character_flag = beard_trait_no_beard

vladwyn_8106_startup = {
    effect = {
        every_player = {
            limit = {
                this = character:Vladwyn_1
                game_start_date = 8106.4.19
                NOT = { has_character_flag = vladwyn_sentinels_path_resolved }
                NOT = { has_character_flag = vladwyn_sentinels_path_queued }
            }

            # -------------------------------------------------
            # FAILSAFE:
            # Tristan MUST begin as a Landless Adventurer.
            # If AGOT/startup has reduced him to a courtier,
            # remove him from that court and rebuild his camp.
            # -------------------------------------------------
            if = {
                limit = {
                    has_dlc_feature = roads_to_power
                    NOT = { is_landless_adventurer = yes }
                }

                move_to_pool = yes

                create_landless_adventurer_title_effect = {
                    REASON = flag:historical
                    FLAVOR_CHAR = this
                }
            }

            add_character_flag = vladwyn_sentinels_path_queued

            trigger_event = {
                id = segrellion_sentinel.0001
                days = 1
            }
        }
    }
}


# =========================================================
# THE UNICORN RETURNS
# DAEMON VS CLIFFORD - SHORT BATTLE FAILSAFE
#
# Root = victorious combat side
# =========================================================

segrellion_unicorn_returns_clifford_battle_check = {

	trigger = {

		character:Segrellion_60 ?= {

			is_alive = yes

			has_character_flag = segrellion_unicorn_returns_war_active

			NOT = {
				has_character_flag = segrellion_clifford_duel_triggered
			}

			NOT = {
				has_character_flag = segrellion_killed_clifford_event_done
			}
		}

		character:Stormflame_13 ?= {
			is_alive = yes
		}

		# Daemon must be on the victorious side.
		OR = {

			any_side_commander = {
				this = character:Segrellion_60
			}

			any_side_knight = {
				this = character:Segrellion_60
			}
		}

		# Clifford must really be on the defeated side.
		enemy_side = {

			OR = {

				any_side_commander = {
					this = character:Stormflame_13
				}

				any_side_knight = {
					this = character:Stormflame_13
				}
			}
		}
	}

	effect = {

		character:Segrellion_60 = {

			add_character_flag = segrellion_clifford_duel_triggered

			trigger_event = {
				id = segrellion_winds.0301
				days = 1
			}
		}
	}
}

#
# on_army_enter_province:
# root = owner of the army
# scope:army = army
# =========================================================

segrellion_unicorn_returns_beacon_arrival_check = {

	trigger = {

		this = character:Segrellion_60

		is_alive = yes

		has_character_flag = segrellion_unicorn_returns_war_active

		NOT = {
			has_character_flag = segrellion_beacon_surrender_demanded
		}

		character:Stormflame_3 ?= {
			is_alive = yes
		}

		scope:army ?= {

			army_owner = character:Segrellion_60

			location = title:b_castle_farring.title_province
		}
	}


	effect = {

		add_character_flag = segrellion_beacon_surrender_demanded

		remove_character_modifier = segrellion_beacon_awaits_modifier

		trigger_event = {
			id = segrellion_winds.0401
			days = 1
		}
	}
}

segrellion_unicorn_returns_beacon_arrival_fallback = {

	trigger = {

		character:Segrellion_60 ?= {

			is_alive = yes

			has_character_flag = segrellion_unicorn_returns_war_active

			NOT = {
				has_character_flag = segrellion_beacon_surrender_demanded
			}

			any_army = {
				location = title:b_castle_farring.title_province
			}
		}

		character:Stormflame_3 ?= {
			is_alive = yes
		}
	}


	effect = {

		character:Segrellion_60 = {

			add_character_flag = segrellion_beacon_surrender_demanded

			remove_character_modifier = segrellion_beacon_awaits_modifier

			trigger_event = {
				id = segrellion_winds.0401
				days = 1
			}
		}
	}
}


# =========================================================
# THE BEACON FALLS AFTER LEO REFUSES
# Root of on_siege_completion = character
# =========================================================

segrellion_unicorn_returns_beacon_siege_check = {

	trigger = {

		location = title:b_castle_farring.title_province

		character:Segrellion_60 ?= {

			is_alive = yes

			has_character_flag = segrellion_unicorn_returns_war_active

			has_character_flag = segrellion_leo_refused_beacon_surrender

			NOT = {
				has_character_flag = segrellion_beacon_stormed_resolution
			}
		}

		OR = {

			this = character:Segrellion_60

			is_at_war_with = character:Stormflame_3

			liege ?= {
				is_at_war_with = character:Stormflame_3
			}

			top_liege ?= {
				is_at_war_with = character:Stormflame_3
			}
		}
	}

	effect = {

		character:Segrellion_60 = {

			add_character_flag = segrellion_beacon_stormed_resolution

			segrellion_capture_stormflame_court_effect = yes

			trigger_event = {
				id = segrellion_winds.0404
				days = 1
			}

			var:segrellion_unicorn_returns_war ?= {
				end_war = attacker
			}
		}
	}
}

# =========================================================
# THE BEACON - YEARLY CASTLE FLAVOR
# =========================================================

segrellion_beacon_flavor_yearly_check = {

	effect = {

		title:c_farring_cross.holder ?= {

			if = {
				limit = {

					is_alive = yes
					is_adult = yes

					title:b_castle_farring.title_province = {
						has_building_or_higher = segrellion_beacon_01
					}
				}


				# =================================================
				# DRAGONSPEAR'S PLANS
				#
				# The plans remain useful to whoever controls the
				# Beacon until the final expansion is completed.
				# =================================================

				if = {
					limit = {

						title:b_castle_farring.title_province = {

							has_variable = segrellion_dragonspear_plans_studied

							NOT = {
								has_building_or_higher = segrellion_beacon_05
							}
						}

						NOT = {
							has_character_modifier = segrellion_dragonspear_plans_modifier
						}
					}

					add_character_modifier = {
						modifier = segrellion_dragonspear_plans_modifier
						years = 2
					}
				}

				else_if = {
					limit = {

						has_character_modifier = segrellion_dragonspear_plans_modifier

						title:b_castle_farring.title_province = {
							has_building_or_higher = segrellion_beacon_05
						}
					}

					remove_character_modifier = segrellion_dragonspear_plans_modifier
				}


				# =================================================
				# BEACON EVENT
				#
				# 35% yearly chance = roughly one every 2.8 years.
				# =================================================

				random = {
					chance = 35

					segrellion_beacon_flavor_pool_effect = yes
				}
			}
		}
	}
}

# =========================================================
# DAEMON + ARIANNE DYNASTIC PACT
#
# First child: Segrellion
# Second child: Martell
# =========================================================

segrellion_daemon_arianne_children = {

	trigger = {

		character:Segrellion_60 ?= {
			has_character_flag = segrellion_daemon_arianne_dynastic_pact
		}

		character:Martell_9 ?= {
			has_character_flag = segrellion_daemon_arianne_dynastic_pact
		}

		scope:father = {
			this = character:Segrellion_60
		}

		scope:mother = {
			this = character:Martell_9
		}
	}


	effect = {

		character:Martell_9 = {
			house = {
				save_scope_as = segrellion_arianne_house
			}
		}


		# =================================================
		# FIRST CHILD -> SEGRELLION
		# =================================================

		if = {

			limit = {

				character:Segrellion_60 = {

					NOT = {
						has_character_flag = segrellion_arianne_first_child_born
					}
				}
			}

			root = {
				set_house = house:house_Segrellion
			}

			character:Segrellion_60 = {
				add_character_flag = segrellion_arianne_first_child_born
			}
		}


		# =================================================
		# SECOND CHILD -> MARTELL
		# =================================================

		else_if = {

			limit = {

				character:Segrellion_60 = {

					NOT = {
						has_character_flag = segrellion_arianne_second_child_born
					}
				}
			}

			root = {
				set_house = scope:segrellion_arianne_house
			}

			character:Segrellion_60 = {
				add_character_flag = segrellion_arianne_second_child_born
			}
		}
	}
}
