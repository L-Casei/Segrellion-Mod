# =========================================================
# DAENERYS AI - INVASION FAILSAFE
# =========================================================

segrellion_daenerys_ai_invasion_failsafe = {

	effect = {

		character:Targaryen_9 ?= {

			if = {

				limit = {

					is_alive = yes
					is_ai = yes

					game_start_date >= 8300.8.20
					game_start_date < 8301.1.1

					has_character_flag = segrellion_daenerys_invasion_committed

					NOT = {
						has_character_flag = segrellion_daenerys_landed_in_westeros
					}
				}


				# =================================================
				# AI FLEET FINANCING
				#
				# Required fleet contributions cost 2300 gold.
				# AI receives one emergency treasury injection.
				# =================================================

				if = {

					limit = {

						has_character_flag = segrellion_daenerys_fleet_project_active

						NOT = {
							has_character_flag = segrellion_daenerys_dragon_fleet_completed
						}

						NOT = {
							has_character_flag = segrellion_daenerys_ai_fleet_funding_failsafe_used
						}
					}

					add_gold = 3000

					add_character_flag = segrellion_daenerys_ai_fleet_funding_failsafe_used
				}


				# =================================================
				# COMPLETED FLEET MUST LEAD TO WESTEROS
				# =================================================

				if = {

					limit = {

						has_character_flag = segrellion_daenerys_dragon_fleet_completed

						has_character_flag = segrellion_daenerys_camped_at_pentos

						NOT = {
							has_character_flag = segrellion_daenerys_crossing_committed
						}

						NOT = {
							has_character_flag = segrellion_daenerys_landing_chosen
						}
					}


					segrellion_daenerys_apply_crossing_capacity_effect = yes

					remove_character_flag = segrellion_daenerys_waiting_at_pentos

					add_character_flag = segrellion_daenerys_crossing_committed


					trigger_event = {
						id = segrellion_daenerys_invasion.0010
						days = 1
					}
				}
			}
		}
	}
}

# =========================================================
# LEO NEVER LEAVES THE BEACON
# =========================================================

segrellion_unicorn_returns_leo_beacon_lock = {

	trigger = {

		character:Segrellion_60 ?= {
			has_character_flag = segrellion_unicorn_returns_war_active
		}

		character:Stormflame_3 ?= {
			is_alive = yes
			NOT = {
				is_imprisoned = yes
			}
		}
	}


	effect = {

		character:Stormflame_3 = {

			set_knight_status = forbid

			# Never allow him to command the Stormflame field host.
			var:segrellion_stormflame_war_host ?= {

				if = {
					limit = {

						army_commander ?= {
							this = character:Stormflame_3
						}
					}


					# Clifford is the canonical field commander
					# while he remains alive and available.
					if = {
						limit = {
							character:Stormflame_13 ?= {
								is_alive = yes
								is_imprisoned = no
							}
						}

						assign_commander = character:Stormflame_13
					}


					# If Clifford is dead, Eldon takes command.
					else_if = {
						limit = {
							character:Stormflame_12 ?= {
								is_alive = yes
								is_imprisoned = no
							}
						}

						assign_commander = character:Stormflame_12
					}


					else = {
						remove_commander = yes
					}
				}
			}
		}
	}
}

# Arianne is reserved for Daemon's or Young Griff's scripted proposal
# during the opening five years of the 300 AC bookmark. The trait blocks
# normal marriage interactions; this guard also reverses any unrelated
# event marriage that bypasses the interaction checks.
segrellion_arianne_marriage_guard = {
	effect = {
		if = {
			limit = {
				game_start_date >= 8300.8.20
				game_start_date < 8301.1.1
				current_date < 8305.8.20
				OR = {
					AND = {
						this = character:Martell_9
						scope:spouse != character:Segrellion_60
						scope:spouse != character:IAF_Young_Griff_1
					}
					AND = {
						scope:spouse = character:Martell_9
						this != character:Segrellion_60
						this != character:IAF_Young_Griff_1
					}
				}
			}

			if = {
				limit = { this = character:Martell_9 }
				divorce = scope:spouse
			}
			else = {
				scope:spouse = { divorce = root }
			}

			character:Martell_9 = {
				if = {
					limit = { NOT = { has_trait = refusing_marriage } }
					add_trait = refusing_marriage
				}
				add_character_flag = segrellion_arianne_marriage_lock_active
			}
		}
		else_if = {
			limit = {
				game_start_date >= 8300.8.20
				game_start_date < 8301.1.1
				current_date < 8305.8.20
				OR = {
					AND = {
						this = character:Martell_9
						OR = {
							scope:spouse = character:Segrellion_60
							scope:spouse = character:IAF_Young_Griff_1
						}
					}
					AND = {
						scope:spouse = character:Martell_9
						OR = {
							this = character:Segrellion_60
							this = character:IAF_Young_Griff_1
						}
					}
				}
			}

			character:Martell_9 = {
				if = {
					limit = { has_trait = refusing_marriage }
					remove_trait = refusing_marriage
				}
				remove_character_flag = segrellion_arianne_marriage_lock_active
				add_character_flag = segrellion_arianne_event_marriage_completed
			}
		}
	}

# =========================================================
# REVERSE ANY UNAUTHORIZED MARRIAGE THAT BYPASSES
# refusing_marriage THROUGH AN EVENT OR OTHER SCRIPT.
# =========================================================

segrellion_young_griff_daemon_marriage_guard = {

	effect = {

		# Absolute Young Griff marriage protection.
		# This scans his REAL current spouses instead of trusting
		# the marriage event that created them.

		segrellion_young_griff_hard_marriage_guard_effect = yes


		# =================================================
		# YOUNG GRIFF
		#
		# Any marriage made while his scripted marriage
		# storyline is still unresolved MUST have been
		# explicitly authorized by one of our events.
		#
		# This does not depend on refusing_marriage or on
		# the lock-added bookkeeping flag.
		# =================================================

		if = {

			limit = {

				OR = {

					this = character:IAF_Young_Griff_1

					scope:spouse = character:IAF_Young_Griff_1
				}


				character:IAF_Young_Griff_1 = {

					NOT = {
						has_character_flag = segrellion_young_griff_story_marriage_authorized
					}


					NOT = {
						has_character_flag = segrellion_young_griff_margaery_story_resolved
					}
				}
			}


			# =============================================
			# YOUNG GRIFF IS ROOT
			# =============================================

			if = {

				limit = {

					this = character:IAF_Young_Griff_1
				}


				divorce = scope:spouse
			}


			# =============================================
			# YOUNG GRIFF IS THE SPOUSE
			# =============================================

			else_if = {

				limit = {

					scope:spouse = character:IAF_Young_Griff_1
				}


				scope:spouse = {

					divorce = root
				}
			}


			# =============================================
			# RESTORE THE LOCK IMMEDIATELY
			# =============================================

			character:IAF_Young_Griff_1 = {

				if = {

					limit = {

						is_alive = yes

						is_married = no


						NOT = {
							has_trait = refusing_marriage
						}
					}


					add_trait = refusing_marriage
				}


				add_character_flag = segrellion_young_griff_story_marriage_lock_added

				add_character_flag = segrellion_young_griff_random_marriage_blocked
			}
		}


		# =================================================
		# DAEMON
		#
		# Visenya is explicitly excluded because her own
		# Black Unicorn marriage guard owns that storyline.
		# =================================================

		if = {

			limit = {

				OR = {

					this = character:Segrellion_60

					scope:spouse = character:Segrellion_60
				}


				character:Segrellion_60 = {

					has_character_flag = segrellion_daemon_story_marriage_lock_added


					NOT = {
						has_character_flag = segrellion_daemon_story_marriage_authorized
					}
				}


				NOT = {

					OR = {

						AND = {

							this = character:Segrellion_60
							scope:spouse = character:Black_Unicorn_1
						}


						AND = {

							this = character:Black_Unicorn_1
							scope:spouse = character:Segrellion_60
						}
					}
				}
			}


			if = {

				limit = {
					this = character:Segrellion_60
				}

				divorce = scope:spouse
			}


			else = {

				scope:spouse = {
					divorce = root
				}
			}
		}


		segrellion_apply_winds_story_marriage_locks_effect = yes
	}
}

			}
		}

			}

				this = character:Baratheon_3
			}
		}


		# =================================================
		# RAMSAY IS IN THE BATTLE
		#
		# Require a participant on the losing combat side.
		# =================================================

		OR = {

			enemy_side = {

				OR = {

					any_side_commander = {

						has_character_flag = segrellion_ramsay_northern_takeover
					}


					any_side_knight = {

						has_character_flag = segrellion_ramsay_northern_takeover
					}
				}
			}

		}
	}


	effect = {

		title:e_the_north.holder = {

			save_scope_as = segrellion_ramsay_battle_prisoner
		}


		character:Baratheon_3 = {

			imprison = {

				target = scope:segrellion_ramsay_battle_prisoner

				type = dungeon
			}


			add_character_flag = segrellion_stannis_defeated_ramsay_personally


			# =============================================
			# THIS IS THE ACTUAL NORTHERN WAR SAVED BY THE CB
			# =============================================

				end_war = attacker
			}
		}


		scope:segrellion_ramsay_battle_prisoner = {

			add_character_flag = segrellion_ramsay_captured_by_stannis_in_battle
		}
	}
}

		# =================================================
		# LEAVE IRON THRONE WARS
		# =================================================

		every_character_war = {

			limit = {


				# This war actually targets the Iron Throne.

				casus_belli = {

					any_target_title = {

						this = title:h_the_iron_throne
					}
				}


				# Never try removing a primary war leader.

				NOT = {

					primary_attacker = scope:segrellion_new_north_holder
				}

				NOT = {

					primary_defender = scope:segrellion_new_north_holder
				}
			}


			remove_participant = scope:segrellion_new_north_holder
		}


		# =================================================
		# THE NORTH MUST BE INDEPENDENT
		# =================================================

		if = {

			limit = {

				NOT = {

					is_independent_ruler = yes
				}
			}


			create_title_and_vassal_change = {

				type = independency

				save_scope_as = segrellion_north_independence_change

				add_claim_on_loss = no
			}


			becomes_independent = {

				change = scope:segrellion_north_independence_change
			}


			resolve_title_and_vassal_change = scope:segrellion_north_independence_change
		}
	}
}

# together with their direct great bannermen.
#
# This is deliberately dynamic:
# Jon, Sansa, Rickon or another Northern ruler all work.
# The same applies to the current ruler of the Vale.
#
# If AGOT's megawar system initially places one of these
# rulers on the Iron Throne's side, remove them from that
# side first and force them onto Stannis' side.
# =========================================================

# =========================================================
# STANNIS - IRON THRONE WAR SUPPORT
#
# Once Jon has been restored to the North and has pledged
# himself to Stannis' claim:
#
# - Jon joins Stannis
# - House Manderly joins Stannis
# - House Dustin joins Stannis
#
# Their existing unrelated wars do not matter.
# =========================================================

segrellion_stannis_iron_throne_war_setup = {

	trigger = {

		exists = scope:war


		scope:war = {

			primary_attacker = character:Baratheon_3


			casus_belli = {

				any_target_title = {

					this = title:h_the_iron_throne
				}
			}
		}
	}


	effect = {


		# =================================================
		# SAVE THE WAR
		# =================================================

		character:Baratheon_3 = {

			set_variable = {

				name = segrellion_stannis_iron_throne_war

				value = scope:war
			}

			add_character_flag = segrellion_stannis_iron_throne_war_started
		}


		# =================================================
		# JON SNOW
		# RESURRECTED VERSION
		# =================================================

		if = {

			limit = {

				has_global_variable = segrellion_resurrected_jon


				global_var:segrellion_resurrected_jon ?= {

					is_alive = yes
					is_landed = yes

					has_title = title:e_the_north

					has_character_flag = segrellion_jon_restored_to_winterfell

					has_character_flag = segrellion_jon_pledged_to_stannis_iron_throne
				}
			}


			global_var:segrellion_resurrected_jon = {

				save_scope_as = segrellion_stannis_jon_ally
			}


			scope:war = {

				# AGOT megawars may have placed Jon
				# on the wrong side first.

				if = {

					limit = {

						is_defender = scope:segrellion_stannis_jon_ally
					}

					remove_participant = scope:segrellion_stannis_jon_ally
				}


				if = {

					limit = {

						NOT = {
							is_attacker = scope:segrellion_stannis_jon_ally
						}
					}

					add_attacker = scope:segrellion_stannis_jon_ally
				}
			}


			scope:segrellion_stannis_jon_ally = {

				if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_stannis_throne_support_event_seen
						}
					}

					add_character_flag = segrellion_stannis_throne_support_event_seen

					trigger_event = {
						id = segrellion_stannis.0600
						days = 1
					}
				}
			}
		}


		# =================================================
		# HISTORICAL JON FALLBACK
		# =================================================

		else_if = {

			limit = {

				character:Stark_7 ?= {

					is_alive = yes
					is_landed = yes

					has_title = title:e_the_north

					has_character_flag = segrellion_jon_restored_to_winterfell

					has_character_flag = segrellion_jon_pledged_to_stannis_iron_throne
				}
			}


			character:Stark_7 = {

				save_scope_as = segrellion_stannis_jon_ally
			}


			scope:war = {

				if = {

					limit = {

						is_defender = scope:segrellion_stannis_jon_ally
					}

					remove_participant = scope:segrellion_stannis_jon_ally
				}


				if = {

					limit = {

						NOT = {
							is_attacker = scope:segrellion_stannis_jon_ally
						}
					}

					add_attacker = scope:segrellion_stannis_jon_ally
				}
			}


			scope:segrellion_stannis_jon_ally = {

				if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_stannis_throne_support_event_seen
						}
					}

					add_character_flag = segrellion_stannis_throne_support_event_seen

					trigger_event = {
						id = segrellion_stannis.0600
						days = 1
					}
				}
			}
		}


		# =================================================
		# MANDERLY + DUSTIN
		#
		# ONLY if Jon actually rules the North and has
		# pledged himself to Stannis.
		# =================================================

		title:e_the_north.holder ?= {

			if = {

				limit = {

					is_alive = yes

					has_character_flag = segrellion_jon_restored_to_winterfell

					has_character_flag = segrellion_jon_pledged_to_stannis_iron_throne
				}


				every_vassal = {

					limit = {

						is_alive = yes
						is_landed = yes


						OR = {

							house = house:house_Manderly

							house = house:house_Dustin
						}
					}


					save_temporary_scope_as = segrellion_stannis_northern_ally


					scope:war = {

						# If AGOT initially threw them onto
						# the Iron Throne's side, remove them.

						if = {

							limit = {

								is_defender = scope:segrellion_stannis_northern_ally
							}

							remove_participant = scope:segrellion_stannis_northern_ally
						}


						# Then force them onto Stannis' side.

						if = {

							limit = {

								NOT = {
									is_attacker = scope:segrellion_stannis_northern_ally
								}
							}

							add_attacker = scope:segrellion_stannis_northern_ally
						}
					}


					add_character_flag = segrellion_northern_lord_supported_stannis


					if = {

						limit = {

							NOT = {
								has_character_flag = segrellion_stannis_throne_support_event_seen
							}
						}

						add_character_flag = segrellion_stannis_throne_support_event_seen

						trigger_event = {
							id = segrellion_stannis.0610
							days = 1
						}
					}
				}
			}
		}
	}
}


# =========================================================
# JON SNOW - RESURRECTION FAILSAFE
#
# Guarantees two things:
#
# 1. If the scheduled mutiny event was lost during startup,
#    Whispers in the Snow is rescheduled.
#
# 2. If the post-death handoff event was lost, Melisandre
#    restarts it.
# =========================================================

segrellion_jon_resurrection_progression = {

    effect = {

        # =================================================
        # JON IS STILL ALIVE
        # =================================================

        character:Stark_7 ?= {

            if = {

                limit = {

                    game_start_date = 8300.8.20

                    is_alive = yes

                    has_title = title:k_the_wall


                    has_global_variable = segrellion_jon_resurrection_story_enabled


                    NOT = {
                        has_global_variable = segrellion_resurrected_jon
                    }


                    NOT = {
                        has_character_flag = segrellion_jon_mutiny_pending
                    }


                    NOT = {
                        has_character_flag = segrellion_jon_mutiny_fallback_scheduled
                    }
                }


                # Prevent another monthly duplicate while
                # the replacement event is queued.

                add_character_flag = {

                    flag = segrellion_jon_mutiny_fallback_scheduled

                    days = 30
                }


                trigger_event = {

                    id = segrellion_jon_resurrection.0001

                    days = 1
                }
            }
        }


        # =================================================
        # JON IS DEAD BUT THE HANDOFF WAS LOST
        # =================================================

        # Recovery is handled by segrellion_jon_body_in_ice_failsafe below.

    }
}

# =========================================================
# JON SNOW - BODY IN THE ICE FAILSAFE
#
# If the direct delayed event somehow disappears,
# the current Lord Commander receives it on the next pulse.
# =========================================================

segrellion_jon_body_in_ice_failsafe = {

    trigger = {

        has_global_variable = segrellion_jon_resurrection_pending


        character:Stark_7 ?= {

            is_alive = no

            has_character_flag = segrellion_jon_murdered_by_watch
        }


        title:k_the_wall = {
            exists = holder
        }


        title:k_the_wall.holder = {

            is_alive = yes

            NOT = {
                has_character_flag = segrellion_body_in_ice_seen
            }

            NOT = {
                has_character_flag = segrellion_body_in_ice_queued
            }
        }


        NOT = {
            has_global_variable = segrellion_resurrected_jon
        }
    }


    effect = {

        title:k_the_wall.holder = {

            add_character_flag = segrellion_jon_resurrection_chain_started


            add_character_flag = {

                flag = segrellion_body_in_ice_queued

                days = 10
            }


            trigger_event = {

                id = segrellion_jon_resurrection.0100

                days = 1
            }
        }
    }
}

# TOMMEN BARATHEON - HARD NO OFFENSIVE WARS
#
# Startup / save recovery guard.
# Real-time wars are blocked by:
# segrellion_tommen_war_guard_on_actions.txt
# =========================================================

segrellion_tommen_peace_guard = {

	effect = {

		character:Baratheon_7 ?= {

			if = {

				limit = {

					is_alive = yes
					is_ai = yes

					game_start_date >= 8300.8.20
					game_start_date < 8301.1.1
				}


				save_scope_as = segrellion_tommen_war_guard


				if = {

					limit = {

						NOT = {
							has_character_flag = sworn_peace
						}
					}

					add_character_flag = sworn_peace
				}


				add_character_flag = segrellion_tommen_scripted_wars_only


				every_character_war = {

					limit = {

						primary_attacker = scope:segrellion_tommen_war_guard
					}

					end_war = invalidated
				}
			}
		}
	}
}

# JON SNOW - WAIT FOR STANNIS

segrellion_jon_wait_for_stannis_progression = {

	effect = {

		if = {

			limit = {

				has_global_variable = segrellion_resurrected_jon
			}


			global_var:segrellion_resurrected_jon ?= {

				if = {

					limit = {

						is_alive = yes

						has_title = title:k_the_wall

						has_character_flag = segrellion_jon_waiting_for_stannis_northern_result

						NOT = {
							has_character_flag = segrellion_jon_stannis_result_resolved
						}
					}


					# =========================================
					# STANNIS WON
					#
					# His existing 1010 event already offers
					# Jon Winterfell and the North.
					# =========================================

					if = {

						limit = {

							character:Baratheon_3 ?= {

								is_alive = yes

								has_character_flag = segrellion_stannis_northern_campaign_victory
							}


							NOT = {
								has_character_flag = segrellion_jon_stannis_north_offer_seen
							}

							NOT = {
								has_character_flag = segrellion_jon_stannis_north_offer_scheduled
							}
						}


						add_character_flag = { flag = segrellion_jon_stannis_north_offer_scheduled days = 7 }


						trigger_event = {

							id = segrellion_stannis.1010

							days = 3
						}
					}


					# =========================================
					# STANNIS LOST OR DIED
					# =========================================

					else_if = {

						limit = {

							OR = {

								character:Baratheon_3 = {

									has_character_flag = segrellion_stannis_northern_campaign_failed
								}


								NOT = {

									character:Baratheon_3 = {

										is_alive = yes
									}
								}
							}
						}


						remove_character_flag = segrellion_jon_waiting_for_stannis_northern_result

						add_character_flag = segrellion_jon_stannis_result_resolved

						add_character_flag = segrellion_jon_marching_on_winterfell


						trigger_event = {

							id = segrellion_jon_resurrection.0500

							days = { 3 7 }
						}
					}
				}
			}
		}
	}
}

segrellion_ramsay_war_lock_progression = {

	effect = {

		# =================================================
		# SAVE-COMPATIBLE RAMSAY SCOPE REPAIR
		# =================================================

		if = {

			limit = {

				NOT = {
					has_global_variable = segrellion_ramsay_bolton
				}


				title:e_the_north.holder ?= {

					is_alive = yes

					has_character_flag = segrellion_ramsay_northern_takeover
				}
			}


			set_global_variable = {

				name = segrellion_ramsay_bolton

				value = title:e_the_north.holder
			}
		}

		if = {

			limit = {

				has_global_variable = segrellion_ramsay_bolton

				global_var:segrellion_ramsay_bolton ?= {

					is_alive = yes

					has_character_flag = segrellion_ramsay_northern_takeover
				}
			}


			global_var:segrellion_ramsay_bolton = {

				add_character_flag = segrellion_ramsay_war_locked


				if = {

					limit = {

						NOT = {
							has_character_flag = sworn_peace
						}
					}

					add_character_flag = sworn_peace
				}


				save_scope_as = segrellion_war_locked_ramsay
			}


			# If another system has already inserted Ramsay
			# into an unrelated war as a secondary participant,
			# remove him.

			scope:segrellion_war_locked_ramsay = {

				every_character_war = {

					if = {

						limit = {

							primary_attacker != scope:segrellion_war_locked_ramsay

							primary_defender != scope:segrellion_war_locked_ramsay
						}


						remove_participant = scope:segrellion_war_locked_ramsay
					}
				}
			}
		}
	}
}

# RAMSAY BOLTON - NEW WAR BLOCK
# =========================================================

segrellion_ramsay_war_lock_on_start = {

	effect = {

		# =================================================
		# RAMSAY MUST NEVER START A WAR
		# =================================================

		if = {

			limit = {

				scope:attacker ?= {

					has_character_flag = segrellion_ramsay_war_locked
				}
			}


			scope:war = {

				end_war = white_peace
			}
		}


		# =================================================
		# RAMSAY CANNOT BE CALLED INTO THIS WAR
		#
		# Mark him as already called before anyone can
		# actually send the interaction.
		# =================================================

		if = {

			limit = {

				has_global_variable = segrellion_ramsay_bolton


				global_var:segrellion_ramsay_bolton ?= {

					is_alive = yes

					has_character_flag = segrellion_ramsay_war_locked
				}
			}


			scope:war = {

				if = {

					limit = {

						NOT = {

							is_participant = global_var:segrellion_ramsay_bolton
						}
					}


					set_called_to = global_var:segrellion_ramsay_bolton
				}
			}
		}
	}
}

# =========================================================
# CALL THE BANNERS - VASSAL JOINS THE ENEMY
#
# ROOT = character who just joined the war
# scope:war = war they joined
# =========================================================

segrellion_banner_enemy_join_check = {

	effect = {

		if = {

			limit = {

				is_alive = yes
				is_landed = yes

				exists = liege

				liege = {

					is_alive = yes
					is_landed = yes
				}

				scope:war ?= {

					is_civil_war = no
				}
			}


			liege = {
				save_scope_as = segrellion_banner_liege
			}


			if = {

				limit = {

					scope:war = {

						is_war_leader = scope:segrellion_banner_liege


						OR = {

							# Vassal joined defenders while liege leads attackers.
							AND = {

								is_attacker = scope:segrellion_banner_liege

								any_war_defender = {
									this = root
								}
							}


							# Vassal joined attackers while liege leads defenders.
							AND = {

								is_defender = scope:segrellion_banner_liege

								any_war_attacker = {
									this = root
								}
							}
						}
					}
				}


				segrellion_banner_joined_enemy_effect = yes
			}
		}
	}
}
