# =========================================================
# SEGRELLION RELIC SEARCH - TRAVEL STATE
# =========================================================


# =========================================================
# A RELIC JOURNEY ACTUALLY BEGINS
# =========================================================

on_travel_plan_start = {
	on_actions = {
		segrellion_relic_on_travel_plan_start
	}
}

segrellion_relic_on_travel_plan_start = {

	trigger = {

		has_character_flag = segrellion_legendary_relic_search_active

		OR = {
			has_character_flag = segrellion_relic_pending_tyrosh
			has_character_flag = segrellion_relic_pending_oldtown
			has_character_flag = segrellion_relic_pending_winterfell
			has_character_flag = segrellion_relic_pending_castle_black
			has_character_flag = segrellion_relic_pending_shadow_tower
			has_character_flag = segrellion_relic_pending_skagos
			has_character_flag = segrellion_relic_pending_nightfort
		}
	}

	effect = {

		# This flag means the character actually pressed
		# Start Journey and left on the expedition.
		add_character_flag = segrellion_relic_journey_started

		remove_character_flag = segrellion_legendary_relic_search_paused
		remove_character_flag = segrellion_relic_continuation_planner
	}
}


# =========================================================
# AN ACTUAL RELIC JOURNEY IS CANCELLED
# =========================================================

on_travel_plan_cancel = {
	on_actions = {
		segrellion_relic_on_travel_plan_cancel
	}
}

segrellion_relic_on_travel_plan_cancel = {

	trigger = {

		has_character_flag = segrellion_legendary_relic_search_active
		has_character_flag = segrellion_relic_journey_started
	}

	effect = {
		remove_character_flag = segrellion_relic_journey_started
	}

	events = {
		segrellion_relic.0090
	}
}
