on_birth_child = {
	on_actions = {
		segrellion_canon_trait_on_birth
	}
}

on_birthday = {
	on_actions = {
		segrellion_canon_child_personality_on_birthday
		segrellion_vhalyrax_birthday_growth
		segrellion_check_dragonriding_house_on_birthday
	}
}

segrellion_canon_trait_on_birth = {
	effect = {
		scope:child = {
			segrellion_assign_canon_trait_from_identity_effect = yes
			segrellion_assign_later_canon_trait_from_identity_effect = yes
		}
		segrellion_try_force_later_canon_marriages_effect = yes
	}
}

segrellion_canon_child_personality_on_birthday = {
	effect = {
		segrellion_assign_canon_child_personality_effect = yes
		segrellion_assign_later_canon_child_personality_effect = yes
		segrellion_try_force_later_canon_marriages_effect = yes
		iaf_try_start_later_segrellion_canon_story_effect = yes
	}
}

segrellion_vhalyrax_birthday_growth = {
	effect = {
		segrellion_vhalyrax_growth_effect = yes
	}
}


segrellion_check_dragonriding_house_on_birthday = {
	effect = {
		if = {
			limit = {
				current_date >= 8120.1.1
			}

			house:house_Segrellion = {
				if = {
					limit = {
						NOT = { has_house_modifier = dragonrider_house_modifier }
					}
					add_house_modifier = {
						modifier = dragonrider_house_modifier
					}
				}
			}

			# Opcional
			house:house_Stormflame = {
				if = {
					limit = {
						NOT = { has_house_modifier = dragonrider_house_modifier }
					}
					add_house_modifier = {
						modifier = dragonrider_house_modifier
					}
				}
			}
		}
	}
}
