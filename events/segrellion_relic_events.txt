	type = character_event

					scope:relic_tyrosh_host = {
						opinion = {
							target = root

				}

				desc = segrellion_relic.1005.desc.hostile
			}

			triggered_desc = {
				trigger = {
					scope:relic_tyrosh_host = {
						opinion = {
							target = root
							value >= 75
						}
					}
				}

			}

			desc = segrellion_relic.1005.desc_cautious
		}
	}

	theme = diplomacy

	override_background = {
		reference = throne_room
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:relic_tyrosh_host
		animation = personality_bold
	}


	# Hostile: opinion below 0.
	option = {

		trigger = {
			scope:relic_tyrosh_host = {
				opinion = {
					target = root
					value < 0
				}
			}
		}

		name = segrellion_relic.1005.a

		trigger_event = {
			id = segrellion_relic.1091
			days = 1
		}
	}


	# Neutral/cautious: 0-74.
	option = {

		trigger = {
			scope:relic_tyrosh_host = {
				opinion = {
					target = root
					value >= 0
				}

				opinion = {
					target = root
					value < 75
				}
			}
		}

		name = segrellion_relic.1005.b

		trigger_event = {
			id = segrellion_relic.1010
			days = { 4 7 }
		}
	}


	# Excellent relations: 75+.
	option = {

		trigger = {
			scope:relic_tyrosh_host = {

				}
			}
		}

		name = segrellion_relic.1005.c

		}

# =========================================================
# 5010 - DOES THE SWORD WISH TO BE FOUND?
# =========================================================

segrellion_relic.5010 = {

	type = character_event

	title = segrellion_relic.5010.t
	desc = segrellion_relic.5010.desc

	theme = mystery

	override_background = {
		reference = dungeon
	}

	left_portrait = {
		character = root
		animation = fear
	}

	option = {

		name = segrellion_relic.5010.a

		custom_tooltip = segrellion_relic.5010.a.tt


		# =====================================================
		# ONLY WINTERFELL CAN LEAD TO ICEBRINGER
		# =====================================================

		if = {

			limit = {
				has_character_flag = segrellion_relic_route_winterfell
			}


			# =====================================================
			# FULL ACCEPTANCE
			# =====================================================

			if = {

				limit = {
					segrellion_icebringer_accepted_trigger = yes
				}

				random_list = {

					90 = {
						trigger_event = segrellion_relic.5100
					}

					10 = {
						trigger_event = segrellion_relic.5200
					}
				}
			}


			# =====================================================
			# PARTIAL ACCEPTANCE
			# =====================================================

			else_if = {

				limit = {
					segrellion_icebringer_partial_trigger = yes
				}

				random_list = {

					65 = {
						trigger_event = segrellion_relic.5100
					}

					35 = {
						trigger_event = segrellion_relic.5200
					}
				}
			}


			# =====================================================
			# REJECTED
			# =====================================================

			else = {

				random_list = {

					15 = {
						trigger_event = segrellion_relic.5100
					}

					85 = {
						trigger_event = segrellion_relic.5200
					}
				}
			}
		}


		# Defensive fallback.
		# Even an old save/debug chain can NEVER recover Icebringer
		# through Tyrosh or Oldtown.
		else = {
			trigger_event = segrellion_relic.5200
		}
	}
}
