# =========================================================
# 5010 - DOES THE SWORD WISH TO BE FOUND?
# =========================================================

segrellion_relic.5010 = {

	type = character_event

	title = segrellion_relic.5010.t
	desc = segrellion_relic.5010.desc

	theme = mystery

	override_background = {
		reference = dungeon
	}

	left_portrait = {
		character = root
		animation = fear
	}

	option = {

		name = segrellion_relic.5010.a

		custom_tooltip = segrellion_relic.5010.a.tt


		# =====================================================
		# FULL ACCEPTANCE
		# =====================================================

		if = {

			limit = {
				segrellion_icebringer_accepted_trigger = yes
			}


			# Winterfell: strongest trail.
			if = {

				limit = {
					has_character_flag = segrellion_relic_route_winterfell
				}

				random_list = {

					90 = {
						trigger_event = segrellion_relic.5100
					}

					10 = {
						trigger_event = segrellion_relic.5200
					}
				}
			}


			# Oldtown: strong archival evidence.
			else_if = {

				limit = {
					has_character_flag = segrellion_relic_route_oldtown
				}

				random_list = {

					85 = {
						trigger_event = segrellion_relic.5100
					}

					15 = {
						trigger_event = segrellion_relic.5200
					}
				}
			}


			# Tyrosh: longest and most uncertain trail.
			else = {

				random_list = {

					80 = {
						trigger_event = segrellion_relic.5100
					}

					20 = {
						trigger_event = segrellion_relic.5200
					}
				}
			}
		}


		# =====================================================
		# PARTIAL ACCEPTANCE
		# =====================================================

		else_if = {

			limit = {
				segrellion_icebringer_partial_trigger = yes
			}


			if = {

				limit = {
					has_character_flag = segrellion_relic_route_winterfell
				}

				random_list = {

					65 = {
						trigger_event = segrellion_relic.5100
					}

					35 = {
						trigger_event = segrellion_relic.5200
					}
				}
			}


			else_if = {

				limit = {
					has_character_flag = segrellion_relic_route_oldtown
				}

				random_list = {

					50 = {
						trigger_event = segrellion_relic.5100
					}

					50 = {
						trigger_event = segrellion_relic.5200
					}
				}
			}


			else = {

				random_list = {

					40 = {
						trigger_event = segrellion_relic.5100
					}

					60 = {
						trigger_event = segrellion_relic.5200
					}
				}
			}
		}


		# =====================================================
		# REJECTED
		# =====================================================

		else = {


			if = {

				limit = {
					has_character_flag = segrellion_relic_route_winterfell
				}

				random_list = {

					15 = {
						trigger_event = segrellion_relic.5100
					}

					85 = {
						trigger_event = segrellion_relic.5200
					}
				}
			}


			else_if = {

				limit = {
					has_character_flag = segrellion_relic_route_oldtown
				}

				random_list = {

					10 = {
						trigger_event = segrellion_relic.5100
					}

					90 = {
						trigger_event = segrellion_relic.5200
					}
				}
			}


			else = {

				random_list = {

					5 = {
						trigger_event = segrellion_relic.5100
					}

					95 = {
						trigger_event = segrellion_relic.5200
					}
				}
			}
		}
	}
}

		}

		segrellion_refresh_icebringer_affinity_effect = yes

		remove_character_flag = segrellion_relic_route_oldtown
		remove_character_flag = segrellion_relic_route_winterfell

		add_character_flag = segrellion_recovered_icebringer
	}
}
