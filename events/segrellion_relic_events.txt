		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:relic_tyrosh_host
		animation = personality_bold
	}


	# Hostile: opinion below 0.
	option = {

		trigger = {
			scope:relic_tyrosh_host = {
				opinion = {
					target = root
					value < 0
				}
			}
		}

		name = segrellion_relic.1005.a

		trigger_event = {
			id = segrellion_relic.1091
			days = 1
		}
	}


	# Neutral/cautious: 0-74.
	option = {

		trigger = {
			scope:relic_tyrosh_host = {
				opinion = {
					target = root
					value >= 0
				}

				opinion = {
					target = root
					value < 75
				}
			}
		}

		name = segrellion_relic.1005.b

		trigger_event = {
			id = segrellion_relic.1010
			days = { 4 7 }
		}
	}


	# Excellent relations: 75+.
	option = {

		trigger = {
			scope:relic_tyrosh_host = {

					target = root

				}
			}
		}
