namespace = agot_choose_dragonpit_event


agot_choose_dragonpit_event.0001 = {
	type = character_event
	title = agot_choose_dragonpit_event.0001.t

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:rider }
				desc = agot_choose_dragonpit_event.0001.desc_rider
			}

			desc = agot_choose_dragonpit_event.0001.desc
		}
	}

	theme = dragon


	widget = {
		is_shown = { is_ai = no }
		gui = "agot_dragonpit_selection_three_options"
		container = "custom_widgets_container"
	}


	left_portrait = {
		character = root
		animation = thinking
	}


	right_portrait = {
		character = scope:head_dragonkeeper
		animation = throne_room_writer
	}


	immediate = {

		# Head Dragonkeeper
		if = {
			limit = {
				employs_court_position = head_dragonkeeper_court_position
			}

			court_position:head_dragonkeeper_court_position = {
				save_scope_as = head_dragonkeeper
			}
		}


		# Current rider of the dragon
		if = {
			limit = {
				scope:dragon = {
					exists = var:current_rider
				}
			}

			scope:dragon = {
				var:current_rider = {
					save_scope_as = rider
				}
			}
		}


		# ========================================================
		# FIND EVERY ACTIVE DRAGONPIT OWNED BY THE CHARACTER
		# ========================================================

		every_held_county = {
			limit = {

				OR = {

					# Generic constructed Dragonpit
					any_county_province = {
						has_holding_type = castle_holding
						has_building_or_higher = generic_dragon_pit_01
					}


					# Historical / special Dragonpits
					any_county_province = {
						OR = {

							# King's Landing Dragonpit
							has_building_or_higher = dragonpit_01

							# Restored Dragonpit ruins
							has_building = dragonpit_ruins_03

							# Dragonstone
							has_building = agot_dragonmont_01

							# =================================================
							# SEGRELLION - CAVERNS OF VHAL
							# Level II+
							# =================================================

							has_building_or_higher = segrellion_vhal_dragonpit_02
						}
					}
				}

				# Must actually have a Dragonkeeper Order
				has_variable = has_dragonkeeper_order
			}


			root = {
				add_to_variable_list = {
					name = possible_titles
					target = prev
				}
			}
		}


		# Used by AGOT's selection widget
		ordered_in_list = {
			variable = possible_titles

			root = {
				set_variable = {
					name = selected_title
					value = prev
				}
			}

			save_scope_as = selected_title
		}
	}


	# ============================================================
	# CHOOSE DRAGONPIT
	# ============================================================

	option = {
		name = agot_choose_dragonpit_event.0001.a

		trigger = {
			exists = var:selected_title
		}


		show_as_unavailable = {

			custom_tooltip = {
				text = CANT_SENT_TO_DRAGONPIT_TT

				scope:dragon = {
					owns_story_of_type = story_dragon_at_war
				}
			}


			custom_tooltip = {
				text = CANT_SENT_TO_DRAGONPIT_TRAVELING_TT

				scope:dragon = {

					trigger_if = {
						limit = {
							exists = var:current_rider
							exists = var:current_rider.current_travel_plan
						}

						var:current_rider.current_travel_plan = {
							has_travel_option = dragon_flight_option
						}
					}

					exists = current_travel_plan
				}
			}
		}


		if = {
			limit = {
				exists = var:selected_title
			}


			var:selected_title = {
				save_scope_as = dragon_pit_location
			}


			custom_tooltip = {
				text = agot_choose_dragonpit_event.0001.a.tt


				# ------------------------------------------------
				# OWNED DRAGON
				# ------------------------------------------------

				if = {
					limit = {
						scope:dragon = {
							has_character_flag = owned_dragon
						}
					}

					agot_send_to_dragonpit_effect = {
						DRAGON = scope:dragon
						DRAGONPIT_HOLDER = scope:dragon_pit_location.holder
						DRAGONPIT_COUNTY = scope:dragon_pit_location
					}
				}


				# ------------------------------------------------
				# WILD / UNOWNED DRAGON
				# ------------------------------------------------

				else = {

					agot_force_wild_dragon_into_dragonpit_effect = {
						DRAGON = scope:dragon
						DRAGONPIT_HOLDER = scope:dragon_pit_location.holder
						DRAGONPIT_COUNTY = scope:dragon_pit_location
					}
				}
			}


			# Dragon housed outside the capital Dragonpit
			if = {
				limit = {
					NOT = {
						var:selected_title = capital_county
					}
				}

				scope:dragon = {
					add_character_flag = dragon_put_in_different_pit
				}
			}


			# Rider is based somewhere else
			if = {
				limit = {

					exists = scope:dragon.var:current_rider

					trigger_if = {
						limit = {
							scope:dragon.var:current_rider = {
								is_landed = yes
							}
						}

						NOT = {
							scope:dragon_pit_location = scope:dragon.var:current_rider.capital_county
						}
					}


					trigger_else = {

						trigger_if = {
							limit = {
								exists = scope:dragon.var:current_rider.liege_or_court_owner
								exists = scope:dragon.var:current_rider.liege_or_court_owner.capital_county
							}

							NOT = {
								scope:dragon_pit_location = scope:dragon.var:current_rider.liege_or_court_owner.capital_county
							}
						}
					}
				}

				custom_tooltip = agot_choose_dragonpit_event.0001.a.tt_2
			}
		}
	}


	# ============================================================
	# RELEASE DRAGON
	# ============================================================

	option = {
		name = agot_choose_dragonpit_event.0001.b

		trigger = {
			scope:dragon = {
				has_character_flag = in_dragonpit
			}

			scope:dragon = {
				exists = var:pitted_dragon_location
			}
		}


		scope:dragon.var:pitted_dragon_location = {
			save_scope_as = dragon_pit_location
		}


		agot_remove_from_dragonpit_effect = {
			DRAGON = scope:dragon
			DRAGONPIT_HOLDER = root
			DRAGONPIT_COUNTY = scope:dragon_pit_location
		}
	}


	# ============================================================
	# LEAVE DRAGON WHERE IT IS
	# ============================================================

	option = {
		name = agot_choose_dragonpit_event.0001.c

		trigger = {
			scope:dragon = {
				has_character_flag = in_dragonpit
			}
		}
	}


	# ============================================================
	# CANCEL
	# ============================================================

	option = {
		name = agot_choose_dragonpit_event.0001.d

		trigger = {
			scope:dragon = {
				NOT = {
					has_character_flag = in_dragonpit
				}
			}
		}
	}


	after = {
		remove_variable = selected_title
		clear_variable_list = possible_titles
	}
}
