namespace = segrellion_relic_bridge


# =========================================================
# 0001 - THREE ROADS NORTH
# =========================================================

segrellion_relic_bridge.0001 = {

	type = character_event

	title = segrellion_relic.3900.t
	desc = segrellion_relic.3900.desc

	theme = travel

	override_background = {
		reference = wilderness
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}


	# =====================================================
	# CASTLE BLACK
	# =====================================================

	option = {

		name = segrellion_relic.3900.a

		custom_tooltip = segrellion_relic.3900.a.trail_tt

		segrellion_clear_relic_pending_destination_effect = yes
		add_character_flag = segrellion_relic_pending_castle_black

		if = {
			limit = {
				exists = current_travel_plan
			}

			current_travel_plan = {
				add_travel_waypoint = title:c_castle_black.title_province
				resume_travel_plan = yes
			}
		}

		else = {
			trigger_event = {
				id = segrellion_relic_bridge.0002
				days = 1
			}
		}
	}


	# =====================================================
	# SHADOW TOWER
	# =====================================================

	option = {

		name = segrellion_relic.3900.b

		custom_tooltip = segrellion_relic.3900.b.trail_tt

		segrellion_clear_relic_pending_destination_effect = yes
		add_character_flag = segrellion_relic_pending_shadow_tower

		if = {
			limit = {
				exists = current_travel_plan
			}

			current_travel_plan = {
				add_travel_waypoint = title:c_the_shadow_tower.title_province
				resume_travel_plan = yes
			}
		}

		else = {
			trigger_event = {
				id = segrellion_relic_bridge.0002
				days = 1
			}
		}
	}


	# =====================================================
	# SKAGOS
	# =====================================================

	option = {

		name = segrellion_relic.3900.c

		custom_tooltip = segrellion_relic.3900.c.trail_tt

		segrellion_clear_relic_pending_destination_effect = yes
		add_character_flag = segrellion_relic_pending_skagos

		if = {
			limit = {
				exists = current_travel_plan
			}

			current_travel_plan = {
				add_travel_waypoint = title:c_skane.title_province
				resume_travel_plan = yes
			}
		}

		else = {
			trigger_event = {
				id = segrellion_relic_bridge.0002
				days = 1
			}
		}
	}
}


# =========================================================
# 0002 - BEGIN NEXT NORTHERN JOURNEY
# =========================================================

segrellion_relic_bridge.0002 = {

	type = character_event
	hidden = yes

	trigger = {

		has_character_flag = segrellion_legendary_relic_search_active

		OR = {
			has_character_flag = segrellion_relic_pending_castle_black
			has_character_flag = segrellion_relic_pending_shadow_tower
			has_character_flag = segrellion_relic_pending_skagos
			has_character_flag = segrellion_relic_pending_nightfort
		}
	}

	immediate = {

		segrellion_resume_relic_travel_effect = yes
	}
}

# =========================================================
# 0003 - NORTHERN ARRIVAL DISPATCHER
# =========================================================

segrellion_relic_bridge.0003 = {

	type = character_event
	hidden = yes

	trigger = {
		has_character_flag = segrellion_legendary_relic_search_active
	}

	immediate = {

		if = {
			limit = {
				has_character_flag = segrellion_relic_pending_winterfell
			}

			trigger_event = {
				id = segrellion_relic.3000
			}
		}

		else_if = {
			limit = {
				has_character_flag = segrellion_relic_pending_castle_black
			}

			trigger_event = {
				id = segrellion_relic.4000
			}
		}

		else_if = {
			limit = {
				has_character_flag = segrellion_relic_pending_shadow_tower
			}

			trigger_event = {
				id = segrellion_relic.6000
			}
		}

		else_if = {
			limit = {
				has_character_flag = segrellion_relic_pending_skagos
			}

			trigger_event = {
				id = segrellion_relic.7000
			}
		}

		else_if = {
			limit = {
				has_character_flag = segrellion_relic_pending_nightfort
			}

			trigger_event = {
				id = segrellion_relic.5000
			}
		}
	}
}
