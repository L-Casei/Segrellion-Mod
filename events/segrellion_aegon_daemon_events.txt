namespace = segrellion_aegon_daemon


# =========================================================
# 0001 - THE FLAMELANDS QUESTION
#
# ROOT = YOUNG GRIFF
#
# Fires shortly after Aegon takes the Iron Throne.
# =========================================================

segrellion_aegon_daemon.0001 = {

	type = character_event

	title = segrellion_aegon_daemon.0001.t


	desc = {

		first_valid = {


			# =================================================
			# DAEMON HAD ALREADY PLEDGED HIMSELF TO AEGON
			# =================================================

			triggered_desc = {

				trigger = {
					segrellion_aegon_daemon_mutual_support_trigger = yes
				}

				desc = segrellion_aegon_daemon.0001.desc_friend
			}


			# =================================================
			# NO PRIOR OATH
			# =================================================

			desc = segrellion_aegon_daemon.0001.desc_hostile
		}
	}


	theme = diplomacy


	trigger = {

		this = character:IAF_Young_Griff_1

		is_alive = yes

		has_title = title:h_the_iron_throne


		NOT = {
			has_character_flag = segrellion_young_griff_flamelands_submission_resolved
		}


		character:Segrellion_60 ?= {

			is_alive = yes


			OR = {

				has_title = title:e_flamelands

				has_title = title:k_flamelands
			}


			NOT = {
				has_character_flag = segrellion_daemon_aegon_status_resolved
			}
		}
	}


	on_trigger_fail = {

		remove_character_flag = segrellion_young_griff_daemon_submission_late_flamelands

		remove_character_flag = segrellion_young_griff_flamelands_submission_queued

		remove_character_flag = segrellion_young_griff_flamelands_submission_story_started

		remove_character_flag = segrellion_young_griff_daemon_submission_retry_cooldown

		remove_character_flag = segrellion_young_griff_flamelands_submission_chain_active


		character:Segrellion_60 ?= {

			remove_character_flag = segrellion_daemon_aegon_status_scheduled
		}
	}

	immediate = {

		remove_character_flag = segrellion_young_griff_daemon_submission_late_flamelands

		remove_character_flag = segrellion_young_griff_flamelands_submission_queued

		add_character_flag = segrellion_young_griff_flamelands_submission_story_started


		character:Segrellion_60 = {

			remove_character_flag = segrellion_daemon_aegon_status_scheduled
		}
	}


	left_portrait = {

		character = root

		animation = thinking
	}


	right_portrait = {

		character = character:Segrellion_60

		animation = personality_bold
	}


	# =====================================================
	# PEACEFUL REQUEST
	#
	# Daemon previously swore himself to Aegon's cause.
	#
	# The oath is stored on BOTH sides:
	#
	# Daemon:
	# segrellion_pledged_to_young_griff
	#
	# Aegon:
	# segrellion_young_griff_daemon_oath_secured
	#
	# Either one is enough here so a missing mirrored flag
	# cannot murder the storyline.
	# =====================================================

	option = {

		name = segrellion_aegon_daemon.0001.a

		custom_tooltip = segrellion_aegon_daemon.0001.a.tt

		trigger = {
			segrellion_aegon_daemon_mutual_support_trigger = yes
		}

		character:Segrellion_60 = {

			add_character_flag = segrellion_daemon_aegon_peaceful_request_pending


			trigger_event = {

				id = segrellion_aegon_daemon.0002

				days = { 2 4 }
			}
		}


		ai_chance = {

			base = 100
		}
	}


	# =====================================================
	# ULTIMATUM
	#
	# Only if there was never an oath between them.
	# =====================================================

	option = {

		name = segrellion_aegon_daemon.0001.b

		custom_tooltip = segrellion_aegon_daemon.0001.b.tt

		trigger = {

			NOT = {
				segrellion_aegon_daemon_mutual_support_trigger = yes
			}
		}

		character:Segrellion_60 = {

			add_character_flag = segrellion_daemon_aegon_ultimatum_pending


			trigger_event = {

				id = segrellion_aegon_daemon.0003

				days = { 2 4 }
			}
		}


		ai_chance = {

			base = 80
		}
	}


	# =====================================================
	# ALLOW THE FLAMELANDS TO REMAIN INDEPENDENT
	#
	# A human Aegon can always choose this.
	#
	# AI Aegon cannot choose this if Daemon had already
	# pledged himself to Aegon's cause.
	# =====================================================

	option = {

		name = segrellion_aegon_daemon.0001.c

		custom_tooltip = segrellion_aegon_daemon.0001.c.tt


		trigger = {

			OR = {

				is_ai = no

				NOT = {
					segrellion_aegon_daemon_mutual_support_trigger = yes
				}
			}
		}


		character:Segrellion_60 = {

			if = {

				limit = {

					liege ?= character:IAF_Young_Griff_1
				}


				becomes_independent = yes
			}


			add_character_flag = segrellion_daemon_aegon_status_resolved

			add_character_flag = segrellion_daemon_independence_tolerated_by_aegon
		}


		add_character_flag = segrellion_young_griff_flamelands_submission_resolved

		remove_character_flag = segrellion_young_griff_flamelands_submission_chain_active


		ai_chance = {

			base = 20
		}
	}
}


# =========================================================
# 0002 - THE KING'S REQUEST
#
# Root = Daemon
# Peaceful request.
# =========================================================

segrellion_aegon_daemon.0002 = {

	type = letter_event

	sender = character:IAF_Young_Griff_1

	opening = segrellion_aegon_daemon.0002.opening

	title = segrellion_aegon_daemon.0002.t

	desc = segrellion_aegon_daemon.0002.desc

	theme = diplomacy


	trigger = {

		this = character:Segrellion_60

		is_alive = yes


		OR = {

			has_title = title:e_flamelands

			has_title = title:k_flamelands
		}


		has_character_flag = segrellion_daemon_aegon_peaceful_request_pending


		NOT = {
			has_character_flag = segrellion_daemon_aegon_status_resolved
		}


		character:IAF_Young_Griff_1 ?= {

			is_alive = yes

			has_title = title:h_the_iron_throne

			segrellion_aegon_daemon_mutual_support_trigger = yes
		}
	}


	on_trigger_fail = {

		remove_character_flag = segrellion_daemon_aegon_peaceful_request_pending


		character:IAF_Young_Griff_1 ?= {

			remove_character_flag = segrellion_young_griff_flamelands_submission_story_started

			remove_character_flag = segrellion_young_griff_daemon_submission_retry_cooldown

			remove_character_flag = segrellion_young_griff_flamelands_submission_chain_active
		}
	}	


	left_portrait = {

		character = root

		animation = thinking
	}


	right_portrait = {

		character = character:IAF_Young_Griff_1

		animation = personality_rational
	}


	# =====================================================
	# ACCEPT
	# =====================================================

	option = {

		name = segrellion_aegon_daemon.0002.a

		custom_tooltip = segrellion_aegon_daemon.0002.a.tt

		remove_character_flag = segrellion_daemon_aegon_peaceful_request_pending

		# If AGOT has not already attached Daemon to Aegon,
		# perform the actual fealty transfer now.

		segrellion_daemon_swear_fealty_to_aegon_effect = yes


		add_character_flag = segrellion_daemon_aegon_fealty_accepted

		add_character_flag = segrellion_daemon_aegon_status_resolved


		character:IAF_Young_Griff_1 = {

			add_character_flag = segrellion_young_griff_flamelands_submission_resolved

			add_character_flag = segrellion_young_griff_flamelands_peacefully_submitted

			remove_character_flag = segrellion_young_griff_flamelands_submission_chain_active
		}


		# Do NOT appoint Hand/Regent in the same tick as the
		# liege transfer. The appointment event does it once
		# the political state has fully settled.

		if = {

			limit = {

				OR = {

					has_character_flag = segrellion_pledged_to_young_griff


					character:IAF_Young_Griff_1 = {

						has_character_flag = segrellion_young_griff_daemon_oath_secured
					}


					character:IAF_Young_Griff_1 = {

						has_character_flag = segrellion_young_griff_supports_daemon
					}
				}
			}


			add_character_flag = {

				flag = segrellion_daemon_aegon_appointment_story_scheduled

				days = 30
			}


			trigger_event = {

				id = segrellion_aegon_daemon.0006

				days = 1
			}

			character:IAF_Young_Griff_1 = {

				trigger_event = {

					id = segrellion_aegon_daemon.0018

					days = 1
				}
			}
		}


		ai_chance = {

			base = 50


			modifier = {

				add = 25

				has_character_flag = segrellion_pledged_to_young_griff


				character:IAF_Young_Griff_1 = {

					has_character_flag = segrellion_young_griff_supports_daemon
				}


				opinion = {

					target = character:IAF_Young_Griff_1

					value >= 25
				}
			}
		}
	}


	# =====================================================
	# REFUSE
	# =====================================================

	option = {

		name = segrellion_aegon_daemon.0002.b

		custom_tooltip = segrellion_aegon_daemon.0002.b.tt

		remove_character_flag = segrellion_daemon_aegon_peaceful_request_pending

		# If both men fought for one another, AI Daemon
		# will NEVER reject Aegon's peaceful settlement.

		trigger = {

			OR = {

				is_ai = no


				NOR = {

					has_character_flag = segrellion_pledged_to_young_griff


					character:IAF_Young_Griff_1 = {

						has_character_flag = segrellion_young_griff_daemon_oath_secured
					}


					character:IAF_Young_Griff_1 = {

						has_character_flag = segrellion_young_griff_supports_daemon
					}
				}
			}
		}

		if = {

			limit = {
				liege ?= character:IAF_Young_Griff_1
			}

			becomes_independent = yes
		}

		add_character_flag = segrellion_daemon_aegon_fealty_refused


		character:IAF_Young_Griff_1 = {

			trigger_event = {

				id = segrellion_aegon_daemon.0004

				days = { 2 4 }
			}
		}


		ai_chance = {

			base = 50


			modifier = {

				add = -25


				has_character_flag = segrellion_pledged_to_young_griff


				character:IAF_Young_Griff_1 = {

					has_character_flag = segrellion_young_griff_supports_daemon
				}


				opinion = {

					target = character:IAF_Young_Griff_1

					value >= 25
				}
			}
		}
	}
}
