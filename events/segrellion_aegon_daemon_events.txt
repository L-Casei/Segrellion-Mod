namespace = segrellion_aegon_daemon


# =========================================================
# 0001 - THE FLAMELANDS QUESTION
#
# Root = Young Griff
# =========================================================

segrellion_aegon_daemon.0001 = {

	type = character_event

	title = segrellion_aegon_daemon.0001.t


	desc = {

		first_valid = {


			# Daemon actually fought for Aegon's crown.

			triggered_desc = {

				trigger = {

					character:Segrellion_60 = {

						has_character_flag = segrellion_pledged_to_young_griff
					}
				}

				desc = segrellion_aegon_daemon.0001.desc_friend
			}


			# They reached this point without Daemon
			# fighting for Aegon's throne.

			desc = segrellion_aegon_daemon.0001.desc_hostile
		}
	}


	theme = diplomacy


	trigger = {

		this = character:IAF_Young_Griff_1

		is_alive = yes

		has_title = title:h_the_iron_throne


		character:Segrellion_60 ?= {

			is_alive = yes

			is_independent_ruler = yes

			has_title = title:e_flamelands


			NOT = {
				has_character_flag = segrellion_daemon_aegon_status_resolved
			}
		}
	}


	on_trigger_fail = {

		character:Segrellion_60 ?= {

			remove_character_flag = segrellion_daemon_aegon_status_scheduled
		}
	}


	immediate = {

		character:Segrellion_60 = {

			remove_character_flag = segrellion_daemon_aegon_status_scheduled
		}


		add_character_flag = segrellion_young_griff_flamelands_submission_story_started
	}


	left_portrait = {

		character = root

		animation = thinking
	}


	right_portrait = {

		character = character:Segrellion_60

		animation = personality_bold
	}


	# =====================================================
	# PEACEFUL REQUEST
	#
	# Available because Daemon fought for Aegon.
	# =====================================================

	option = {

		name = segrellion_aegon_daemon.0001.a

		custom_tooltip = segrellion_aegon_daemon.0001.a.tt


		trigger = {

			character:Segrellion_60 = {

				has_character_flag = segrellion_pledged_to_young_griff
			}
		}


		character:Segrellion_60 = {

			add_character_flag = segrellion_daemon_aegon_peaceful_request_pending


			trigger_event = {

				id = segrellion_aegon_daemon.0002

				days = { 2 5 }
			}
		}


		ai_chance = {
			base = 90
		}
	}


	# =====================================================
	# ULTIMATUM
	#
	# Daemon did NOT fight for Aegon.
	# =====================================================

	option = {

		name = segrellion_aegon_daemon.0001.b

		custom_tooltip = segrellion_aegon_daemon.0001.b.tt


		trigger = {

			character:Segrellion_60 = {

				NOT = {
					has_character_flag = segrellion_pledged_to_young_griff
				}
			}
		}


		character:Segrellion_60 = {

			add_character_flag = segrellion_daemon_aegon_ultimatum_pending


			trigger_event = {

				id = segrellion_aegon_daemon.0003

				days = { 2 5 }
			}
		}


		ai_chance = {
			base = 80
		}
	}


	# =====================================================
	# LEAVE THE FLAMELANDS INDEPENDENT
	# =====================================================

	option = {

		name = segrellion_aegon_daemon.0001.c

		custom_tooltip = segrellion_aegon_daemon.0001.c.tt


		character:Segrellion_60 = {

			add_character_flag = segrellion_daemon_aegon_status_resolved

			add_character_flag = segrellion_daemon_independence_tolerated_by_aegon
		}


		add_character_flag = segrellion_young_griff_flamelands_submission_resolved


		ai_chance = {
			base = 20
		}
	}
}


# =========================================================
# 0002 - THE KING'S REQUEST
#
# Root = Daemon
# Peaceful request.
# =========================================================

segrellion_aegon_daemon.0002 = {

	type = letter_event

	sender = character:IAF_Young_Griff_1

	opening = segrellion_aegon_daemon.0002.opening

	title = segrellion_aegon_daemon.0002.t

	desc = segrellion_aegon_daemon.0002.desc

	theme = diplomacy


	trigger = {

		this = character:Segrellion_60

		is_alive = yes

		is_independent_ruler = yes

		has_title = title:e_flamelands

		has_character_flag = segrellion_daemon_aegon_peaceful_request_pending


		character:IAF_Young_Griff_1 ?= {

			is_alive = yes

			has_title = title:h_the_iron_throne
		}
	}


	on_trigger_fail = {

		remove_character_flag = segrellion_daemon_aegon_peaceful_request_pending
	}


	immediate = {

		remove_character_flag = segrellion_daemon_aegon_peaceful_request_pending
	}


	left_portrait = {

		character = root

		animation = thinking
	}


	right_portrait = {

		character = character:IAF_Young_Griff_1

		animation = personality_rational
	}


	# =====================================================
	# ACCEPT
	# =====================================================

	option = {

		name = segrellion_aegon_daemon.0002.a

		custom_tooltip = segrellion_aegon_daemon.0002.a.tt


		segrellion_daemon_swear_fealty_to_aegon_effect = yes


		add_character_flag = segrellion_daemon_aegon_fealty_accepted

		add_character_flag = segrellion_daemon_aegon_status_resolved


		character:IAF_Young_Griff_1 = {

			add_character_flag = segrellion_young_griff_flamelands_submission_resolved

			add_character_flag = segrellion_young_griff_flamelands_peacefully_submitted
		}


		ai_chance = {

			base = 50


			# Exact 75 / 25:
			#
			# Aegon helped Daemon.
			# Daemon helped Aegon.
			# Daemon likes Aegon.

			modifier = {

				add = 25


				has_character_flag = segrellion_pledged_to_young_griff


				character:IAF_Young_Griff_1 = {
					has_character_flag = segrellion_young_griff_supports_daemon
				}


				opinion = {

					target = character:IAF_Young_Griff_1

					value >= 25
				}
			}
		}
	}


	# =====================================================
	# REFUSE
	# =====================================================

	option = {

		name = segrellion_aegon_daemon.0002.b

		custom_tooltip = segrellion_aegon_daemon.0002.b.tt


		add_character_flag = segrellion_daemon_aegon_fealty_refused


		character:IAF_Young_Griff_1 = {

			trigger_event = {

				id = segrellion_aegon_daemon.0004

				days = { 2 4 }
			}
		}


		ai_chance = {

			base = 50


			modifier = {

				add = -25


				has_character_flag = segrellion_pledged_to_young_griff


				character:IAF_Young_Griff_1 = {
					has_character_flag = segrellion_young_griff_supports_daemon
				}


				opinion = {

					target = character:IAF_Young_Griff_1

					value >= 25
				}
			}
		}
	}
}
