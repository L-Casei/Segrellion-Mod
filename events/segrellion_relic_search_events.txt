segrellion_relic.1005 = {

	type = character_event

	immediate = {
		title:c_tyrosh.holder ?= {
			save_scope_as = relic_tyrosh_host
		}
	}

	title = segrellion_relic.1005.t

	desc = {
		first_valid = {

			triggered_desc = {
				trigger = {
					scope:relic_tyrosh_host = {
						opinion = {
							target = root
							value < 0
						}
					}
				}

				desc = segrellion_relic.1005.desc.hostile
			}

			triggered_desc = {
				trigger = {
					scope:relic_tyrosh_host = {
						opinion = {
							target = root
							value >= 75
						}
					}
				}

				desc = segrellion_relic.1005.desc_warm
			}

			desc = segrellion_relic.1005.desc_cautious
		}
	}

	theme = diplomacy

	override_background = {
		reference = throne_room
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:relic_tyrosh_host
		animation = personality_bold
	}


	# Hostile: opinion below 0.
	option = {

		trigger = {
			scope:relic_tyrosh_host = {
				opinion = {
					target = root
					value < 0
				}
			}
		}

		name = segrellion_relic.1005.a

		trigger_event = {
			id = segrellion_relic.1091
			
		}
	}


	# Neutral/cautious: 0-74.
	option = {

		trigger = {
			scope:relic_tyrosh_host = {
				opinion = {
					target = root
					value >= 0
				}

				opinion = {
					target = root
					value < 75
				}
			}
		}

		name = segrellion_relic.1005.b

		trigger_event = {
			id = segrellion_relic.1010
			days = { 4 7 }
		}
	}


	# Excellent relations: 75+.
	option = {

		trigger = {
			scope:relic_tyrosh_host = {
				opinion = {
					target = root
					value >= 75
				}
			}
		}

		name = segrellion_relic.1005.c

		trigger_event = {
			id = segrellion_relic.1050
			days = 2
		}
	}
}

segrellion_relic.6010 = {

	type = character_event

	title = segrellion_relic.6010.t
	desc = segrellion_relic.6010.desc

	theme = learning

	override_background = {
		reference = segrellion_beyond_the_wall_blizzard
	}

	left_portrait = {
		character = root
		animation = thinking
	}

	option = {

		name = segrellion_relic.6010.a

		hidden_effect = {

			if = {
				limit = {
					NOT = {
						has_global_variable = segrellion_dragonspear_black_gauntlet
					}
				}

				if = {
					limit = {
						dynasty = dynasty:dynn_Segrellion
					}

					random_list = {

						25 = {
							trigger_event = {
								id = segrellion_relic.6020
								
							}
						}

						75 = {
							trigger_event = {
								id = segrellion_relic.6030
								
							}
						}
					}
				}

				else = {

					random_list = {

						5 = {
							trigger_event = {
								id = segrellion_relic.6020
								
							}
						}

						95 = {
							trigger_event = {
								id = segrellion_relic.6030
								
							}
						}
					}
				}
			}

			else = {

				trigger_event = {
					id = segrellion_relic.6030
					
				}
			}
		}
	}
}

segrellion_relic.7000 = {

	type = character_event

	trigger = {
		has_character_flag = segrellion_legendary_relic_search_active
		has_character_flag = segrellion_relic_pending_skagos
	}

	title = segrellion_relic.7000.t
	desc = segrellion_relic.7000.desc

	theme = travel

	override_background = {
		reference = segrellion_barren_expanse
	}

	immediate = {

		remove_character_flag = segrellion_relic_pending_skagos
		remove_character_flag = segrellion_legendary_relic_search_paused
	}

	left_portrait = {
		character = root
		animation = worry
	}

	option = {

		name = segrellion_relic.7000.a

		custom_tooltip = {
			text = segrellion_relic.7000.a.chance_tt
		}

		trigger_event = {
			id = segrellion_relic.7010
			days = { 2 4 }
		}
	}
}

# 7030 - DRINK
# =========================================================

segrellion_relic.7030 = {

	type = character_event

	title = segrellion_relic.7030.t
	desc = segrellion_relic.7030.desc

	theme = faith

	override_background = {
		reference = segrellion_icebringer_fountain
	}

	left_portrait = {
		character = root
		animation = worry
	}

	option = {

		name = segrellion_relic.7030.a

		custom_tooltip = segrellion_relic.7030.a.tt

		hidden_effect = {

			remove_character_flag = segrellion_relic_skagos_success_roll

			random = {

				chance = segrellion_relic_skagos_legendary_chance

				add_character_flag = segrellion_relic_skagos_success_roll
			}

			if = {
				limit = {
					has_character_flag = segrellion_relic_skagos_success_roll

					NOT = {
						has_global_variable = segrellion_icebringer
					}
				}

				remove_character_flag = segrellion_relic_skagos_success_roll

				trigger_event = {
					id = segrellion_relic.7100
					
				}
			}

			else = {

				remove_character_flag = segrellion_relic_skagos_success_roll

				trigger_event = {
					id = segrellion_relic.7200
					
				}
			}
		}
	}
}
