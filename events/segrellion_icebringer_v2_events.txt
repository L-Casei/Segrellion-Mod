# =========================================================
# 6300 - PROCESSION DISPATCHER
#
# Original pool:
#   Prophet / Lava    45
#   Intruder / Mob    45
#   Library           10
#
# Unique results disappear after being seen.
#
# When both normal failures have been seen:
#   Repeating Loop    90
#   Library           10
#
# Once Library has also been discovered:
# the reward cannot be obtained again.
# =========================================================

segrellion_icebringer.6300 = {
	type = character_event
	hidden = yes

	immediate = {

		# =================================================
		# LIBRARY ALREADY DISCOVERED
		# =================================================

		if = {
			limit = {
				has_character_flag = segrellion_doom_seen_library
			}

			if = {
				limit = {
					NOT = {
						has_character_flag = segrellion_doom_seen_procession_prophet
					}

					NOT = {
						has_character_flag = segrellion_doom_seen_procession_intruder
					}
				}

				random_list = {
					50 = {
						trigger_event = segrellion_icebringer.6301
					}

					50 = {
						trigger_event = segrellion_icebringer.6302
					}
				}
			}

			else_if = {
				limit = {
					has_character_flag = segrellion_doom_seen_procession_prophet

					NOT = {
						has_character_flag = segrellion_doom_seen_procession_intruder
					}
				}

				trigger_event = segrellion_icebringer.6302
			}

			else_if = {
				limit = {
					NOT = {
						has_character_flag = segrellion_doom_seen_procession_prophet
					}

					has_character_flag = segrellion_doom_seen_procession_intruder
				}

				trigger_event = segrellion_icebringer.6301
			}

			else = {
				trigger_event = segrellion_icebringer.6305
			}
		}


		# =================================================
		# LIBRARY NOT YET DISCOVERED
		# =================================================

		else = {

			# =============================================
			# NOTHING SEEN
			# =============================================

			if = {
				limit = {
					NOT = {
						has_character_flag = segrellion_doom_seen_procession_prophet
					}

					NOT = {
						has_character_flag = segrellion_doom_seen_procession_intruder
					}
				}

				random_list = {
					45 = {
						trigger_event = segrellion_icebringer.6301
					}

					45 = {
						trigger_event = segrellion_icebringer.6302
					}

					10 = {
						trigger_event = segrellion_icebringer.6303
					}
				}
			}


			# =============================================
			# PROPHET ALREADY SEEN
			# =============================================

			else_if = {
				limit = {
					has_character_flag = segrellion_doom_seen_procession_prophet

					NOT = {
						has_character_flag = segrellion_doom_seen_procession_intruder
					}
				}

				random_list = {
					45 = {
						trigger_event = segrellion_icebringer.6302
					}

					10 = {
						trigger_event = segrellion_icebringer.6303
					}
				}
			}


			# =============================================
			# INTRUDER ALREADY SEEN
			# =============================================

			else_if = {
				limit = {
					NOT = {
						has_character_flag = segrellion_doom_seen_procession_prophet
					}

					has_character_flag = segrellion_doom_seen_procession_intruder
				}

				random_list = {
					45 = {
						trigger_event = segrellion_icebringer.6301
					}

					10 = {
						trigger_event = segrellion_icebringer.6303
					}
				}
			}


			# =============================================
			# BOTH FALSE UNIQUE RESULTS SEEN
			#
			# Library stays rare instead of becoming 100%.
			# =============================================

			else = {

				random_list = {
					90 = {
						trigger_event = segrellion_icebringer.6305
					}

					10 = {
						trigger_event = segrellion_icebringer.6303
					}
				}
			}
		}
	}
}

# 6570 - HEART DISPATCHER
# =========================================================

	type = character_event
	hidden = yes

	immediate = {

		random_list = {

			50 = {
				trigger_event = segrellion_icebringer.6571
			}

			50 = {

				if = {
					limit = {
						has_character_flag = segrellion_stage3_heart_favor_earned
					}

					trigger_event = segrellion_icebringer.6595
				}

				else = {
					trigger_event = segrellion_icebringer.6580
				}
			}
		}
	}
}

# =========================================================
# 6580 - THE JUDGMENT
# =========================================================

segrellion_icebringer.6580 = {
	type = character_event

	title = segrellion_icebringer.6580.t
	desc = segrellion_icebringer.6580.desc

	override_background = {
		reference = dungeon
	}

	theme = intrigue

	left_portrait = {
		character = root
		animation = worry
	}

	immediate = {

		# Every new Judgment trial begins at zero.
		set_variable = {
			name = segrellion_heart_refusal_count
			value = 0
		}
	}

	# Kill the starving thief.
	option = {
		name = segrellion_icebringer.6580.a

		remove_variable = segrellion_heart_refusal_count

		trigger_event = segrellion_icebringer.6592
	}


	# Kill the alleged rebel.
	option = {
		name = segrellion_icebringer.6580.b

		remove_variable = segrellion_heart_refusal_count

		trigger_event = segrellion_icebringer.6593
	}


	# Sadistic only.
	option = {
		name = segrellion_icebringer.6580.c

		trigger = {
			has_trait = sadistic
		}

		add_internal_flag = special

		remove_variable = segrellion_heart_refusal_count

		trigger_event = segrellion_icebringer.6594
	}


	# TRUE HEART PATH.
	option = {
		name = segrellion_icebringer.6580.d

		change_variable = {
			name = segrellion_heart_refusal_count
			add = 1
		}

		trigger_event = segrellion_icebringer.6581
	}
}


# =========================================================
# 6581 - CHOOSE
#
# Repeats until the player has refused TEN times.
#
# Initial refusal in 6580 = refusal #1.
# This event therefore needs nine additional refusals.
# =========================================================

segrellion_icebringer.6581 = {
	type = character_event

	title = segrellion_icebringer.6581.t
	desc = segrellion_icebringer.6581.desc

	override_background = {
		reference = dungeon
	}

	theme = intrigue

	left_portrait = {
		character = root
		animation = worry
	}


	option = {
		name = segrellion_icebringer.6581.a

		remove_variable = segrellion_heart_refusal_count

		trigger_event = segrellion_icebringer.6592
	}


	option = {
		name = segrellion_icebringer.6581.b

		remove_variable = segrellion_heart_refusal_count

		trigger_event = segrellion_icebringer.6593
	}


	option = {
		name = segrellion_icebringer.6581.c

		trigger = {
			has_trait = sadistic
		}

		add_internal_flag = special

		remove_variable = segrellion_heart_refusal_count

		trigger_event = segrellion_icebringer.6594
	}


	option = {
		name = segrellion_icebringer.6581.d

		change_variable = {
			name = segrellion_heart_refusal_count
			add = 1
		}


		if = {
			limit = {
				var:segrellion_heart_refusal_count >= 10
			}

			remove_variable = segrellion_heart_refusal_count

			trigger_event = segrellion_icebringer.6591
		}


		else = {
			trigger_event = {
				id = segrellion_icebringer.6581
				days = 1
			}
		}
	}
}
