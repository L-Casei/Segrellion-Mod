# =========================================================
# 6100 - GIRL DISPATCHER
#
# FIRST VISIT:
#   Father          59
#   Other Self      40
#   True Path        1 + character modifiers
#
# SEEN RESULTS ARE REMOVED.
#
# Once both false unique results have been seen:
#   Loop            99
#   True Path        1 + character modifiers
#
# This prevents content exhaustion from making success 100%.
# =========================================================

segrellion_icebringer.6100 = {
	type = character_event
	hidden = yes

	immediate = {

		# =================================================
		# NEITHER FALSE RESULT HAS BEEN SEEN
		# =================================================

		if = {
			limit = {
				NOT = {
					has_character_flag = segrellion_doom_seen_girl_father
				}

				NOT = {
					has_character_flag = segrellion_doom_seen_girl_other_self
				}
			}

			random_list = {

				59 = {
					trigger_event = segrellion_icebringer.6101
				}

				40 = {
					trigger_event = segrellion_icebringer.6102
				}

				# =========================================
				# TRUE PATH
				# =========================================

				1 = {

					# -------------------------------------
					# ERA
					# -------------------------------------

					modifier = {
						factor = 0.10
						current_date < 8200.1.1
					}

					modifier = {
						factor = 0.25
						current_date >= 8200.1.1
						current_date < 8250.1.1
					}

					modifier = {
						factor = 0.50
						current_date >= 8250.1.1
						current_date < 8280.1.1
					}

					# 280-294 AC = base factor 1.

					modifier = {
						factor = 2.00
						current_date >= 8295.1.1
					}


					# -------------------------------------
					# PERSONALITY
					# -------------------------------------

					modifier = {
						factor = 1.25
						has_trait = brave
					}

					modifier = {
						factor = 1.35
						has_trait = honest
					}

					modifier = {
						factor = 1.20
						has_trait = just
					}

					modifier = {
						factor = 1.10
						has_trait = stubborn
					}


					# -------------------------------------
					# SUPERNATURAL SENSITIVITY
					# -------------------------------------

					modifier = {
						factor = 1.50
						has_trait = dragon_dreams
					}


					# -------------------------------------
					# THE CANON GUARDIAN
					#
					# Strongly favored, but NOT automatic.
					# -------------------------------------

					modifier = {
						factor = 10
						has_trait = segrellion_canon_daemon_bastard_unicorn_modifier
					}

					trigger_event = segrellion_icebringer.6103
				}
			}
		}


		# =================================================
		# FATHER ALREADY SEEN
		# =================================================

		else_if = {
			limit = {
				has_character_flag = segrellion_doom_seen_girl_father

				NOT = {
					has_character_flag = segrellion_doom_seen_girl_other_self
				}
			}

			random_list = {

				40 = {
					trigger_event = segrellion_icebringer.6102
				}

				1 = {

					modifier = {
						factor = 0.10
						current_date < 8200.1.1
					}

					modifier = {
						factor = 0.25
						current_date >= 8200.1.1
						current_date < 8250.1.1
					}

					modifier = {
						factor = 0.50
						current_date >= 8250.1.1
						current_date < 8280.1.1
					}

					modifier = {
						factor = 2.00
						current_date >= 8295.1.1
					}

					modifier = {
						factor = 1.25
						has_trait = brave
					}

					modifier = {
						factor = 1.35
						has_trait = honest
					}

					modifier = {
						factor = 1.20
						has_trait = just
					}

					modifier = {
						factor = 1.10
						has_trait = stubborn
					}

					modifier = {
						factor = 1.50
						has_trait = dragon_dreams
					}

					modifier = {
						factor = 10
						has_trait = segrellion_canon_daemon_bastard_unicorn_modifier
					}

					trigger_event = segrellion_icebringer.6103
				}
			}
		}


		# =================================================
		# OTHER SELF ALREADY SEEN
		# =================================================

		else_if = {
			limit = {
				NOT = {
					has_character_flag = segrellion_doom_seen_girl_father
				}

				has_character_flag = segrellion_doom_seen_girl_other_self
			}

			random_list = {

				59 = {
					trigger_event = segrellion_icebringer.6101
				}

				1 = {

					modifier = {
						factor = 0.10
						current_date < 8200.1.1
					}

					modifier = {
						factor = 0.25
						current_date >= 8200.1.1
						current_date < 8250.1.1
					}

					modifier = {
						factor = 0.50
						current_date >= 8250.1.1
						current_date < 8280.1.1
					}

					modifier = {
						factor = 2.00
						current_date >= 8295.1.1
					}

					modifier = {
						factor = 1.25
						has_trait = brave
					}

					modifier = {
						factor = 1.35
						has_trait = honest
					}

					modifier = {
						factor = 1.20
						has_trait = just
					}

					modifier = {
						factor = 1.10
						has_trait = stubborn
					}

					modifier = {
						factor = 1.50
						has_trait = dragon_dreams
					}

					modifier = {
						factor = 10
						has_trait = segrellion_canon_daemon_bastard_unicorn_modifier
					}

					trigger_event = segrellion_icebringer.6103
				}
			}
		}


		# =================================================
		# BOTH UNIQUE FALSE OUTCOMES HAVE BEEN SEEN
		#
		# The dream begins looping.
		# =================================================

		else = {

			random_list = {

				99 = {
					trigger_event = segrellion_icebringer.6104
				}

				1 = {

					modifier = {
						factor = 0.10
						current_date < 8200.1.1
					}

					modifier = {
						factor = 0.25
						current_date >= 8200.1.1
						current_date < 8250.1.1
					}

					modifier = {
						factor = 0.50
						current_date >= 8250.1.1
						current_date < 8280.1.1
					}

					modifier = {
						factor = 2.00
						current_date >= 8295.1.1
					}

					modifier = {
						factor = 1.25
						has_trait = brave
					}

					modifier = {
						factor = 1.35
						has_trait = honest
					}

					modifier = {
						factor = 1.20
						has_trait = just
					}

					modifier = {
						factor = 1.10
						has_trait = stubborn
					}

					modifier = {
						factor = 1.50
						has_trait = dragon_dreams
					}

					modifier = {
						factor = 10
						has_trait = segrellion_canon_daemon_bastard_unicorn_modifier
					}

					trigger_event = segrellion_icebringer.6103
				}
			}
		}
	}
}

# =========================================================
# THE RED PROCESSION
# =========================================================


# =========================================================
# 6300 - PROCESSION DISPATCHER
#
# Original pool:
#   Prophet / Lava    45
#   Intruder / Mob    45
#   Library           10
#
# Unique results disappear after being seen.
#
# When both normal failures have been seen:
#   Repeating Loop    90
#   Library           10
#
# Once Library has also been discovered:
# the reward cannot be obtained again.
# =========================================================

segrellion_icebringer.6300 = {
	type = character_event
	hidden = yes

	immediate = {

		# =================================================
		# LIBRARY ALREADY DISCOVERED
		# =================================================

		if = {
			limit = {
				has_character_flag = segrellion_doom_seen_library
			}

			if = {
				limit = {
					NOT = {
						has_character_flag = segrellion_doom_seen_procession_prophet
					}

					NOT = {
						has_character_flag = segrellion_doom_seen_procession_intruder
					}
				}

				random_list = {
					50 = {
						trigger_event = segrellion_icebringer.6301
					}

					50 = {
						trigger_event = segrellion_icebringer.6302
					}
				}
			}

			else_if = {
				limit = {
					has_character_flag = segrellion_doom_seen_procession_prophet

					NOT = {
						has_character_flag = segrellion_doom_seen_procession_intruder
					}
				}

				trigger_event = segrellion_icebringer.6302
			}

			else_if = {
				limit = {
					NOT = {
						has_character_flag = segrellion_doom_seen_procession_prophet
					}

					has_character_flag = segrellion_doom_seen_procession_intruder
				}

				trigger_event = segrellion_icebringer.6301
			}

			else = {
				trigger_event = segrellion_icebringer.6305
			}
		}


		# =================================================
		# LIBRARY NOT YET DISCOVERED
		# =================================================

		else = {

			# =============================================
			# NOTHING SEEN
			# =============================================

			if = {
				limit = {
					NOT = {
						has_character_flag = segrellion_doom_seen_procession_prophet
					}

					NOT = {
						has_character_flag = segrellion_doom_seen_procession_intruder
					}
				}

				random_list = {
					45 = {
						trigger_event = segrellion_icebringer.6301
					}

					45 = {
						trigger_event = segrellion_icebringer.6302
					}

					10 = {
						trigger_event = segrellion_icebringer.6303
					}
				}
			}


			# =============================================
			# PROPHET ALREADY SEEN
			# =============================================

			else_if = {
				limit = {
					has_character_flag = segrellion_doom_seen_procession_prophet

					NOT = {
						has_character_flag = segrellion_doom_seen_procession_intruder
					}
				}

				random_list = {
					45 = {
						trigger_event = segrellion_icebringer.6302
					}

					10 = {
						trigger_event = segrellion_icebringer.6303
					}
				}
			}


			# =============================================
			# INTRUDER ALREADY SEEN
			# =============================================

			else_if = {
				limit = {
					NOT = {
						has_character_flag = segrellion_doom_seen_procession_prophet
					}

					has_character_flag = segrellion_doom_seen_procession_intruder
				}

				random_list = {
					45 = {
						trigger_event = segrellion_icebringer.6301
					}

					10 = {
						trigger_event = segrellion_icebringer.6303
					}
				}
			}


			# =============================================
			# BOTH FALSE UNIQUE RESULTS SEEN
			#
			# Library stays rare instead of becoming 100%.
			# =============================================

			else = {

				random_list = {
					90 = {
						trigger_event = segrellion_icebringer.6305
					}

					10 = {
						trigger_event = segrellion_icebringer.6303
					}
				}
			}
		}
	}
}

#
# Fixed weights always preserved:
# Light vision 35
# Torture      60
# Madness       5
#
# If a result has already been seen, its weight is replaced
# by a loop variant, so the RNG does not become easier.

	type = character_event
	hidden = yes

	immediate = {
		random_list = {

			35 = {
				if = {
					limit = {
						has_character_flag = segrellion_island_seen_tower_light
					}

					trigger_event = segrellion_icebringer.6417
				}
				else = {
					if = {
						limit = {
							segrellion_icebringer_destined_guardian_trigger = yes
						}

						trigger_event = segrellion_icebringer.6413
					}
					else = {
						trigger_event = segrellion_icebringer.6414
					}
				}
			}

			60 = {
				if = {
					limit = {
						has_character_flag = segrellion_island_seen_tower_torture
					}

					trigger_event = segrellion_icebringer.6418
				}
				else = {
					trigger_event = segrellion_icebringer.6415
				}
			}

			5 = {
				if = {
					limit = {
						has_character_flag = segrellion_island_seen_tower_madness
					}

					trigger_event = segrellion_icebringer.6419
				}
				else = {
					trigger_event = segrellion_icebringer.6416
				}
			}
		}
	}
}

# =========================================================
# 6423 - RESIST DISPATCHER
#
# Visible correct choice, but two hidden outcomes:
#
# Partial success 95
# True path       1 + modifiers
#
# If partial success has already been seen, it becomes a
# loop event with the same fixed weight, so true path does
# not inflate automatically.
# =========================================================

segrellion_icebringer.6423 = {
	type = character_event
	hidden = yes

	immediate = {
		random_list = {

			95 = {
				if = {
					limit = {
						has_character_flag = segrellion_island_seen_resisted_temptation
					}

					trigger_event = segrellion_icebringer.6427
				}
				else = {
					trigger_event = segrellion_icebringer.6424
				}
			}

			1 = {

				# -----------------------------------------
				# ERA
				# -----------------------------------------

				modifier = {
					factor = 0.10
					current_date < 8200.1.1
				}

				modifier = {
					factor = 0.25
					current_date >= 8200.1.1
					current_date < 8250.1.1
				}

				modifier = {
					factor = 0.50
					current_date >= 8250.1.1
					current_date < 8280.1.1
				}

				modifier = {
					factor = 2.00
					current_date >= 8295.1.1
				}

				# -----------------------------------------
				# PERSONALITY / PURPOSE
				# -----------------------------------------

				modifier = {
					factor = 1.35
					has_trait = honest
				}

				modifier = {
					factor = 1.25
					has_trait = just
				}

				modifier = {
					factor = 1.15
					has_trait = temperate
				}

				modifier = {
					factor = 1.20
					has_trait = chaste
				}

				modifier = {
					factor = 1.25
					has_trait = brave
				}

				modifier = {
					factor = 1.50
					has_trait = dragon_dreams
				}

				# -----------------------------------------
				# DESTINED ONES
				# -----------------------------------------

				modifier = {
					factor = 10
					segrellion_icebringer_destined_guardian_trigger = yes
				}

				trigger_event = segrellion_icebringer.6425
			}
		}
	}
}
