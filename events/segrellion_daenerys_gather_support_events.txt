# 0200 - TYRION: THE REACH
# =========================================================

segrellion_daenerys_support.0200 = {

	type = character_event

	title = segrellion_daenerys_support.0200.t
	desc = segrellion_daenerys_support.0200.desc

	theme = diplomacy

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	override_background = {
		reference = council_chamber
	}


	immediate = {

		# =================================================
		# FIND OLENNA THROUGH THE CURRENT REACH FAMILY
		# =================================================

		if = {

			limit = {

				title:e_the_reach.holder ?= {

					mother ?= {
						is_alive = yes
					}
				}
			}


			title:e_the_reach.holder.mother = {
				save_scope_as = segrellion_daenerys_support_olenna
			}
		}


		else_if = {

			limit = {

				title:e_the_reach.holder ?= {

					father ?= {

						mother ?= {
							is_alive = yes
						}
					}
				}
			}


			title:e_the_reach.holder.father.mother = {
				save_scope_as = segrellion_daenerys_support_olenna
			}
		}
	}


	left_portrait = {
		character = root
		animation = thinking
	}


	right_portrait = {
		character = character:Lannister_8
		animation = personality_rational
	}


	option = {

		name = segrellion_daenerys_support.0200.a


		if = {

			limit = {
				exists = scope:segrellion_daenerys_support_olenna
			}


			scope:segrellion_daenerys_support_olenna = {

				trigger_event = {

					id = segrellion_daenerys_support.0210

					days = { 3 7 }
				}
			}
		}


		else = {

			title:e_the_reach.holder = {

				add_character_flag = segrellion_reach_direct_contact_for_daenerys


				trigger_event = {

					id = segrellion_daenerys_support.0210

					days = { 3 7 }
				}
			}
		}
	}
}

	type = letter_event

	theme = diplomacy


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	sender = character:Targaryen_9

	title = segrellion_daenerys_support.0410.t
	desc = segrellion_daenerys_support.0410.desc


	# =====================================================
	# SUPPORT DAENERYS
	# =====================================================

	option = {

		name = segrellion_daenerys_support.0410.a

		custom_tooltip = segrellion_daenerys_support.0410.a.tt


		add_character_flag = segrellion_dorne_secretly_supports_daenerys

		remove_character_flag = segrellion_dorne_supports_young_griff


		character:Targaryen_9 = {

			add_character_flag = segrellion_daenerys_secret_dornish_support_against_aegon

			add_character_flag = segrellion_daenerys_dorne_answer_yes


			trigger_event = {

				id = segrellion_daenerys_support.0420

				days = 2
			}
		}


		ai_chance = {

			base = 45


			# =================================================
			# Aegon revealed as Blackfyre
			# =================================================

			modifier = {

				add = 90

				character:IAF_Young_Griff_1 = {
					has_character_flag = segrellion_young_griff_blackfyre_revealed
				}
			}


			# =================================================
			# Aegon already politically lost Dorne
			# =================================================

			modifier = {

				add = 80

				character:IAF_Young_Griff_1 = {
					has_character_flag = segrellion_young_griff_lost_dorne_over_margaery
				}
			}


			# =================================================
			# Internal history state.
			#
			# We still use this mechanically, but do not
			# expose Arianne-specific alternate-history
			# warnings to Daenerys.
			# =================================================

			modifier = {

				add = 50

				character:Martell_9 ?= {
					has_character_flag = segrellion_arianne_rejects_margaery_second_wife
				}
			}


			# =================================================
			# Aegon is not the ruling king
			# =================================================

			modifier = {

				add = 40

				title:h_the_iron_throne.holder != character:IAF_Young_Griff_1
			}


			# =================================================
			# CURRENT DORNISH RULER DISLIKES AEGON
			# =================================================

			modifier = {

				add = 30

				opinion = {
					target = character:IAF_Young_Griff_1
					value <= -25
				}
			}


			modifier = {

				add = 40

				opinion = {
					target = character:IAF_Young_Griff_1
					value <= -50
				}
			}


			# =================================================
			# CURRENT DORNISH RULER LIKES DAENERYS
			# =================================================

			modifier = {

				add = 25

				opinion = {
					target = character:Targaryen_9
					value >= 25
				}
			}


			modifier = {

				add = 35

				opinion = {
					target = character:Targaryen_9
					value >= 50
				}
			}
		}
	}


	# =====================================================
	# REFUSE DAENERYS
	#
	# Refusal no longer automatically means "Dorne belongs
	# to Aegon". It simply means Sunspear declines her offer.
	# =====================================================

	option = {

		name = segrellion_daenerys_support.0410.b

		custom_tooltip = segrellion_daenerys_support.0410.b.tt


		character:Targaryen_9 = {

			add_character_flag = segrellion_daenerys_dorne_answer_no


			trigger_event = {

				id = segrellion_daenerys_support.0420

				days = 2
			}
		}


		ai_chance = {

			base = 55


			modifier = {

				add = 30

				opinion = {
					target = character:IAF_Young_Griff_1
					value >= 25
				}
			}


			modifier = {

				add = 50

				opinion = {
					target = character:IAF_Young_Griff_1
					value >= 50
				}
			}


			modifier = {

				add = 25

				opinion = {
					target = character:Targaryen_9
					value <= -25
				}
			}
		}
	}
}
