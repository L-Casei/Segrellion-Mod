segrellion_daenerys_support.0200 = {

	type = character_event

	title = segrellion_daenerys_support.0200.t
	desc = segrellion_daenerys_support.0200.desc

	theme = diplomacy


	immediate = {

		# =================================================
		# FIND OLENNA THROUGH THE CURRENT REACH FAMILY
		# =================================================

		if = {

			limit = {

				title:k_the_reach.holder ?= {

					mother ?= {
						is_alive = yes
					}
				}
			}


			title:k_the_reach.holder.mother = {
				save_scope_as = segrellion_daenerys_support_olenna
			}
		}


		else_if = {

			limit = {

				title:k_the_reach.holder ?= {

					father ?= {

						mother ?= {
							is_alive = yes
						}
					}
				}
			}


			title:k_the_reach.holder.father.mother = {
				save_scope_as = segrellion_daenerys_support_olenna
			}
		}
	}


	left_portrait = {
		character = root
		animation = thinking
	}


	right_portrait = {
		character = character:Lannister_8
		animation = personality_rational
	}


	option = {

		name = segrellion_daenerys_support.0200.a


		if = {

			limit = {
				exists = scope:segrellion_daenerys_support_olenna
			}


			scope:segrellion_daenerys_support_olenna = {

				trigger_event = {

					id = segrellion_daenerys_support.0210

					days = { 3 7 }
				}
			}
		}


		else = {

			title:k_the_reach.holder = {

				add_character_flag = segrellion_reach_direct_contact_for_daenerys


				trigger_event = {

					id = segrellion_daenerys_support.0210

					days = { 3 7 }
				}
			}
		}
	}
}

	type = letter_event

	theme = diplomacy

	sender = character:Targaryen_9

	desc = segrellion_daenerys_support.0410.desc

	# SUPPORT DAENERYS
	#
	# HARD BLOCK:
	# - Arianne currently married to Aegon
	# - OR Arianne has a living child by Aegon
	# =====================================================

	option = {

		name = segrellion_daenerys_support.0410.a


		trigger = {

			NOT = {

				character:IAF_Young_Griff_1 ?= {

					any_spouse = {
						this = character:Martell_9
					}
				}
			}


			NOT = {

				character:Martell_9 ?= {

					any_child = {

						is_alive = yes

						father = character:IAF_Young_Griff_1
					}
				}
			}
		}


		add_character_flag = segrellion_dorne_secretly_supports_daenerys

		remove_character_flag = segrellion_dorne_supports_young_griff


		character:Targaryen_9 = {

			add_character_flag = segrellion_daenerys_secret_dornish_support_against_aegon

			add_character_flag = segrellion_daenerys_dorne_answer_yes


			trigger_event = {

				id = segrellion_daenerys_support.0420

				days = 2
			}
		}


		ai_chance = {

			base = 45


			modifier = {

				add = 90

				character:IAF_Young_Griff_1 = {
					has_character_flag = segrellion_young_griff_blackfyre_revealed
				}
			}


			modifier = {

				add = 80

				character:IAF_Young_Griff_1 = {
					has_character_flag = segrellion_young_griff_lost_dorne_over_margaery
				}
			}


			modifier = {

				add = 50

				character:Martell_9 ?= {
					has_character_flag = segrellion_arianne_rejects_margaery_second_wife
				}
			}


			modifier = {

				add = 40

				title:h_the_iron_throne.holder != character:IAF_Young_Griff_1
			}
		}
	}


	# =====================================================
	# REFUSE
	# =====================================================

	option = {

		name = segrellion_daenerys_support.0410.b


		if = {

			limit = {

				OR = {

					character:IAF_Young_Griff_1 ?= {

						any_spouse = {
							this = character:Martell_9
						}
					}


					character:Martell_9 ?= {

						any_child = {

							is_alive = yes

							father = character:IAF_Young_Griff_1
						}
					}
				}
			}


			add_character_flag = segrellion_dorne_supports_young_griff
		}


		character:Targaryen_9 = {

			add_character_flag = segrellion_daenerys_dorne_answer_no


			trigger_event = {

				id = segrellion_daenerys_support.0420

				days = 2
			}
		}


		ai_chance = {

			base = 55


			modifier = {

				add = 1000

				OR = {

					character:IAF_Young_Griff_1 ?= {

						any_spouse = {
							this = character:Martell_9
						}
					}


					character:Martell_9 ?= {

							is_alive = yes
