	type = character_event
	title = segrellion_sentinel.0001.t
	desc = segrellion_sentinel.0001.desc

		character = scope:tristan

	}

		name = segrellion_sentinel.0001.a
		custom_tooltip = segrellion_sentinel.0001.a.best_friend_tt

		if = {

				target = scope:tristan
				reason = best_friend_childhood_oath
				copy_reason = friend
			}

		scope:tristan = {
			add_trait = refusing_marriage

		name = segrellion_sentinel.0001.b
		trigger = {
			has_dlc_feature = roads_to_power
			has_multiple_players = no
		}
		custom_tooltip = segrellion_sentinel.0001.b.tt

		hidden_effect = {
			scope:tristan = {
				add_character_flag = vladwyn_landless_route_unlocked
				create_landless_adventurer_title_effect = {

					FLAVOR_CHAR = root
				}
			}
		}
		add_internal_flag = special
		set_player_character = scope:tristan
		scope:tristan = {
			create_landless_adventurer_title_tooltip_effect = yes
		}
	}
}
