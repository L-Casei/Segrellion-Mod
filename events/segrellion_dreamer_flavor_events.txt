# =========================================================
# 0101 - THE STRANGER'S EYES
# =========================================================

segrellion_dreamer.0101 = {

	type = character_event
	
	title = segrellion_dreamer.0101.t
	desc = segrellion_dreamer.0101.desc

	theme = intrigue

	override_background = { reference = segrellion_dreamer_tier_2_event_background }

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}



	left_portrait = {
		character = root
		animation = shock
	}

	immediate = {

		segrellion_gain_dreamer_xp_effect = yes


		# Select a REAL living human character.
		# Their identity is deliberately hidden from the player.

		random_living_character = {

			limit = {

				is_human = yes
				is_adult = yes

				NOT = {
					this = root
				}
			}

			save_scope_as = dream_stranger
		}


		# Save the actual person for a future callback.
		if = {
			limit = {
				exists = scope:dream_stranger
			}

			set_variable = {
				name = segrellion_dreamer_stranger_identity
				value = scope:dream_stranger
			}
		}


		# Rarely this vision is unusually clear.
		# Total: 2 XP instead of 1.

		random = {
			chance = 10

			segrellion_gain_dreamer_xp_effect = yes
		}
	}

	option = {

		name = segrellion_dreamer.0101.a

		custom_tooltip = segrellion_dreamer_gain_xp_tt

		add_stress = 5

		add_character_modifier = {
			modifier = segrellion_dreamer_borrowed_eyes_modifier
			days = 30
		}
	}
}

# =========================================================
# 0105 - THE RAVEN WITHOUT EYES
# =========================================================

segrellion_dreamer.0105 = {

	type = character_event
	override_background = { reference = segrellion_dreamer_tier_2_event_background }

	title = segrellion_dreamer.0105.t

	desc = {

		first_valid = {

			triggered_desc = {

				trigger = {
					exists = scope:dream_voice
				}

				desc = segrellion_dreamer.0105.desc_known
			}

			desc = segrellion_dreamer.0105.desc
		}
	}

	theme = intrigue

	left_portrait = {
		character = root
		animation = worry
	}

	immediate = {

		segrellion_gain_dreamer_xp_effect = yes


		# First preference:
		# someone personally important to the Dreamer.

		if = {
			limit = {

				any_relation = {

					type = friend
					type = best_friend
					type = lover
					type = soulmate
					type = rival
					type = nemesis

					is_alive = yes
				}
			}

			random_relation = {

				type = friend
				type = best_friend
				type = lover
				type = soulmate
				type = rival
				type = nemesis

				limit = {
					is_alive = yes
				}

				save_scope_as = dream_voice
			}
		}


		# Otherwise use living family.

		else_if = {
			limit = {

				any_close_or_extended_family_member = {

					is_alive = yes

					NOT = {
						this = root
					}
				}
			}

			random_close_or_extended_family_member = {

				limit = {

					is_alive = yes

					NOT = {
						this = root
					}
				}

				save_scope_as = dream_voice
			}
		}


		if = {
			limit = {
				exists = scope:dream_voice
			}

			set_variable = {
				name = segrellion_dreamer_raven_voice
				value = scope:dream_voice
			}
		}
	}

	option = {

		name = segrellion_dreamer.0105.a

		custom_tooltip = segrellion_dreamer_gain_xp_tt

		add_stress = 5
	}
}

		random_living_character = {

			limit = {

				is_human = yes
				is_adult = yes

				this != root

				has_trait = segrellion_dreamer_trait

				has_trait_xp = {
					trait = segrellion_dreamer_trait
					value >= 25
				}
			}

		name = segrellion_dreamer.4002.a

		add_character_modifier = {
			modifier = segrellion_dreamer_shared_dream_modifier
			years = 5
		}


		if = {

			limit = {
				exists = scope:other_dreamer
			}

			scope:other_dreamer = {

				trigger_event = {
					id = segrellion_dreamer.4102
					days = { 1 3 }
				}
			}
		}
	}

	option = {

		name = segrellion_dreamer.4002.b

		custom_tooltip = segrellion_dreamer.4002.b.tt

# 4114 - THE FUTURE REMEMBERS
# =========================================================

segrellion_dreamer.4114 = {

	type = character_event

	override_background = {
		reference = segrellion_dreamer_tier_4_event_background
	}

	title = segrellion_dreamer.4114.t

	desc = {

		first_valid = {

			triggered_desc = {

				trigger = {
					has_character_flag = segrellion_dreamer_chose_future_war
				}

				desc = segrellion_dreamer.4114.desc_war
			}

			triggered_desc = {

				trigger = {
					has_character_flag = segrellion_dreamer_chose_future_crown
				}

				desc = segrellion_dreamer.4114.desc_crown
			}

			triggered_desc = {

				trigger = {
					has_character_flag = segrellion_dreamer_chose_future_ashes
				}

				desc = segrellion_dreamer.4114.desc_ashes
			}

			desc = segrellion_dreamer.4114.desc
		}
	}

	theme = learning

	left_portrait = {
		character = root
		animation = worry
	}

	option = {

		name = segrellion_dreamer.4114.a


		if = {

			limit = {
				has_character_flag = segrellion_dreamer_chose_future_war
			}

			add_character_modifier = {
				modifier = segrellion_dreamer_future_war_modifier
				years = 10
			}
		}

		else_if = {

			limit = {
				has_character_flag = segrellion_dreamer_chose_future_crown
			}

			add_character_modifier = {
				modifier = segrellion_dreamer_future_crown_modifier
				years = 10
			}
		}

		else_if = {

			limit = {
				has_character_flag = segrellion_dreamer_chose_future_ashes
			}

			add_character_modifier = {
				modifier = segrellion_dreamer_future_ashes_modifier
				years = 10
			}
		}


		remove_character_flag = segrellion_dreamer_chose_future_war
		remove_character_flag = segrellion_dreamer_chose_future_crown
		remove_character_flag = segrellion_dreamer_chose_future_ashes
	}
}
