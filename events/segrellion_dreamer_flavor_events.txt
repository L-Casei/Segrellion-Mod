# =========================================================
# 0101 - THE STRANGER'S EYES
# =========================================================

segrellion_dreamer.0101 = {

	type = character_event
	override_background = { reference = segrellion_dreamer_event_background }

	title = segrellion_dreamer.0101.t
	desc = segrellion_dreamer.0101.desc

	theme = intrigue

	left_portrait = {
		character = root
		animation = shock
	}

	immediate = {

		segrellion_gain_dreamer_xp_effect = yes


		# Select a REAL living human character.
		# Their identity is deliberately hidden from the player.

		random_living_character = {

			limit = {

				is_human = yes
				is_adult = yes

				NOT = {
					this = root
				}
			}

			save_scope_as = dream_stranger
		}


		# Save the actual person for a future callback.
		if = {
			limit = {
				exists = scope:dream_stranger
			}

			set_variable = {
				name = segrellion_dreamer_stranger_identity
				value = scope:dream_stranger
			}
		}


		# Rarely this vision is unusually clear.
		# Total: 2 XP instead of 1.

		random = {
			chance = 10

			segrellion_gain_dreamer_xp_effect = yes
		}
	}

	option = {

		name = segrellion_dreamer.0101.a

		custom_tooltip = segrellion_dreamer_gain_xp_tt

		add_stress = 5

		add_character_modifier = {
			modifier = segrellion_dreamer_borrowed_eyes_modifier
			days = 30
		}
	}
}

# =========================================================
# 0105 - THE RAVEN WITHOUT EYES
# =========================================================

segrellion_dreamer.0105 = {

	type = character_event
	override_background = { reference = segrellion_dreamer_event_background }

	title = segrellion_dreamer.0105.t

	desc = {

		first_valid = {

			triggered_desc = {

				trigger = {
					exists = scope:dream_voice
				}

				desc = segrellion_dreamer.0105.desc_known
			}

			desc = segrellion_dreamer.0105.desc
		}
	}

	theme = intrigue

	left_portrait = {
		character = root
		animation = worry
	}

	immediate = {

		segrellion_gain_dreamer_xp_effect = yes


		# First preference:
		# someone personally important to the Dreamer.

		if = {
			limit = {

				any_relation = {

					type = friend
					type = best_friend
					type = lover
					type = soulmate
					type = rival
					type = nemesis

					is_alive = yes
				}
			}

			random_relation = {

				type = friend
				type = best_friend
				type = lover
				type = soulmate
				type = rival
				type = nemesis

				limit = {
					is_alive = yes
				}

				save_scope_as = dream_voice
			}
		}


		# Otherwise use living family.

		else_if = {
			limit = {

				any_close_or_extended_family_member = {

					is_alive = yes

					NOT = {
						this = root
					}
				}
			}

			random_close_or_extended_family_member = {

				limit = {

					is_alive = yes

					NOT = {
						this = root
					}
				}

				save_scope_as = dream_voice
			}
		}


		if = {
			limit = {
				exists = scope:dream_voice
			}

			set_variable = {
				name = segrellion_dreamer_raven_voice
				value = scope:dream_voice
			}
		}
	}

	option = {

		name = segrellion_dreamer.0105.a

		custom_tooltip = segrellion_dreamer_gain_xp_tt

		add_stress = 5
	}
}
