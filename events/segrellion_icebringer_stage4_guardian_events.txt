	title = segrellion_icebringer.7000.t

	override_background = {
		reference = segrellion_vhal_island

		animation = wonder

# =========================================================
# 7030 - THE CLOAKED MAN
# =========================================================

segrellion_icebringer.7030 = {
	type = character_event

	immediate = {
		segrellion_gain_dreamer_xp_effect = yes
	}

	title = segrellion_icebringer.7030.t

	desc = {

		desc = segrellion_icebringer.7030.desc

		triggered_desc = {
			trigger = {
				has_character_flag = segrellion_guardian_ignored_fisherman
				has_character_flag = segrellion_guardian_executed_ale_woman
			}

			desc = segrellion_icebringer.7030.desc_darkness
		}
	}

	theme = physical_health

	override_background = {
		reference = segrellion_vhalerion_throne_room
	}

	left_portrait = {
		character = root
		animation = shock
	}


	# -----------------------------------------------------
	# LISTEN TO TRISTAN
	# -----------------------------------------------------

	option = {
		name = segrellion_icebringer.7030.a

		trigger_event = segrellion_icebringer.7031
	}


	# -----------------------------------------------------
	# KILL HIM IMMEDIATELY
	#
	# If the first two escalations were also chosen:
	# -> Corrupted / Darkness route.
	#
	# Otherwise:
	# -> terrible Guardian score result.
	# -----------------------------------------------------

	option = {
		name = segrellion_icebringer.7030.b

		if = {
			limit = {
				has_character_flag = segrellion_guardian_ignored_fisherman
				has_character_flag = segrellion_guardian_executed_ale_woman
			}

			add_character_flag = segrellion_guardian_darkness_path

			trigger_event = segrellion_icebringer.7050
		}

		else = {

			change_variable = {
				name = segrellion_guardian_score
				add = -3
			}

			trigger_event = segrellion_icebringer.7032
		}
	}
}

			limit = {
				var:segrellion_guardian_score <= 0
			}

			trigger_event = segrellion_icebringer.7071
		}


		# =================================================
		# 1 - 4
		# =================================================

		else_if = {
			limit = {
				var:segrellion_guardian_score < 5
			}

			trigger_event = segrellion_icebringer.7072
		}


		# =================================================
		# 5 - 9
		# =================================================

		else_if = {
			limit = {
				var:segrellion_guardian_score < 10
			}

			trigger_event = segrellion_icebringer.7073
		}


		# =================================================
		# PERFECT 10
		# =================================================

		else = {

			trigger_event = segrellion_icebringer.7074
		}
	}
}

		hidden_effect = {

			if = {
				limit = {
					exists = global_var:segrellion_icebringer
				}

				remove_global_variable = segrellion_icebringer
			}

			remove_variable = segrellion_guardian_score

			remove_character_flag = segrellion_icebringer_awakening_chain_active
		}
	}
