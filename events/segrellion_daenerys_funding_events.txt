# =========================================================
# 0350 - DAENERYS ASKS DAEMON
#
# ROOT = Daemon
# =========================================================

segrellion_daenerys_funding.0350 = {

	type = character_event

	title = segrellion_daenerys_funding.0350.t

	desc = {
		first_valid = {

			triggered_desc = {

				trigger = {
					has_character_flag = segrellion_daemon_funding_existing_pact
				}

				desc = segrellion_daenerys_funding.0350.desc_pact
			}


			triggered_desc = {

				trigger = {
					has_character_flag = segrellion_daemon_funding_locked_to_aegon
				}

				desc = segrellion_daenerys_funding.0350.desc_aegon
			}


			desc = segrellion_daenerys_funding.0350.desc
		}
	}


	theme = diplomacy


	trigger = {

		this = character:Segrellion_60

		is_alive = yes

		character:Targaryen_9 ?= {
			is_alive = yes
		}
	}


	immediate = {

		title:d_masseys_hook.holder ?= {
			save_scope_as = maric_massey
		}


		if = {

			limit = {
				NOT = {
					exists = scope:maric_massey
				}
			}

			title:c_stonedance.holder ?= {
				save_scope_as = maric_massey
			}
		}


		character:Targaryen_9 = {
			segrellion_daenerys_update_fleet_shortfall_effect = yes
		}


		# =================================================
		# EXISTING DAENERYS PACT
		# =================================================

		if = {

			limit = {

				OR = {

					has_character_flag = segrellion_pledged_to_daenerys

					character:Targaryen_9 = {
						has_character_flag = segrellion_daenerys_supports_daemon
					}

					character:Targaryen_9 = {
						has_character_flag = segrellion_marriage_promised_to_daemon
					}
				}
			}


			add_character_flag = segrellion_daemon_funding_existing_pact
		}


		# =================================================
		# LOYAL TO YOUNG GRIFF
		#
		# If Aegon is dead, this no longer blocks Daemon.
		# If Aegon has been exposed as Blackfyre, Daemon may
		# betray him and negotiate with Daenerys.
		# =================================================

		else_if = {

			limit = {

				character:IAF_Young_Griff_1 ?= {

					is_alive = yes

					NOT = {
						has_character_flag = segrellion_young_griff_blackfyre_revealed
					}
				}


				OR = {

					has_character_flag = segrellion_pledged_to_young_griff

					has_relation_best_friend = character:IAF_Young_Griff_1


					AND = {

						has_relation_friend = character:IAF_Young_Griff_1

						is_allied_to = character:IAF_Young_Griff_1
					}


					character:IAF_Young_Griff_1 = {

						OR = {
							has_character_flag = segrellion_young_griff_daemon_oath_secured
							has_character_flag = segrellion_young_griff_supports_daemon
						}
					}
				}
			}


			add_character_flag = segrellion_daemon_funding_locked_to_aegon
		}


		else = {

			add_character_flag = segrellion_daemon_funding_can_bargain
		}
	}


	left_portrait = {
		character = root
		animation = thinking
	}


	right_portrait = {
		character = character:Targaryen_9
		animation = personality_rational
	}


	# =====================================================
	# EXISTING PACT: ACCEPT
	# 90%
	# =====================================================

	option = {

		name = segrellion_daenerys_funding.0350.a
		custom_tooltip = segrellion_daenerys_funding.0350.a.tt


		trigger = {
			has_character_flag = segrellion_daemon_funding_existing_pact
		}


		remove_character_flag = segrellion_daemon_funding_existing_pact


		character:Targaryen_9 = {

			trigger_event = {
				id = segrellion_daenerys_funding.0360
				days = 1
			}
		}


		ai_chance = {
			base = 90
		}
	}


	# =====================================================
	# EXISTING PACT: BREAK IT
	# 10%
	# =====================================================

	option = {

		name = segrellion_daenerys_funding.0350.b
		custom_tooltip = segrellion_daenerys_funding.0350.b.tt


		trigger = {
			has_character_flag = segrellion_daemon_funding_existing_pact
		}


		remove_character_flag = segrellion_daemon_funding_existing_pact


		add_character_modifier = {
			modifier = segrellion_daemon_broke_daenerys_funding_pact_modifier
			years = 10
		}


		character:Targaryen_9 = {

			add_opinion = {
				target = root
				modifier = segrellion_daemon_broke_daenerys_funding_pact_opinion
			}

			add_character_flag = segrellion_daenerys_daemon_funding_refused_after_pact
		}


		scope:maric_massey ?= {

			add_opinion = {
				target = root
				modifier = segrellion_massey_daemon_broke_funding_pact_opinion
			}
		}


		ai_chance = {
			base = 10
		}
	}


	# =====================================================
	# YOUNG GRIFF LOYALTY: REFUSE
	# =====================================================

	option = {

		name = segrellion_daenerys_funding.0350.c
		custom_tooltip = segrellion_daenerys_funding.0350.c.tt


		trigger = {
			has_character_flag = segrellion_daemon_funding_locked_to_aegon
		}


		remove_character_flag = segrellion_daemon_funding_locked_to_aegon

		add_character_flag = segrellion_daemon_refused_daenerys_funding_for_aegon


		character:Targaryen_9 = {
			add_character_flag = segrellion_daenerys_daemon_funding_refused
		}


		ai_chance = {
			base = 100
		}
	}


	# =====================================================
	# DEMAND MARRIAGE
	# =====================================================

	option = {

		name = segrellion_daenerys_funding.0350.d
		custom_tooltip = segrellion_daenerys_funding.0350.d.tt


		trigger = {

			has_character_flag = segrellion_daemon_funding_can_bargain

			is_married = no

			character:Targaryen_9 = {
				is_married = no
			}
		}


		remove_character_flag = segrellion_daemon_funding_can_bargain


		character:Targaryen_9 = {

			add_character_flag = segrellion_daenerys_daemon_funding_marriage_pending

			trigger_event = {
				id = segrellion_daenerys_funding.0370
				days = 1
			}
		}


		ai_chance = {
			base = 35
		}
	}


	# =====================================================
	# COUNCIL + AUTONOMY + MARRIAGE PREFERENCE
	# =====================================================

	option = {
