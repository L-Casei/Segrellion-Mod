# =========================================================
# 2000 - THE REACH
# ROOT = DAENERYS
# =========================================================

segrellion_daenerys_support_v2.2000 = {

	type = character_event

	title = segrellion_daenerys_support_v2.2000.t

	desc = {
		first_valid = {

			triggered_desc = {

				trigger = {

					character:Tyrell_14 ?= {

						father ?= {

							mother ?= {
								is_alive = yes
							}
						}
					}
				}

				desc = segrellion_daenerys_support_v2.2000.desc_olenna
			}

			desc = segrellion_daenerys_support_v2.2000.desc_lord
		}
	}

	theme = diplomacy

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	override_background = {
		reference = council_chamber
	}

	immediate = {

		add_character_flag = segrellion_daenerys_support_negotiation_pending

		# Olenna is Margaery's paternal grandmother.
		# This is much safer than deriving her through
		# whichever character happens to hold the Reach.

		if = {

			limit = {

				character:Tyrell_14 ?= {

					father ?= {

						mother ?= {
							is_alive = yes
						}
					}
				}
			}

			character:Tyrell_14.father.mother = {
				save_scope_as = segrellion_reach_contact
			}
		}

		else = {

			title:k_the_reach.holder = {
				save_scope_as = segrellion_reach_contact
			}
		}

		set_variable = {
			name = segrellion_reach_contact
			value = scope:segrellion_reach_contact
		}
	}

	left_portrait = {
		character = root
		animation = thinking
	}

	right_portrait = {
		character = character:Lannister_8
		animation = personality_rational
	}

	option = {

		name = segrellion_daenerys_support_v2.2000.a

		custom_tooltip = segrellion_daenerys_support_v2.reply_weeks_tt

		var:segrellion_reach_contact = {

			trigger_event = {
				id = segrellion_daenerys_support_v2.2010
				days = { 14 24 }
			}
		}

		ai_chance = {
			base = 100
		}
	}
}
