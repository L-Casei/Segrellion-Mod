segrellion_dance.0120 = {
	type = character_event
	title = segrellion_dance.0120.t
	desc = segrellion_dance.0120.desc
	theme = realm

	override_background = {
		reference = bedchamber
	}

	left_portrait = {
		character = root
		animation = sadness
	}

	trigger = {
		this = character:Segrellion_1
		character:Targaryen_60 ?= { is_alive = yes }
		character:Targaryen_65 ?= { is_alive = yes }
		character:Targaryen_63 ?= { is_alive = yes }
		NOT = { has_character_flag = segrellion_dance_viserys_death_processed }
	}

	immediate = {
		add_character_flag = segrellion_dance_viserys_death_processed
		add_character_flag = segrellion_dance_green_coup_active

		character:Targaryen_60 = {
			death = {
				death_reason = death_ill
			}
		}

		# The Greens immediately seize the royal apparatus.
		# AGOT history gives Aegon II these exact two titles.
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = segrellion_dance_green_title_change
			add_claim_on_loss = yes
		}

		title:h_the_iron_throne = {
			change_title_holder_include_vassals = {
				holder = character:Targaryen_65
				change = scope:segrellion_dance_green_title_change
			}
		}

		title:e_the_crownlands = {
			change_title_holder_include_vassals = {
				holder = character:Targaryen_65
				change = scope:segrellion_dance_green_title_change
			}
		}

		resolve_title_and_vassal_change = scope:segrellion_dance_green_title_change

		# Rhaenyra's claim is not erased by the coup.
		character:Targaryen_63 = {
			if = {
				limit = {
					NOT = { has_claim_on = title:h_the_iron_throne }
				}
				add_pressed_claim = title:h_the_iron_throne
			}
			if = {
				limit = {
					NOT = { has_claim_on = title:e_the_crownlands }
				}
				add_pressed_claim = title:e_the_crownlands
			}
			add_character_flag = segrellion_dance_rhaenyra_dispossessed
		}

		character:Targaryen_65 = {
			add_character_flag = segrellion_dance_aegon_proclaimed
		}
	}

	option = {
		name = segrellion_dance.0120.a

		add_stress = 15
		add_character_flag = segrellion_dance_alex_knows_viserys_dead

		trigger_event = {
			id = segrellion_dance.0130
			days = 1
		}
	}
}
