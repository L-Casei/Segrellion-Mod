# 128 AC - THE DANCE OF THE DRAGONS
# ALEX SEGRELLION PRELUDE
#
# This first block deliberately stops before Viserys dies.
# It establishes Alex's history, relationships and political
# posture so the succession crisis can react to player choices.

segrellion_dance.0001 = {
	type = character_event
	title = segrellion_dance.0001.t
	desc = segrellion_dance.0001.desc
	theme = realm

	override_background = {
		reference = segrellion_vhalerion_throne_room
	}

		character = root
		animation = personality_honorable

		this = character:Segrellion_1

		add_character_flag = segrellion_dance_128_intro_seen

		}

			limit = {
