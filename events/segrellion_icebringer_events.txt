# =========================================================
# BLACK DOOR
#
# DAEMON SEGRELLION:
# Always reaches the true event.
#
# EVERYONE ELSE:
# Two false endings and one extremely rare true ending.
# =========================================================

segrellion_icebringer.5200 = {
	type = character_event
	hidden = yes

	immediate = {


		# -----------------------------------------------------
		# THE PROPHESIED GUARDIAN
		#
		# Daemon Segrellion always reaches the true path.
		# -----------------------------------------------------

		if = {
			limit = {
				has_trait = segrellion_canon_daemon_bastard_unicorn_modifier
			}

			trigger_event = segrellion_icebringer.5213
		}


		# -----------------------------------------------------
		# EVERYONE ELSE
		# -----------------------------------------------------

		else = {

			random_list = {


				# ---------------------------------------------
				# FALSE ENDING I
				# ---------------------------------------------

				100 = {
					trigger_event = segrellion_icebringer.5211
				}


				# ---------------------------------------------
				# FALSE ENDING II
				# ---------------------------------------------

				100 = {
					trigger_event = segrellion_icebringer.5212
				}


				# ---------------------------------------------
				# TRUE ENDING
				#
				# Base weight = 1
				#
				# Against 200 failure weight.
				# Roughly 0.5% before modifiers.
				# ---------------------------------------------

				1 = {

					# -----------------------------------------
					# ERA
					#
					# The closer the world moves toward the
					# prophetic age, the more willing the
					# blade becomes to answer.
					# -----------------------------------------

					modifier = {
						factor = 0.10

						current_date < 8200.1.1
					}

					modifier = {
						factor = 0.25

						current_date >= 8200.1.1
						current_date < 8250.1.1
					}

					modifier = {
						factor = 0.50

						current_date >= 8250.1.1
						current_date < 8280.1.1
					}

					# 8280-8294:
					# Base factor remains 1.

					modifier = {
						factor = 2.00

						current_date >= 8295.1.1
					}


					# -----------------------------------------
					# PERSONALITY
					# -----------------------------------------

					modifier = {
						factor = 1.25
						has_trait = brave
					}

					modifier = {
						factor = 1.35
						has_trait = honest
					}

					modifier = {
						factor = 1.20
						has_trait = just
					}

					modifier = {
						factor = 1.10
						has_trait = stubborn
					}


					# -----------------------------------------
					# PROPHECY / SUPERNATURAL SENSITIVITY
					# -----------------------------------------

					modifier = {
						factor = 1.50
						has_trait = dragon_dreams
					}


					trigger_event = segrellion_icebringer.5213
				}
			}
		}
	}
}
