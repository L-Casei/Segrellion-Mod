# ---------------------------------------------------------

segrellion_icebringer.1001 = {
	type = character_event
	title = segrellion_icebringer.1001.t
	desc = segrellion_icebringer.1001.desc
	theme = war

	override_background = {
		reference = army_camp
	}

	left_portrait = {
		character = root
		animation = shock
	}

	trigger = {
		has_character_flag = segrellion_icebringer_state_rejected

		any_equipped_character_artifact = {
			has_variable = segrellion_icebringer_artifact
		}
	}

	option = {
		name = segrellion_icebringer.1001.a

		add_stress = 10
		add_prestige = -15
	}

	option = {
		name = segrellion_icebringer.1001.b

		add_stress = 25

			modifier = segrellion_icebringer_burned_hand_modifier
			days = 180
		}

		add_martial_lifestyle_xp = 50
	}
}
