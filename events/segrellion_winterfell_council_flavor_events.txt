# ---------------------------------------------------------
# 6001 - A QUESTION OF PRECEDENCE
# ---------------------------------------------------------

segrellion_winterfell_council.6001 = {

	type = activity_event

	title = segrellion_winterfell_council.6001.t
	desc = segrellion_winterfell_council.6001.desc

	theme = realm

	override_background = {
		reference = throne_room
	}

	cooldown = {
		years = 1
	}


	trigger = {

		this = scope:host

		scope:activity = {

			any_attending_character = {

				count >= 2

				this != scope:host

				is_alive = yes
				is_adult = yes
				is_landed = yes
			}
		}
	}


	immediate = {

		scope:activity = {

			random_attending_character = {

				limit = {

					this != scope:host

					is_alive = yes
					is_adult = yes
					is_landed = yes
				}

				save_scope_as = council_lord_a
			}


			random_attending_character = {

				limit = {

					this != scope:host
					this != scope:council_lord_a

					is_alive = yes
					is_adult = yes
					is_landed = yes
				}

				save_scope_as = council_lord_b
			}
		}
	}


	left_portrait = {

		character = scope:council_lord_a
		animation = personality_bold
	}


	right_portrait = {

		character = scope:council_lord_b
		animation = disgust
	}


	option = {

		name = segrellion_winterfell_council.6001.a

		custom_tooltip = segrellion_winterfell_council.6001.a.tt


		scope:council_lord_a = {

			add_opinion = {

				target = scope:host

				modifier = segrellion_council_supported_me_opinion
			}


			change_variable = {

				name = segrellion_council_king_support
				add = 1
			}
		}


		scope:council_lord_b = {

			add_opinion = {

				target = scope:host

				modifier = segrellion_council_slighted_me_opinion
			}
		}


		add_prestige = 50
	}


	option = {

		name = segrellion_winterfell_council.6001.b

		custom_tooltip = segrellion_winterfell_council.6001.b.tt


		scope:council_lord_b = {

			add_opinion = {

				target = scope:host

				modifier = segrellion_council_supported_me_opinion
			}


			change_variable = {

				name = segrellion_council_king_support
				add = 1
			}
		}


		scope:council_lord_a = {

			add_opinion = {

				target = scope:host

				modifier = segrellion_council_slighted_me_opinion
			}
		}


		add_prestige = 50
	}


	option = {

		name = segrellion_winterfell_council.6001.c

		custom_tooltip = segrellion_winterfell_council.6001.c.tt


		scope:council_lord_a = {

			add_opinion = {

				target = scope:council_lord_b

				modifier = segrellion_council_reconciled_us_opinion
			}

			add_opinion = {

				target = scope:host

				modifier = segrellion_council_respected_me_opinion
			}
		}


		scope:council_lord_b = {

			add_opinion = {

				target = scope:council_lord_a

				modifier = segrellion_council_reconciled_us_opinion
			}

			add_opinion = {

				target = scope:host

				modifier = segrellion_council_respected_me_opinion
			}
		}


		add_character_modifier = {

			modifier = segrellion_council_mediator_modifier
			years = 3
		}


		add_prestige = 75


		scope:activity = {

			change_variable = {

				name = segrellion_council_independence_support
				add = 1
			}
		}
	}
}
