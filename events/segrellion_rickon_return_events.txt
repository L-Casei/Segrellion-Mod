# =========================================================
# 0000 - HIDDEN ROUTER / WATCHDOG
# ROOT = RICKON
# =========================================================

segrellion_rickon_return.0000 = {
	hidden = yes

	trigger = {
		this = character:Stark_Dummy_11
		is_alive = yes
		NOT = { has_character_flag = segrellion_rickon_north_return_resolved }
	}

	immediate = {
		remove_character_flag = segrellion_rickon_return_watch_queued

		title:e_flamelands.holder ?= {
			if = {
				limit = {
					segrellion_rickon_return_flamelands_protector_trigger = yes
				}
				save_scope_as = segrellion_rickon_current_flamelands_protector
			}
		}

		# -------------------------------------------------
		# SANSA CURRENTLY SHELTERS RICKON
		# -------------------------------------------------

		if = {
			limit = {
				has_character_flag = segrellion_rickon_with_sansa_riverlands
				character:Stark_Dummy_8 ?= {
					is_alive = yes
					NOT = { segrellion_rickon_sansa_holds_riverlands_trigger = yes }
				}
				title:e_flamelands.holder ?= {
					segrellion_rickon_return_flamelands_protector_trigger = yes
				}
				NOT = { has_character_flag = segrellion_rickon_return_to_beacon_queued }
			}

			add_character_flag = {
				flag = segrellion_rickon_return_to_beacon_queued
				days = 60
			}

			title:e_flamelands.holder = {
				trigger_event = {
					id = segrellion_rickon_return.0090
					days = 1
				}
			}
		}

		# -------------------------------------------------
		# SANSA HAS WON THE RIVERLANDS
		# -------------------------------------------------

		else_if = {
			limit = {
				title:e_flamelands.holder ?= {
					OR = {
						this = character:Segrellion_60
						this = character:Black_Unicorn_1
					}
				}
				has_character_flag = segrellion_rickon_sheltered_with_sansa_at_beacon
				NOT = { has_character_flag = segrellion_rickon_with_sansa_riverlands }
				NOT = { has_character_flag = segrellion_rickon_travelling_to_sansa }
				character:Stark_Dummy_8 ?= {
					segrellion_rickon_sansa_holds_riverlands_trigger = yes
				}
			}

			character:Stark_Dummy_8 = {
				trigger_event = {
					id = segrellion_rickon_return.0080
					days = 1
				}
			}
		}

		# -------------------------------------------------
		# NORMAL BEACON -> NORTH ROUTE
		# -------------------------------------------------

		else_if = {
			limit = {
				NOR = {
					has_character_flag = segrellion_rickon_travelling_north
					has_character_flag = segrellion_rickon_waiting_white_harbor
					has_character_flag = segrellion_rickon_return_request_pending
					has_character_flag = segrellion_rickon_travelling_to_sansa
					has_character_flag = segrellion_rickon_returning_to_beacon
				}

				OR = {
					NOT = { has_character_flag = segrellion_rickon_kept_as_beacon_ward }
					is_adult = yes
				}

				exists = scope:segrellion_rickon_current_flamelands_protector

				OR = {
					is_courtier_of = scope:segrellion_rickon_current_flamelands_protector
					location = title:b_castle_farring.title_province
				}

				OR = {
					title:e_the_north.holder ?= {
						segrellion_rickon_return_jon_king_trigger = yes
					}
					AND = {
						scope:segrellion_rickon_current_flamelands_protector = character:Segrellion_60
						character:Segrellion_60 = {
							segrellion_rickon_sansa_beacon_option_trigger = yes
						}
					}
				}
			}

			add_character_flag = {
				flag = segrellion_rickon_return_prompt_queued
				days = 30
			}

			scope:segrellion_rickon_current_flamelands_protector = {
				trigger_event = {
					id = segrellion_rickon_return.0010
					days = 1
				}
			}
		}
	}
}

segrellion_rickon_return.0010 = {
	type = character_event

	title = segrellion_rickon_return.0010.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					character:Stark_Dummy_11 = {
						is_adult = yes
						has_character_flag = segrellion_rickon_kept_as_beacon_ward
					}
				}
				desc = segrellion_rickon_return.0010.desc_adult
			}

			triggered_desc = {
				trigger = {
					this = character:Segrellion_60
					segrellion_rickon_sansa_beacon_option_trigger = yes
				}
				desc = segrellion_rickon_return.0010.desc_sansa
			}

			triggered_desc = {
				trigger = { this = character:Black_Unicorn_1 }
				desc = segrellion_rickon_return.0010.desc_visenya
			}

			desc = segrellion_rickon_return.0010.desc_daemon
		}
	}

	theme = diplomacy

	override_background = {
		reference = segrellion_the_beacon_throne_room
	}

	override_header_background = {
		reference = "gfx/interface/window_event/event_header_blue.dds"
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	trigger = {
		segrellion_rickon_return_flamelands_protector_trigger = yes

		character:Stark_Dummy_11 ?= {
			is_alive = yes
			has_character_flag = segrellion_rickon_return_prompt_queued
		}
	}

	on_trigger_fail = {
		character:Stark_Dummy_11 ?= {
			remove_character_flag = segrellion_rickon_return_prompt_queued
		}
	}

	immediate = {
		character:Stark_Dummy_11 = {
			remove_character_flag = segrellion_rickon_return_prompt_queued
		}

		if = {
			limit = { exists = cp:councillor_castellan }
			cp:councillor_castellan = {
				save_scope_as = segrellion_rickon_return_castellan
			}
		}
		else = {
			save_scope_as = segrellion_rickon_return_castellan
		}
	}

	left_portrait = {
		character = scope:segrellion_rickon_return_castellan
		animation = personality_rational
	}

	right_portrait = {
		character = character:Stark_Dummy_11
		animation = child_personality_bold
	}

	# SEND RICKON TO JON
	option = {
		name = segrellion_rickon_return.0010.a
		custom_tooltip = segrellion_rickon_return.0010.a.tt

		trigger = {
			title:e_the_north.holder ?= {
				segrellion_rickon_return_jon_king_trigger = yes
			}
		}

		character:Stark_Dummy_11 = {
			add_character_flag = segrellion_rickon_return_request_pending
			remove_character_flag = segrellion_rickon_kept_as_beacon_ward
		}

		if = {
			limit = { this = character:Segrellion_60 }
			add_character_flag = segrellion_daemon_sent_rickon_offer_to_jon
			title:e_the_north.holder = {
				trigger_event = {
					id = segrellion_rickon_return.0020
					days = { 14 30 }
				}
			}
		}
		else = {
			add_character_flag = segrellion_visenya_sent_rickon_offer_to_jon
			title:e_the_north.holder = {
				trigger_event = {
					id = segrellion_rickon_return.0021
					days = { 14 30 }
				}
			}
		}

		ai_chance = {
			base = 75
		}
	}

	# KEEP RICKON AS A WARD UNTIL ADULTHOOD
	option = {
		name = segrellion_rickon_return.0010.b
		custom_tooltip = segrellion_rickon_return.0010.b.tt

		trigger = {
			character:Stark_Dummy_11 = {
				is_adult = no
			}
		}

		character:Stark_Dummy_11 = {
			add_character_flag = segrellion_rickon_kept_as_beacon_ward
			remove_character_flag = segrellion_rickon_return_request_pending
			remove_character_flag = segrellion_rickon_education_chain_ended
			remove_character_flag = segrellion_rickon_education_paused_by_departure

			if = {
				limit = { root = character:Segrellion_60 }
				add_character_flag = segrellion_rickon_under_daemon_protection
				remove_character_flag = segrellion_rickon_tutor_visenya
			}
			else = {
				add_character_flag = segrellion_rickon_under_visenya_protection
				remove_character_flag = segrellion_rickon_tutor_daemon
			}

			set_relation_guardian = root
			segrellion_rickon_ensure_education_watch_effect = yes
		}

		ai_chance = {
			base = 20
		}
	}

	# KEEP BOTH STARK CHILDREN AT THE BEACON
	option = {
		name = segrellion_rickon_return.0010.c
		custom_tooltip = segrellion_rickon_return.0010.c.tt

		trigger = {
			this = character:Segrellion_60
			segrellion_rickon_sansa_beacon_option_trigger = yes
		}

		character:Stark_Dummy_11 = {
			add_character_flag = segrellion_rickon_sheltered_with_sansa_at_beacon
			remove_character_flag = segrellion_rickon_return_request_pending
			add_character_flag = segrellion_rickon_under_daemon_protection
			remove_character_flag = segrellion_rickon_education_chain_ended
			remove_character_flag = segrellion_rickon_education_paused_by_departure
			set_relation_guardian = character:Segrellion_60
			segrellion_rickon_ensure_education_watch_effect = yes
		}

		character:Stark_Dummy_8 = {
			add_character_flag = segrellion_sansa_sheltered_with_rickon_at_beacon
			if = {
				limit = { is_ruler = no }
				set_employer = character:Segrellion_60
			}
			set_location = title:b_castle_farring.title_province
		}

		ai_chance = {
			base = 5
		}
	}
}

	trigger = {
		this = character:Stark_Dummy_11

	}

	immediate = {
		if = {
			limit = {
				title:e_the_north.holder ?= {
					segrellion_rickon_return_jon_king_trigger = yes
				}
			}

			title:c_white_harbor.holder ?= {
				if = {
					limit = { is_alive = yes }
					trigger_event = {
						id = segrellion_rickon_return.0051

					}
				}
			}

			title:e_the_north.holder = {
				trigger_event = {
					id = segrellion_rickon_return.0060
					days = 1
				}
			}
		}
		else = {
			# Jon ceased to be King in the North while Rickon was
			# already travelling. Manderly keeps the boy safe and
			# the route waits rather than teleporting him elsewhere.
			trigger_event = {
				id = segrellion_rickon_return.0050
				days = 60
			}
		}
	}
}

				target = character:Black_Unicorn_1

					limit = {

						type = favor_hook
