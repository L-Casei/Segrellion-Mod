	type = character_event

	title = segrellion_castle_lore.1000.t
	desc = segrellion_castle_lore.1000.desc

	override_background = {
		reference = segrellion_vhal_inside_event_background
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	left_portrait = root


	immediate = {

		if = {

			limit = {
				NOT = {
					exists = var:segrellion_lore_vhal_visit
				}
			}

			segrellion_add_lore_effect = {
				AMOUNT = 2
				VISITED_VARIABLE = segrellion_lore_vhal_visit
			}
		}
	}

		name = segrellion_castle_lore.1000.a

		trigger_event = {
			id = segrellion_castle_lore.1010
		}
	}

		name = segrellion_castle_lore.1000.b

		trigger_event = {
			id = segrellion_castle_lore.1020
		}

	option = {

		name = segrellion_castle_lore.1000.c

		trigger_event = {
			id = segrellion_castle_lore.1030
