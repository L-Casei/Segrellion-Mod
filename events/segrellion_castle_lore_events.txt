segrellion_castle_lore.1032 = {

	type = character_event

	title = segrellion_castle_lore.1032.t
	desc = segrellion_castle_lore.1032.desc
	theme = travel

	override_background = {
		reference = segrellion_vhal_inside_event_background
	}


	option = {

		name = segrellion_castle_lore.1032.a
		custom_tooltip = segrellion_castle_lore.return_to_vhal_tt

		add_prestige = 300

		hidden_effect = {

			if = {

				limit = {
					NOT = {
						exists = var:segrellion_lore_vhal_caves
					}
				}

				if = {

					limit = {

						NOT = {
							exists = global_var:segrellion_vhalyrax_lost_egg
						}
					}


					create_artifact_dragon_egg_yellow_effect = {
						OWNER = root
					}


					scope:newly_created_artifact ?= {

						set_artifact_name = segrellion_vhalyrax_lost_egg_name

						set_artifact_description = segrellion_vhalyrax_lost_egg_description


						set_variable = {
							name = segrellion_vhalyrax_lost_egg_artifact
							value = yes
						}


						set_variable = {
							name = historical_unique_artifact
							value = yes
						}


						set_global_variable = {
							name = segrellion_vhalyrax_lost_egg
							value = this
						}
					}
				}

				segrellion_add_lore_effect = {
					AMOUNT = 3
					VISITED_VARIABLE = segrellion_lore_vhal_caves
				}
			}
		}

		trigger_event = {
			id = segrellion_castle_lore.1000
		}
	}
}
