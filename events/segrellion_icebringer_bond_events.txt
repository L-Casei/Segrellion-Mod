namespace = segrellion_icebringer_bond


# =========================================================
# 0000 - BOND AWAKENING DISPATCHER
# =========================================================

segrellion_icebringer_bond.0000 = {

	hidden = yes


	trigger = {

		is_alive = yes

		NOT = {
			has_trait = segrellion_icebringer_bearer_trait
		}

		any_character_artifact = {
			has_variable = segrellion_icebringer_artifact
		}
	}


	on_trigger_fail = {

		remove_character_flag = segrellion_icebringer_bond_event_queued
	}


	immediate = {

		random_list = {

			34 = {
				trigger_event = segrellion_icebringer_bond.0001
			}

			33 = {
				trigger_event = segrellion_icebringer_bond.0002
			}

			33 = {
				trigger_event = segrellion_icebringer_bond.0003
			}
		}
	}
}

# =========================================================
# 0099 - OWNERSHIP WATCHDOG
#
# Follows Icebringer from holder to holder every 30 days.
# Quarterly/yearly on-actions below remain as recovery in
# case succession or death interrupts this event chain.
# =========================================================

segrellion_icebringer_bond.0099 = {

	hidden = yes


	trigger = {
		is_alive = yes
	}


	immediate = {

		remove_character_flag = segrellion_icebringer_bond_watch_running


		# First update the old/current root. If Icebringer has
		# moved away, their permanent trait becomes inactive.

		segrellion_icebringer_bond_maintenance_effect = yes


		# Then find the actual current owner and move the
		# watcher to them.

		if = {

			limit = {
				exists = global_var:segrellion_icebringer
			}


			global_var:segrellion_icebringer = {

				artifact_owner ?= {

					segrellion_icebringer_bond_maintenance_effect = yes
				}
			}
		}
	}
}
