# =========================================================
# 0000 - THE FALLEN CROWN
#
# ROOT = DAENERYS
#
# The ruler who held the Iron Throne when Daenerys won
# is brought before the new queen.
# =========================================================

segrellion_daenerys_post_throne.0000 = {

	type = character_event

	title = segrellion_daenerys_post_throne.0000.t

	desc = segrellion_daenerys_post_throne.0000.desc

	theme = crown

	override_background = {

		reference = throne_room
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	widget = {

		gui = "event_window_widget_segrellion_daenerys_royal_tyranny"

		container = "custom_widgets_container"

		controller = default


		is_shown = {

			this = character:Targaryen_9

			has_variable = segrellion_daenerys_pacification_tyranny
		}
	}


	trigger = {

		this = character:Targaryen_9

		is_alive = yes

		has_title = title:h_the_iron_throne

		has_character_flag = segrellion_daenerys_defeated_throne_holder_fate_pending

		has_variable = segrellion_daenerys_defeated_throne_holder


		var:segrellion_daenerys_defeated_throne_holder = {

			is_alive = yes

			this != root
		}
	}


	on_trigger_fail = {

		remove_character_flag = segrellion_daenerys_defeated_throne_holder_event_scheduled

		remove_character_flag = segrellion_daenerys_defeated_throne_holder_fate_pending

		add_character_flag = segrellion_daenerys_defeated_throne_holder_fate_resolved

		remove_variable = segrellion_daenerys_defeated_throne_holder


		segrellion_start_daenerys_post_throne_effect = yes
	}


	immediate = {

		remove_character_flag = segrellion_daenerys_defeated_throne_holder_event_scheduled

		segrellion_daenerys_cancel_agot_wars_aftermath_effect = yes

		segrellion_daenerys_finalize_defeated_throne_holder_effect = yes


		var:segrellion_daenerys_defeated_throne_holder = {

			save_scope_as = segrellion_daenerys_fallen_king
		}


		if = {

			limit = {

				scope:segrellion_daenerys_fallen_king = {

					is_imprisoned = no
				}
			}


			imprison = {

				target = scope:segrellion_daenerys_fallen_king

				type = house_arrest
			}
		}
	}


	left_portrait = {

		character = root

		animation = personality_bold
	}


	right_portrait = {

		character = scope:segrellion_daenerys_fallen_king

		animation = fear
	}


	# =====================================================
	# EXECUTION
	#
	# CANON / AI ROUTE
	# =====================================================

	option = {

		name = segrellion_daenerys_post_throne.0000.a


		hidden_effect = {

			scope:segrellion_daenerys_fallen_king = {

				death = {

					death_reason = death_execution

					killer = root
				}
			}


			add_character_flag = segrellion_daenerys_executed_defeated_throne_holder

			remove_character_flag = segrellion_daenerys_defeated_throne_holder_fate_pending

			add_character_flag = segrellion_daenerys_defeated_throne_holder_fate_resolved

			remove_variable = segrellion_daenerys_defeated_throne_holder


			segrellion_daenerys_cancel_agot_wars_aftermath_effect = yes

			segrellion_start_daenerys_post_throne_effect = yes
		}


		ai_chance = {
			base = 100
		}
	}


	# =====================================================
	# EXILE
	#
	# PLAYER ALTERNATIVE
	# =====================================================

	option = {

		name = segrellion_daenerys_post_throne.0000.b

		custom_tooltip = segrellion_daenerys_post_throne.0000.b.tt


		hidden_effect = {

			scope:segrellion_daenerys_fallen_king = {

				if = {

					limit = {
						is_imprisoned = yes
					}

					release_from_prison = yes
				}


				banish = yes

				add_character_flag = segrellion_exiled_by_daenerys_after_throne


				if = {

					limit = {

						is_landed = no

						title:c_pentos = {
							exists = holder
						}
					}


					set_employer = title:c_pentos.holder

					set_location = title:c_pentos.title_province
				}
			}


			add_character_flag = segrellion_daenerys_exiled_defeated_throne_holder

			remove_character_flag = segrellion_daenerys_defeated_throne_holder_fate_pending

			add_character_flag = segrellion_daenerys_defeated_throne_holder_fate_resolved

			remove_variable = segrellion_daenerys_defeated_throne_holder


			segrellion_daenerys_cancel_agot_wars_aftermath_effect = yes

			segrellion_start_daenerys_post_throne_effect = yes
		}


		ai_chance = {
			base = 0
		}
	}
}

segrellion_daenerys_post_throne.0001 = {

	type = character_event

	title = segrellion_daenerys_post_throne.0001.t

	desc = segrellion_daenerys_post_throne.0001.desc

	theme = crown

	override_background = {

		reference = throne_room
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	widget = {

		gui = "event_window_widget_segrellion_daenerys_royal_tyranny"

		container = "custom_widgets_container"

		controller = default


		is_shown = {

			this = character:Targaryen_9

			has_variable = segrellion_daenerys_pacification_tyranny
		}
	}


	trigger = {

		this = character:Targaryen_9

		is_alive = yes

		has_title = title:h_the_iron_throne

		has_character_flag = segrellion_daenerys_post_throne_started
	}

	on_trigger_fail = {

		remove_character_flag = segrellion_daenerys_post_throne_opening_scheduled
	}


	immediate = {

		# Final safety cleanup in case AGOT created its
		# generic aftermath story during the war-resolution tick.

		segrellion_daenerys_cancel_agot_wars_aftermath_effect = yes


		remove_character_flag = segrellion_daenerys_post_throne_opening_scheduled

		add_character_flag = segrellion_daenerys_post_throne_opening_seen
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}


	option = {

		name = segrellion_daenerys_post_throne.0001.a


		# =================================================
		# DAENERYS PROMISED THE HANDSHIP DURING GATHER SUPPORT
		# =================================================

		if = {

			limit = {

				has_character_flag = segrellion_daenerys_has_post_throne_hand_promise
			}


			add_character_flag = {

				flag = segrellion_daenerys_kingsguard_event_scheduled

				days = 30
			}


			trigger_event = {

				id = segrellion_daenerys_post_throne.0003
			}
		}


		# =================================================
		# TYRION MAY BECOME HAND
		# =================================================

		else_if = {

			limit = {

				character:Lannister_8 ?= {

					is_alive = yes
				}
			}


			add_character_flag = {

				flag = segrellion_daenerys_hand_event_scheduled

				days = 30
			}


			trigger_event = {

				id = segrellion_daenerys_post_throne.0002
			}
		}


		else = {

			add_character_flag = {

				flag = segrellion_daenerys_kingsguard_event_scheduled

				days = 30
			}


			trigger_event = {

				id = segrellion_daenerys_post_throne.0003
			}
		}
	}
}

# 0010 - WESTERLANDS ROUTER
# =========================================================

segrellion_daenerys_post_throne.0010 = {

	type = character_event

	hidden = yes


	trigger = {

		this = character:Targaryen_9

		is_alive = yes

		has_title = title:h_the_iron_throne

		NOT = {
			has_character_flag = segrellion_daenerys_westerlands_resolved
		}

		title:e_the_westerlands = {
			exists = holder
		}
	}

	on_trigger_fail = {

		remove_character_flag = segrellion_daenerys_westerlands_issue_scheduled
	}


	immediate = {

		remove_character_flag = segrellion_daenerys_westerlands_issue_scheduled

		title:e_the_westerlands.holder = {
			save_scope_as = segrellion_westerlands_holder
		}


		# =================================================
		# JOFFREY VLADWYN
		# =================================================

		if = {

			limit = {
				scope:segrellion_westerlands_holder = character:Vladwyn_5
			}


			trigger_event = {
				id = segrellion_daenerys_post_throne.0040
				
			}
		}


		# =================================================
		# CLOSE LANNISTER FAMILY
		# =================================================

		else_if = {

			limit = {

				OR = {

					scope:segrellion_westerlands_holder = character:Lannister_6

					scope:segrellion_westerlands_holder = character:Lannister_7

					scope:segrellion_westerlands_holder = character:Baratheon_6

					scope:segrellion_westerlands_holder = character:Baratheon_7
				}
			}


			trigger_event = {
				id = segrellion_daenerys_post_throne.0020
				
			}
		}


		# =================================================
		# ANOTHER LANNISTER
		# =================================================

		else_if = {

			limit = {

				scope:segrellion_westerlands_holder = {
					dynasty = dynasty:dynn_Lannister
				}
			}


			trigger_event = {
				id = segrellion_daenerys_post_throne.0030
				
			}
		}


		# =================================================
		# UNEXPECTED HOLDER
		# =================================================

		else = {

			trigger_event = {
				id = segrellion_daenerys_post_throne.0035
				
			}
		}
	}
}
