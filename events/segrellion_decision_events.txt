		character = root

	}

	option = {
		name = segrellion_decision.0004.a
		set_realm_capital = title:c_farring_cross
		title:k_flamelands = { set_capital_county = title:c_farring_cross }
		title:c_farring_cross = { set_title_name = segrellion_the_beacon_name }
		title:b_castle_farring = { set_title_name = segrellion_the_beacon_name }
		title:d_wendwater_forest = { set_title_name = segrellion_blue_forest_name }
		title:b_castle_farring.title_province = {
			if = {
				limit = { NOT = { has_holding_type = castle_holding } }
				set_holding_type = castle_holding
			}
			if = {
				limit = { NOT = { has_building_or_higher = castle_01 } }
				add_building = castle_01
			}
			if = {
				limit = { has_special_building_slot = no }
				add_special_building_slot = segrellion_beacon_01
			}
			if = {
				limit = { NOT = { has_building_or_higher = segrellion_beacon_01 } }
				add_special_building = segrellion_beacon_01
			}
		}
		title:c_farring_cross = {
			change_development_level = 4
			add_county_modifier = {
				modifier = segrellion_heart_of_the_flamelands_county_modifier
				years = 20
			}
		}
		title:b_castle_farring.title_province = {
			add_province_modifier = {
				modifier = segrellion_heart_of_the_flamelands_province_modifier
			}
		}
		add_prestige = 1000
		add_legitimacy = 250
		dynasty = { add_dynasty_prestige = 500 }
		add_character_flag = segrellion_established_the_beacon
	}
}

				}
			}
			every_held_title = {
				limit = {
					title_tier = duchy

						NOT = { exists = kingdom }
						kingdom = {
							NOT = { is_in_list = segrellion_paramountcy_moved_kingdoms }
						}
					}
				}
				set_de_jure_liege_title = title:k_flamelands
			}
			every_sub_realm_county = {
				limit = {
					exists = duchy
					OR = {
						NOT = { exists = kingdom }

							NOT = { is_in_list = segrellion_paramountcy_moved_kingdoms }
						}
					}
					duchy = {
						save_temporary_scope_as = segrellion_paramountcy_test_duchy
					}
					root = {
						completely_controls = scope:segrellion_paramountcy_test_duchy
					}
				}
				if = {
					limit = {
						NOT = {
							duchy = {
								is_in_list = segrellion_paramountcy_additional_duchies
							}
						}
