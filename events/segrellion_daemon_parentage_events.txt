namespace = segrellion_daemon_parentage


# =========================================================
# 0001
# HUGH FINDS DAENA'S LETTER AMONG SYMON'S PAPERS
# =========================================================

segrellion_daemon_parentage.0001 = {

	type = character_event

	title = segrellion_daemon_parentage.0001.t
	desc = segrellion_daemon_parentage.0001.desc

	theme = intrigue

	trigger = {
		this = character:Segrellion_43

		character:Segrellion_60 = {
			is_alive = yes

			NOT = {
				has_character_flag = segrellion_daemon_parentage_known
			}
		}
	}

	left_portrait = {
		character = root
		animation = contemplation
	}

	immediate = {
		add_character_flag = segrellion_daemon_letter_found
	}


	# =====================================================
	# OPTION A
	#
	# Hugh obeys Daena and Symon's wishes.
	# He gives the letter to Daemon UNOPENED.
	#
	# Hugh never learns the actual secret.
	# =====================================================

	option = {

		name = segrellion_daemon_parentage.0001.a

		custom_tooltip = segrellion_daemon_parentage.0001.a.tt

		ai_chance = {

			base = 50

			modifier = {
				add = 40
				has_trait = honest
			}

			modifier = {
				add = 25
				has_trait = just
			}

			modifier = {
				add = 20
				has_trait = compassionate
			}
		}

		add_character_flag = segrellion_daemon_letter_given_unopened

		character:Segrellion_60 = {

			trigger_event = {
				id = segrellion_daemon_parentage.0002
				days = { 1 3 }
			}
		}
	}


	# =====================================================
	# OPTION B
	#
	# Hugh breaks the seal and reads Daena's letter.
	# Then he tells Daemon.
	#
	# Both Hugh and Daemon know.
	# =====================================================

	option = {

		name = segrellion_daemon_parentage.0001.b

		custom_tooltip = segrellion_daemon_parentage.0001.b.tt

		ai_chance = {

			base = 50

			modifier = {
				add = 40
				has_trait = scholar
			}

			modifier = {
				add = 20
				has_trait = cynical
			}

			modifier = {
				add = 10
				has_trait = arbitrary
			}

			modifier = {
				add = 15
				has_trait = deceitful
			}
		}

		add_character_flag = segrellion_hugh_read_daena_letter

		character:Segrellion_60 = {

			trigger_event = {
				id = segrellion_daemon_parentage.0003
				days = { 1 3 }
			}
		}
	}
}


# =========================================================
# 0002
# DAEMON RECEIVES THE SEALED LETTER
#
# ONLY DAEMON LEARNS THE TRUTH.
# =========================================================

segrellion_daemon_parentage.0002 = {

	type = character_event

	title = segrellion_daemon_parentage.0002.t
	desc = segrellion_daemon_parentage.0002.desc

	theme = family

	trigger = {
		this = character:Segrellion_60

		NOT = {
			has_character_flag = segrellion_daemon_parentage_known
		}
	}

	left_portrait = {
		character = root
		animation = shock
	}

	immediate = {

		# -------------------------------------------------
		# Create the actual CK3/AGOT heritage secret.
		#
		# Daemon's history already contains:
		#
		# father = Segrellion_39
		# real_father = Targaryen_1
		#
		# AGOT recognises secret_disputed_heritage for
		# non-Jon royal bastard claimants.
		# -------------------------------------------------

		add_secret = {
			type = secret_disputed_heritage
			target = root
		}

		add_character_flag = segrellion_daemon_parentage_known
		add_character_flag = segrellion_daemon_parentage_known_daemon_only
	}

	option = {

		name = segrellion_daemon_parentage.0002.a

		custom_tooltip = segrellion_daemon_parentage.0002.a.tt
		custom_tooltip = segrellion_daemon_parentage.reveal_tt
	}
}


# =========================================================
# 0003
# HUGH READ THE LETTER FIRST
#
# HUGH TELLS DAEMON.
# BOTH KNOW THE SECRET.
# =========================================================

segrellion_daemon_parentage.0003 = {

	type = character_event

	title = segrellion_daemon_parentage.0003.t
	desc = segrellion_daemon_parentage.0003.desc

	theme = family

	trigger = {
		this = character:Segrellion_60

		character:Segrellion_43 = {
			is_alive = yes
			has_character_flag = segrellion_hugh_read_daena_letter
		}

		NOT = {
			has_character_flag = segrellion_daemon_parentage_known
		}
	}

	left_portrait = {
		character = character:Segrellion_43
		animation = contemplation
	}

	right_portrait = {
		character = root
		animation = shock
	}

	immediate = {

		# Create Daemon's disputed heritage secret.
		add_secret = {
			type = secret_disputed_heritage
			target = root
		}

		add_character_flag = segrellion_daemon_parentage_known
		add_character_flag = segrellion_daemon_parentage_known_with_hugh


		# Hugh read the letter.
		# Reveal the SAME secret to him.

		random_known_secret = {
			limit = {
				secret_type = secret_disputed_heritage
				secret_target = root
			}

			reveal_to = character:Segrellion_43
		}


		# Hugh has already told Daemon.
		# Prevent AGOT from later offering Hugh a redundant
		# "tell Daemon who he really is" interaction/event.

		character:Segrellion_43 = {
			add_character_flag = told_true_parentage
			add_character_flag = segrellion_hugh_knows_daemon_parentage
		}
	}

	option = {

		name = segrellion_daemon_parentage.0003.a

		custom_tooltip = segrellion_daemon_parentage.0003.a.tt
		custom_tooltip = segrellion_daemon_parentage.reveal_tt
	}
}
