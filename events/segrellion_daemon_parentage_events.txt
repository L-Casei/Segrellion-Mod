namespace = segrellion_daemon_parentage


# =========================================================
# 0001 - HUGH FINDS DAENA'S LETTER
# =========================================================

segrellion_daemon_parentage.0001 = {

	type = character_event

	title = segrellion_daemon_parentage.0001.t
	desc = segrellion_daemon_parentage.0001.desc

	theme = intrigue

	left_portrait = root


	# =====================================================
	# A - GIVE IT TO DAEMON UNOPENED
	#
	# Hugh obeys Symon's instructions.
	# Daemon learns the truth alone.
	# =====================================================

	option = {

		name = segrellion_daemon_parentage.0001.a

		custom_tooltip = segrellion_daemon_parentage.0001.a.tt

		ai_chance = {
			base = 10
		}

		add_character_flag = segrellion_daemon_letter_given_unopened

		character:Segrellion_60 = {

			trigger_event = {
				id = segrellion_daemon_parentage.0002
				days = 2
			}
		}
	}


	# =====================================================
	# B - READ IT, THEN TELL DAEMON
	#
	# Hugh breaks the seal, learns the truth,
	# and intends to tell Daemon immediately.
	# =====================================================

	option = {

		name = segrellion_daemon_parentage.0001.b

		custom_tooltip = segrellion_daemon_parentage.0001.b.tt

		ai_chance = {
			base = 10
		}

		add_character_flag = segrellion_hugh_parentage_reveal_immediately

		trigger_event = {
			id = segrellion_daemon_parentage.0004
			days = 1
		}
	}


	# =====================================================
	# C - READ IT, BUT KEEP THE TRUTH
	#
	# Canon-favoured AI route.
	# Hugh learns the truth but does not tell Daemon yet.
	# =====================================================

	option = {

		name = segrellion_daemon_parentage.0001.c

		custom_tooltip = segrellion_daemon_parentage.0001.c.tt

		ai_chance = {
			base = 80
		}

		add_character_flag = segrellion_hugh_parentage_withhold_intent

		trigger_event = {
			id = segrellion_daemon_parentage.0004
			days = 1
		}
	}
}


# =========================================================
# 0002 - DAEMON RECEIVES THE SEALED LETTER
#
# Hugh never read it.
# Only Daemon learns the truth.
# =========================================================

segrellion_daemon_parentage.0002 = {

	type = character_event

	title = segrellion_daemon_parentage.0002.t
	desc = segrellion_daemon_parentage.0002.desc

	theme = family

	trigger = {

		this = character:Segrellion_60

		NOT = {
			has_character_flag = segrellion_daemon_parentage_known
		}
	}

	left_portrait = root


	immediate = {

		# Daemon reads Daena's letter himself and becomes
		# the sole knower of his AGOT heritage secret.

		segrellion_create_daemon_parentage_secret_effect = yes


		# Our own narrative bookkeeping.

		add_character_flag = segrellion_daemon_parentage_known
		add_character_flag = segrellion_daemon_parentage_known_daemon_only
	}


	option = {

		name = segrellion_daemon_parentage.0002.a

		custom_tooltip = segrellion_daemon_parentage.0002.a.tt
		custom_tooltip = segrellion_daemon_parentage.reveal_tt
	}
}


# =========================================================
# 0003 - DAEMON IS TOLD THE TRUTH BY HUGH
#
# Used both for:
# - Hugh revealing immediately
# - Hugh revealing years later through the decision
# =========================================================

segrellion_daemon_parentage.0003 = {

	type = character_event

	title = segrellion_daemon_parentage.0003.t
	desc = segrellion_daemon_parentage.0003.desc

	theme = family


	trigger = {

		this = character:Segrellion_60

		character:Segrellion_43 ?= {

			is_alive = yes

			has_character_flag = segrellion_hugh_read_daena_letter
		}

		NOT = {
			has_character_flag = segrellion_daemon_parentage_known
		}
	}


	left_portrait = character:Segrellion_43
	right_portrait = root


	immediate = {

		add_character_flag = segrellion_daemon_parentage_known
		add_character_flag = segrellion_daemon_parentage_known_with_hugh

		character:Segrellion_43 = {

			add_character_flag = segrellion_hugh_knows_daemon_parentage
			add_character_flag = segrellion_hugh_revealed_daemon_parentage
		}
	}


	option = {

		name = segrellion_daemon_parentage.0003.a

		custom_tooltip = segrellion_daemon_parentage.0003.a.tt
		custom_tooltip = segrellion_daemon_parentage.reveal_tt
	}
}


# =========================================================
# 0004 - HUGH BREAKS THE SEAL
#
# HUGH POV.
#
# This is the actual moment Hugh learns that
# Aerys II was Daemon's biological father.
# =========================================================

segrellion_daemon_parentage.0004 = {

	type = character_event

	title = segrellion_daemon_parentage.0004.t
	desc = segrellion_daemon_parentage.0004.desc

	theme = family


	trigger = {

		this = character:Segrellion_43

		OR = {
			has_character_flag = segrellion_hugh_parentage_reveal_immediately
			has_character_flag = segrellion_hugh_parentage_withhold_intent
		}

		character:Segrellion_60 ?= {

			is_alive = yes

			NOT = {
				has_character_flag = segrellion_daemon_parentage_known
			}
		}
	}


	left_portrait = root
	right_portrait = character:Segrellion_60


	immediate = {

		# Hugh now personally knows what was in Daena's letter.
		#
		# We deliberately DO NOT create the actual CK3 secret
		# here if Hugh intends to keep silent.
		#
		# Otherwise AGOT's own royal-bastard mechanics could
		# attempt to expose the information before our custom
		# story decides that Hugh has told Daemon.

		add_character_flag = segrellion_hugh_read_daena_letter
		add_character_flag = segrellion_hugh_knows_daemon_parentage
	}

	# =====================================================
	# TELL DAEMON NOW
	# =====================================================

	option = {

		name = segrellion_daemon_parentage.0004.a

		trigger = {
			has_character_flag = segrellion_hugh_parentage_reveal_immediately
		}

		custom_tooltip = segrellion_daemon_parentage.0004.a.tt

		remove_character_flag = segrellion_hugh_parentage_reveal_immediately


		# Hugh now possesses the REAL CK3 heritage secret.
		segrellion_create_daemon_parentage_secret_effect = yes


		# Daemon privately learns that same secret.
		random_known_secret = {

			limit = {

				secret_type = secret_agot_disputed_heritage

				secret_target = {
					this = character:Segrellion_60
				}
			}

			reveal_to = character:Segrellion_60
		}


		add_character_flag = told_true_parentage
		add_character_flag = segrellion_hugh_revealed_daemon_parentage


		character:Segrellion_60 = {

			trigger_event = {
				id = segrellion_daemon_parentage.0003
				days = 2
			}
		}
	}


	# =====================================================
	# KEEP THE TRUTH FOR NOW
	# =====================================================

	option = {

		name = segrellion_daemon_parentage.0004.b

		trigger = {
			has_character_flag = segrellion_hugh_parentage_withhold_intent
		}

		custom_tooltip = segrellion_daemon_parentage.0004.b.tt


		remove_character_flag = segrellion_hugh_parentage_withhold_intent

		add_character_flag = segrellion_hugh_withholding_daemon_parentage


		# =================================================
		# PLAYER HUGH
		#
		# Give him the REAL secret immediately.
		#
		# AGOT's native "Expose True Parentage" decision
		# can now detect Daemon.
		# =================================================

		if = {

			limit = {
				is_ai = no
			}

			segrellion_create_daemon_parentage_secret_effect = yes
			custom_tooltip = segrellion_daemon_parentage.reveal_decision_unlocked_tt
		}


		# =================================================
		# AI HUGH
		#
		# Do NOT create the Secret yet.
		#
		# Otherwise AGOT's own AI could decide to expose it
		# before our intended 302 AC timing.
		# =================================================

		else = {

			add_character_flag = segrellion_hugh_waiting_to_reveal_daemon_parentage

			# Start checking whether Hugh may hand the secret
			# over to AGOT's native true-parentage system.
			trigger_event = {
				id = segrellion_daemon_parentage.0005
				days = 1
			}
		}
	}
}

# =========================================================
# 0005 - AI HUGH MAY REVEAL DAEMON'S PARENTAGE
#
# Hugh chose to keep Daena's secret.
#
# Before 302 AC:
#   Hugh knows the truth narratively, but does NOT possess
#   the actual AGOT heritage secret, preventing premature
#   AI exposure.
#
# From 302 AC onward:
#   Hugh receives the same AGOT heritage-secret type used
#   by Howland Reed for Jon Snow.
#
# From that point onward AGOT's native
# Expose True Parentage mechanics can handle the reveal.
# =========================================================

segrellion_daemon_parentage.0005 = {

	type = character_event
	hidden = yes


	trigger = {

		this = character:Segrellion_43

		is_ai = yes

		has_character_flag = segrellion_hugh_waiting_to_reveal_daemon_parentage

		character:Segrellion_60 ?= {

			is_alive = yes

			NOT = {
				has_character_flag = segrellion_daemon_parentage_known
			}
		}
	}


	immediate = {

		# =================================================
		# 302 AC OR LATER
		#
		# Hugh now becomes a REAL knower of Daemon's
		# AGOT heritage secret, just as Howland Reed
		# knows Jon Snow's.
		# =================================================

		if = {

			limit = {
				current_date >= 302.1.1
			}

			remove_character_flag = segrellion_hugh_waiting_to_reveal_daemon_parentage

			segrellion_create_daemon_parentage_secret_effect = yes
		}


		# =================================================
		# BEFORE 302 AC
		#
		# Check again later.
		# Only Hugh runs this, so this is negligible.
		# =================================================

		else = {

			trigger_event = {
				id = segrellion_daemon_parentage.0005
				days = 30
			}
		}
	}
}
