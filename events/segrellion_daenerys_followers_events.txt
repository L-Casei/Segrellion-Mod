namespace = segrellion_daenerys_followers

# Hidden post-start repair. AGOT's adventurer initialization first removes
# ordinary courtiers and then opens "The Roads We Walk". Running this after
# time advances guarantees that the historical characters become employees
# of Daenerys' camp after that cleanup has finished.
segrellion_daenerys_followers.0001 = {
	type = character_event
	hidden = yes

	trigger = {
		this = character:Targaryen_9
		is_alive = yes
		game_start_date >= 8300.8.20
		game_start_date < 8301.1.1
		current_date < 8301.1.1
	}

	immediate = {
		add_character_flag = segrellion_daenerys_followers_delayed_event_fired
		segrellion_setup_daenerys_followers_effect = yes
	}
}
