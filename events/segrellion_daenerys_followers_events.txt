namespace = segrellion_daenerys_followers


# =========================================================
# 0001 - DELAYED HISTORICAL FOLLOWER SETUP
#
# Root = Daenerys
#
# IMPORTANT:
# This event deliberately runs after normal game-start
# initialization instead of participating in the zero-tick
# on_action chain.
#
# DEBUG VERSION:
# Keep this event visible until the system is confirmed.
# =========================================================

segrellion_daenerys_followers.0001 = {

    immediate = {

    debug_log = "[SEGRELLION DANY DEBUG] FOLLOWER EVENT FIRED"
    debug_log_date = yes
    debug_log_scopes = yes
    # todo el código que YA FUNCIONA

	type = character_event

	title = segrellion_daenerys_followers.0001.t
	desc = segrellion_daenerys_followers.0001.desc

	theme = stewardship


	trigger = {

		this = character:Targaryen_9
		is_alive = yes

		game_start_date >= 8300.8.20
		game_start_date < 8301.1.1
	}


	immediate = {

		add_character_flag = segrellion_daenerys_followers_delayed_event_fired


		# =================================================
		# REMEMBER THE SECOND SONS OWNER
		#
		# Tyrion is known to begin inside their camp.
		# We capture his employer BEFORE detaching him.
		#
		# This also lets us identify Jorah without having
		# to blindly trust Mormont_3 vs Mormont_8.
		# =================================================

		character:Lannister_8 ?= {

			employer ?= {
				save_scope_as = segrellion_second_sons_owner
			}
		}


		# =================================================
		# JORAH MORMONT
		#
		# Recent revisions have used both Mormont_3 and
		# Mormont_8.
		#
		# Only take a candidate if he currently belongs
		# to the exact same camp Tyrion belonged to.
		# =================================================

		if = {

			limit = {

				exists = scope:segrellion_second_sons_owner

				character:Mormont_3 ?= {

					is_alive = yes

					employer = scope:segrellion_second_sons_owner
				}
			}


			# Completely detach him from the Second Sons.

			character:Mormont_3 = {
				move_to_pool = yes
			}


			# Daenerys recruits him from the pool.

			add_courtier = character:Mormont_3


			character:Mormont_3 = {

				if = {
					limit = {
						is_courtier_of = root
					}

					return_to_court = yes
				}
			}
		}


		if = {

			limit = {

				exists = scope:segrellion_second_sons_owner

				character:Mormont_8 ?= {

					is_alive = yes

					employer = scope:segrellion_second_sons_owner
				}
			}


			character:Mormont_8 = {
				move_to_pool = yes
			}


			add_courtier = character:Mormont_8


			character:Mormont_8 = {

				if = {
					limit = {
						is_courtier_of = root
					}

					return_to_court = yes
				}
			}
		}


		# =================================================
		# TYRION LANNISTER
		# =================================================

		character:Lannister_8 ?= {

			if = {

				limit = {
					is_alive = yes
					exists = employer
				}

				move_to_pool = yes
			}
		}


		if = {

			limit = {

				character:Lannister_8 ?= {
					is_alive = yes
				}
			}

			add_courtier = character:Lannister_8
		}


		character:Lannister_8 ?= {

			if = {

				limit = {
					is_alive = yes
					is_courtier_of = root
				}

				return_to_court = yes
			}
		}


		# =================================================
		# BARRISTAN SELMY
		# =================================================

		character:Selmy_3 ?= {

			if = {

				limit = {
					is_alive = yes
					exists = employer
				}

				move_to_pool = yes
			}
		}


		if = {

			limit = {

				character:Selmy_3 ?= {
					is_alive = yes
				}
			}

			add_courtier = character:Selmy_3
		}


		character:Selmy_3 ?= {

			if = {

				limit = {
					is_alive = yes
					is_courtier_of = root
				}

				return_to_court = yes
			}
		}


		# =================================================
		# MISSANDEI
		# =================================================

		character:IAF_Missandei_1 ?= {

			if = {

				limit = {
					is_alive = yes
					exists = employer
				}

				move_to_pool = yes
			}
		}


		if = {

			limit = {

				character:IAF_Missandei_1 ?= {
					is_alive = yes
				}
			}

			add_courtier = character:IAF_Missandei_1
		}


		character:IAF_Missandei_1 ?= {

			if = {

				limit = {
					is_alive = yes
					is_courtier_of = root
				}

				return_to_court = yes
			}
		}


		# =================================================
		# TORGO NUDHO
		# =================================================

		character:IAF_Torgo_Nudho_1 ?= {

			if = {

				limit = {
					is_alive = yes
					exists = employer
				}

				move_to_pool = yes
			}
		}


		if = {

			limit = {

				character:IAF_Torgo_Nudho_1 ?= {
					is_alive = yes
				}
			}

			add_courtier = character:IAF_Torgo_Nudho_1
		}


		character:IAF_Torgo_Nudho_1 ?= {

			if = {

				limit = {
					is_alive = yes
					is_courtier_of = root
				}

				return_to_court = yes
			}
		}


		# =================================================
		# TYRION BECOMES THE SECOND
		# =================================================

		if = {

			limit = {

				character:Lannister_8 ?= {

					is_alive = yes

					is_courtier_of = root
				}


				NOT = {

					any_court_position_holder = {
						type = second_camp_officer
					}
				}


				is_valid_to_hire_court_position_type = second_camp_officer
			}


			appoint_court_position = {

				recipient = character:Lannister_8
				court_position = second_camp_officer
			}
		}


		add_character_flag = segrellion_daenerys_followers_delayed_completed
	}


	option = {
		name = segrellion_daenerys_followers.0001.a
	}
}
