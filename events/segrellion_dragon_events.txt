segrellion_dragon.0001 = {
	type = character_event
	title = segrellion_dragon.0001.t
	desc = segrellion_dragon.0001.desc
	theme = dragon
	override_background = { reference = segrellion_vhalerion_throne_room }

	left_portrait = {
		character = root
		animation = personality_rational
	}

	immediate = {
		every_character_artifact = {
			limit = { has_variable = segrellion_dragon_egg_artifact }
			remove_variable = dud_egg
			set_variable = { name = agot_artifact_cannot_gift value = yes }
			save_scope_as = segrellion_dragon_egg
		}
	}

	option = {
		name = segrellion_dragon.0001.a
		ai_chance = { base = 100 }
		remove_character_flag = segrellion_ancient_egg_decision_pending
		add_character_flag = segrellion_tending_ancient_egg
		segrellion_dragon_advance_egg_progress_effect = { VALUE = 1 }
		add_learning_lifestyle_xp = 100
		add_prestige = 75
		add_character_modifier = {
			modifier = segrellion_keeper_of_the_ancient_egg_modifier
			years = 3
		}
		trigger_event = {
			id = segrellion_dragon.0002
			days = 730
		}
	}

	option = {
		name = segrellion_dragon.0001.b
		ai_chance = { base = 0 }
		segrellion_dragon_pause_chain_effect = yes
	}
}

segrellion_dragon.0005 = {
	type = character_event
	window = agot_dragon_character_event
	title = segrellion_dragon.0005.t
	desc = segrellion_dragon.0005.desc
	theme = dragon
	override_background = { reference = bedchamber }

	left_portrait = {
		character = root
		animation = amazement
	}

	right_portrait = {
		character = scope:dragon
		animation = dragon_head
		camera = camera_dragon_event_standing_baby
		outfit_tags = { linear_camera_zoom }
	}

	immediate = {
		hidden_effect = {
			every_character_artifact = {
				limit = { has_variable = segrellion_dragon_egg_artifact }
				save_scope_as = segrellion_dragon_egg
			}
			if = {
				limit = { exists = scope:segrellion_dragon_egg }
				segrellion_spawn_vhalyrax_from_egg_effect = {
					OWNER = root
					EGG = scope:segrellion_dragon_egg
				}
			}
		}
	}

	option = {
		name = segrellion_dragon.0005.a
		ai_chance = { base = 100 }
		add_prestige = 1000
		add_prestige_experience = 1000
		add_legitimacy = 750
		dynasty = {
			add_dynasty_prestige = 750
			while = {
				limit = { dynasty_prestige_level < 3 }
				add_dynasty_prestige_level = 1
			}
			add_dynasty_modifier = {
				modifier = segrellion_a_dragon_reborn_dynasty_modifier
				years = 10
			}
		}
		custom_tooltip = segrellion_house_gains_dragonriding_house_tt
		hidden_effect = {
			house:house_Segrellion = {
				if = {
					limit = {
						NOT = { has_house_modifier = dragonrider_house_modifier }
					}
					add_house_modifier = dragonrider_house_modifier
				}
			}
		}
		add_stress = -60
		add_character_modifier = {
			modifier = segrellion_a_quiet_companion_modifier
			years = 10
		}
		hidden_effect = {
			remove_character_modifier = segrellion_keeper_of_the_ancient_egg_modifier
			remove_character_modifier = segrellion_dragon_dreams_modifier
			remove_character_modifier = segrellion_haunted_by_the_white_shadow_modifier
			remove_character_flag = segrellion_tending_ancient_egg
			remove_character_flag = vhalyrax_hatching_imminent
			add_character_flag = vhalyrax_hatched_secretly
			add_character_flag = segrellion_knows_vhalyrax_hatched
			character:Segrellion_2 ?= {
				if = {
					limit = { is_alive = yes }
					add_character_flag = segrellion_knows_vhalyrax_hatched
				}
			}
			character:Segrellion_3 ?= {
				if = {
					limit = { is_alive = yes }
					add_character_flag = segrellion_knows_vhalyrax_hatched
				}
			}
			character:Vladwyn_1 ?= {
				if = {
					limit = { is_alive = yes }
					add_character_flag = segrellion_knows_vhalyrax_hatched
				}
			}
		}
	}
}
