			VALUE = 12
		}
	}

	title = segrellion_icebringer.8000.t
	desc = segrellion_icebringer.8000.desc

	theme = learning

	override_background = {
		reference = segrellion_vhalerion_throne_room
	}

		character = root
		animation = wonder
	}

	option = {
		name = segrellion_icebringer.8000.a

		add_character_flag = segrellion_icebringer_stage5_entry_reached

		remove_character_flag = segrellion_icebringer_awakening_chain_active
	}
}



# =========================================================
# 8001 - TRUEST WARRIOR STAGE V ENTRY
# =========================================================

segrellion_icebringer.8001 = {
	type = character_event

	immediate = {
		segrellion_gain_icebringer_bond_xp_effect = {
			VALUE = 12
		}
	}

	title = segrellion_icebringer.8001.t
	desc = segrellion_icebringer.8001.desc

	theme = learning

	override_background = {
		reference = segrellion_vhalerion_throne_room
	}

	left_portrait = {
		character = root
		animation = wonder
	}

	option = {
		name = segrellion_icebringer.8001.a

		if = {
			limit = {
				NOT = {
					has_trait = segrellion_truest_warrior_trait
				}
			}

			add_trait = segrellion_truest_warrior_trait
		}

		add_character_flag = segrellion_icebringer_stage5_entry_reached

		remove_character_flag = segrellion_icebringer_awakening_chain_active
	}
}



# =========================================================
# 8200 - CORRUPTED ICEBRINGER
#
# DARKNESS STAGE V
# =========================================================

segrellion_icebringer.8200 = {

	type = character_event


	immediate = {

		segrellion_gain_icebringer_bond_xp_effect = {
			VALUE = 12
		}
	}


	title = segrellion_icebringer.8200.t
	desc = segrellion_icebringer.8200.desc

	theme = death


	override_background = {
		reference = segrellion_darkness
	}


	left_portrait = {
		character = root
		animation = shock
	}


	option = {

		name = segrellion_icebringer.8200.a

		custom_tooltip = segrellion_icebringer.8200.a.tt


		# =================================================
		# THE BLADE IS CORRUPTED
		# =================================================

		segrellion_corrupt_icebringer_effect = yes


		# =================================================
		# STAGE V
		# =================================================

		add_character_flag = segrellion_icebringer_stage5_entry_reached
		add_character_flag = segrellion_icebringer_stage5_darkness


		# The temporary lock has fulfilled its purpose.
		remove_character_flag = segrellion_icebringer_stage5_corrupted_locked


		remove_character_flag = segrellion_icebringer_awakening_chain_active
	}
}
