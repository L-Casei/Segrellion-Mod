	type = character_event

	title = segrellion_icebringer.8001.t
	desc = segrellion_icebringer.8001.desc

	theme = learning

	override_background = {
		reference = segrellion_vhalerion_throne_room
	}

	left_portrait = {
		character = root
		animation = wonder
	}

	option = {
		name = segrellion_icebringer.8001.a

		custom_tooltip = segrellion_truest_warrior_gain_tt

		add_character_flag = segrellion_icebringer_stage5_entry_reached

		remove_character_flag = segrellion_icebringer_awakening_chain_active
	}
}
