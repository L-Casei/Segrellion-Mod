namespace = segrellion_dragonlore

segrellion_dragonlore.0001 = {
	type = character_event
	title = segrellion_dragonlore.0001.t
	desc = segrellion_dragonlore.0001.desc
	theme = dragon
	left_portrait = root

	trigger = {
		always = no
	}

	option = {
		name = segrellion_dragonlore.0001.a

		custom_tooltip = {
			text = segrellion_dragonlore_increase_tt
			if = {
				limit = { exists = var:dragonlore }
				change_variable = {
					name = dragonlore
					add = 2
				}
			}
			else = {
				set_variable = {
					name = dragonlore
					value = 2
				}
			}
			set_variable = segrellion_firstflame_dragonlore_found
			add_character_flag = segrellion_firstflame_dragonlore_found
		}
	}
}
