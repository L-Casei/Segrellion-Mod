# =========================================================
# GREAT RECORD HALLS
# =========================================================

segrellion_vhal_investigation.0300 = {

	hidden = yes

	immediate = {

		# Restore artifact references before deciding whether
		# the physical relic can still be found.
		segrellion_find_sword_of_the_sentinel_effect = yes
		segrellion_setup_dragonspear_lance_history_effect = yes


		# Daemon always finds Daena's letter on his first
		# investigation of the Records if he does not yet
		# know his true parentage.
		if = {

			limit = {

				this = character:Segrellion_60

				NOT = {
					has_character_flag = segrellion_daemon_parentage_known
				}

				NOT = {
					has_character_flag = segrellion_daena_letter_found_in_vhal_records
				}
			}

			trigger_event = {
				id = segrellion_vhal_investigation.0390
			}
		}

		else = {

			random_list = {

				60 = {

					trigger_event = {
						id = segrellion_vhal_investigation.0310
					}
				}

				25 = {

					trigger = {

						NOT = {
							exists = global_var:segrellion_old_valyria_history_book
						}
					}

					trigger_event = {
						id = segrellion_vhal_investigation.0320
					}
				}

				15 = {

					trigger = {

						OR = {

							NOT = {
								exists = global_var:segrellion_blue_flower_diadem
							}

							NOT = {
								exists = global_var:segrellion_alysanne_memorandum
							}

							NOT = {
								exists = global_var:segrellion_sword_of_the_sentinel_global
							}

							NOT = {
								exists = global_var:segrellion_dragonspear_lance
							}
						}
					}

					trigger_event = {
						id = segrellion_vhal_investigation.0330
					}
				}
			}
		}
	}
}

	type = character_event

	title = segrellion_vhal_investigation.0390.t
	desc = segrellion_vhal_investigation.0390.desc

	theme = intrigue

	override_background = {
		reference = segrellion_vhal_inside_event_background
	}

	trigger = {

		this = character:Segrellion_60

		NOT = {
			has_character_flag = segrellion_daemon_parentage_known

			has_character_flag = segrellion_daena_letter_found_in_vhal_records

	immediate = {

		add_character_flag = segrellion_daena_letter_found_in_vhal_records

		add_character_flag = segrellion_daemon_parentage_known
		add_character_flag = segrellion_daemon_parentage_known_daemon_only

		segrellion_daemon_try_start_parentage_identity_chain_effect = yes
	}

	option = {

		name = segrellion_vhal_investigation.0390.a
	}
