# =========================================================
# 0010 - FOR THE WATCH
#
# Jon is genuinely murdered.
#
# BEFORE his death:
#
# 1. Save Longclaw.
# 2. Choose a Black Brother favourable to Jon.
# 3. Mark the mutineers.
# 4. Give the Watch temporarily to the loyal successor.
# 5. DIRECTLY schedule The Body in the Ice on him.
#
# Melisandre never becomes event root.
# =========================================================

segrellion_jon_resurrection.0010 = {

    type = character_event

    title = segrellion_jon_resurrection.0010.t
    desc = segrellion_jon_resurrection.0010.desc

    theme = death


    override_background = {
        reference = segrellion_castle_black_great_hall
    }


    override_icon = {
        reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
    }


    trigger = {

        this = character:Stark_7

        is_alive = yes

        has_title = title:k_the_wall

        has_character_flag = segrellion_jon_mutiny_pending

        has_global_variable = segrellion_jon_resurrection_story_enabled

        NOT = {
            has_global_variable = segrellion_resurrected_jon
        }
    }


    immediate = {

        # =================================================
        # SAVE LONGCLAW
        #
        # We save Jon's equipped weapon itself.
        # When he dies CK3 may give it to his heir.
        # The global variable continues pointing to the
        # exact same artifact object.
        # =================================================

        random_equipped_character_artifact = {

            limit = {
                artifact_slot_type = primary_armament
            }


            set_global_variable = {

                name = segrellion_jon_longclaw_artifact

                value = this
            }
        }


        # =================================================
        # SAVE EVERY WATCH TITLE JON HOLDS
        # =================================================

        every_held_title = {
            add_to_list = segrellion_jon_watch_titles
        }

        # =================================================
        # MARK MUTINEERS
        #
        # These characters will later be executed by Jon.
        # =================================================

        every_vassal_or_below = {

            limit = {

                is_alive = yes
                is_adult = yes

                has_trait = nightswatch

                this != root


                opinion = {
                    target = root
                    value <= -10
                }
            }


            add_character_flag = segrellion_jon_mutineer
        }


        every_courtier = {

            limit = {

                is_alive = yes
                is_adult = yes

                has_trait = nightswatch

                this != root


                opinion = {
                    target = root
                    value <= -10
                }
            }


            add_character_flag = segrellion_jon_mutineer
        }


        # =================================================
        # CHOOSE THE ACTING LORD COMMANDER
        #
        # IMPORTANT:
        # - Must already belong to the Night's Watch.
        # - Must be UNLANDED.
        # - Must NOT be one of Jon's mutineers.
        # - Prefer men who actually liked Jon.
        #
        # This naturally excludes commanders such as
        # Denys Mallister and Cotter Pyke while also
        # excluding hostile conspirators such as Bowen Marsh
        # if he has been marked as a mutineer.
        # =================================================

        ordered_courtier = {

            limit = {

                is_alive = yes
                is_adult = yes

                has_trait = nightswatch

                is_ruler = no

                NOT = {
                    has_character_flag = segrellion_jon_mutineer
                }

                opinion = {
                    target = root
                    value >= 20
                }
            }

            order_by = diplomacy

            position = 0

            check_range_bounds = no

            save_scope_as = jon_loyal_successor
        }


        # =================================================
        # FALLBACK: NEUTRAL BUT LOYAL BLACK BROTHER
        # =================================================

        if = {

            limit = {

                NOT = {
                    exists = scope:jon_loyal_successor
                }
            }


            ordered_courtier = {

                limit = {

                    is_alive = yes
                    is_adult = yes

                    has_trait = nightswatch

                    is_ruler = no

                    NOT = {
                        has_character_flag = segrellion_jon_mutineer
                    }

                    opinion = {
                        target = root
                        value >= 0
                    }
                }

                order_by = diplomacy

                position = 0

                check_range_bounds = no

                save_scope_as = jon_loyal_successor
            }
        }


        # =================================================
        # FINAL FALLBACK
        #
        # Any unlanded non-mutineer Black Brother.
        # =================================================

        if = {

            limit = {

                NOT = {
                    exists = scope:jon_loyal_successor
                }
            }


            ordered_courtier = {

                limit = {

                    is_alive = yes
                    is_adult = yes

                    has_trait = nightswatch

                    is_ruler = no

                    NOT = {
                        has_character_flag = segrellion_jon_mutineer
                    }
                }

                order_by = diplomacy

                position = 0

                check_range_bounds = no

                save_scope_as = jon_loyal_successor
            }
        }


        # =================================================
        # REPRESENTATIVE KILLER
        # =================================================

        ordered_vassal_or_below = {

            limit = {

                is_alive = yes

                has_character_flag = segrellion_jon_mutineer
            }


            order_by = intrigue

            position = 0

            check_range_bounds = no


            save_scope_as = jon_mutineer
        }


        if = {

            limit = {

                NOT = {
                    exists = scope:jon_mutineer
                }
            }


            ordered_courtier = {

                limit = {

                    is_alive = yes

                    has_trait = nightswatch

                    this != root
                    this != scope:jon_loyal_successor
                }


                order_by = intrigue

                position = 0

                check_range_bounds = no


                add_character_flag = segrellion_jon_mutineer

                save_scope_as = jon_mutineer
            }
        }
    }


    left_portrait = {

        character = root

        animation = shock
    }


    option = {

		name = segrellion_jon_resurrection.0010.a


		# =================================================
		# PREPARE SUCCESSION BEFORE JON DIES
		# =================================================

		hidden_effect = {

			add_character_flag = segrellion_jon_murdered_by_watch


			set_global_variable = {
				name = segrellion_jon_resurrection_pending
				value = yes
			}


			set_global_variable = {
				name = segrellion_jon_loyal_lord_commander
				value = character:Mallister_6
			}


			# =================================================
			# ACTING LORD COMMANDER
			# =================================================

			set_global_variable = {

				name = segrellion_jon_loyal_lord_commander

				value = scope:jon_loyal_successor
			}


			# =================================================
			# ENSURE HE IS A BLACK BROTHER
			# =================================================

			scope:jon_loyal_successor = {

				if = {

					limit = {

						NOT = {
							has_trait = nightswatch
						}
					}

					add_trait = nightswatch
				}
			}


			# =================================================
			# TRANSFER EVERY WATCH TITLE JON PERSONALLY HOLDS
			# =================================================

			create_title_and_vassal_change = {

				type = granted

				save_scope_as = jon_watch_transfer

				add_claim_on_loss = no
			}


			title:k_the_wall = {

				change_title_holder_include_vassals = {

					holder = scope:jon_loyal_successor

					change = scope:jon_watch_transfer

					government_base = root
				}
			}


			every_in_list = {

				list = segrellion_jon_watch_titles

				limit = {

					this != title:k_the_wall
				}


				change_title_holder = {

					holder = scope:jon_loyal_successor

					change = scope:jon_watch_transfer
				}
			}


			resolve_title_and_vassal_change = scope:jon_watch_transfer


			# =================================================
			# ACTING LORD COMMANDER SETUP
			# =================================================

			scope:jon_loyal_successor = {

				change_government = nights_watch_government

				set_primary_title_to = title:k_the_wall

				set_realm_capital = title:c_castle_black

				set_location = title:c_castle_black.title_province


				add_character_flag = segrellion_jon_resurrection_chain_started

				add_character_flag = segrellion_jon_acting_lord_commander


				add_character_flag = {

					flag = segrellion_body_in_ice_queued

					days = 12
				}


				trigger_event = {

					id = segrellion_jon_resurrection.0100

					days = 10
				}
			}


			# =================================================
			# PLAYER CONTINUES AS THE ACTING LORD COMMANDER
			# =================================================

			if = {

				limit = {
					is_ai = no
				}


				root = {

					set_player_character = scope:jon_loyal_successor
				}
			}


			# =================================================
			# TRANSFER THE NIGHT'S WATCH
			# =================================================

			create_title_and_vassal_change = {

				type = granted

				save_scope_as = jon_watch_transfer

				add_claim_on_loss = no
			}


			title:k_the_wall = {

				change_title_holder_include_vassals = {

					holder = character:Mallister_6

					change = scope:jon_watch_transfer

					government_base = root
				}
			}


			every_in_list = {

				list = segrellion_jon_watch_titles

				limit = {
					this != title:k_the_wall
				}


				change_title_holder = {

					holder = character:Mallister_6

					change = scope:jon_watch_transfer
				}
			}


			resolve_title_and_vassal_change = scope:jon_watch_transfer


			# =================================================
			# FORCE THE CORRECT GOVERNMENT AFTER LANDING DENYS
			# =================================================

			character:Mallister_6 = {

				change_government = nights_watch_government

				set_primary_title_to = title:k_the_wall

				set_realm_capital = title:c_castle_black

				set_location = title:c_castle_black.title_province


				add_character_flag = segrellion_jon_resurrection_chain_started


				add_character_flag = {

					flag = segrellion_body_in_ice_queued

					days = 12
				}


				# ONLY ONE COPY OF BODY IN THE ICE

				trigger_event = {

					id = segrellion_jon_resurrection.0100

					days = 10
				}
			}


			# =================================================
			# PLAYER CONTINUES AS DENYS
			# =================================================

			if = {

				limit = {
					is_ai = no
				}


				root = {
					set_player_character = character:Mallister_6
				}
			}
		}


		# =================================================
		# VISIBLE DEATH
		# =================================================

		if = {

			limit = {
				exists = scope:jon_mutineer
			}


			death = {

				death_reason = death_murder

				killer = scope:jon_mutineer
			}
		}

		else = {

			death = {
				death_reason = death_murder
			}
		}


		ai_chance = {
			base = 100
		}
	}
}

					has_variable = segrellion_stannis_northern_war
				}
			}


			trigger_event = {

				id = segrellion_jon_resurrection.0400

				days = 1
			}
		}


		# =============================================
		# STANNIS IS DEAD / DEFEATED / WAR IS OVER
		# =============================================

		else = {

			trigger_event = {

				id = segrellion_jon_resurrection.0500

				days = 1
			}
		}


		ai_chance = {
			base = 95
		}
	}


	# =====================================================
	# REMAIN LORD COMMANDER
	# =====================================================

	option = {

		name = segrellion_jon_resurrection.0300.b

		custom_tooltip = segrellion_jon_resurrection.0300.b.tt


		add_character_flag = segrellion_jon_post_resurrection_path_chosen

		add_character_flag = segrellion_jon_remained_lord_commander


		ai_chance = {
			base = 5
		}
	}
}
