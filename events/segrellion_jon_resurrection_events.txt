# =========================================================
# 0001 - WHISPERS IN THE SNOW
#
# Fallback event.
#
# Organic AGOT murder schemes have already had several
# months to kill Jon.
#
# If Jon is STILL Lord Commander, the canonical mutiny
# begins.
# =========================================================

segrellion_jon_resurrection.0001 = {

    type = character_event

    title = segrellion_jon_resurrection.0001.t
    desc = segrellion_jon_resurrection.0001.desc

    theme = intrigue


    override_background = {
        reference = segrellion_castle_black_great_hall
    }


    override_icon = {
        reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
    }


    trigger = {

        this = character:Stark_7

        is_alive = yes

        has_title = title:k_the_wall


        has_global_variable = segrellion_jon_resurrection_story_enabled


        NOT = {
            has_global_variable = segrellion_resurrected_jon
        }


        NOT = {
            has_character_flag = segrellion_jon_mutiny_pending
        }
    }


    left_portrait = {

        character = root

        animation = thinking
    }


    option = {

        name = segrellion_jon_resurrection.0001.a


        add_character_flag = segrellion_jon_mutiny_pending


        trigger_event = {

            id = segrellion_jon_resurrection.0010

            days = { 5 10 }
        }


        ai_chance = {
            base = 100
        }
    }
}


# =========================================================
# 0010 - FOR THE WATCH
#
# Jon is genuinely killed.
#
# This event is now THE canonical source of Jon's death
# for the Segrellion 300 AC storyline.
#
# Before killing Jon, Melisandre is given a delayed hidden
# router. Therefore the resurrection chain no longer
# depends exclusively on title succession on_actions.
# =========================================================

segrellion_jon_resurrection.0010 = {

    type = character_event

    title = segrellion_jon_resurrection.0010.t
    desc = segrellion_jon_resurrection.0010.desc

    theme = death


    override_background = {
        reference = segrellion_castle_black_great_hall
    }


    override_icon = {
        reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
    }


    trigger = {

        this = character:Stark_7

        is_alive = yes

        has_title = title:k_the_wall

        has_character_flag = segrellion_jon_mutiny_pending


        has_global_variable = segrellion_jon_resurrection_story_enabled


        NOT = {
            has_global_variable = segrellion_resurrected_jon
        }
    }


    immediate = {

        # =================================================
        # FIND A REPRESENTATIVE MUTINEER
        #
        # Prefer a landed/member-of-the-Watch character.
        # The actual event represents several brothers
        # attacking Jon.
        # =================================================

        ordered_vassal_or_below = {

            limit = {

                is_alive = yes
                is_adult = yes

                has_trait = nightswatch

                this != root
            }

            order_by = intrigue

            position = 0

            check_range_bounds = no

            save_scope_as = jon_mutineer
        }


        if = {

            limit = {

                NOT = {
                    exists = scope:jon_mutineer
                }
            }


            ordered_courtier = {

                limit = {

                    is_alive = yes
                    is_adult = yes

                    has_trait = nightswatch

                    this != root
                }

                order_by = intrigue

                position = 0

                check_range_bounds = no

                save_scope_as = jon_mutineer
            }
        }
    }


    left_portrait = {

        character = root

        animation = shock
    }


    option = {

        name = segrellion_jon_resurrection.0010.a


        # =================================================
        # MARK THE STORY BEFORE DEATH
        # =================================================

        add_character_flag = segrellion_jon_murdered_by_watch


        set_global_variable = {

            name = segrellion_jon_resurrection_pending

            value = yes
        }


        # =================================================
        # START THE RESURRECTION ROUTER BEFORE JON DIES
        #
        # Root of 0090 will be Melisandre, so Jon's death
        # cannot invalidate the queued event.
        # =================================================

        character:Melisandre_1 ?= {

            trigger_event = {

                id = segrellion_jon_resurrection.0090

                days = 2
            }
        }


        # =================================================
        # JON DIES
        # =================================================

        if = {

            limit = {
                exists = scope:jon_mutineer
            }


            death = {

                death_reason = death_murder

                killer = scope:jon_mutineer
            }
        }


        else = {

            death = {

                death_reason = death_murder
            }
        }


        ai_chance = {
            base = 100
        }
    }
}



# =========================================================
# 0090 - RESURRECTION HANDOFF
#
# HIDDEN
#
# ROOT = Melisandre
#
# This event waits until CK3 has finished Night's Watch
# succession and then gives the resurrection storyline
# directly to the new Lord Commander.
#
# If succession has not resolved yet, it retries.
# =========================================================

segrellion_jon_resurrection.0090 = {

    hidden = yes


    trigger = {

        this = character:Melisandre_1

        is_alive = yes


        character:Stark_7 = {

            is_alive = no

            has_character_flag = segrellion_jon_murdered_by_watch
        }


        has_global_variable = segrellion_jon_resurrection_pending


        NOT = {
            has_global_variable = segrellion_jon_resurrection_chain_handed_off
        }


        NOT = {
            has_global_variable = segrellion_resurrected_jon
        }
    }


    immediate = {

        # =================================================
        # A NEW LORD COMMANDER EXISTS
        # =================================================

        if = {

            limit = {

                title:k_the_wall = {
                    exists = holder
                }


                title:k_the_wall.holder = {

                    is_alive = yes

                    this != character:Stark_7
                }
            }


            title:k_the_wall.holder = {

                add_character_flag = segrellion_jon_resurrection_chain_started


                save_scope_as = resurrected_jon_lord_commander


                trigger_event = {

                    id = segrellion_jon_resurrection.0100

                    days = 1
                }
            }


            set_global_variable = {

                name = segrellion_jon_resurrection_chain_handed_off

                value = yes
            }


            set_global_variable = {

                name = segrellion_jon_resurrection_lord_commander

                value = scope:resurrected_jon_lord_commander
            }
        }


        # =================================================
        # SUCCESSION HAS NOT FINISHED
        #
        # Try again shortly.
        # =================================================

        else = {

            trigger_event = {

                id = segrellion_jon_resurrection.0090

                days = 2
            }
        }
    }
}

# =========================================================
# 0200 - THE WOLF OPENS HIS EYES
#
# GLOBAL PLAYER EVENT.
#
# Every human player can continue normally or switch
# immediately to the resurrected Jon.
# =========================================================

segrellion_jon_resurrection.0200 = {

    type = character_event

    title = segrellion_jon_resurrection.0200.t
    desc = segrellion_jon_resurrection.0200.desc

    theme = faith


    override_background = {
        reference = segrellion_castle_black_great_hall
    }


    override_icon = {
        reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
    }


    trigger = {

        is_ai = no

        has_global_variable = segrellion_resurrected_jon


        global_var:segrellion_resurrected_jon ?= {

            is_alive = yes

            is_playable_character = yes
        }
    }


    immediate = {

        global_var:segrellion_resurrected_jon = {
            save_scope_as = resurrected_jon
        }


        title:k_the_wall.holder ?= {
            save_scope_as = resurrected_jon_lord_commander
        }


        character:Melisandre_1 ?= {
            save_scope_as = melisandre
        }
    }


    left_portrait = {

        character = scope:resurrected_jon

        animation = thinking
    }


    right_portrait = {

        character = scope:melisandre

        animation = personality_rational
    }


    # =====================================================
    # CONTINUE CURRENT GAME
    # =====================================================

    option = {

        name = segrellion_jon_resurrection.0200.a
    }


    # =====================================================
    # PLAY AS JON
    # =====================================================

    option = {

        name = segrellion_jon_resurrection.0200.b


        trigger = {

            scope:resurrected_jon = {

                is_alive = yes

                is_ai = yes

                is_playable_character = yes
            }
        }


        custom_tooltip = segrellion_jon_resurrection.0200.b.tt


        hidden_effect = {

            root = {

                set_player_character = scope:resurrected_jon
            }
        }
    }
}
