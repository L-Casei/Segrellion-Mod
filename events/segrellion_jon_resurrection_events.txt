# =========================================================
# 0010 - FOR THE WATCH
#
# Jon is genuinely murdered.
#
# BEFORE his death:
#
# 1. Save Longclaw.
# 2. Choose a Black Brother favourable to Jon.
# 3. Mark the mutineers.
# 4. Give the Watch temporarily to the loyal successor.
# 5. DIRECTLY schedule The Body in the Ice on him.
#
# Melisandre never becomes event root.
# =========================================================

segrellion_jon_resurrection.0010 = {

    type = character_event

    title = segrellion_jon_resurrection.0010.t
    desc = segrellion_jon_resurrection.0010.desc

    theme = death


    override_background = {
        reference = segrellion_castle_black_great_hall
    }


    override_icon = {
        reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
    }


    trigger = {

        this = character:Stark_7

        is_alive = yes

        has_title = title:k_the_wall

        has_character_flag = segrellion_jon_mutiny_pending

        has_global_variable = segrellion_jon_resurrection_story_enabled

        NOT = {
            has_global_variable = segrellion_resurrected_jon
        }
    }


    immediate = {

        # =================================================
        # SAVE LONGCLAW
        #
        # We save Jon's equipped weapon itself.
        # When he dies CK3 may give it to his heir.
        # The global variable continues pointing to the
        # exact same artifact object.
        # =================================================

        random_equipped_character_artifact = {

            limit = {
                artifact_slot_type = primary_armament
            }


            set_global_variable = {

                name = segrellion_jon_longclaw_artifact

                value = this
            }
        }


        # =================================================
        # SAVE EVERY WATCH TITLE JON HOLDS
        # =================================================

        every_held_title = {
            add_to_list = segrellion_jon_watch_titles
        }


        # =================================================
        # CHOOSE TEMPORARY LORD COMMANDER
        #
        # Prefer someone who actually likes Jon.
        # =================================================

        ordered_vassal_or_below = {

            limit = {

                is_alive = yes
                is_adult = yes

                has_trait = nightswatch

                this != root


                opinion = {
                    target = root
                    value >= 25
                }
            }


            order_by = diplomacy

            position = 0

            check_range_bounds = no


            save_scope_as = jon_loyal_successor
        }


        # Courtier fallback.

        if = {

            limit = {

                NOT = {
                    exists = scope:jon_loyal_successor
                }
            }


            ordered_courtier = {

                limit = {

                    is_alive = yes
                    is_adult = yes

                    has_trait = nightswatch

                    this != root


                    opinion = {
                        target = root
                        value >= 25
                    }
                }


                order_by = diplomacy

                position = 0

                check_range_bounds = no


                save_scope_as = jon_loyal_successor
            }
        }


        # Neutral fallback.

        if = {

            limit = {

                NOT = {
                    exists = scope:jon_loyal_successor
                }
            }


            ordered_vassal_or_below = {

                limit = {

                    is_alive = yes
                    is_adult = yes

                    has_trait = nightswatch

                    this != root
                }


                order_by = diplomacy

                position = 0

                check_range_bounds = no


                save_scope_as = jon_loyal_successor
            }
        }


        if = {

            limit = {

                NOT = {
                    exists = scope:jon_loyal_successor
                }
            }


            ordered_courtier = {

                limit = {

                    is_alive = yes
                    is_adult = yes

                    has_trait = nightswatch

                    this != root
                }


                order_by = diplomacy

                position = 0

                check_range_bounds = no


                save_scope_as = jon_loyal_successor
            }
        }


        # =================================================
        # MARK MUTINEERS
        #
        # These characters will later be executed by Jon.
        # =================================================

        every_vassal_or_below = {

            limit = {

                is_alive = yes
                is_adult = yes

                has_trait = nightswatch

                this != root
                this != scope:jon_loyal_successor


                opinion = {
                    target = root
                    value <= -10
                }
            }


            add_character_flag = segrellion_jon_mutineer
        }


        every_courtier = {

            limit = {

                is_alive = yes
                is_adult = yes

                has_trait = nightswatch

                this != root
                this != scope:jon_loyal_successor


                opinion = {
                    target = root
                    value <= -10
                }
            }


            add_character_flag = segrellion_jon_mutineer
        }


        # =================================================
        # REPRESENTATIVE KILLER
        # =================================================

        ordered_vassal_or_below = {

            limit = {

                is_alive = yes

                has_character_flag = segrellion_jon_mutineer
            }


            order_by = intrigue

            position = 0

            check_range_bounds = no


            save_scope_as = jon_mutineer
        }


        if = {

            limit = {

                NOT = {
                    exists = scope:jon_mutineer
                }
            }


            ordered_courtier = {

                limit = {

                    is_alive = yes

                    has_trait = nightswatch

                    this != root
                    this != scope:jon_loyal_successor
                }


                order_by = intrigue

                position = 0

                check_range_bounds = no


                add_character_flag = segrellion_jon_mutineer

                save_scope_as = jon_mutineer
            }
        }
    }


    left_portrait = {

        character = root

        animation = shock
    }


    option = {

        name = segrellion_jon_resurrection.0010.a


        add_character_flag = segrellion_jon_murdered_by_watch


        set_global_variable = {

            name = segrellion_jon_resurrection_pending

            value = yes
        }


        # =================================================
        # SAVE TEMPORARY LORD COMMANDER
        # =================================================

        set_global_variable = {

            name = segrellion_jon_loyal_lord_commander

            value = scope:jon_loyal_successor
        }


        # =================================================
        # TRANSFER NIGHT'S WATCH
        # =================================================

        create_title_and_vassal_change = {

            type = granted

            save_scope_as = jon_watch_transfer

            add_claim_on_loss = no
        }


        title:k_the_wall = {

            change_title_holder_include_vassals = {

                holder = scope:jon_loyal_successor

                change = scope:jon_watch_transfer

                government_base = root
            }
        }


        every_in_list = {

            list = segrellion_jon_watch_titles


            limit = {
                this != title:k_the_wall
            }


            change_title_holder = {

                holder = scope:jon_loyal_successor

                change = scope:jon_watch_transfer
            }
        }


        resolve_title_and_vassal_change = scope:jon_watch_transfer


        # =================================================
        # ABSOLUTELY FORCE NIGHT'S WATCH GOVERNMENT
        #
        # Do not let CK3 infer Feudal.
        # =================================================

        scope:jon_loyal_successor = {

            change_government = nights_watch_government

            set_realm_capital = title:c_castle_black

            add_character_flag = segrellion_jon_resurrection_chain_started


            add_character_flag = {

                flag = segrellion_body_in_ice_queued

                days = 10
            }


            # =============================================
            # DIRECT EVENT HANDOFF
            #
            # THIS is the important fix.
            # =============================================

            trigger_event = {

                id = segrellion_jon_resurrection.0100

                days = 2
            }
        }


        # =================================================
        # JON DIES
        # =================================================

        if = {

            limit = {
                exists = scope:jon_mutineer
            }


            death = {

                death_reason = death_murder

                killer = scope:jon_mutineer
            }
        }


        else = {

            death = {
                death_reason = death_murder
            }
        }


        ai_chance = {
            base = 100
        }
    }
}

				type = wildling_raiders

				men = 1200
			}


			# =====================================================
			# FREE FOLK RIDERS
			# =====================================================

			men_at_arms = {

				type = wildling_cavalry

				men = 300
			}


			# =====================================================
			# REMAINING FREE FOLK / VOLUNTEERS
			# =====================================================

			location = title:c_castle_black.title_province

			war = scope:jon_stannis_northern_war


			save_scope_as = jon_returned_host

		}


		scope:jon_returned_host ?= {
			assign_commander = root
		}

		}
	}

segrellion_jon_winterfell_campaign_cb = {

	icon = invasion

	group = invasion


	combine_into_one = yes

	should_show_war_goal_subview = yes


	mutually_exclusive_titles = {
		always = yes
	}


	allow_hostages = no


	allowed_for_character = {

		has_character_flag = segrellion_canonical_current_jon

		has_character_flag = segrellion_jon_winterfell_cb_authorized
	}


	allowed_against_character = {

		scope:defender = title:e_the_north.holder
	}


	target_titles = all

	target_title_tier = empire

	target_de_jure_regions_above = no


	ignore_effect = change_title_holder


	valid_to_start = {

		scope:target = {
			this = title:e_the_north
		}
	}


	should_invalidate = {

		OR = {

			scope:attacker = {
				is_alive = no
			}

			scope:defender = {

				NOT = {
					has_title = title:e_the_north
				}
			}
		}
	}


	cost = {
	}


	# =====================================================
	# WAR DECLARATION
	# =====================================================

	on_declaration = {

		on_declared_war = yes


		scope:attacker = {

			remove_character_flag = segrellion_jon_winterfell_cb_authorized

			add_character_flag = segrellion_jon_winterfell_campaign_active


			set_variable = {

				name = segrellion_jon_winterfell_war

				value = scope:war
			}


			# =================================================
			# JON'S RETURNED HOST
			# =================================================

			spawn_army = {

				name = segrellion_jon_returned_host


				men_at_arms = {

					type = bowmen

					men = 600
				}


				men_at_arms = {

					type = pikemen_unit

					men = 400
				}


				levies = 4000


				inheritable = no


				location = title:c_castle_black.title_province

				origin = title:c_castle_black.title_province


				war = scope:war

				war_keep_on_attacker_victory = yes


				save_scope_as = jon_winterfell_host
			}
		}


		scope:jon_winterfell_host ?= {
			assign_commander = scope:attacker
		}


		# =================================================
		# THE NORTH RISES
		#
		# No AI vote.
		# No opinion requirement.
		#
		# Whoever currently represents the great Northern
		# houses abandons the Bolton regime and joins Jon.
		#
		# This remains true even if they previously lost
		# alongside Stannis.
		# =================================================

		scope:defender = {

			every_vassal = {

				limit = {

					is_alive = yes

					is_landed = yes

					this != scope:attacker

					this != scope:defender
				}


				save_temporary_scope_as = jon_northern_supporter


				scope:war = {

					if = {

						limit = {
							is_defender = scope:jon_northern_supporter
						}

						remove_participant = scope:jon_northern_supporter
					}


					if = {

						limit = {

							NOT = {
								is_attacker = scope:jon_northern_supporter
							}
						}


						add_attacker = scope:jon_northern_supporter
					}
				}


				add_character_flag = segrellion_northern_lord_supported_returned_jon
			}
		}
	}


	# =====================================================
	# VICTORY
	# =====================================================

	on_victory_desc = {
		desc = segrellion_jon_winterfell_campaign_victory_desc
	}


	on_victory = {

		scope:attacker = {

			remove_character_flag = segrellion_jon_winterfell_campaign_active

			remove_variable = segrellion_jon_winterfell_war

			add_character_flag = segrellion_jon_winterfell_campaign_victory


			ep3_become_landed_warning_effect = {

				TITLE_RECEIVER = scope:attacker

				TITLE = title:e_the_north
			}


			create_title_and_vassal_change = {

				type = conquest

				save_scope_as = jon_north_restoration_change

				add_claim_on_loss = no
			}


			title:e_the_north = {

				change_title_holder_include_vassals = {

					holder = scope:attacker

					change = scope:jon_north_restoration_change

					government_base = scope:defender
				}
			}


			title:k_winterfell = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:jon_north_restoration_change
				}
			}


			title:d_winterfell = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:jon_north_restoration_change
				}
			}


			title:c_winterfell = {

				change_title_holder = {

					holder = scope:attacker

					change = scope:jon_north_restoration_change
				}
			}


			resolve_title_and_vassal_change = scope:jon_north_restoration_change


			set_primary_title_to = title:e_the_north

			set_realm_capital = title:c_winterfell


			ep3_become_landed_cleanup_effect = yes
		}


		# Ramsay / current enemy is captured.

		if = {

			limit = {

				scope:defender = {

					is_alive = yes

					is_imprisoned = no
				}
			}


			scope:attacker = {

				imprison = {

					target = scope:defender

					type = dungeon
				}
			}
		}


		add_truce_attacker_victory_effect = yes
	}


	# =====================================================
	# WHITE PEACE
	# =====================================================

	on_white_peace_desc = {
		desc = segrellion_jon_winterfell_campaign_white_peace_desc
	}


	on_white_peace = {

		scope:attacker = {

			remove_character_flag = segrellion_jon_winterfell_campaign_active

			remove_variable = segrellion_jon_winterfell_war

			add_character_flag = segrellion_jon_winterfell_campaign_failed
		}


		add_truce_white_peace_effect = yes
	}


	# =====================================================
	# DEFEAT
	# =====================================================

	on_defeat_desc = {
		desc = segrellion_jon_winterfell_campaign_defeat_desc
	}


	on_defeat = {

		scope:attacker = {

			remove_character_flag = segrellion_jon_winterfell_campaign_active

			remove_variable = segrellion_jon_winterfell_war

			add_character_flag = segrellion_jon_winterfell_campaign_failed
		}


		add_truce_attacker_defeat_effect = yes
	}


	on_invalidated = {

		scope:attacker = {

			remove_character_flag = segrellion_jon_winterfell_cb_authorized

			remove_character_flag = segrellion_jon_winterfell_campaign_active

			remove_variable = segrellion_jon_winterfell_war
		}
	}


	transfer_behavior = transfer


	on_primary_attacker_death = invalidate

	on_primary_defender_death = inherit


	attacker_allies_inherit = yes

	defender_allies_inherit = yes


	war_name = "segrellion_jon_winterfell_campaign_war_name"

	war_name_base = "segrellion_jon_winterfell_campaign_war_name_base"

	cb_name = "segrellion_jon_winterfell_campaign_cb_name"


	interface_priority = 100


	use_de_jure_wargoal_only = yes


	attacker_wargoal_percentage = 0.8


	max_attacker_score_from_battles = 75

	max_defender_score_from_occupation = 150

	max_attacker_score_from_occupation = 150
}
