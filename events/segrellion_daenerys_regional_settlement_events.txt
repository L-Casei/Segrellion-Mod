# =========================================================
# 0100 - DORNE
# =========================================================

segrellion_daenerys_post_throne.0100 = {

	type = character_event

	title = segrellion_daenerys_post_throne.0100.t


	desc = {

		first_valid = {

			# DORNE HELPED DAENERYS
			triggered_desc = {

				trigger = {

					OR = {

						has_character_flag = segrellion_daenerys_secret_dornish_support_against_aegon

						title:e_dorne.holder ?= {
							has_character_flag = segrellion_dorne_secretly_supports_daenerys
						}
					}
				}

				desc = segrellion_daenerys_post_throne.0100.desc_ally
			}


			# AEGON + ARIANNE HAVE A LIVING CHILD
			triggered_desc = {

				trigger = {
					has_variable = segrellion_daenerys_dorne_aegon_child
				}

				desc = segrellion_daenerys_post_throne.0100.desc_child
			}


			# ARIANNE / DORNE SUPPORTED AEGON
			triggered_desc = {

				trigger = {

					OR = {

						character:IAF_Young_Griff_1 = {
							has_character_flag = segrellion_young_griff_married_arianne
						}

						character:Martell_9 = {
							has_character_flag = segrellion_young_griff_married_arianne
						}

						character:IAF_Young_Griff_1 = {
							has_character_flag = segrellion_young_griff_dorne_support_secured
						}

						title:e_dorne.holder ?= {
							has_character_flag = segrellion_dorne_supports_young_griff
						}
					}
				}

				desc = segrellion_daenerys_post_throne.0100.desc_aegon
			}


			desc = segrellion_daenerys_post_throne.0100.desc
		}
	}


	theme = crown

	override_background = {
		reference = segrellion_red_keep_throne_room_event_background
	}


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		this = character:Targaryen_9

		is_alive = yes

		has_title = title:h_the_iron_throne

		has_character_flag = segrellion_daenerys_westerlands_resolved

		NOT = {
			has_character_flag = segrellion_daenerys_dorne_resolved
		}

		title:e_dorne = {
			exists = holder
		}
	}


	immediate = {

		remove_character_flag = segrellion_daenerys_dorne_issue_scheduled

		remove_variable = segrellion_daenerys_dorne_aegon_child


		title:e_dorne.holder = {
			save_scope_as = segrellion_dorne_holder
		}


		# =================================================
		# FIND A LIVING CHILD OF AEGON + ARIANNE
		# =================================================

		character:Martell_9 ?= {

			random_child = {

				limit = {

					is_alive = yes

					father = character:IAF_Young_Griff_1
				}


				save_scope_as = segrellion_aegon_arianne_child
			}
		}


		if = {

			limit = {
				exists = scope:segrellion_aegon_arianne_child
			}


			set_variable = {

				name = segrellion_daenerys_dorne_aegon_child

				value = scope:segrellion_aegon_arianne_child
			}
		}
	}


	left_portrait = {
		character = root
		animation = personality_rational
	}


	right_portrait = {
		character = scope:segrellion_dorne_holder
		animation = worry
	}


	# =====================================================
	# MERCY + LOYALTY
	# =====================================================

	option = {

		name = segrellion_daenerys_post_throne.0100.a


		if = {

			limit = {
				has_variable = segrellion_daenerys_dorne_aegon_child
			}

			segrellion_daenerys_spare_dornish_aegon_child_effect = yes
		}


		add_character_flag = segrellion_daenerys_dorne_mercy_offered


		if = {

			limit = {
				scope:segrellion_dorne_holder = root
			}


			segrellion_daenerys_dorne_resolved_effect = yes
		}


		else = {

			scope:segrellion_dorne_holder = {

				trigger_event = {
					id = segrellion_daenerys_post_throne.0110
				}
			}
		}


		ai_chance = {
			base = 70
		}
	}


	# =====================================================
	# DORNE HELPED DAENERYS
	# MERCY + COUNCIL POSITION
	# =====================================================

	option = {

		name = segrellion_daenerys_post_throne.0100.b


		trigger = {

			OR = {

				has_character_flag = segrellion_daenerys_secret_dornish_support_against_aegon

				scope:segrellion_dorne_holder = {
					has_character_flag = segrellion_dorne_secretly_supports_daenerys
				}

				scope:segrellion_dorne_holder = {
					is_allied_to = root
				}
			}
		}


		segrellion_daenerys_spare_dornish_aegon_child_effect = yes

		segrellion_daenerys_dorne_swears_fealty_effect = yes


		scope:segrellion_dorne_holder = {

			add_opinion = {

				target = root

				modifier = segrellion_daenerys_rewarded_loyal_dorne_opinion
			}


			trigger_event = {
				id = segrellion_daenerys_post_throne.0112
			}
		}


		segrellion_change_daenerys_pacification_tyranny_effect = {
			VALUE = -5
		}


		segrellion_daenerys_dorne_resolved_effect = yes


		ai_chance = {
			base = 100
		}
	}


	# =====================================================
	# FEED AEGON'S CHILD TO DROGON
	# =====================================================

	option = {

		name = segrellion_daenerys_post_throne.0100.c


		trigger = {

			has_variable = segrellion_daenerys_dorne_aegon_child

			character:dragon_drogon ?= {
				is_alive = yes
			}
		}


		segrellion_daenerys_devour_dornish_aegon_child_effect = yes

		segrellion_daenerys_dorne_swears_fealty_effect = yes

		segrellion_daenerys_dorne_resolved_effect = yes


		ai_chance = {

			base = 1

			modifier = {
				factor = 5
				has_trait = sadistic
			}

			modifier = {
				factor = 3
				has_trait = callous
			}

			modifier = {
				factor = 3
				has_trait = wrathful
			}
		}
	}


	# =====================================================
	# NOTHING SPECIAL HAPPENED IN DORNE
	# =====================================================

	option = {

		name = segrellion_daenerys_post_throne.0100.d


		trigger = {

			NOR = {

				has_variable = segrellion_daenerys_dorne_aegon_child

				has_character_flag = segrellion_daenerys_secret_dornish_support_against_aegon

				character:IAF_Young_Griff_1 = {
					has_character_flag = segrellion_young_griff_married_arianne
				}

				character:IAF_Young_Griff_1 = {
					has_character_flag = segrellion_young_griff_dorne_support_secured
				}
			}
		}


		segrellion_daenerys_dorne_swears_fealty_effect = yes

		segrellion_daenerys_dorne_resolved_effect = yes


		ai_chance = {
			base = 100
		}
	}
}

segrellion_daenerys_post_throne.0300 = {

	type = character_event

	hidden = yes


	trigger = {

		this = character:Targaryen_9

		is_alive = yes

			has_character_flag = segrellion_daenerys_riverlands_resolved
		}

			exists = holder
		}
	}


	immediate = {

		remove_character_flag = segrellion_daenerys_riverlands_issue_scheduled


		title:e_the_riverlands.holder = {
			save_scope_as = segrellion_riverlands_holder
		}


		# =================================================
		# JON HOLDS THE RIVERLANDS
		# DEFER EVERYTHING TO HIS STORY
		# =================================================

		if = {

			limit = {

				scope:segrellion_riverlands_holder = {
					segrellion_is_current_jon_trigger = yes
				}
