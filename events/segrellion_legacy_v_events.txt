# 5010 - CHOOSE THE FAVOR
# ============================================================

segrellion_legacy.5010 = {
	type = character_event
	title = segrellion_legacy.5010.t
	desc = segrellion_legacy.5010.desc
	theme = diplomacy

	left_portrait = {
		character = root
		animation = personality_honorable
	}

	right_portrait = {
		character = scope:segrellion_favor_target
	}


	# --------------------------------------------------------
	# GOLD
	# --------------------------------------------------------

	option = {
		trigger = {
			scope:segrellion_favor_target = {
				gold >= 500
			}
		}

		name = segrellion_legacy.5010.gold

		add_character_flag = {
			flag = segrellion_great_house_favor_cooldown
			years = 2
		}

		save_scope_as = segrellion_requester

		scope:segrellion_favor_target = {
			trigger_event = segrellion_legacy.5100
		}
	}


	# --------------------------------------------------------
	# WAR
	# Prioritises defensive wars.
	# --------------------------------------------------------

	option = {
		trigger = {
			any_character_war = {
				OR = {
					primary_attacker = root
					primary_defender = root
				}
			}
		}

		name = segrellion_legacy.5010.war

		if = {
			limit = {
				any_character_war = {
					primary_defender = root
				}
			}

			random_character_war = {
				limit = {
					primary_defender = root
				}
				save_scope_as = segrellion_requested_war
			}
		}

		else = {
			random_character_war = {
				limit = {
					primary_attacker = root
				}
				save_scope_as = segrellion_requested_war
			}
		}

		add_character_flag = {
			flag = segrellion_great_house_favor_cooldown
			years = 2
		}

		save_scope_as = segrellion_requester

		scope:segrellion_favor_target = {
			trigger_event = segrellion_legacy.5110
		}
	}


	# --------------------------------------------------------
	# MARRIAGE PROMISE
	# Gives a favor hook which may be spent through the
	# normal Arrange Marriage interaction.
	# --------------------------------------------------------

	option = {
		trigger = {
			can_add_hook = {
				target = scope:segrellion_favor_target
				type = favor_hook
			}
		}

		name = segrellion_legacy.5010.marriage

		add_character_flag = {
			flag = segrellion_great_house_favor_cooldown
			years = 2
		}

		save_scope_as = segrellion_requester

		scope:segrellion_favor_target = {
			trigger_event = segrellion_legacy.5120
		}
	}


	option = {
		name = segrellion_legacy.5010.back
		trigger_event = segrellion_legacy.5001
	}

	option = {
		name = segrellion_legacy.cancel
	}
}

					has_trait = arbitrary
				}
			}

				add = -20
				has_trait = honest
				scope:segrellion_requester = {
					has_trait = deceitful
				}
			}


			# MONEY-SPECIFIC PERSONALITY

			modifier = {
				add = 35

			}

			modifier = {
				add = -50

			}

			}

			}

				}
			}

				}

		}

# 5110 - WAR REQUEST
# ============================================================

segrellion_legacy.5110 = {
	type = character_event
	title = segrellion_legacy.5110.t
	desc = segrellion_legacy.5110.desc
	theme = battle

	left_portrait = {
		character = scope:segrellion_requester
	}

	right_portrait = {
		character = root
	}


	option = {
		name = segrellion_legacy.accept_war

		scope:segrellion_requested_war = {

			if = {
				limit = {
					is_attacker = scope:segrellion_requester
				}

				add_attacker = root
			}

			else_if = {
				limit = {
					is_defender = scope:segrellion_requester
				}

				add_defender = root
			}
		}

		scope:segrellion_requester = {
			trigger_event = segrellion_legacy.5210
		}

		ai_chance = {
			base = 40

			modifier = {
				add = 60
				opinion = {
					target = scope:segrellion_requester
					value >= 80
				}
			}

			modifier = {
				add = 30
				opinion = {
					target = scope:segrellion_requester
					value >= 50
				}
			}

			modifier = {
				add = -30
				opinion = {
					target = scope:segrellion_requester
					value < 0
				}
			}

			modifier = {
				add = 50
				has_relation_friend = scope:segrellion_requester
			}

			modifier = {
				add = -80
				has_relation_rival = scope:segrellion_requester
			}

			modifier = {
				factor = 0
				has_relation_nemesis = scope:segrellion_requester
			}


			# Personality

			modifier = {
				add = 30
				has_trait = brave
			}

			modifier = {
				add = -50
				has_trait = craven
			}

			modifier = {
				add = 15
				has_trait = ambitious
			}

			modifier = {
				add = -15
				has_trait = content
			}

			modifier = {
				add = 10
				has_trait = brave

				scope:segrellion_requester = {
					has_trait = brave
				}
			}

			modifier = {
				add = -20
				has_trait = brave

				scope:segrellion_requester = {
					has_trait = craven
				}
			}


			# Defensive wars receive much more sympathy.

			modifier = {
				add = 30

				scope:segrellion_requested_war = {
					primary_defender = scope:segrellion_requester
				}
			}


			# Already fighting elsewhere.

			modifier = {
				add = -30
				is_at_war = yes
			}


			modifier = {
				add = 15

				scope:segrellion_requester = {
					diplomacy >= 15
				}
			}

			modifier = {
				add = 20

				scope:segrellion_requester = {
					legitimacy_level >= 5
				}
			}
		}
	}


	option = {
		name = segrellion_legacy.refuse_war

		scope:segrellion_requester = {
			trigger_event = segrellion_legacy.5211
		}

		ai_chance = {
			base = 60
		}
	}
}
