namespace = segrellion_legacy


# ============================================================
# 5110
# THE UNICORN CALLS
#
# ROOT = Lord Paramount receiving the request
# scope:segrellion_great_house_caller = Segrellion House Head
# ============================================================

segrellion_legacy.5110 = {

	type = character_event

	title = segrellion_legacy.5110.t
	desc = segrellion_legacy.5110.desc

	theme = battle


	trigger = {

		is_alive = yes
		is_ruler = yes

		NOT = {
			is_at_war_with = scope:segrellion_great_house_caller
		}

		scope:segrellion_great_house_caller ?= {

			is_alive = yes
			is_at_war = yes

			any_character_war = {

				segrellion_legacy_v_callable_war_trigger = {
					CALLER = scope:segrellion_great_house_caller
					TARGET = root
				}
			}
		}
	}


	left_portrait = {
		character = scope:segrellion_great_house_caller
		animation = personality_bold
	}

	right_portrait = {
		character = root
		animation = personality_rational
	}


	# ========================================================
	# ACCEPT
	# ========================================================

	option = {

		name = segrellion_legacy.5110.accept


		# Keep this lord available for the response event.
		save_scope_as = segrellion_answering_great_lord


		# ====================================================
		# JOIN EVERY ELIGIBLE WAR
		#
		# If already in one of the wars, that war did not pass
		# the callable trigger and is simply ignored.
		# ====================================================

		scope:segrellion_great_house_caller = {

			every_character_war = {

				limit = {

					segrellion_legacy_v_callable_war_trigger = {
						CALLER = scope:segrellion_great_house_caller
						TARGET = root
					}
				}


				# ---------------------------------------------
				# CALLER IS AN ATTACKER
				# ---------------------------------------------

				if = {

					limit = {
						is_attacker = scope:segrellion_great_house_caller
					}

					set_called_to = root
					add_attacker = root
				}


				# ---------------------------------------------
				# CALLER IS A DEFENDER
				# ---------------------------------------------

				else_if = {

					limit = {
						is_defender = scope:segrellion_great_house_caller
					}

					set_called_to = root
					add_defender = root
				}
			}
		}


		scope:segrellion_great_house_caller = {
			trigger_event = segrellion_legacy.5111
		}


		# ====================================================
		# AI
		# ====================================================

		ai_chance = {

			base = 50


			# =================================================
			# OPINION OF THE SEGRELLION CALLER
			# =================================================

			modifier = {
				add = 50

				opinion = {
					target = scope:segrellion_great_house_caller
					value >= 80
				}
			}

			modifier = {
				add = 30

				opinion = {
					target = scope:segrellion_great_house_caller
					value >= 50
				}
			}

			modifier = {
				add = 15

				opinion = {
					target = scope:segrellion_great_house_caller
					value >= 25
				}
			}

			modifier = {
				add = -25

				opinion = {
					target = scope:segrellion_great_house_caller
					value < 0
				}
			}

			modifier = {
				add = -50

				opinion = {
					target = scope:segrellion_great_house_caller
					value <= -50
				}
			}


			# =================================================
			# PERSONAL RELATIONSHIP WITH CALLER
			# =================================================

			modifier = {
				add = 30
				has_relation_friend = scope:segrellion_great_house_caller
			}

			modifier = {
				add = 55
				has_relation_best_friend = scope:segrellion_great_house_caller
			}

			modifier = {
				add = 35
				is_close_family_of = scope:segrellion_great_house_caller
			}

			modifier = {
				add = 30
				is_allied_to = scope:segrellion_great_house_caller
			}

			modifier = {
				add = -50
				has_relation_rival = scope:segrellion_great_house_caller
			}

			modifier = {
				factor = 0.10
				has_relation_nemesis = scope:segrellion_great_house_caller
			}


			# =================================================
			# PERSONALITY
			# =================================================

			modifier = {
				add = 20
				has_trait = brave
			}

			modifier = {
				add = -30
				has_trait = craven
			}

			modifier = {
				add = 10
				has_trait = ambitious
			}

			modifier = {
				add = -10
				has_trait = content
			}

			modifier = {
				add = 10
				has_trait = loyal
			}

			modifier = {
				add = -10
				has_trait = paranoid
			}


			# =================================================
			# SHARED PERSONALITY
			# =================================================

			modifier = {

				add = 12

				has_trait = brave

				scope:segrellion_great_house_caller = {
					has_trait = brave
				}
			}

			modifier = {

				add = 10

				has_trait = just

				scope:segrellion_great_house_caller = {
					has_trait = just
				}
			}

			modifier = {

				add = 8

				has_trait = honest

				scope:segrellion_great_house_caller = {
					has_trait = honest
				}
			}

			modifier = {

				add = 8

				has_trait = compassionate

				scope:segrellion_great_house_caller = {
					has_trait = compassionate
				}
			}

			modifier = {

				add = 8

				has_trait = ambitious

				scope:segrellion_great_house_caller = {
					has_trait = ambitious
				}
			}


			# =================================================
			# DEFENDING HOUSE SEGRELLION
			#
			# Lords are more willing to intervene if at least
			# one of the wars is defensive.
			# =================================================

			modifier = {

				add = 25

				scope:segrellion_great_house_caller = {

					any_character_war = {

						segrellion_legacy_v_callable_war_trigger = {
							CALLER = scope:segrellion_great_house_caller
							TARGET = root
						}

						is_defender = scope:segrellion_great_house_caller
					}
				}
			}


			# =================================================
			# THE LORD IS ALREADY BUSY
			# =================================================

			modifier = {
				add = -20
				is_at_war = yes
			}


			# =================================================
			# CALLER'S DIPLOMACY
			# =================================================

			modifier = {

				add = 10

				scope:segrellion_great_house_caller = {
					diplomacy >= 15
				}
			}

			modifier = {

				add = 10

				scope:segrellion_great_house_caller = {
					diplomacy >= 20
				}
			}


			# =================================================
			# CALLER'S LEGITIMACY
			# =================================================

			modifier = {

				add = 10

				scope:segrellion_great_house_caller = {
					legitimacy_level >= 4
				}
			}

			modifier = {

				add = 10

				scope:segrellion_great_house_caller = {
					legitimacy_level >= 5
				}
			}


			# =================================================
			# CULTURE / FAITH
			# =================================================

			modifier = {
				add = 5
				culture = scope:segrellion_great_house_caller.culture
			}

			modifier = {
				add = 5
				faith = scope:segrellion_great_house_caller.faith
			}


			# =================================================
			# ENEMY MILITARY STRENGTH
			#
			# If any eligible enemy is stronger than this
			# Lord Paramount, intervention becomes less
			# attractive.
			# =================================================

			modifier = {

				add = -20

				scope:segrellion_great_house_caller = {

					any_character_war = {

						segrellion_legacy_v_callable_war_trigger = {
							CALLER = scope:segrellion_great_house_caller
							TARGET = root
						}

						OR = {

							AND = {

								is_attacker = scope:segrellion_great_house_caller

								primary_defender = {
									current_military_strength > root.current_military_strength
								}
							}

							AND = {

								is_defender = scope:segrellion_great_house_caller

								primary_attacker = {
									current_military_strength > root.current_military_strength
								}
							}
						}
					}
				}
			}


			# Enemy at least ~50% stronger.
			modifier = {

				add = -30

				scope:segrellion_great_house_caller = {

					any_character_war = {

						segrellion_legacy_v_callable_war_trigger = {
							CALLER = scope:segrellion_great_house_caller
							TARGET = root
						}

						OR = {

							AND = {

								is_attacker = scope:segrellion_great_house_caller

								primary_defender = {
									current_military_strength >= root.one_and_a_half_times_current_military_strength
								}
							}

							AND = {

								is_defender = scope:segrellion_great_house_caller

								primary_attacker = {
									current_military_strength >= root.one_and_a_half_times_current_military_strength
								}
							}
						}
					}
				}
			}


			# At least one enemy is weaker than the lord being called.
			modifier = {

				add = 12

				scope:segrellion_great_house_caller = {

					any_character_war = {

						segrellion_legacy_v_callable_war_trigger = {
							CALLER = scope:segrellion_great_house_caller
							TARGET = root
						}

						OR = {

							AND = {

								is_attacker = scope:segrellion_great_house_caller

								primary_defender = {
									current_military_strength < root.current_military_strength
								}
							}

							AND = {

								is_defender = scope:segrellion_great_house_caller

								primary_attacker = {
									current_military_strength < root.current_military_strength
								}
							}
						}
					}
				}
			}


			# =================================================
			# FRIEND / FAMILY OF AN ENEMY
			#
			# Major reluctance.
			# =================================================

			modifier = {

				factor = 0.25

				scope:segrellion_great_house_caller = {

					any_character_war = {

						segrellion_legacy_v_callable_war_trigger = {
							CALLER = scope:segrellion_great_house_caller
							TARGET = root
						}

						OR = {

							AND = {

								is_attacker = scope:segrellion_great_house_caller

								primary_defender = {

									OR = {
										has_relation_friend = root
										has_relation_best_friend = root
										has_relation_lover = root
										has_relation_soulmate = root
										is_close_family_of = root
									}
								}
							}

							AND = {

								is_defender = scope:segrellion_great_house_caller

								primary_attacker = {

									OR = {
										has_relation_friend = root
										has_relation_best_friend = root
										has_relation_lover = root
										has_relation_soulmate = root
										is_close_family_of = root
									}
								}
							}
						}
					}
				}
			}


			# =================================================
			# ALLIED TO AN ENEMY
			# =================================================

			modifier = {

				factor = 0.30

				scope:segrellion_great_house_caller = {

					any_character_war = {

						segrellion_legacy_v_callable_war_trigger = {
							CALLER = scope:segrellion_great_house_caller
							TARGET = root
						}

						OR = {

							AND = {

								is_attacker = scope:segrellion_great_house_caller

								primary_defender = {
									is_allied_to = root
								}
							}

							AND = {

								is_defender = scope:segrellion_great_house_caller

								primary_attacker = {
									is_allied_to = root
								}
							}
						}
					}
				}
			}


			# =================================================
			# RIVAL / NEMESIS OF AN ENEMY
			# =================================================

			modifier = {

				add = 50

				scope:segrellion_great_house_caller = {

					any_character_war = {

						segrellion_legacy_v_callable_war_trigger = {
							CALLER = scope:segrellion_great_house_caller
							TARGET = root
						}

						OR = {

							AND = {

								is_attacker = scope:segrellion_great_house_caller

								primary_defender = {
									has_relation_rival = root
								}
							}

							AND = {

								is_defender = scope:segrellion_great_house_caller

								primary_attacker = {
									has_relation_rival = root
								}
							}
						}
					}
				}
			}

			modifier = {

				add = 80

				scope:segrellion_great_house_caller = {

					any_character_war = {

						segrellion_legacy_v_callable_war_trigger = {
							CALLER = scope:segrellion_great_house_caller
							TARGET = root
						}

						OR = {

							AND = {

								is_attacker = scope:segrellion_great_house_caller

								primary_defender = {
									has_relation_nemesis = root
								}
							}

							AND = {

								is_defender = scope:segrellion_great_house_caller

								primary_attacker = {
									has_relation_nemesis = root
								}
							}
						}
					}
				}
			}


			# =================================================
			# WAR AGAINST THEIR LIEGE
			#
			# Not mathematically impossible, but extremely
			# unlikely.
			# =================================================

			modifier = {

				factor = 0.08

				root = {
					exists = liege
				}

				scope:segrellion_great_house_caller = {

					any_character_war = {

						segrellion_legacy_v_callable_war_trigger = {
							CALLER = scope:segrellion_great_house_caller
							TARGET = root
						}

						OR = {

							AND = {

								is_attacker = scope:segrellion_great_house_caller

								primary_defender = {
									this = root.liege
								}
							}

							AND = {

								is_defender = scope:segrellion_great_house_caller

								primary_attacker = {
									this = root.liege
								}
							}
						}
					}
				}
			}
		}
	}


	# ========================================================
	# REFUSE
	# ========================================================

	option = {

		name = segrellion_legacy.5110.refuse

		save_scope_as = segrellion_answering_great_lord

		scope:segrellion_great_house_caller = {
			trigger_event = segrellion_legacy.5112
		}

		ai_chance = {
			base = 50
		}
	}
}
