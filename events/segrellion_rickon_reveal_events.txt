namespace = segrellion_rickon_reveal


# =========================================================
# 0001 - RICKON IS WITH DAEMON
# =========================================================

segrellion_rickon_reveal.0001 = {

	type = letter_event

	sender = character:Segrellion_60

	opening = segrellion_rickon_reveal.0001.opening


	desc = {

		first_valid = {

			triggered_desc = {

				trigger = {
					this = character:Bolton_5
				}

				desc = segrellion_rickon_reveal.0001.desc_ramsay
			}


			triggered_desc = {

				trigger = {
					this = character:Stark_7
				}

				desc = segrellion_rickon_reveal.0001.desc_jon
			}


			triggered_desc = {

				trigger = {

					title:e_the_north.holder ?= {
						this = root
					}

					this != character:Bolton_5
					this != character:Stark_7
				}

				desc = segrellion_rickon_reveal.0001.desc_ruler
			}


			desc = segrellion_rickon_reveal.0001.desc
		}
	}


	option = {
		name = segrellion_rickon_reveal.0001.a
	}
}
