segrellion_beacon.1001 = {

	type = character_event

	}

	override_background = {
		reference = segrellion_the_beacon
	}

	left_portrait = root


	option = {

		name = segrellion_beacon.1001.a

		trigger = {
			gold >= 150
		}

		remove_short_term_gold = 150
		add_prestige = 150
		add_stewardship_lifestyle_xp = 125

		title:c_farring_cross = {
			add_county_modifier = {
				modifier = segrellion_blue_gardens_restored_modifier

			}
		}

		if = {
			limit = {
				has_royal_court = yes
			}

			change_current_court_grandeur = 5
		}

		random = {

			chance = 8

			custom_tooltip = segrellion_beacon_old_object_found_tt

			segrellion_beacon_add_lore_effect = {
				AMOUNT = 1
			}

			add_prestige = 50
		}
	}


	option = {

		name = segrellion_beacon.1001.b

		add_stress = -15

		title:c_farring_cross = {
			add_county_modifier = {
				modifier = segrellion_blue_gardens_wild_modifier
				years = 5
			}
		}
	}


	option = {

		name = segrellion_beacon.1001.c

		add_gold = 75
		add_prestige = -75

		title:c_farring_cross = {
			change_county_control = 5
		}

		dynasty:dynn_Segrellion = {

			every_dynasty_member = {

				limit = {
					is_alive = yes
					this != root
				}

				add_opinion = {
					target = root

				}
			}
		}
	}
