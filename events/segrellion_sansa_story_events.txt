# =========================================================
# 0100 - YOHN ROYCE'S FIRST LETTER
# Root = Daemon
# =========================================================

segrellion_sansa.0100 = {

	type = letter_event

	sender = title:c_runestone.holder

	opening = segrellion_sansa.0100.opening

	title = segrellion_sansa.0100.t
	desc = segrellion_sansa.0100.desc

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		this = character:Segrellion_60

		is_alive = yes

		has_character_flag = segrellion_vale_support_pact
		has_character_flag = segrellion_unicorn_returns_completed

		NOT = {
			has_character_flag = segrellion_sansa_royce_first_letter_seen
		}

		exists = character:Stark_Dummy_8
		character:Stark_Dummy_8 = {
			age >= 16
			is_alive = yes
			NOT = {
				has_character_flag = segrellion_sansa_identity_revealed
			}
		}

		exists = title:c_runestone.holder
		title:c_runestone.holder = {
			is_alive = yes
		}
	}


	immediate = {

		remove_character_flag = segrellion_sansa_royce_first_letter_scheduled

		add_character_flag = segrellion_sansa_royce_first_letter_seen


		set_global_variable = {

			name = segrellion_sansa_yohn_royce

			value = title:c_runestone.holder
		}
	}


	# =====================================================
	# YES
	# AI = 90%
	# =====================================================

	option = {

		name = segrellion_sansa.0100.a

		add_character_flag = segrellion_sansa_royce_preliminary_agreement
		add_character_flag = { flag = segrellion_sansa_royce_second_letter_scheduled days = 75 }


		trigger_event = {

			id = segrellion_sansa.0110

			days = { 30 60 }
		}


		ai_chance = {
			base = 90
		}
	}


	# =====================================================
	# NO
	# AI = 10%
	# =====================================================

	option = {

		name = segrellion_sansa.0100.b

		add_character_flag = segrellion_sansa_royce_route_refused


		ai_chance = {
			base = 10
		}
	}
}

# =========================================================
# 0140 - THE PRICE OF A BANNER
# =========================================================

segrellion_sansa.0140 = {

	type = character_event

	title = segrellion_sansa.0140.t

	desc = {

		first_valid = {

			triggered_desc = {
				trigger = {
					has_character_flag = segrellion_sansa_price_marriage_offer
				}

				desc = segrellion_sansa.0140.desc_marriage
			}

			triggered_desc = {
				trigger = {
					has_character_flag = segrellion_sansa_price_debt_offer
				}

				desc = segrellion_sansa.0140.desc_debt
			}
		}
	}

	theme = diplomacy

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	trigger = {

		this = character:Segrellion_60
		is_alive = yes

		has_character_flag = segrellion_sansa_war_price_requested

		NOT = {
			has_character_flag = segrellion_sansa_war_appeal_resolved
		}

		character:Stark_Dummy_8 ?= {
			is_alive = yes
		}

		has_variable = segrellion_sansa_vale_war
	}

	immediate = {

		remove_character_flag = segrellion_sansa_price_marriage_offer
		remove_character_flag = segrellion_sansa_price_debt_offer

		if = {
			limit = {

				is_married = no

				character:Stark_Dummy_8 = {
					is_married = no
				}
			}

			if = {
				limit = {
					has_character_flag = segrellion_vale_debt_active
				}

				random_list = {

					90 = {
						add_character_flag = segrellion_sansa_price_marriage_offer
					}

					10 = {
						add_character_flag = segrellion_sansa_price_debt_offer
					}
				}
			}

			else = {
				add_character_flag = segrellion_sansa_price_marriage_offer
			}
		}

		else = {
			add_character_flag = segrellion_sansa_price_debt_offer
		}
	}

	left_portrait = {
		character = root
		animation = thinking
	}

	right_portrait = {
		character = character:Stark_Dummy_8
		animation = personality_rational
	}


	# =====================================================
	# SANSA OFFERS MARRIAGE
	# =====================================================

	option = {

		name = segrellion_sansa.0140.a

		trigger = {
			has_character_flag = segrellion_sansa_price_marriage_offer
		}

		custom_tooltip = segrellion_sansa.0140.a.tt

		remove_character_flag = segrellion_sansa_war_price_requested

		add_character_flag = segrellion_sansa_war_appeal_resolved
		add_character_flag = segrellion_sansa_marriage_for_vale_war

		segrellion_daemon_marry_sansa_for_vale_war_effect = yes

		segrellion_daemon_join_sansa_vale_war_effect = yes
	}


	# =====================================================
	# SANSA CLEARS THE VALE DEBT
	# =====================================================

	option = {

		name = segrellion_sansa.0140.b

		trigger = {
			has_character_flag = segrellion_sansa_price_debt_offer
		}

		custom_tooltip = segrellion_sansa.0140.b.tt

		remove_character_flag = segrellion_sansa_war_price_requested

		add_character_flag = segrellion_sansa_war_appeal_resolved
		add_character_flag = segrellion_sansa_debt_price_for_vale_war

		segrellion_sansa_clear_vale_debt_effect = yes

		segrellion_daemon_join_sansa_vale_war_effect = yes
	}
}

				add = 1000

# 0600 - BAELISH'S PLAN
#
# Sansa reaches sixteen without being extracted by Royce.
# Root = Lord of the Riverlands / Baelish
# =========================================================

segrellion_sansa.0600 = {

	type = character_event

	title = segrellion_sansa.0600.t
	desc = segrellion_sansa.0600.desc

	theme = intrigue

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		always = no
		is_alive = yes

		dynasty = dynasty:dynn_Baelish

		has_title = title:k_the_riverlands


		character:Stark_Dummy_8 ?= {

			is_alive = yes

			age >= 16

			NOT = {
				has_character_flag = segrellion_sansa_identity_revealed
			}
		}


		exists = global_var:segrellion_resurrected_jon


		global_var:segrellion_resurrected_jon = {

			is_alive = yes

			has_title = title:e_the_north
		}
	}


	immediate = {

		remove_character_flag = segrellion_baelish_sansa_choice_scheduled

		add_character_flag = segrellion_baelish_sansa_choice_resolved

		add_character_flag = segrellion_baelish_stannis_murder_plot_prepared
	}


	left_portrait = {

		character = root
		animation = scheming
	}


	right_portrait = {

		character = character:Stark_Dummy_8
		animation = personality_content
	}


	# =====================================================
	# SEND SANSA TO JON
	# =====================================================

	option = {

		name = segrellion_sansa.0600.a

		custom_tooltip = segrellion_sansa.0600.a.tt


		character:Stark_Dummy_8 = {

			segrellion_reveal_sansa_stark_effect = yes

			add_character_flag = segrellion_sansa_sent_to_jon


			if = {

				limit = {
					is_ruler = no
				}

				set_employer = global_var:segrellion_resurrected_jon
			}


			set_location = global_var:segrellion_resurrected_jon.capital_province
		}


		if = {

			limit = {

				can_add_hook = {

					target = global_var:segrellion_resurrected_jon

					type = favor_hook
				}
			}

			add_hook = {

				type = favor_hook

				target = global_var:segrellion_resurrected_jon
			}
		}


		global_var:segrellion_resurrected_jon = {

			trigger_event = {

				id = segrellion_sansa.0610

				days = { 2 5 }
			}
		}
	}


	# =====================================================
	# KEEP ALAYNE HIDDEN
	#
	# This is the branch we can continue later with
	# Baelish's larger political plan.
	# =====================================================

	option = {

		name = segrellion_sansa.0600.b

		add_character_flag = segrellion_baelish_kept_sansa_hidden
	}
}
