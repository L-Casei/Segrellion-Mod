# =========================================================
# 0001 - THE QUEEN ACROSS THE SEA
#
# First major crossroads for Daenerys.
#
# FIRST APPEARANCE:
#   A - Commit to Westeros.
#   B - Wait and reconsider later.
#   C - Reject Westeros and turn toward the Free Cities.
#
# SECOND APPEARANCE:
#   A - Commit to Westeros.
#   B - Abandon all special storylines and play freely.
#   C - Reject Westeros and turn toward the Free Cities.
# =========================================================

segrellion_daenerys_invasion.0001 = {

	type = character_event

	title = segrellion_daenerys_invasion.0001.t


	# =====================================================
	# DESCRIPTION
	# Changes after Daenerys has already delayed once.
	# =====================================================

	desc = {

		first_valid = {

			triggered_desc = {

				trigger = {
					has_character_flag = segrellion_daenerys_invasion_delayed_once
				}

				desc = segrellion_daenerys_invasion.0001.desc_second
			}

			desc = segrellion_daenerys_invasion.0001.desc
		}
	}


	theme = crown


	override_background = {
		reference = segrellion_daenerys_crossroads
	}


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		this = character:Targaryen_9

		is_alive = yes


		# Westeros path already permanently abandoned.
		NOT = {
			has_character_flag = segrellion_daenerys_invasion_abandoned
		}


		# Player explicitly chose to abandon all special stories.
		NOT = {
			has_character_flag = segrellion_daenerys_special_story_abandoned
		}


		# Free Cities path has already been chosen.
		NOT = {
			has_character_flag = segrellion_daenerys_free_cities_path
		}


		# Westeros path has already begun.
		NOT = {
			has_character_flag = segrellion_daenerys_invasion_committed
		}


		# Peaceful AI never enters the special invasion storyline.
		OR = {

			is_ai = no

			NOT = {
				has_game_rule = segrellion_daenerys_aggression_peaceful
			}
		}
	}


	left_portrait = {

		character = root

		animation = thinking
	}


	# =====================================================
	# OPTION A
	# WESTEROS
	# =====================================================

	option = {

		name = segrellion_daenerys_invasion.0001.a

		custom_tooltip = segrellion_daenerys_invasion.0001.a.tt


		ai_chance = {
			base = 100
		}


		remove_character_flag = segrellion_daenerys_invasion_delayed
		remove_character_flag = segrellion_daenerys_invasion_delayed_once

		add_character_flag = segrellion_daenerys_invasion_committed


		trigger_event = {

			id = segrellion_daenerys_invasion.0002

			days = { 3 7 }
		}
	}


	# =====================================================
	# OPTION B - FIRST APPEARANCE
	#
	# NOT YET.
	#
	# Daenerys remains completely free to wander.
	# Event returns once after 6-12 months.
	# =====================================================

	option = {

		name = segrellion_daenerys_invasion.0001.b

		trigger = {

			NOT = {
				has_character_flag = segrellion_daenerys_invasion_delayed_once
			}
		}

		custom_tooltip = segrellion_daenerys_invasion.0001.b.tt


		# AI timing is already controlled by the Game Rule.
		# This exists primarily for the player.
		ai_chance = {
			base = 0
		}


		add_character_flag = segrellion_daenerys_invasion_delayed
		add_character_flag = segrellion_daenerys_invasion_delayed_once


		trigger_event = {

			id = segrellion_daenerys_invasion.0001

			days = { 180 360 }
		}
	}


	# =====================================================
	# OPTION B - SECOND APPEARANCE
	#
	# NO SPECIAL STORY.
	#
	# Daenerys keeps her normal claims and can simply be
	# played as a normal landless character.
	# No more custom Westeros/Essos path events.
	# =====================================================

	option = {

		name = segrellion_daenerys_invasion.0001.b_second

		trigger = {

			has_character_flag = segrellion_daenerys_invasion_delayed_once
		}

		custom_tooltip = segrellion_daenerys_invasion.0001.b_second.tt


		ai_chance = {
			base = 0
		}


		remove_character_flag = segrellion_daenerys_invasion_delayed
		remove_character_flag = segrellion_daenerys_invasion_delayed_once
		remove_character_flag = segrellion_daenerys_invasion_committed


		# Stops the Westeros special chain.
		add_character_flag = segrellion_daenerys_invasion_abandoned


		# More general flag.
		# Future Daenerys story systems should respect this.
		add_character_flag = segrellion_daenerys_special_story_abandoned


		# Useful semantic flag for debugging/future content.
		add_character_flag = segrellion_daenerys_wandering_freely
	}


	# =====================================================
	# OPTION C
	# REJECT WESTEROS
	#
	# Abandons Iron Throne ambitions permanently.
	# Keeps Dragonstone.
	# Sets up future Free Cities storyline.
	# =====================================================

	option = {

		name = segrellion_daenerys_invasion.0001.c

		custom_tooltip = segrellion_daenerys_invasion.0001.c.tt


		ai_chance = {
			base = 0
		}


		remove_character_flag = segrellion_daenerys_invasion_delayed
		remove_character_flag = segrellion_daenerys_invasion_delayed_once
		remove_character_flag = segrellion_daenerys_invasion_committed


		# =================================================
		# REMOVE EVERY CLAIM EXCEPT DRAGONSTONE
		# =================================================

		every_claim = {

			limit = {

				this != title:c_dragonstone
			}

			root = {
				remove_claim = prev
			}
		}


		# Westeros is permanently rejected.
		add_character_flag = segrellion_daenerys_invasion_abandoned
		add_character_flag = segrellion_daenerys_westeros_rejected


		# =================================================
		# FUTURE FREE CITIES PATH
		#
		# Intentionally does NOTHING yet.
		#
		# When Essos is sufficiently implemented we can use
		# these flags as the entry point to the next story.
		# =================================================

		add_character_flag = segrellion_daenerys_free_cities_path
		add_character_flag = segrellion_daenerys_free_cities_path_pending
	}
}

		animation = personality_bold

# =========================================================
# 0016 - DEFERRED BEACON MARRIAGE CHECK
#
# Hidden monthly check.
# Waits until both Daenerys and Daemon are at peace.
# =========================================================

segrellion_daenerys_invasion.0016 = {

	type = character_event
	hidden = yes


	trigger = {

		this = character:Targaryen_9

		has_character_flag = segrellion_daenerys_marriage_deferred_at_beacon
	}


	immediate = {

		# -------------------------------------------------
		# BOTH ARE ALIVE, UNMARRIED, STILL PROMISED,
		# AND BOTH ARE NOW AT PEACE.
		# -------------------------------------------------

		if = {

			limit = {

				is_alive = yes
				is_married = no
				is_at_war = no

				has_character_flag = segrellion_marriage_promised_to_daemon


				character:Segrellion_60 ?= {

					is_alive = yes
					is_married = no
					is_at_war = no

					has_character_flag = segrellion_daenerys_marriage_promised
				}
			}


			remove_character_flag = segrellion_daenerys_marriage_deferred_at_beacon


			character:Segrellion_60 = {

				remove_character_flag = segrellion_daenerys_marriage_deferred_at_beacon

				add_character_flag = segrellion_daenerys_marriage_event_pending


				trigger_event = {

					id = segrellion_support.4000

					days = { 3 7 }
				}
			}
		}


		# -------------------------------------------------
		# STILL VALID, BUT WAR IS NOT OVER.
		# CHECK AGAIN IN 30 DAYS.
		# -------------------------------------------------

		else_if = {

			limit = {

				is_alive = yes
				is_married = no

				has_character_flag = segrellion_marriage_promised_to_daemon


				character:Segrellion_60 ?= {

					is_alive = yes
					is_married = no

					has_character_flag = segrellion_daenerys_marriage_promised
				}
			}


			trigger_event = {

				id = segrellion_daenerys_invasion.0016

				days = 30
			}
		}


		# -------------------------------------------------
		# PROMISE CAN NO LONGER BE FULFILLED.
		# -------------------------------------------------

		else = {

			remove_character_flag = segrellion_daenerys_marriage_deferred_at_beacon
			remove_character_flag = segrellion_marriage_promised_to_daemon


			character:Segrellion_60 ?= {

				remove_character_flag = segrellion_daenerys_marriage_deferred_at_beacon
				remove_character_flag = segrellion_daenerys_marriage_event_pending
				remove_character_flag = segrellion_daenerys_marriage_promised
			}
		}
	}
}

# 0018 - DAENERYS REACHES WESTEROS
#
# Common physical arrival point for EVERY landing route.
# =========================================================

segrellion_daenerys_invasion.0018 = {

	hidden = yes


	trigger = {

		this = character:Targaryen_9

		is_alive = yes

		has_character_flag = segrellion_daenerys_landing_chosen
	}


	immediate = {

		# =================================================
		# DAENERYS HAS PHYSICALLY REACHED WESTEROS
		# =================================================

		if = {

			limit = {

				NOT = {
					has_character_flag = segrellion_daenerys_landed_in_westeros
				}
			}


			add_character_flag = segrellion_daenerys_landed_in_westeros
		}


		# =================================================
		# BEACON
		# =================================================

		if = {

			limit = {
				has_character_flag = segrellion_daenerys_landing_beacon
			}

			trigger_event = {

				id = segrellion_daenerys_invasion.0015

				days = 1
			}
		}


		# =================================================
		# ALL OTHER LANDINGS
		# =================================================

		else = {

			trigger_event = {

				id = segrellion_daenerys_invasion.0020

				days = 1
			}
		}
	}
}
