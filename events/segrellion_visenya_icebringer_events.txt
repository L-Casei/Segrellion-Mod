# =========================================================
# 1000 - FIRST REQUEST DISPATCHER
#
# THREE POSSIBLE OPENINGS:
#
# 1010 = Daemon already wielded Icebringer on Skagos
# 1020 = Visenya agrees, but warns him
# 1030 = Visenya refuses
# =========================================================

segrellion_visenya_icebringer.1000 = {

	type = character_event
	hidden = yes


	immediate = {

		# =================================================
		# PATH I
		#
		# Daemon defeated Visenya and already held
		# Icebringer in his hands.
		# She KNOWS the blade accepted him.
		# =================================================

		if = {

			limit = {

				has_character_flag = segrellion_black_unicorn_daemon_won_duel

				has_character_flag = segrellion_black_unicorn_saw_icebringer
			}


			trigger_event = {
				id = segrellion_visenya_icebringer.1010
			}
		}


		# =================================================
		# NORMAL ROUTE
		#
		# Whether Visenya immediately accepts depends
		# privately on how she currently feels about Daemon.
		# No probability is shown to the player.
		# =================================================

		else = {

			if = {

				limit = {

					character:Black_Unicorn_1 = {

						opinion = {
							target = root
							value >= 70
						}
					}
				}


				random_list = {

					60 = {
						trigger_event = {
							id = segrellion_visenya_icebringer.1020
						}
					}

					40 = {
						trigger_event = {
							id = segrellion_visenya_icebringer.1030
						}
					}
				}
			}


			else_if = {

				limit = {

					character:Black_Unicorn_1 = {

						opinion = {
							target = root
							value >= 35
						}
					}
				}


				random_list = {

					40 = {
						trigger_event = {
							id = segrellion_visenya_icebringer.1020
						}
					}

					60 = {
						trigger_event = {
							id = segrellion_visenya_icebringer.1030
						}
					}
				}
			}


			else = {

				random_list = {

					20 = {
						trigger_event = {
							id = segrellion_visenya_icebringer.1020
						}
					}

					80 = {
						trigger_event = {
							id = segrellion_visenya_icebringer.1030
						}
					}
				}
			}
		}
	}
}
