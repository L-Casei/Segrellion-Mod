namespace = segrellion_beacon_dragonpit

# =========================================================
# FINANCE THE DRAGONPIT OF KNIGHT'S BRIDGE
# =========================================================

segrellion_beacon_dragonpit.0100 = {

	type = character_event

	title = segrellion_beacon_dragonpit.0100.t
	desc = segrellion_beacon_dragonpit.0100.desc
	theme = stewardship_domain_focus

	immediate = {

		title:c_farring_cross.holder ?= {

			random_great_project = {
				limit = {
					great_project_type = segrellion_construct_beacon_dragonpit
					great_project_construction_phase_is = planned
				}

				save_scope_as = segrellion_beacon_project

				random_contribution = {
					limit = {
						contribution_id = materials
						contribution_is_funded = no
					}
					save_scope_as = segrellion_beacon_materials_contribution
				}

				random_contribution = {
					limit = {
						contribution_id = workforce
						contribution_is_funded = no
					}
					save_scope_as = segrellion_beacon_workforce_contribution
				}

				random_contribution = {
					limit = {
						contribution_id = dragonkeeper_quarters
						contribution_is_funded = no
					}
					save_scope_as = segrellion_beacon_dragonkeepers_contribution
				}
			}
		}
	}

	# Stone, iron and vaulting.
	option = {
		name = segrellion_beacon_dragonpit.0100.a

		trigger = {
			exists = scope:segrellion_beacon_materials_contribution
		}

		show_as_unavailable = {
			gold < 600
		}

		custom_tooltip = segrellion_finance_beacon_dragonpit_materials_tt

		hidden_effect = {
			fund_great_project_contribution = {
				contribution = scope:segrellion_beacon_materials_contribution
				check_can_contribute = no
			}
		}

		ai_chance = {
			base = 100
		}
	}

	# Workforce.
	option = {
		name = segrellion_beacon_dragonpit.0100.b

		trigger = {
			exists = scope:segrellion_beacon_workforce_contribution
		}

		show_as_unavailable = {
			gold < 450
		}

		custom_tooltip = segrellion_finance_beacon_dragonpit_workforce_tt

		hidden_effect = {
			fund_great_project_contribution = {
				contribution = scope:segrellion_beacon_workforce_contribution
				check_can_contribute = no
			}
		}

		ai_chance = {
			base = 100
		}
	}

	# Dragonkeeper halls.
	option = {
		name = segrellion_beacon_dragonpit.0100.c

		trigger = {
			exists = scope:segrellion_beacon_dragonkeepers_contribution
		}

		show_as_unavailable = {
			OR = {
				gold < 300
				prestige < 100
			}
		}

		custom_tooltip = segrellion_finance_beacon_dragonpit_dragonkeepers_tt

		hidden_effect = {
			fund_great_project_contribution = {
				contribution = scope:segrellion_beacon_dragonkeepers_contribution
				check_can_contribute = no
			}
		}

		ai_chance = {
			base = 100
		}
	}

	# Leave without funding anything.
	option = {
		name = segrellion_beacon_dragonpit.0100.d
		custom_tooltip = segrellion_finance_beacon_dragonpit_cancel_tt

		ai_chance = {
			base = 0
		}
	}
}
