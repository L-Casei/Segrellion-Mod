# =========================================================
# 4000 - CASTLE BLACK
# =========================================================

segrellion_relic.4000 = {

	type = character_event

	trigger = {
		has_character_flag = segrellion_legendary_relic_search_active
		has_character_flag = segrellion_relic_pending_castle_black
	}

	title = segrellion_relic.4000.t
	desc = segrellion_relic.4000.desc

	theme = diplomacy

	override_background = {
		reference = segrellion_castle_black_great_hall
	}

	immediate = {

		remove_character_flag = segrellion_relic_pending_castle_black
		remove_character_flag = segrellion_legendary_relic_search_paused

		title:c_castle_black.holder ?= {
			save_scope_as = relic_watch_host
		}
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:relic_watch_host
		animation = personality_rational
	}

	option = {

		name = segrellion_relic.4000.a

		trigger_event = {
			id = segrellion_relic.4010
			
		}
	}
}

	override_background = {
		reference = segrellion_castle_black_great_hall
	}

	left_portrait = {
		character = root
		animation = thinking
	}

	option = {

		name = segrellion_relic.4010.a

		add_character_flag = segrellion_relic_knows_nightfort

		hidden_effect = {

			segrellion_clear_relic_pending_destination_effect = yes

			add_character_flag = segrellion_relic_pending_nightfort

			if = {
				limit = {
					exists = current_travel_plan
				}

				current_travel_plan = {
					complete_travel_plan = yes
				}
			}

			trigger_event = {
				id = segrellion_relic_bridge.0002
				days = 1
			}
		}
	}
}
