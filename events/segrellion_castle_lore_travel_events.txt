namespace = segrellion_castle_lore_travel


segrellion_castle_lore_travel.0001 = {

	type = character_event
	hidden = yes


	trigger = {

		is_alive = yes

		exists = current_travel_plan

		current_travel_plan.travel_plan_owner = root
	}


	immediate = {

		# =================================================
		# VHAL
		# Province 4427
		# Historically established from 18 April 106.
		# =================================================

		if = {

			limit = {

				current_date >= 8106.4.18

				location = province:4427

				NOT = {
					exists = var:segrellion_lore_vhal_visit
				}
			}


			trigger_event = {
				id = segrellion_castle_lore.1000
			}
		}


		# =================================================
		# THE BEACON
		# Historically established from 28 January 144.
		# =================================================

		if = {

			limit = {

				current_date >= 8144.1.28

				location = title:b_castle_farring.title_province

				NOT = {
					exists = var:segrellion_lore_beacon_visit
				}
			}


			trigger_event = {
				id = segrellion_castle_lore.2000
			}
		}


		# =================================================
		# WHITE HALL
		# Completed and granted as a seat on 1 January 164.
		# =================================================

		if = {

			limit = {

				current_date >= 8164.1.1

				location = title:b_benthill_dale.title_province

				NOT = {
					exists = var:segrellion_lore_white_hall_visit
				}
			}


			trigger_event = {
				id = segrellion_castle_lore.3000
			}
		}
	}
}
