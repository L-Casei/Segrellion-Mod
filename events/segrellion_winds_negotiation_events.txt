# =========================================================
# 1000 - STANNIS BARATHEON
# =========================================================

segrellion_support.1000 = {

	type = character_event

	title = segrellion_support.1000.t
	desc = segrellion_support.1000.desc

	theme = diplomacy

	override_background = {
		reference = throne_room
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		has_character_flag = segrellion_negotiation_pending_stannis
	}


	immediate = {

		# We have arrived at Dragonstone.
		remove_character_flag = segrellion_negotiation_pending_stannis

		character:Baratheon_3 ?= {

			save_scope_as = support_stannis

			random_child = {

				limit = {
					is_alive = yes
					is_female = yes
					is_married = no
				}

				save_scope_as = support_shireen
			}
		}
	}


	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:support_stannis
		animation = personality_rational
	}


	# =====================================================
	# ACCEPT
	# =====================================================

	option = {

		name = segrellion_support.1000.a

		ai_chance = {
			base = 95
		}

		trigger = {
			exists = scope:support_shireen
			is_married = no
		}

		marry = scope:support_shireen

		segrellion_grant_stannis_support_effect = yes


		# ---------------------------------------------
		# YOUNG GRIFF BREAKS WITH DAEMON
		# ---------------------------------------------

		character:IAF_Young_Griff_1 ?= {

			remove_character_flag = segrellion_young_griff_supports_daemon

			add_opinion = {
				target = root
				modifier = segrellion_chose_stannis_over_young_griff_opinion
			}
		}


		# ---------------------------------------------
		# STANNIS APPROVES
		# ---------------------------------------------

		scope:support_stannis = {

			add_opinion = {
				target = root
				modifier = segrellion_swore_to_stannis_opinion
			}
		}
	}


	# =====================================================
	# REFUSE
	# =====================================================

	option = {

		name = segrellion_support.1000.b

		ai_chance = {
			base = 5
		}

		segrellion_clear_troop_negotiation_effect = yes
	}
}

# =========================================================
# 3020 - HIS ANSWER
# Root = Daenerys
# Final decision.
# =========================================================

segrellion_support.3020 = {

	type = character_event

	title = segrellion_support.3020.t

	desc = {

		triggered_desc = {

			trigger = {
				character:Segrellion_60 = {
					has_character_flag = segrellion_daemon_offered_marriage_to_daenerys
				}
			}

			desc = segrellion_support.3020.desc.marriage
		}

		triggered_desc = {

			trigger = {
				character:Segrellion_60 = {

					has_character_flag = segrellion_daemon_offered_oath_to_daenerys

					NOT = {
						has_character_flag = segrellion_daemon_offered_marriage_to_daenerys
					}
				}
			}

			desc = segrellion_support.3020.desc.oath
		}
	}

	theme = diplomacy

	override_background = {
		reference = wilderness
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	left_portrait = {
		character = root
		animation = thinking
	}

	right_portrait = {
		character = character:Segrellion_60
		animation = personality_rational
	}


	# =====================================================
	# ACCEPT MARRIAGE + OATH
	# =====================================================

	option = {

		name = segrellion_support.3020.a

		trigger = {

			character:Segrellion_60 = {
				has_character_flag = segrellion_daemon_offered_marriage_to_daenerys
			}

			is_married = no
		}

		ai_chance = {

			base = 35

			modifier = {
				add = 35

				opinion = {
					target = character:Segrellion_60
					value >= 75
				}
			}

			modifier = {
				add = 20

				opinion = {
					target = character:Segrellion_60
					value >= 40
				}
			}

			modifier = {
				add = 15
				is_attracted_to_gender_of = character:Segrellion_60
			}

			modifier = {
				add = 15
				has_trait = ambitious
			}

			modifier = {
				add = 15
				has_character_flag = segrellion_daenerys_invasion_committed
			}
		}

		add_character_flag = segrellion_marriage_promised_to_daemon
		add_character_flag = segrellion_daenerys_supports_daemon

		character:Segrellion_60 = {

			add_character_flag = segrellion_daenerys_marriage_promised

			remove_character_flag = segrellion_daemon_offered_oath_to_daenerys
			remove_character_flag = segrellion_daemon_offered_marriage_to_daenerys

			segrellion_grant_daenerys_support_effect = yes

			trigger_event = {
				id = segrellion_support.3030
				days = 1
			}
		}
	}


	# =====================================================
	# REFUSE MARRIAGE BUT ACCEPT HIS OATH
	# =====================================================

	option = {

		name = segrellion_support.3020.b

		trigger = {

			character:Segrellion_60 = {
				has_character_flag = segrellion_daemon_offered_marriage_to_daenerys
			}
		}

		ai_chance = {

			base = 50

			modifier = {
				add = 20
				has_trait = ambitious
			}

			modifier = {
				add = 15
				has_character_flag = segrellion_daenerys_invasion_committed
			}
		}


		# Remember that Daenerys specifically rejected
		# the proposed marriage, not Daemon's alliance.

		add_character_flag = segrellion_daenerys_rejected_daemon_marriage
		add_character_flag = segrellion_daenerys_supports_daemon


		character:Segrellion_60 = {

			add_character_flag = segrellion_daenerys_rejected_marriage

			remove_character_flag = segrellion_daemon_offered_oath_to_daenerys
			remove_character_flag = segrellion_daemon_offered_marriage_to_daenerys


			segrellion_grant_daenerys_support_effect = yes


			trigger_event = {
				id = segrellion_support.3032
				days = 1
			}
		}
	}


	# =====================================================
	# ACCEPT THE ORDINARY OATH
	# =====================================================

	option = {

		name = segrellion_support.3020.c

		trigger = {

			character:Segrellion_60 = {

				has_character_flag = segrellion_daemon_offered_oath_to_daenerys

				NOT = {
					has_character_flag = segrellion_daemon_offered_marriage_to_daenerys
				}
			}
		}

		ai_chance = {

			base = 70

			modifier = {
				add = 25
				has_trait = ambitious
			}

			modifier = {
				add = 20
				has_character_flag = segrellion_daenerys_invasion_committed
			}

			modifier = {
				add = 20

				opinion = {
					target = character:Segrellion_60
					value >= 40
				}
			}
		}

		add_character_flag = segrellion_daenerys_supports_daemon

		character:Segrellion_60 = {

			remove_character_flag = segrellion_daemon_offered_oath_to_daenerys

			segrellion_grant_daenerys_support_effect = yes

			trigger_event = {
				id = segrellion_support.3031
				days = 1
			}
		}
	}


	# =====================================================
	# REJECT EVERYTHING
	# Always possible.
	# =====================================================

	option = {

		name = segrellion_support.3020.d

		ai_chance = {

			base = 25

			modifier = {
				add = 40

				opinion = {
					target = character:Segrellion_60
					value < 0
				}
			}

			modifier = {
				add = 25
				has_trait = paranoid
			}

			modifier = {
				add = 30
				has_character_flag = segrellion_daenerys_invasion_abandoned
			}
		}

		add_character_flag = segrellion_daenerys_rejected_daemon

		character:Segrellion_60 = {

			remove_character_flag = segrellion_daemon_offered_oath_to_daenerys
			remove_character_flag = segrellion_daemon_offered_marriage_to_daenerys

			segrellion_clear_troop_negotiation_effect = yes

			trigger_event = {
				id = segrellion_support.3090
				days = 1
			}
		}
	}
}

# =========================================================
# 5000 - VHALERION'S CHOICE
# =========================================================

segrellion_support.5000 = {

	type = character_event

	title = segrellion_support.5000.t
	desc = segrellion_support.5000.desc

	theme = realm

	override_background = {
		reference = segrellion_vhalerion_throne_room
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		has_character_flag = segrellion_vhal_reunification_event_pending

		title:d_vhalerion = {
			holder = root
		}

		character:Segrellion_60 ?= {
			is_alive = yes
			has_title = title:e_flamelands
		}
	}


	left_portrait = {
		character = root
		animation = thinking
	}

	right_portrait = {
		character = character:Segrellion_60
		animation = personality_bold
	}


	# =====================================================
	# RETURN TO THE FLAMELANDS
	# =====================================================

	option = {

		name = segrellion_support.5000.a

		remove_character_flag = segrellion_vhal_reunification_event_pending

		add_character_flag = segrellion_vhal_reunification_resolved
		add_character_flag = segrellion_vhal_rejoined_flamelands


		# Restore the historical de jure relationship.
		title:d_vhalerion = {
			set_de_jure_liege_title = title:k_flamelands
		}


		# Hugh / current Lord of Vhal becomes Daemon's vassal.
		set_liege = character:Segrellion_60


		character:Segrellion_60 = {

			add_opinion = {
				target = root
				modifier = segrellion_vhal_returned_to_daemon_opinion
			}
		}
	}


	# =====================================================
	# REMAIN APART
	# =====================================================

	option = {

		name = segrellion_support.5000.b

		remove_character_flag = segrellion_vhal_reunification_event_pending
