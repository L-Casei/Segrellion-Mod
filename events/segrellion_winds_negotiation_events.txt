# =========================================================
# 1000 - STANNIS BARATHEON
# =========================================================

segrellion_support.1000 = {

	type = character_event

	title = segrellion_support.1000.t
	desc = segrellion_support.1000.desc

	theme = diplomacy

	override_background = {
		reference = throne_room
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		has_character_flag = segrellion_negotiation_pending_stannis
	}


	immediate = {

		# We have arrived at Dragonstone.
		remove_character_flag = segrellion_negotiation_pending_stannis

		character:Baratheon_3 ?= {

			save_scope_as = support_stannis

			random_child = {

				limit = {
					is_alive = yes
					is_female = yes
					is_married = no
				}

				save_scope_as = support_shireen
			}
		}
	}


	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:support_stannis
		animation = personality_rational
	}


	# =====================================================
	# ACCEPT
	# =====================================================

	option = {

		name = segrellion_support.1000.a

		ai_chance = {
			base = 95
		}

		trigger = {
			exists = scope:support_shireen
			is_married = no
		}

		marry = scope:support_shireen

		segrellion_grant_stannis_support_effect = yes


		# ---------------------------------------------
		# YOUNG GRIFF REACTS TO DAEMON CHOOSING STANNIS
		# ---------------------------------------------

		character:IAF_Young_Griff_1 ?= {

			if = {

				limit = {

					is_alive = yes

					NOT = {
						has_character_flag = segrellion_young_griff_daemon_choice_resolved
					}
				}

				trigger_event = {
					id = segrellion_young_griff.1000
					days = 1
				}
			}
		}


		# ---------------------------------------------
		# STANNIS APPROVES
		# ---------------------------------------------

		scope:support_stannis = {

			add_opinion = {
				target = root
				modifier = segrellion_swore_to_stannis_opinion
			}
		}
	}


	# =====================================================
	# REFUSE
	# =====================================================

	option = {

		name = segrellion_support.1000.b

		ai_chance = {
			base = 5
		}

		add_character_flag = segrellion_stannis_support_route_closed
		segrellion_clear_troop_negotiation_effect = yes
	}
}

# =========================================================
# 1990 - ARRIVAL AT SUNSPEAR
# Hidden router.
#
# Checks whether Arianne's situation changed while Daemon
# was travelling to Dorne.
# =========================================================

segrellion_support.1990 = {

	hidden = yes

	trigger = {
		this = character:Segrellion_60
		has_character_flag = segrellion_negotiation_pending_dorne
	}

	immediate = {

		# ---------------------------------------------
		# ARIANNE MARRIED WHILE DAEMON WAS TRAVELLING
		# ---------------------------------------------

		if = {

			limit = {

				character:Martell_9 ?= {
					is_alive = yes
					is_married = yes
				}
			}

			remove_character_flag = segrellion_negotiation_pending_dorne

			trigger_event = {
				id = segrellion_support.2050
				days = 1
			}
		}


		# ---------------------------------------------
		# ORIGINAL BARGAIN IS STILL POSSIBLE
		# ---------------------------------------------

		else_if = {

			limit = {

				character:Martell_9 ?= {
					is_alive = yes
					is_married = no
				}
			}

			# 2000 will remove the pending flag itself.
			trigger_event = {
				id = segrellion_support.2000
				days = 1
			}
		}


		# ---------------------------------------------
		# ARIANNE IS NO LONGER AVAILABLE AT ALL
		# ---------------------------------------------

		else = {

			remove_character_flag = segrellion_negotiation_pending_dorne

			trigger_event = {
				id = segrellion_support.2051
				days = 1
			}
		}
	}
}


# =========================================================
# 2000 - DORNE
# =========================================================

segrellion_support.2000 = {

	type = character_event

	title = segrellion_support.2000.t
	desc = segrellion_support.2000.desc

	theme = diplomacy

	override_background = {
		reference = throne_room
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {
		has_character_flag = segrellion_negotiation_pending_dorne		
	}


	immediate = {

		# We have arrived at Sunspear.
		remove_character_flag = segrellion_negotiation_pending_dorne

		character:Martell_9 ?= {

			save_scope_as = support_arianne

			father ?= {
				save_scope_as = support_doran
			}
		}
	}


	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:support_doran
		animation = personality_rational
	}

	lower_right_portrait = {
		character = scope:support_arianne
		animation = personality_content
	}


	# =====================================================
	# ACCEPT
	# =====================================================

	option = {

		name = segrellion_support.2000.a

		ai_chance = {
			base = 95
		}

		trigger = {
			is_married = no

			scope:support_arianne = {
				is_married = no
			}
		}

		scope:support_arianne = {
			if = {
				limit = { has_trait = refusing_marriage }
				remove_trait = refusing_marriage
			}
		}

		marry = scope:support_arianne

		add_character_flag = segrellion_daemon_arianne_dynastic_pact

		scope:support_arianne = {
			add_character_flag = segrellion_daemon_arianne_dynastic_pact
		}

		segrellion_grant_dorne_support_effect = yes


		# ---------------------------------------------
		# YOUNG GRIFF REACTS TO THE DORNISH PACT
		# ---------------------------------------------

		character:IAF_Young_Griff_1 ?= {

			if = {

				limit = {

					is_alive = yes

					NOT = {
						has_character_flag = segrellion_young_griff_daemon_choice_resolved
					}
				}

				trigger_event = {
					id = segrellion_young_griff.3000
					days = 1
				}
			}
		}
	}


	# =====================================================
	# REFUSE
	# =====================================================

	option = {

		name = segrellion_support.2000.b

		ai_chance = {
			base = 5
		}

		add_character_flag = segrellion_dorne_support_route_closed
		segrellion_clear_troop_negotiation_effect = yes
	}
}

# Root = Daenerys
# Final decision.
# =========================================================

	type = character_event

	title = segrellion_support.3020.t

	desc = {

		triggered_desc = {

			trigger = {
				character:Segrellion_60 = {
					has_character_flag = segrellion_daemon_offered_marriage_to_daenerys
				}
			}

			desc = segrellion_support.3020.desc.marriage
		}

		triggered_desc = {

			trigger = {
				character:Segrellion_60 = {

					has_character_flag = segrellion_daemon_offered_oath_to_daenerys

					NOT = {
						has_character_flag = segrellion_daemon_offered_marriage_to_daenerys

	}

	right_portrait = {
		character = character:Segrellion_60
		animation = personality_rational
	}


	# =====================================================
	# ACCEPT MARRIAGE + OATH
	# =====================================================

		name = segrellion_support.3020.a

		trigger = {

			character:Segrellion_60 = {
				has_character_flag = segrellion_daemon_offered_marriage_to_daenerys
			}

			is_married = no
		}

		ai_chance = {

			base = 35

			modifier = {
				add = 35

				opinion = {
					target = character:Segrellion_60
					value >= 75
				}

				add = 20

				opinion = {
					target = character:Segrellion_60
					value >= 40
				}
			}

				add = 15

			}

			value = var:segrellion_vale_debt

		}

	}

	# No new interest. Another demand arrives next year.
	# =====================================================

		name = segrellion_support.6200.b

			id = segrellion_support.6200

		}
	}
