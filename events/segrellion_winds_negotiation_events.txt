# =========================================================
# 1000 - STANNIS BARATHEON
# =========================================================

segrellion_support.1000 = {

	type = character_event

	title = segrellion_support.1000.t
	desc = segrellion_support.1000.desc

	theme = diplomacy

	override_background = {
		reference = throne_room
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		has_character_flag = segrellion_negotiation_pending_stannis
	}


	immediate = {

		# We have arrived at Dragonstone.
		remove_character_flag = segrellion_negotiation_pending_stannis

		character:Baratheon_3 ?= {

			save_scope_as = support_stannis

			any_child = {

				limit = {
					is_alive = yes
					is_female = yes
					is_married = no
				}

				save_scope_as = support_shireen
			}
		}
	}


	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:support_stannis
		animation = personality_rational
	}


	# =====================================================
	# ACCEPT
	# =====================================================

	option = {

		name = segrellion_support.1000.a

		trigger = {
			exists = scope:support_shireen
			is_married = no
		}

		marry = scope:support_shireen

		segrellion_grant_stannis_support_effect = yes


		# ---------------------------------------------
		# YOUNG GRIFF BREAKS WITH DAEMON
		# ---------------------------------------------

		character:IAF_Young_Griff_1 ?= {

			remove_character_flag = segrellion_young_griff_supports_daemon

			add_opinion = {
				target = root
				modifier = segrellion_chose_stannis_over_young_griff_opinion
			}
		}


		# ---------------------------------------------
		# STANNIS APPROVES
		# ---------------------------------------------

		scope:support_stannis = {

			add_opinion = {
				target = root
				modifier = segrellion_swore_to_stannis_opinion
			}
		}
	}


	# =====================================================
	# REFUSE
	# =====================================================

	option = {

		name = segrellion_support.1000.b

		segrellion_clear_troop_negotiation_effect = yes
	}
}

# =========================================================
# 3010 - DAENERYS CONSIDERS THE MARRIAGE
# Root = Daenerys
# =========================================================

segrellion_support.3010 = {

	type = character_event

	title = segrellion_support.3010.t
	desc = segrellion_support.3010.desc

	theme = marriage

	override_background = {
		reference = wilderness
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		this = character:Targaryen_9

		is_alive = yes
		is_married = no

		character:Segrellion_60 ?= {
			is_alive = yes
			is_married = no
		}
	}


	left_portrait = {
		character = root
		animation = thinking
	}

	right_portrait = {
		character = character:Segrellion_60
		animation = personality_rational
	}


	# =====================================================
	# ACCEPT THE MARRIAGE
	# =====================================================

	option = {

		name = segrellion_support.3010.a


		ai_chance = {

			base = 30


			# -----------------------------------------
			# OPINION
			# -----------------------------------------

			modifier = {
				add = 35

				opinion = {
					target = character:Segrellion_60
					value >= 75
				}
			}

			modifier = {
				add = 20

				opinion = {
					target = character:Segrellion_60
					value >= 40
				}
			}

			modifier = {
				add = -30

				opinion = {
					target = character:Segrellion_60
					value < 0
				}
			}


			# -----------------------------------------
			# SEXUAL ATTRACTION IS POSSIBLE
			# -----------------------------------------

			modifier = {
				add = 15

				is_attracted_to_gender_of = character:Segrellion_60
			}


			# -----------------------------------------
			# DAEMON IS ATTRACTIVE
			# -----------------------------------------

			modifier = {

				add = 15

				character:Segrellion_60 = {

					OR = {
						has_trait = beauty_good_1
						has_trait = beauty_good_2
						has_trait = beauty_good_3
					}
				}
			}


			# -----------------------------------------
			# SHARED AMBITION
			# -----------------------------------------

			modifier = {

				add = 15

				has_trait = ambitious

				character:Segrellion_60 = {
					has_trait = ambitious
				}
			}


			# -----------------------------------------
			# SHARED PATIENCE
			# -----------------------------------------

			modifier = {

				add = 10

				has_trait = patient

				character:Segrellion_60 = {
					has_trait = patient
				}
			}


			# -----------------------------------------
			# SHARED BOLD / COMMANDING CHARACTER
			# -----------------------------------------

			modifier = {

				add = 10

				has_trait = authoritative

				character:Segrellion_60 = {
					has_trait = authoritative
				}
			}
		}


		character:Segrellion_60 = {

			segrellion_grant_daenerys_support_effect = yes

			add_character_flag = segrellion_daenerys_marriage_promised

			trigger_event = {
				id = segrellion_support.3020
				days = 1
			}
		}

		add_character_flag = segrellion_marriage_promised_to_daemon
	}


	# =====================================================
	# REFUSE MARRIAGE, KEEP THE MILITARY PACT
	# =====================================================

	option = {

		name = segrellion_support.3010.b

		ai_chance = {
			base = 50
		}

		character:Segrellion_60 = {

			segrellion_grant_daenerys_support_effect = yes

			trigger_event = {
				id = segrellion_support.3030
				days = 1
			}
		}
	}
}

# Daemon has taken the Flamelands.

	type = character_event

	title = segrellion_support.4000.t
	desc = segrellion_support.4000.desc

	theme = marriage

	override_background = {
		reference = segrellion_the_beacon_throne_room
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		this = character:Segrellion_60

		has_title = title:e_flamelands

		has_character_flag = segrellion_daenerys_marriage_promised

		is_married = no

		character:Targaryen_9 ?= {
			is_alive = yes
			is_married = no
		}
	}

		character = root
		animation = personality_bold
	}

	right_portrait = {
		character = character:Targaryen_9
		animation = personality_bold
	}


	option = {

		name = segrellion_support.4000.a

		marry = character:Targaryen_9

		remove_character_flag = segrellion_daenerys_marriage_promised
		remove_character_flag = segrellion_daenerys_marriage_event_pending

		character:Targaryen_9 = {
			remove_character_flag = segrellion_marriage_promised_to_daemon
		}

		add_character_flag = segrellion_married_daenerys_after_restoration
	}
}


# =========================================================
# 5000 - VHALERION'S CHOICE
# =========================================================

segrellion_support.5000 = {

	type = character_event

	title = segrellion_support.5000.t
	desc = segrellion_support.5000.desc

	theme = realm

	override_background = {
		reference = segrellion_vhalerion_throne_room
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		has_character_flag = segrellion_vhal_reunification_event_pending

		title:d_vhalerion = {
			holder = root
		}

		character:Segrellion_60 ?= {
			is_alive = yes
			has_title = title:e_flamelands
		}
	}


	left_portrait = {
		character = root
		animation = thinking
	}

	right_portrait = {
		character = character:Segrellion_60
		animation = personality_bold
	}


	# =====================================================
	# RETURN TO THE FLAMELANDS
	# =====================================================

	option = {

		name = segrellion_support.5000.a

		remove_character_flag = segrellion_vhal_reunification_event_pending

		add_character_flag = segrellion_vhal_reunification_resolved
		add_character_flag = segrellion_vhal_rejoined_flamelands


		# Restore the historical de jure relationship.
		title:d_vhalerion = {
			set_de_jure_liege_title = title:k_flamelands
		}


		# Hugh / current Lord of Vhal becomes Daemon's vassal.
		set_liege = character:Segrellion_60


		character:Segrellion_60 = {

			add_opinion = {
				target = root
				modifier = segrellion_vhal_returned_to_daemon_opinion
			}
		}
	}


	# =====================================================
	# REMAIN APART
	# =====================================================

	option = {

		name = segrellion_support.5000.b

		remove_character_flag = segrellion_vhal_reunification_event_pending

		add_character_flag = segrellion_vhal_reunification_resolved
		add_character_flag = segrellion_vhal_refused_daemon


		# Ensure Vhal remains outside the Flamelands.
		title:d_vhalerion = {
			set_de_jure_liege_title = title:h_the_iron_throne
		}


		# Daemon does NOT take this well.
		character:Segrellion_60 = {

			add_opinion = {
				target = root
				modifier = segrellion_vhal_refused_daemon_opinion
			}
		}
	}
}
