# =========================================================
# BLACK UNICORN - PATH III VARIANT ROUTER
#
# Routes unresolved / non-hostile earlier outcomes into a
# softer Beacon confrontation instead of forcing the old
# violent "Steel in the Dark" opening.
# =========================================================

segrellion_black_unicorn.0900 = {

	type = character_event
	hidden = yes

	trigger = {

		this = character:Segrellion_60
		is_alive = yes

		segrellion_black_unicorn_part_iii_ready_trigger = yes
	}

	on_trigger_fail = {

		character:Segrellion_60 ?= {

			if = {

				limit = {

					is_alive = yes

					has_character_flag = segrellion_black_unicorn_post_beacon_timer_v4

					NOT = {
						has_character_flag = segrellion_black_unicorn_post_beacon_resolved
					}

					NOT = {
						has_character_flag = segrellion_black_unicorn_path_iii_variant_started
					}

					exists = character:Black_Unicorn_1

					character:Black_Unicorn_1 = {
						is_alive = yes

						NOT = {
							has_character_flag = segrellion_black_unicorn_visenya_joined_daemon
						}
					}
				}

				trigger_event = {
					id = segrellion_black_unicorn.0900
					days = 30
				}
			}
		}
	}


	immediate = {

		remove_character_flag = segrellion_black_unicorn_post_beacon_queued

		# =====================================================
		# THE TRAIL WENT COLD
		#
		# Daemon receives one last set of rumours before
		# Visenya eventually reaches the Beacon herself.
		# =====================================================

		if = {

			limit = {
				has_character_flag = segrellion_black_unicorn_future_post_war_return
			}

			add_character_flag = segrellion_black_unicorn_path_iii_variant_started
			add_character_flag = segrellion_black_unicorn_path_iii_cold_trail_entry

			trigger_event = {
				id = segrellion_black_unicorn.0901
				days = 1
			}
		}

		# =====================================================
		# UNFINISHED / AMBIGUOUS SKAGOS ENCOUNTER
		# =====================================================

		else_if = {

			limit = {
				segrellion_black_unicorn_part_iii_soft_entry_trigger = yes
			}

			add_character_flag = segrellion_black_unicorn_path_iii_variant_started
			add_character_flag = segrellion_black_unicorn_path_iii_unfinished_business_entry

			trigger_event = {
				id = segrellion_black_unicorn.0910
				days = 1
			}
		}

		# =====================================================
		# OLD HOSTILE PATH III
		#
		# Truly hostile / destiny routes keep Steel in the Dark.
		# =====================================================

		else = {

			trigger_event = {
				id = segrellion_black_unicorn.0600
				days = 1
			}
		}
	}
}

# =========================================================
# 0910 - THE UNICORN AT MIDNIGHT
#
# Peaceful / tense Path III opening.
# Visenya enters by surprise, but not to spill blood.
# =========================================================

segrellion_black_unicorn.0910 = {

	type = character_event

	title = segrellion_black_unicorn.0910.t

	desc = {

		first_valid = {

			# -------------------------------------------------
			# THEY HAVE ALREADY BEEN INTIMATE
			# -------------------------------------------------

			triggered_desc = {

				trigger = {

					OR = {
						has_character_flag = segrellion_black_unicorn_shared_night
						has_character_flag = segrellion_black_unicorn_daemon_slept_with_visenya
					}
				}

				desc = segrellion_black_unicorn.0910.desc_warm
			}

			# -------------------------------------------------
			# VISENYA CHOSE TO REMAIN BEHIND THE FIRST TIME
			# -------------------------------------------------

			triggered_desc = {

				trigger = {
					has_character_flag = segrellion_black_unicorn_visenya_declined_first_return
				}

				desc = segrellion_black_unicorn.0910.desc_declined
			}

			# -------------------------------------------------
			# DAEMON WAS ONCE HER PRISONER / FORCED FROM SKAGOS
			# -------------------------------------------------

			triggered_desc = {

				trigger = {
					has_character_flag = segrellion_black_unicorn_forced_from_skagos
				}

				desc = segrellion_black_unicorn.0910.desc_banished
			}

			# -------------------------------------------------
			# DAEMON WALKED AWAY WITHOUT AN ANSWER
			# -------------------------------------------------

			triggered_desc = {

				trigger = {

					OR = {
						has_character_flag = segrellion_black_unicorn_future_beacon_ambush
						has_character_flag = segrellion_black_unicorn_left_without_truth
						has_character_flag = segrellion_black_unicorn_left_skagos_without_visenya
					}
				}

				desc = segrellion_black_unicorn.0910.desc_tense
			}

			# -------------------------------------------------
			# DAEMON INVESTIGATED THE VALE RUMOURS
			# -------------------------------------------------

			triggered_desc = {

				trigger = {
					has_character_flag = segrellion_black_unicorn_vale_rumours_investigated
				}

				desc = segrellion_black_unicorn.0910.desc_investigated
			}

			# -------------------------------------------------
			# DAEMON IGNORED THEM
			# -------------------------------------------------

			triggered_desc = {

				trigger = {
					has_character_flag = segrellion_black_unicorn_vale_rumours_ignored
				}

				desc = segrellion_black_unicorn.0910.desc_ignored
			}

			# -------------------------------------------------
			# TRUST EXISTS, BUT THEY NEVER SETTLED THE FUTURE
			# -------------------------------------------------

			triggered_desc = {

				trigger = {

					OR = {
						has_character_flag = segrellion_black_unicorn_visenya_trust_earned
						has_character_flag = segrellion_black_unicorn_daemon_knows_visenya_identity
					}
				}

				desc = segrellion_black_unicorn.0910.desc_trusting
			}

			desc = segrellion_black_unicorn.0910.desc
		}
	}

	theme = intrigue

	override_background = {
		reference = bedchamber
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	trigger = {

		this = character:Segrellion_60
		is_alive = yes
		is_imprisoned = no

		has_character_flag = segrellion_black_unicorn_path_iii_variant_started

		NOT = {
			has_character_flag = segrellion_black_unicorn_post_beacon_resolved
		}

		character:Black_Unicorn_1 ?= {
			is_alive = yes
			is_imprisoned = no
		}
	}

	on_trigger_fail = {

		character:Segrellion_60 ?= {

			if = {

				limit = {

					is_alive = yes

					has_character_flag = segrellion_black_unicorn_path_iii_variant_started

					NOT = {
						has_character_flag = segrellion_black_unicorn_post_beacon_resolved
					}

					exists = character:Black_Unicorn_1

					character:Black_Unicorn_1 = {
						is_alive = yes

						NOT = {
							has_character_flag = segrellion_black_unicorn_visenya_joined_daemon
						}
					}
				}

				trigger_event = {
					id = segrellion_black_unicorn.0910
					days = 30
				}
			}
		}
	}

	immediate = {

		add_character_flag = segrellion_black_unicorn_post_beacon_started

		# -----------------------------------------------------
		# DAEMON IS AT THE BEACON
		# -----------------------------------------------------

		current_travel_plan ?= {
			abort_travel_plan = yes
		}

		every_army = {

			if = {

				limit = {

					army_commander ?= {
						this = root
					}
				}

				remove_commander = yes
			}
		}

		set_location = title:b_castle_farring.title_province

		remove_character_flag = segrellion_black_unicorn_beacon_spouse_present

		primary_spouse ?= {

			if = {

				limit = {
					is_alive = yes
					location = title:b_castle_farring.title_province
				}

				save_scope_as = segrellion_daemon_beacon_spouse

				root = {
					add_character_flag = segrellion_black_unicorn_beacon_spouse_present
				}
			}
		}

		# -----------------------------------------------------
		# RICKON AND OSHA REMAIN WITH VISENYA
		# -----------------------------------------------------

		segrellion_black_unicorn_rickon_osha_join_visenya_effect = yes

		# -----------------------------------------------------
		# VISENYA ENTERS THE BEACON PEACEFULLY
		# -----------------------------------------------------

		character:Black_Unicorn_1 = {

			remove_character_flag = segrellion_black_unicorn_lock_camp_skagos

			current_travel_plan ?= {
				abort_travel_plan = yes
			}

			set_location = title:b_castle_farring.title_province

			add_character_flag = segrellion_black_unicorn_infiltrated_beacon
			add_character_flag = segrellion_black_unicorn_peaceful_beacon_arrival
		}

		character:Stark_Dummy_11 ?= {

			if = {

				limit = {
					is_alive = yes
					is_courtier_of = character:Black_Unicorn_1
				}

				set_location = title:b_castle_farring.title_province
				add_character_flag = segrellion_rickon_at_beacon_with_visenya
			}
		}

		character:Stark_hh_13 ?= {

			if = {

				limit = {
					is_alive = yes
					is_courtier_of = character:Black_Unicorn_1
				}

				set_location = title:b_castle_farring.title_province
				add_character_flag = segrellion_osha_at_beacon_with_visenya
			}
		}

		# -----------------------------------------------------
		# REMEMBER THE TONE OF THE MEETING
		# -----------------------------------------------------

		if = {

			limit = {

				OR = {
					has_character_flag = segrellion_black_unicorn_shared_night
					has_character_flag = segrellion_black_unicorn_daemon_slept_with_visenya
					has_character_flag = segrellion_black_unicorn_visenya_trust_earned
					has_character_flag = segrellion_black_unicorn_visenya_declined_first_return
				}
			}

			add_character_flag = segrellion_black_unicorn_path_iii_tone_warm
		}

		else_if = {

			limit = {

				OR = {
					has_character_flag = segrellion_black_unicorn_future_beacon_ambush
					has_character_flag = segrellion_black_unicorn_left_without_truth
					has_character_flag = segrellion_black_unicorn_left_skagos_without_visenya
				}
			}

			add_character_flag = segrellion_black_unicorn_path_iii_tone_tense
		}

		else = {
			add_character_flag = segrellion_black_unicorn_path_iii_tone_guarded
		}
	}

	left_portrait = {
		character = root
		animation = shock
	}

	right_portrait = {
		character = character:Black_Unicorn_1
		animation = personality_bold
	}

	option = {

		name = segrellion_black_unicorn.0910.a
		custom_tooltip = segrellion_black_unicorn.0910.a.tt

		trigger_event = {
			id = segrellion_black_unicorn.0911
			days = 1
		}
	}
}
