# =========================================================
# 1000 - HUGH SETS OUT
# =========================================================

segrellion_hugh_black_unicorn.1000 = {

	type = character_event
	hidden = yes


	trigger = {

		this = character:Segrellion_43

		is_alive = yes

		has_character_flag = hugh_intends_skagos_journey
	}


	immediate = {

		remove_character_flag = hugh_black_unicorn_journey_queued
		add_character_flag = hugh_black_unicorn_journey_started


		location = {
			save_scope_as = hugh_black_unicorn_origin_scope
		}


		if = {

			limit = {
				exists = scope:hugh_black_unicorn_origin_scope
			}


			set_variable = {

				name = hugh_black_unicorn_origin

				value = scope:hugh_black_unicorn_origin_scope
			}
		}


		current_travel_plan ?= {
			abort_travel_plan = yes
		}


		start_travel_plan = {

			destination = title:b_bitter_dryn.title_province

			return_trip = no

			on_arrival_event = segrellion_hugh_black_unicorn.1010

			on_arrival_destinations = first

			on_travel_planner_cancel_event = segrellion_hugh_black_unicorn.1098
		}
	}
}
