				has_trait = compassionate
			}

			modifier = {
				add = 25
				has_trait = just
			}

			modifier = {
				add = 15
				has_trait = brave
			}
		}
	}

# 0010
# CHANGE PATH
# =========================================================

segrellion_knightly_ideal.0010 = {

	type = character_event

	title = segrellion_knightly_ideal.0010.t
	desc = segrellion_knightly_ideal.0010.desc


	# =====================================================
	# BLADE
	# =====================================================

	option = {

		name = segrellion_knightly_ideal.0010.a


		trigger = {

			NOT = {
				has_character_flag = segrellion_knightly_focus_blade
			}

			NOT = {

				has_trait_xp = {
					trait = segrellion_knightly_ideal_trait
					track = knightly_blade
					value >= 100
				}
			}
		}


		custom_tooltip = segrellion_knightly_choose_blade_tt

		segrellion_knightly_ideal_select_blade_effect = yes


		add_character_flag = {
			flag = segrellion_knightly_focus_change_cooldown

segrellion_knightly_ideal.9000 = {

	type = character_event

	title = segrellion_knightly_ideal.9000.t

	desc = {

		first_valid = {


			triggered_desc = {

				trigger = {
					has_character_flag = is_Vladwyn_1
				}

				desc = segrellion_knightly_ideal.9000.desc_tristan
			}


			desc = segrellion_knightly_ideal.9000.desc
		}
	}


	immediate = {

		if = {

			limit = {

				NOT = {
					has_trait = segrellion_truest_warrior_trait
				}
			}

			add_trait = segrellion_truest_warrior_trait
		}
	}


	option = {

		name = segrellion_knightly_ideal.9000.a
	}
}
