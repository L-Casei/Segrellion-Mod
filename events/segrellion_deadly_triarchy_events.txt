# =========================================================
# 0001 - DAEMON SEGRELLION
# =========================================================

segrellion_deadly_triarchy.0001 = {

	type = character_event

	title = segrellion_deadly_triarchy.0001.t
	desc = segrellion_deadly_triarchy.0001.desc

	theme = war


	trigger = {

		this = character:Segrellion_60
		is_alive = yes

		has_character_flag = segrellion_deadly_triarchy_formed


		character:IAF_Young_Griff_1 ?= {

			is_alive = yes

			has_character_flag = segrellion_deadly_triarchy_formed
			has_character_flag = segrellion_young_griff_supports_daemon
		}


		character:Vladwyn_5 ?= {

			is_alive = yes

			has_character_flag = segrellion_deadly_triarchy_formed
		}


		# The foothold in the Stormlands has already been secured.
		title:e_the_stormlands.holder = character:IAF_Young_Griff_1
	}


	left_portrait = {

		character = root
		animation = personality_bold
	}


	option = {

		name = segrellion_deadly_triarchy.0001.a

		custom_tooltip = segrellion_deadly_triarchy.formation.tt

		ai_chance = {
			base = 100
		}
	}
}

# =========================================================
# 0003 - YOUNG GRIFF
# =========================================================

segrellion_deadly_triarchy.0003 = {

	type = character_event

	title = segrellion_deadly_triarchy.0003.t
	desc = segrellion_deadly_triarchy.0003.desc

	theme = war


	trigger = {

		this = character:IAF_Young_Griff_1
		is_alive = yes

		has_character_flag = segrellion_deadly_triarchy_formed
		has_character_flag = segrellion_young_griff_supports_daemon


		character:Segrellion_60 ?= {

			is_alive = yes

			has_character_flag = segrellion_deadly_triarchy_formed
			has_character_flag = segrellion_daemon_supported_by_young_griff
		}


		character:Vladwyn_5 ?= {

			is_alive = yes

			has_character_flag = segrellion_deadly_triarchy_formed
		}


		title:e_the_stormlands.holder = character:IAF_Young_Griff_1
	}


	left_portrait = {

		character = root
		animation = personality_bold
	}


	option = {

		name = segrellion_deadly_triarchy.0003.a

		custom_tooltip = segrellion_deadly_triarchy.formation.tt


		hidden_effect = {

			# =============================================
			# YOUNG GRIFF CANON STORY
			#
			# Once the Stormlands foothold is secured and
			# the Deadly Triarchy is established, Varys
			# makes his first move from King's Landing.
			# =============================================

			if = {

				limit = {

					NOT = {
						has_character_flag = segrellion_young_griff_varys_contacted
					}

					NOT = {
						has_character_flag = segrellion_young_griff_varys_opening_scheduled
					}
				}


				add_character_flag = segrellion_young_griff_varys_opening_scheduled


				trigger_event = {

					id = segrellion_young_griff.0100

					days = 2
				}
			}
		}


		ai_chance = {
			base = 100
		}
	}
}
