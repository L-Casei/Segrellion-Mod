# =========================================================
# 0100 - THE SPIDER'S THREADS
#
# Root = Young Griff
# =========================================================

segrellion_young_griff.0100 = {

	type = character_event

	title = segrellion_young_griff.0100.t
	desc = segrellion_young_griff.0100.desc

	theme = intrigue

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		this = character:IAF_Young_Griff_1

		is_alive = yes


		# 300 AC Winds of Winter bookmark.
		game_start_date >= 8300.8.20
		game_start_date < 8301.1.1


		# Varys' storyline begins only once Aegon's
		# Stormlands coalition has actually formed.
		has_character_flag = segrellion_deadly_triarchy_formed


		title:e_the_stormlands.holder = character:IAF_Young_Griff_1


		NOT = {
			has_character_flag = segrellion_young_griff_varys_contacted
		}
	}


	on_trigger_fail = {

		remove_character_flag = segrellion_young_griff_varys_opening_scheduled
	}


	left_portrait = {

		character = root

		animation = thinking
	}

	right_portrait = {

		character = character:Eunuch_1

		animation = personality_rational
	}


	option = {

		name = segrellion_young_griff.0100.a


		remove_character_flag = segrellion_young_griff_varys_opening_scheduled

		add_character_flag = segrellion_young_griff_varys_contacted

		# Permanent story state for Varys' future operation.
		add_character_flag = segrellion_young_griff_varys_preparing_kings_landing


		add_gold = 500


		add_character_modifier = {

			modifier = segrellion_young_griff_spiders_support_modifier

			years = 5
		}


		segrellion_young_griff_try_schedule_arianne_storms_end_effect = yes
	}
}

segrellion_young_griff.0230 = {

	type = character_event

	title = segrellion_young_griff.0230.t
	desc = segrellion_young_griff.0230.desc

	theme = diplomacy


	trigger = {

		this = character:IAF_Young_Griff_1

		is_alive = yes

		has_character_flag = segrellion_young_griff_arianne_story_started


		NOT = {
			has_character_flag = segrellion_young_griff_arianne_route_resolved
		}


		character:Martell_9 ?= {

			is_alive = yes
		}
	}


	left_portrait = {

		character = root
		animation = personality_bold
	}


	right_portrait = {

		character = character:Martell_9
		animation = personality_bold
	}


	# =====================================================
	# MARRIAGE
	# AI = 40%
	# =====================================================

	option = {

		name = segrellion_young_griff.0230.a

		custom_tooltip = segrellion_young_griff.0230.a.tt


		trigger = {

			is_married = no

			character:Martell_9 ?= {
				is_married = no
			}
		}


		segrellion_young_griff_secure_dorne_support_effect = yes


		add_character_flag = segrellion_young_griff_arianne_route_resolved
		add_character_flag = segrellion_young_griff_married_arianne
		add_character_flag = segrellion_young_griff_marriage_pact_resolved


		add_opinion = {

			target = character:Martell_9

			modifier = segrellion_arianne_chosen_as_queen_opinion
		}


		character:Martell_9 = {

			add_opinion = {

				target = root

				modifier = segrellion_arianne_chosen_as_queen_opinion
			}
		}


		segrellion_young_griff_release_arianne_lock_effect = yes


		character:Martell_9 = {

			add_character_flag = segrellion_arianne_event_marriage_completed
			add_character_flag = segrellion_young_griff_married_arianne
		}


		marry = character:Martell_9


		ai_chance = {
			base = 40
		}
	}


	# =====================================================
	# POLITICAL ALLIANCE WITHOUT MARRIAGE
	# AI = 50%
	# =====================================================

	option = {

		name = segrellion_young_griff.0230.b

		custom_tooltip = segrellion_young_griff.0230.b.tt


		segrellion_young_griff_secure_dorne_support_effect = yes


		add_character_flag = segrellion_young_griff_arianne_route_resolved
		add_character_flag = segrellion_young_griff_arianne_alliance_only


		add_opinion = {

			target = character:Martell_9

			modifier = segrellion_arianne_dornish_alliance_opinion
		}


		character:Martell_9 = {

			add_opinion = {

				target = root

				modifier = segrellion_arianne_dornish_alliance_opinion
			}
		}


		segrellion_young_griff_release_arianne_lock_effect = yes

		segrellion_young_griff_arianne_return_south_effect = yes


		ai_chance = {
			base = 50
		}
	}


	# =====================================================
	# FAILURE
	# AI = 10%
	# =====================================================

	option = {

		name = segrellion_young_griff.0230.c

		custom_tooltip = segrellion_young_griff.0230.c.tt


		add_character_flag = segrellion_young_griff_arianne_route_resolved
		add_character_flag = segrellion_young_griff_dorne_route_closed


		character:Martell_9 = {

			add_character_flag = segrellion_arianne_rejected_young_griff


			add_opinion = {

				target = root

				modifier = segrellion_arianne_insulted_by_aegon_opinion
			}


			father ?= {

				add_character_flag = segrellion_dorne_rejected_young_griff

				add_opinion = {

					target = root

					modifier = segrellion_dorne_rejected_young_griff_opinion
				}
			}
		}


		add_opinion = {

			target = character:Martell_9

			modifier = segrellion_arianne_insulted_by_aegon_opinion
		}


		segrellion_young_griff_release_arianne_lock_effect = yes

		segrellion_young_griff_arianne_return_south_effect = yes


		ai_chance = {
			base = 10
		}

# =========================================================
# 1000 - DAEMON CHOSE STANNIS
# Root = Young Griff
# =========================================================

segrellion_young_griff.1000 = {

	type = character_event

	title = segrellion_young_griff.1000.t
	desc = segrellion_young_griff.1000.desc

	theme = diplomacy

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	trigger = {

		this = character:IAF_Young_Griff_1
		is_alive = yes

		NOT = {
			has_character_flag = segrellion_young_griff_daemon_choice_resolved
		}

		character:Segrellion_60 ?= {
			is_alive = yes
			has_character_flag = segrellion_pledged_to_stannis
		}
	}


	left_portrait = {
		character = root
		animation = thinking
	}

	right_portrait = {
		character = character:Segrellion_60
		animation = personality_bold
	}


	# =====================================================
	# LEAVE DAEMON
	#
	# AI: 100%
	# =====================================================

	option = {
	name = segrellion_young_griff.1000.a

	custom_tooltip = segrellion_young_griff_stannis_abandon_daemon_tt

	hidden_effect = {

		add_character_flag = segrellion_young_griff_daemon_choice_resolved
		add_character_flag = segrellion_young_griff_left_daemon_stannis
		add_character_flag = segrellion_young_griff_canon_path_1
		remove_character_flag = segrellion_young_griff_canon_path_2

		segrellion_young_griff_break_daemon_alliance_effect = yes

		add_opinion = {
			modifier = segrellion_chose_stannis_over_young_griff_opinion
		target = character:Segrellion_60
			}

			segrellion_young_griff_leave_daemon_restoration_effect = yes

			# =====================================================
			# AEGON NOW FOLLOWS HIS OWN ROAD
			#
			# He no longer waits for The Unicorn Returns.
			# Six months later he reassesses his position.
			# =====================================================

			if = {

				limit = {

					NOT = {
						has_character_flag = segrellion_young_griff_independent_throne_path_scheduled
					}

					NOT = {
						has_character_flag = segrellion_young_griff_independent_throne_path_started
					}
				}


				add_character_flag = segrellion_young_griff_independent_throne_path_scheduled


				trigger_event = {

					id = segrellion_young_griff.7900

					months = 6
				}
			}

			character:Segrellion_60 = {
				trigger_event = {
					id = segrellion_young_griff.1100
					days = 1
				}
			}
		}

		ai_chance = {
			base = 100
		}
	}


	# =====================================================
	# KEEP HIS WORD
	#
	# Player-only deviation.
	# AI: 0%
	# =====================================================

	option = {
		name = segrellion_young_griff.1000.b

		custom_tooltip = segrellion_young_griff_stannis_support_daemon_tt

		hidden_effect = {

			add_character_flag = segrellion_young_griff_daemon_choice_resolved
			add_character_flag = segrellion_young_griff_stayed_daemon_stannis
			add_character_flag = segrellion_young_griff_canon_path_2
			remove_character_flag = segrellion_young_griff_canon_path_1

			# Political alliance still ends because Daemon chose Stannis.
			segrellion_young_griff_break_daemon_alliance_effect = yes

			add_opinion = {
				modifier = segrellion_chose_stannis_over_young_griff_opinion
				target = character:Segrellion_60
			}

			# But Aegon keeps his promise to fight for the Flamelands.
			segrellion_young_griff_stay_with_daemon_restoration_effect = yes

			character:Segrellion_60 = {
				trigger_event = {
					id = segrellion_young_griff.1100
					days = 1
				}
			}
		}

		ai_chance = {
			base = 0
		}
	}
}

					has_character_flag = segrellion_young_griff_married_arianne
				}

				trigger = {
					has_character_flag = segrellion_young_griff_arianne_offer_lost
				}

				desc = segrellion_young_griff.4121.desc_lost
			}

			desc = segrellion_young_griff.4121.desc_reject
		}
	}

	theme = marriage

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	trigger = {

		this = character:IAF_Young_Griff_1
		is_alive = yes
	}

	immediate = {

		character:Martell_9 ?= {

			save_scope_as = yg_arianne

			father ?= {
				save_scope_as = yg_doran
			}
		}


		# -------------------------------------------------
		# DORAN ACCEPTED AND THE MATCH REMAINS VALID
		# -------------------------------------------------

		if = {

			limit = {

				has_character_flag = segrellion_young_griff_arianne_offer_accepted

				character:Martell_9 ?= {

					is_alive = yes
					is_married = no
				}

				character:Segrellion_60 ?= {

					NOR = {
						has_character_flag = segrellion_negotiation_pending_dorne
						has_character_flag = segrellion_dorne_marriage_pact
					}
				}
			}

			character:Martell_9 = {

				if = {
					limit = {
						has_trait = refusing_marriage
					}

					remove_trait = refusing_marriage
				}

				remove_character_flag = segrellion_arianne_marriage_lock_active
				remove_character_flag = segrellion_young_griff_arianne_offer_accepted_pending

				add_character_flag = segrellion_arianne_event_marriage_completed
			}

			marry = character:Martell_9

			add_character_flag = segrellion_young_griff_married_arianne
			add_character_flag = segrellion_young_griff_marriage_pact_resolved

			character:Martell_9 = {
				add_character_flag = segrellion_young_griff_married_arianne
			}

		# -------------------------------------------------

			limit = {
				has_character_flag = segrellion_young_griff_arianne_offer_accepted
			}

			id = segrellion_young_griff.6010

		}

# =========================================================
# 6010 - THE BANNERS GATHER
#
# Also verifies that Aegon's Iron Throne war was actually
# created.
#
# Martells + Vladwyn + Daemon if pledged.
# =========================================================

segrellion_young_griff.6010 = {

	type = character_event

	hidden = yes


	trigger = {

		this = character:IAF_Young_Griff_1

		is_alive = yes

		has_character_flag = segrellion_young_griff_daemon_war_resolved
	}


	immediate = {


		# =================================================
		# FIND THE ACTUAL WAR
		# =================================================

		random_character_war = {

			limit = {

				attacker = root


				casus_belli = {

					any_target_title = {

						this = title:h_the_iron_throne
					}
				}
			}


			save_scope_as = young_griff_iron_throne_war
		}


		# =================================================
		# WAR EXISTS
		# =================================================

		if = {

			limit = {

				exists = scope:young_griff_iron_throne_war
			}


			add_character_flag = segrellion_young_griff_iron_throne_war_started

			remove_character_flag = segrellion_young_griff_invasion_event_failed

			remove_character_flag = segrellion_young_griff_invasion_event_scheduled


			# =============================================
			# DORAN MARTELL
			#
			# Dorne ONLY joins if Arianne's Storm's End
			# storyline actually secured its support.
			# =============================================

			character:Martell_9 ?= {

				father ?= {

					if = {

						limit = {

							is_alive = yes

							this != title:h_the_iron_throne.holder


							character:IAF_Young_Griff_1 = {

								has_character_flag = segrellion_young_griff_dorne_support_secured
							}
						}


						save_scope_as = young_griff_doran
					}
				}
			}


			# =============================================
			# JOFFREY VLADWYN
			# =============================================

			character:Vladwyn_5 ?= {

				if = {

					limit = {

						is_alive = yes

						this != title:h_the_iron_throne.holder
					}


					save_scope_as = young_griff_vladwyn
				}
			}


			# =============================================
			# DAEMON
			#
			# ONLY if he actually swore to Aegon.
			# =============================================

			character:Segrellion_60 ?= {

				if = {

					limit = {

						is_alive = yes

						has_character_flag = segrellion_pledged_to_young_griff

						this != title:h_the_iron_throne.holder
					}


					save_scope_as = young_griff_daemon
				}
			}


			# =============================================
			# ADD SUPPORTERS
			# =============================================

			scope:young_griff_iron_throne_war = {


				if = {

					limit = {

						exists = scope:young_griff_doran
					}


					add_attacker = scope:young_griff_doran
				}


				if = {

					limit = {

						exists = scope:young_griff_vladwyn
					}


					add_attacker = scope:young_griff_vladwyn
				}


				if = {

					limit = {

						exists = scope:young_griff_daemon
					}


					add_attacker = scope:young_griff_daemon
				}
			}
		}


		# =================================================
		# WAR DID NOT MATERIALIZE
		#
		# Do NOT freeze the storyline.
		#
		# The monthly failsafe will try again.
		# =================================================

		else = {

			remove_character_flag = segrellion_young_griff_iron_throne_war_started

			remove_character_flag = segrellion_young_griff_invasion_event_scheduled

			add_character_flag = segrellion_young_griff_invasion_event_failed
		}
	}
}
