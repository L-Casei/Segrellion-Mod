# =========================================================
# 1000 - DAEMON CHOSE STANNIS
# Root = Young Griff
# =========================================================

segrellion_young_griff.1000 = {

	type = character_event

	title = segrellion_young_griff.1000.t
	desc = segrellion_young_griff.1000.desc

	theme = diplomacy

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	trigger = {

		this = character:IAF_Young_Griff_1
		is_alive = yes

		NOT = {
			has_character_flag = segrellion_young_griff_daemon_choice_resolved
		}

		character:Segrellion_60 ?= {
			is_alive = yes
			has_character_flag = segrellion_pledged_to_stannis
		}
	}


	immediate = {

		# Daemon has chosen Stannis's cause.
		# Their formal alliance cannot survive this.

		segrellion_young_griff_break_daemon_alliance_effect = yes

		add_opinion = {
			target = character:Segrellion_60
			modifier = segrellion_chose_stannis_over_young_griff_opinion
		}
	}


	left_portrait = {
		character = root
		animation = thinking
	}

	right_portrait = {
		character = character:Segrellion_60
		animation = personality_bold
	}


	# =====================================================
	# LEAVE DAEMON
	#
	# AI: 100%
	# =====================================================

	option = {
	name = segrellion_young_griff.1000.a

	custom_tooltip = segrellion_young_griff_stannis_abandon_daemon_tt

	hidden_effect = {

		add_character_flag = segrellion_young_griff_daemon_choice_resolved
		add_character_flag = segrellion_young_griff_left_daemon_stannis

		segrellion_young_griff_break_daemon_alliance_effect = yes

		add_opinion = {
			modifier = segrellion_chose_stannis_over_young_griff_opinion
		target = character:Segrellion_60
			}

			segrellion_young_griff_leave_daemon_restoration_effect = yes

			character:Segrellion_60 = {
				trigger_event = {
					id = segrellion_young_griff.1100
					days = 1
				}
			}
		}

		ai_chance = {
			base = 100
		}
	}


	# =====================================================
	# KEEP HIS WORD
	#
	# Player-only deviation.
	# AI: 0%
	# =====================================================

	option = {
		name = segrellion_young_griff.1000.b

		custom_tooltip = segrellion_young_griff_stannis_support_daemon_tt

		hidden_effect = {

			add_character_flag = segrellion_young_griff_daemon_choice_resolved
			add_character_flag = segrellion_young_griff_stayed_daemon_stannis

			# Political alliance still ends because Daemon chose Stannis.
			segrellion_young_griff_break_daemon_alliance_effect = yes

			add_opinion = {
				modifier = segrellion_chose_stannis_over_young_griff_opinion
				target = character:Segrellion_60
			}

			# But Aegon keeps his promise to fight for the Flamelands.
			segrellion_young_griff_stay_with_daemon_restoration_effect = yes

			character:Segrellion_60 = {
				trigger_event = {
					id = segrellion_young_griff.1100
					days = 1
				}
			}
		}

		ai_chance = {
			base = 0
		}
	}
}

# =========================================================
# 5010 - THE BLACK DRAGON UNMASKED
# =========================================================

segrellion_young_griff.5010 = {

	type = character_event

	title = segrellion_young_griff.5010.t
	desc = segrellion_young_griff.5010.desc

	theme = crown

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	trigger = {

		this = character:IAF_Young_Griff_1

		is_alive = yes

		has_character_flag = segrellion_young_griff_true_blackfyre

		NOT = {
			has_character_flag = segrellion_young_griff_blackfyre_revealed
		}
	}


	immediate = {

		# =================================================
		# TRUE PARENTAGE
		# Uses AGOT's own Blackfyre fAegon parentage.
		# =================================================

		set_father = character:Mopatis_1
		set_mother = character:Mopatis_hh_1

		set_house = house:house_Blackfyre


		# =================================================
		# THE DISGUISE ENDS
		# =================================================

		if = {

			limit = {
				has_character_modifier = shade_of_the_evening_hair_dye
			}

			remove_character_modifier = shade_of_the_evening_hair_dye
		}


		# =================================================
		# THE TARGARYEN CLAIM WAS FALSE
		# =================================================

		if = {

			limit = {
				has_claim_on = title:h_the_iron_throne
			}

			remove_claim = title:h_the_iron_throne
		}


		# =================================================
		# POLITICAL CONSEQUENCES
		# =================================================

		add_legitimacy = -300

		add_character_modifier = {
			modifier = segrellion_exposed_blackfyre_pretender_modifier
			years = 20
		}


		remove_character_flag = segrellion_young_griff_blackfyre_reveal_scheduled
		remove_character_flag = segrellion_young_griff_blackfyre_truth_circulating

		add_character_flag = segrellion_young_griff_blackfyre_revealed


		# Daenerys takes the deception particularly badly.
		character:Targaryen_9 ?= {

			if = {

				limit = {
					is_alive = yes
				}

				add_opinion = {
					target = root
					modifier = segrellion_exposed_blackfyre_to_daenerys_opinion
				}
			}
		}
	}


	left_portrait = {
		character = root
		animation = shock
	}


	option = {

		name = segrellion_young_griff.5010.a
	}
}
