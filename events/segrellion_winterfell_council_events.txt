# =========================================================
# FINAL COUNCIL RESOLVER
# =========================================================

segrellion_winterfell_council.4900 = {

	type = activity_event

	hidden = yes

	trigger = {

		this = scope:host

		scope:activity = {

			has_activity_type = activity_segrellion_winterfell_council

			has_current_phase = segrellion_winterfell_council_proclamation_phase
		}
	}

	immediate = {

		segrellion_winterfell_council_select_candidate_effect = yes


		# -------------------------------------------------
		# OVERWHELMING NO-CROWN FACTION
		# -------------------------------------------------

		if = {
			limit = {

				scope:activity = {
					var:segrellion_council_no_crown_support >= 12
				}
			}

			trigger_event = {
				id = segrellion_winterfell_council.4950
				days = 1
			}
		}


		# -------------------------------------------------
		# OVERWHELMING LONG NIGHT FACTION
		# -------------------------------------------------

		else_if = {
			limit = {

				scope:activity = {

					var:segrellion_council_wall_support >= 12

					var:segrellion_council_independence_support < 10
				}
			}

			trigger_event = {
				id = segrellion_winterfell_council.4920
				days = 1
			}
		}


		# -------------------------------------------------
		# OVERWHELMING DRAGON ALLIANCE FACTION
		# -------------------------------------------------

		else_if = {
			limit = {

				scope:activity = {

					var:segrellion_council_dragon_support >= 10

					var:segrellion_council_independence_support < 10
				}
			}

			trigger_event = {
				id = segrellion_winterfell_council.4930
				days = 1
			}
		}


		# -------------------------------------------------
		# CROWN A KING
		# -------------------------------------------------

		else_if = {
			limit = {

				scope:activity = {

					var:segrellion_council_independence_support >= 6

					var:segrellion_council_winning_candidate = {

						exists = var:segrellion_council_king_support

						var:segrellion_council_king_support >= 6
					}
				}
			}


			if = {
				limit = {

					this = scope:activity.var:segrellion_council_winning_candidate
				}

				trigger_event = {
					id = segrellion_jon_frey.0020
					days = 1
				}
			}

			else = {

				trigger_event = {
					id = segrellion_winterfell_council.4910
					days = 1
				}
			}
		}


		# -------------------------------------------------
		# INDEPENDENT NORTH, NO KING
		# -------------------------------------------------

		else_if = {
			limit = {
				scope:activity = {
					var:segrellion_council_independence_support >= 5
				}
			}

			trigger_event = {
				id = segrellion_winterfell_council.4940
				days = 1
			}
		}


		# -------------------------------------------------
		# WALL FALLBACK
		# -------------------------------------------------

		else_if = {
			limit = {
				scope:activity = {
					var:segrellion_council_wall_support >= 6
				}
			}

			trigger_event = {
				id = segrellion_winterfell_council.4920
				days = 1
			}
		}


		# -------------------------------------------------
		# DRAGON FALLBACK
		# -------------------------------------------------

		else_if = {
			limit = {
				scope:activity = {
					var:segrellion_council_dragon_support >= 6
				}
			}

			trigger_event = {
				id = segrellion_winterfell_council.4930
				days = 1
			}
		}


		else = {

			trigger_event = {
				id = segrellion_winterfell_council.4950
				days = 1
			}
		}
	}
}
