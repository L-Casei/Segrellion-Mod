				scope:activity = {

				id = segrellion_winterfell_council.4940

			}
		}

			limit = {

				scope:activity = {
					var:segrellion_council_strategy_dragon_won = 1
				}
			}

			trigger_event = {
				id = segrellion_winterfell_council.4930
