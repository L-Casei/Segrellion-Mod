segrellion_intro.0002 = {
	type = character_event
	title = segrellion_intro.0002.t
	desc = segrellion_intro.0002.desc
	theme = diplomacy
	override_header_background = {
		reference = "gfx/interface/window_event/event_header_blue.dds"
	}

	override_background = { reference = agot_dragonstone_throneroom }

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = character:Targaryen_63
		animation = personality_honorable
	}

	trigger = {
		has_character_flag = segrellion_opening_route_dragonstone
		NOT = { has_character_flag = segrellion_chose_stepstones }
		NOT = { has_character_flag = segrellion_returned_to_vhalerion }
	}

	option = {
		
		name = segrellion_intro.0002.a
		ai_chance = { base = 80 }

		add_character_flag = segrellion_chose_stepstones
		add_character_flag = segrellion_stepstones_war_joined
		add_character_flag = segrellion_stepstones_route_active
		add_character_flag = segrellion_stepstones_campaign_active

		add_prestige = 75

		custom_tooltip = segrellion_stepstones_route_started_tt
		
		hidden_effect = {

			if = {

				limit = {

					character:Targaryen_61 ?= {
						is_alive = yes
					}

					title:d_first_stepstones.holder ?= {
						exists = this
					}
				}


				character:Targaryen_61 = {

					# =================================================
					# PRIMARY PATH
					#
					# AGOT's Rogue Prince scenario should already have
					# created the canonical first Stepstones war.
					#
					# This remains only as a failsafe in case the war
					# does not exist for any reason.
					# =================================================

					if = {

						limit = {

							NOT = {

								any_character_war = {

									using_cb = daemon_stepstones_first_war

									primary_attacker ?= prev
								}
							}
						}

						start_war = {

							casus_belli = daemon_stepstones_first_war

							target = title:d_first_stepstones.holder

							claimant = this

							target_title = title:d_first_stepstones
						}
					}


					random_character_war = {

						limit = {

							using_cb = daemon_stepstones_first_war

							primary_attacker ?= prev
						}


						set_variable = {
							name = segrellion_tracked_first_stepstones_war
							value = yes
						}


						# =================================================
						# CANON SAFETY
						#
						# Corlys Velaryon must fight alongside Daemon
						# from the beginning of the first Stepstones war.
						#
						# Normally AGOT's own Rogue Prince setup handles
						# this. This is only a defensive fallback.
						# =================================================

						if = {

							limit = {

								character:Velaryon_31 ?= {
									is_alive = yes
								}

								NOR = {

									any_war_attacker = {
										this = character:Velaryon_31
									}

									any_war_defender = {
										this = character:Velaryon_31
									}
								}
							}

							add_attacker = character:Velaryon_31
						}


						# =================================================
						# ALEX JOINS THE CANONICAL WAR
						# =================================================

						if = {

							limit = {

								NOR = {

									any_war_attacker = {
										this = root
									}

									any_war_defender = {
										this = root
									}
								}
							}

							add_attacker = root
						}


						root = {
							add_character_flag = segrellion_stepstones_war_joined
						}


						save_scope_as = segrellion_stepstones_war
					}
				}


				# =================================================
				# DAEMON'S STEPSTONES VOLUNTEERS
				#
				# Persistent special troops.
				# They are NOT tied to the first war.
				# =================================================

				character:Targaryen_61 = {

					if = {

						limit = {

							NOT = {
								has_character_flag = segrellion_daemon_stepstones_special_troops_granted
							}
						}

						add_character_flag = segrellion_daemon_stepstones_special_troops_granted

						spawn_army = {

							levies = 500

							location = character:Targaryen_61.capital_province
							origin = character:Targaryen_61.capital_province

							name = segrellion_daemon_stepstones_volunteers_army

							inheritable = no
							uses_supply = yes
						}
					}
				}


				# =================================================
				# ALEX'S STEPSTONES VOLUNTEERS
				#
				# Persistent special troops.
				# They remain available after the first war.
				# =================================================

				if = {

					limit = {

						NOT = {
							has_character_flag = segrellion_alex_stepstones_special_troops_granted
						}
					}

					add_character_flag = segrellion_alex_stepstones_special_troops_granted

					spawn_army = {

						levies = 250

						location = root.capital_province
						origin = root.capital_province

						name = segrellion_stepstones_volunteers_army

						inheritable = no
						uses_supply = yes
					}
				}
			}
		}
	}

	option = {
		name = segrellion_intro.0002.b
		ai_chance = { base = 20 }
		add_character_flag = segrellion_returned_to_vhalerion
		custom_tooltip = segrellion_vhalerion_route_continues_tt
		trigger_event = {
			id = segrellion_intro.0003
			days = 21
		}
	}
}

segrellion_intro.0190 = {

	type = character_event

	hidden = yes


	trigger = {

		is_alive = yes

		has_character_flag = segrellion_stepstones_campaign_active
		has_character_flag = segrellion_stepstones_second_war_completed

		NOT = {
			has_character_flag = segrellion_stepstones_story_completed
		}
	}


	immediate = {


		# Daemon died before the campaign was resolved.

		if = {

			limit = {

				character:Targaryen_61 ?= {
					is_alive = no
				}
			}

			add_character_flag = segrellion_stepstones_daemon_failed
			add_character_flag = segrellion_stepstones_aftermath_pending

			trigger_event = {
				id = segrellion_intro.0008
				days = 1
			}
		}


		# A Triarchy counter-war is currently active.
		# Its normal war-result hooks will resolve the story.

		else_if = {

			limit = {

				character:Targaryen_61 ?= {

					any_character_war = {

						using_cb = triarchy_anti_daemon_war

						primary_defender ?= prev
					}
				}
			}

			trigger_event = {
				id = segrellion_intro.0190
				days = 90
			}
		}


		# Daemon has consolidated the Stepstones and no
		# counter-war remains.

		else_if = {

			limit = {

				character:Targaryen_61 ?= {

					is_alive = yes

					has_title = title:k_the_stepstones
				}
			}

			trigger_event = {
				id = segrellion_intro.0103
				days = 1
