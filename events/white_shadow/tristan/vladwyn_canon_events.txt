vladwyn_canon.1000 = {

	}

		character:Vladwyn_1 ?= {

			is_alive = yes
		}
	}


	immediate = {

		character:Vladwyn_1 = {

			add_character_flag = vladwyn_met_elyse_sunglass

			remove_character_flag = vladwyn_elyse_meeting_queued
		}
	}


	option = {

		name = vladwyn_canon.1000.a


		character:Vladwyn_1 = {

			add_prestige = 50
		}


		trigger_event = {
			id = vladwyn_canon.1010
			days = { 60 120 }
		}
	}
}


# =========================================================
# TRISTAN AND ELYSE MARRY
# =========================================================

vladwyn_canon.1010 = {

	type = character_event

	title = vladwyn_canon.1010.t
	desc = vladwyn_canon.1010.desc

	theme = family


	left_portrait = {
		character = character:Vladwyn_1
		animation = personality_honorable
	}


	right_portrait = {
		character = character:IAF_Sunglass_2
		animation = personality_compassionate
	}


	trigger = {

		character:Vladwyn_1 ?= {

			is_alive = yes

			has_character_flag = vladwyn_met_elyse_sunglass

			NOT = {
				has_character_flag = vladwyn_married_elyse_sunglass
			}
		}

		character:IAF_Sunglass_2 ?= {
			is_alive = yes
		}
	}


	immediate = {

		character:Vladwyn_1 = {

			remove_character_flag = vladwyn_elyse_marriage_queued


			# Remove any non-canon marriage which may have
			# happened before the storyline reached 114 AC.

			if = {

				limit = {
					is_married = yes
				}

				every_spouse = {

					limit = {
						NOT = {
							this = character:IAF_Sunglass_2
						}
					}

					divorce = character:Vladwyn_1
				}
			}


			if = {

				limit = {
					exists = betrothed
				}

				break_betrothal = betrothed
			}
		}


		character:IAF_Sunglass_2 = {

			if = {

				limit = {
					is_married = yes
				}

				every_spouse = {

					limit = {
						NOT = {
							this = character:Vladwyn_1
						}
					}

					divorce = character:IAF_Sunglass_2
				}
			}


			if = {

				limit = {
					exists = betrothed
				}

				break_betrothal = betrothed
			}
		}
	}


	option = {

		name = vladwyn_canon.1010.a


		hidden_effect = {

			character:IAF_Sunglass_2 = {

				if = {

					limit = {

						NOT = {
							any_spouse = {
								this = character:Vladwyn_1
							}
						}
					}

					add_spouse = character:Vladwyn_1
				}
			}


			character:Vladwyn_1 = {

				add_character_flag = vladwyn_married_elyse_sunglass
				add_character_flag = vladwyn_canon_family_started
			}
		}
	}
}

# 123 AC
# RHAENYRA'S SWORN SWORD
# =========================================================

vladwyn_canon.1030 = {

	type = character_event

	title = vladwyn_canon.1030.t
	desc = vladwyn_canon.1030.desc

	theme = diplomacy


	left_portrait = {
		character = character:Targaryen_63
		animation = personality_bold
	}


	right_portrait = {
		character = character:Vladwyn_1
		animation = personality_honorable
	}


	trigger = {

		character:Vladwyn_1 ?= {

			is_alive = yes

			has_character_flag = vladwyn_elyse_died

			NOT = {
				has_character_flag = vladwyn_rhaenyra_household_captain
			}
		}

		character:Targaryen_63 ?= {
			is_alive = yes
			is_landed = yes
		}
	}


	immediate = {

		character:Vladwyn_1 = {
			remove_character_flag = vladwyn_rhaenyra_service_queued
		}
	}


	option = {

		name = vladwyn_canon.1030.a


		hidden_effect = {

			character:Vladwyn_1 = {

				set_employer = character:Targaryen_63

				add_character_flag = vladwyn_rhaenyra_sworn_sword
				add_character_flag = vladwyn_rhaenyra_master_at_arms
				add_character_flag = vladwyn_rhaenyra_household_captain


				add_character_modifier = {
					modifier = vladwyn_rhaenyra_household_captain_modifier
				}
			}


			character:Targaryen_63 = {

				assign_councillor_type = {
					type = councillor_marshal
					target = character:Vladwyn_1
				}


				if = {

					limit = {

						NOT = {
							has_relation_friend = character:Vladwyn_1
						}

						NOT = {
							has_relation_best_friend = character:Vladwyn_1
						}
					}

					set_relation_friend = {
						target = character:Vladwyn_1
						reason = friend_shared_interests
					}
				}
			}
		}
	}
}
