vladwyn_adventurer.0001 = {
	type = character_event
	title = vladwyn_adventurer.0001.t
	desc = vladwyn_adventurer.0001.desc
	theme = travel
	override_background = { reference = wilderness }

	left_portrait = {
		character = root
		animation = personality_bold
	}

	option = {
		name = vladwyn_adventurer.0001.a
		add_character_flag = vladwyn_landless_route_unlocked
		add_character_flag = vladwyn_open_road_taken
		add_character_flag = vladwyn_adventurer_events_unlocked
		primary_title ?= {
			save_title_name_change_data = { TRANSFER_VAR = yes }
			reset_title_prefix = yes
			set_title_name = vladwyn_the_honorables_name
		}
		add_prestige = 200
		add_martial_lifestyle_xp = 250
		add_character_modifier = {
			modifier = vladwyn_wandering_knight_modifier
			years = 8
		}
	}
}

vladwyn_adventurer.0005 = {
	type = character_event
	title = vladwyn_adventurer.0005.t
	desc = vladwyn_adventurer.0005.desc
	theme = diplomacy
	override_background = { reference = council_chamber }

	left_portrait = {
		character = root
		animation = personality_bold
	}

	option = {
		name = vladwyn_adventurer.0005.a
		add_character_flag = vladwyn_claimed_blood_of_storm
		add_pressed_claim = title:e_the_stormlands
		add_prestige = 750
		dynasty = { add_dynasty_prestige = 300 }
		add_character_modifier = {
			modifier = vladwyn_blood_of_the_storm_modifier
			years = 10
		}
		if = {
			limit = {
				title:e_the_stormlands = { exists = holder }
			}
			title:e_the_stormlands.holder = {
				add_opinion = {
					target = root
					modifier = vladwyn_claimed_baratheon_blood_opinion
				}
			}
		}
		house:house_BaratheonSE = {
			every_house_member = {
				limit = {
					is_alive = yes
					NOT = { this = root }
				}
				add_opinion = {
					target = root
					modifier = vladwyn_claimed_baratheon_blood_opinion
				}
			}
		}
	}
}
