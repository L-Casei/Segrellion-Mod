# WESTERLANDS
segrellion_daenerys_settlement.0001 = {

	type = character_event

	hidden = yes


	immediate = {

		if = {

			limit = {

				OR = {

					has_character_flag = segrellion_daenerys_tyranny_revolt_pending

					has_character_flag = segrellion_daenerys_tyranny_revolt_active
				}
			}


			trigger_event = {

				id = segrellion_daenerys_settlement.0001

				days = 30
			}
		}


		else_if = {

			limit = {

				this = character:Targaryen_9

				has_title = title:h_the_iron_throne


				NOT = {
					has_character_flag = segrellion_daenerys_lost_tyranny_revolt
				}
			}


			segrellion_daenerys_clear_current_settlement_region_effect = yes

			add_character_flag = segrellion_settlement_region_westerlands

			segrellion_daenerys_route_current_settlement_effect = yes
		}
	}
}

segrellion_daenerys_settlement.0100 = {

	type = character_event

	title = segrellion_daenerys_settlement.0100.t

	desc = segrellion_daenerys_settlement.0100.desc

	theme = diplomacy


	override_header_background = {
		reference = "gfx/interface/window_event/event_header_blue.dds"
	}


	immediate = {

		var:segrellion_daenerys_current_settlement_holder ?= {

			save_scope_as = segrellion_settlement_holder
		}
	}


	left_portrait = {

		character = root

		animation = personality_rational
	}


	right_portrait = {

		character = scope:segrellion_settlement_holder

		animation = personality_bold
	}


	# =====================================================
	# HONOR MARRIAGE
	# =====================================================

	option = {

		name = segrellion_daenerys_settlement.0100.marriage


		trigger = {

			scope:segrellion_settlement_holder = {

				has_character_flag = segrellion_daenerys_promised_marriage_for_support

				is_married = no
			}


			is_married = no
		}


		segrellion_daenerys_current_region_swears_fealty_effect = yes

		segrellion_daenerys_pardon_current_region_supporters_effect = yes


		add_spouse = scope:segrellion_settlement_holder


		scope:segrellion_settlement_holder = {

			remove_character_flag = segrellion_daenerys_promised_marriage_for_support

			add_character_flag = segrellion_daenerys_marriage_promise_honored


			add_opinion = {

				target = root

				modifier = segrellion_daenerys_honored_support_bargain_opinion

				years = 20
			}
		}


		remove_character_flag = segrellion_daenerys_has_post_throne_marriage_promise

		remove_character_flag = segrellion_daenerys_support_marriage_already_promised


		segrellion_daenerys_finish_current_settlement_effect = yes
	}


	# =====================================================
	# HONOR FAVORABLE CONTRACT
	# =====================================================

	option = {

		name = segrellion_daenerys_settlement.0100.contract


		trigger = {

			scope:segrellion_settlement_holder = {

				has_character_flag = segrellion_daenerys_promised_favorable_vassal_contract
			}
		}


		segrellion_daenerys_current_region_swears_fealty_effect = yes

		segrellion_daenerys_honor_favorable_contract_effect = yes

		segrellion_daenerys_pardon_current_region_supporters_effect = yes


		scope:segrellion_settlement_holder = {

			add_opinion = {

				target = root

				modifier = segrellion_daenerys_honored_support_bargain_opinion

				years = 20
			}
		}


		segrellion_daenerys_finish_current_settlement_effect = yes
	}


	# =====================================================
	# HONOR HAND PROMISE
	# =====================================================

	option = {

		name = segrellion_daenerys_settlement.0100.hand


		trigger = {

			scope:segrellion_settlement_holder = {

				has_character_flag = segrellion_daenerys_promised_hand_of_the_queen
			}
		}


		segrellion_daenerys_current_region_swears_fealty_effect = yes

		segrellion_daenerys_pardon_current_region_supporters_effect = yes


		assign_councillor_type = {

			type = councillor_chancellor

			target = scope:segrellion_settlement_holder

			remove_existing_councillor = yes
		}


		scope:segrellion_settlement_holder = {

			remove_character_flag = segrellion_daenerys_promised_hand_of_the_queen

			add_character_flag = segrellion_hand_of_daenerys


			add_opinion = {

				target = root

				modifier = segrellion_daenerys_honored_support_bargain_opinion

				years = 20
			}
		}


		remove_character_flag = segrellion_daenerys_has_post_throne_hand_promise

		remove_character_flag = segrellion_daenerys_hand_already_promised


		segrellion_daenerys_finish_current_settlement_effect = yes
	}


	# =====================================================
	# HONOR COUNCIL PROMISE
	# =====================================================

	option = {

		name = segrellion_daenerys_settlement.0100.council


		trigger = {

			scope:segrellion_settlement_holder = {

				has_character_flag = segrellion_daenerys_promised_council_seat
			}
		}


		segrellion_daenerys_current_region_swears_fealty_effect = yes

		segrellion_daenerys_pardon_current_region_supporters_effect = yes


		trigger_event = {

			id = segrellion_daenerys_settlement.0110
		}
	}


	# =====================================================
	# HONOR FUTURE FAVOR
	# =====================================================

	option = {

		name = segrellion_daenerys_settlement.0100.favor


		trigger = {

			scope:segrellion_settlement_holder = {

				has_character_flag = segrellion_daenerys_owes_future_favor_for_support
			}
		}


		segrellion_daenerys_current_region_swears_fealty_effect = yes

		segrellion_daenerys_pardon_current_region_supporters_effect = yes


		scope:segrellion_settlement_holder = {

			add_hook = {

				target = root

				type = favor_hook
			}


			remove_character_flag = segrellion_daenerys_owes_future_favor_for_support

			add_character_flag = segrellion_daenerys_future_favor_paid


			add_opinion = {

				target = root

				modifier = segrellion_daenerys_honored_support_bargain_opinion

				years = 20
			}
		}


		segrellion_daenerys_finish_current_settlement_effect = yes
	}


	# =====================================================
	# NO PROMISE: 300 GOLD
	# =====================================================

	option = {

		name = segrellion_daenerys_settlement.0100.gold_small


		trigger = {

			gold >= 300


			scope:segrellion_settlement_holder = {

				NOR = {

					has_character_flag = segrellion_daenerys_promised_marriage_for_support

					has_character_flag = segrellion_daenerys_promised_favorable_vassal_contract

					has_character_flag = segrellion_daenerys_promised_hand_of_the_queen

					has_character_flag = segrellion_daenerys_promised_council_seat

					has_character_flag = segrellion_daenerys_owes_future_favor_for_support
				}
			}
		}


		add_gold = -300


		scope:segrellion_settlement_holder = {

			add_gold = 300


			add_opinion = {

				target = root

				modifier = segrellion_daenerys_rewarded_support_opinion

				years = 15
			}
		}


		segrellion_daenerys_current_region_swears_fealty_effect = yes

		segrellion_daenerys_pardon_current_region_supporters_effect = yes

		segrellion_daenerys_finish_current_settlement_effect = yes
	}


	# =====================================================
	# NO PROMISE: 600 GOLD
	# =====================================================

	option = {

		name = segrellion_daenerys_settlement.0100.gold_large


		trigger = {

			gold >= 600


			scope:segrellion_settlement_holder = {

				NOR = {

					has_character_flag = segrellion_daenerys_promised_marriage_for_support

					has_character_flag = segrellion_daenerys_promised_favorable_vassal_contract

					has_character_flag = segrellion_daenerys_promised_hand_of_the_queen

					has_character_flag = segrellion_daenerys_promised_council_seat

					has_character_flag = segrellion_daenerys_owes_future_favor_for_support
				}
			}
		}


		add_gold = -600


		scope:segrellion_settlement_holder = {

			add_gold = 600


			add_opinion = {

				target = root

				modifier = segrellion_daenerys_generously_rewarded_support_opinion

				years = 20
			}
		}


		segrellion_daenerys_current_region_swears_fealty_effect = yes

		segrellion_daenerys_pardon_current_region_supporters_effect = yes

		segrellion_daenerys_finish_current_settlement_effect = yes
	}


	# =====================================================
	# NO PROMISE: JUST THANK THEM
	# =====================================================

	option = {

		name = segrellion_daenerys_settlement.0100.thanks


		trigger = {

			scope:segrellion_settlement_holder = {

				NOR = {

					has_character_flag = segrellion_daenerys_promised_marriage_for_support

					has_character_flag = segrellion_daenerys_promised_favorable_vassal_contract

					has_character_flag = segrellion_daenerys_promised_hand_of_the_queen

					has_character_flag = segrellion_daenerys_promised_council_seat

					has_character_flag = segrellion_daenerys_owes_future_favor_for_support
				}
			}
		}

			add_opinion = {

				target = root

				modifier = segrellion_daenerys_only_thanked_supporter_opinion

			}

# =========================================================

segrellion_daenerys_settlement.0200 = {

	type = character_event

	title = segrellion_daenerys_settlement.0200.t

	desc = segrellion_daenerys_settlement.0200.desc

	theme = crown


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	immediate = {

		var:segrellion_daenerys_current_settlement_holder ?= {

			save_scope_as = segrellion_settlement_holder
		}


		if = {

			limit = {

				scope:segrellion_settlement_holder = {

					is_imprisoned = no
				}
			}


			imprison = {

				target = scope:segrellion_settlement_holder

				type = house_arrest
			}
		}
	}


	left_portrait = {

		character = root

		animation = personality_rational
	}


	right_portrait = {

		character = scope:segrellion_settlement_holder

		animation = worry
	}


	# =====================================================
	# FULL PARDON
	# =====================================================

	option = {

		name = segrellion_daenerys_settlement.0200.pardon


		segrellion_daenerys_pardon_enemy_house_effect = yes

		segrellion_daenerys_finish_current_settlement_effect = yes


		ai_chance = {

			base = 55
		}
	}


	# =====================================================
	# THE WALL
	# =====================================================

	option = {

		name = segrellion_daenerys_settlement.0200.wall


		trigger = {

			scope:segrellion_settlement_holder = {

				is_male = yes
			}
		}


		segrellion_daenerys_send_enemy_holder_to_wall_effect = yes

		segrellion_daenerys_finish_current_settlement_effect = yes


		ai_chance = {

			base = 20
		}
	}


	# =====================================================
	# EXECUTE ONLY THE LORD
	# =====================================================

	option = {

		name = segrellion_daenerys_settlement.0200.execute


		segrellion_daenerys_execute_enemy_holder_effect = yes

		segrellion_daenerys_finish_current_settlement_effect = yes


		ai_chance = {

			base = 20
		}
	}


	# =====================================================
	# DESTROY THE HOUSE
	# =====================================================

	option = {

		name = segrellion_daenerys_settlement.0200.destroy_house


		segrellion_daenerys_purge_enemy_house_effect = yes


		if = {

			limit = {
				is_ai = yes
			}


			segrellion_daenerys_ai_appoint_current_paramount_effect = yes

			segrellion_daenerys_finish_current_settlement_effect = yes
		}


		else = {

			trigger_event = {

				id = segrellion_daenerys_settlement.0300
			}
		}


		ai_chance = {

			base = 5


			modifier = {

				factor = 4

				has_trait = sadistic
			}


			modifier = {

				factor = 2

				has_trait = callous
			}
		}
	}
}
