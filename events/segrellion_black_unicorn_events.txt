# =========================================================
# 0030 - FOLLOWING THE BLACK UNICORN
# =========================================================

segrellion_black_unicorn.0030 = {

	type = character_event

	title = segrellion_black_unicorn.0030.t
	desc = segrellion_black_unicorn.0030.desc

	theme = travel


	override_background = {
		reference = segrellion_the_beacon
	}


	left_portrait = {

		character = root
		animation = personality_rational
	}


	immediate = {


		var:segrellion_black_unicorn_investigator ?= {

			save_scope_as = black_unicorn_investigator
		}
	}


	# =====================================================
	# GO PERSONALLY
	# =====================================================

	option = {

		name = segrellion_black_unicorn.0030.a

		add_internal_flag = special

		custom_tooltip = segrellion_black_unicorn.0030.a.tt


		hidden_effect = {


			add_character_flag = segrellion_black_unicorn_expedition_committed

			add_character_flag = segrellion_black_unicorn_daemon_personal_route


			location = {

				save_scope_as = black_unicorn_origin
			}


			if = {

				limit = {
					exists = scope:black_unicorn_origin
				}


				set_variable = {

					name = segrellion_black_unicorn_origin

					value = scope:black_unicorn_origin
				}
			}


			start_travel_plan = {

				destination = title:b_bitter_dryn.title_province

				return_trip = no

				on_arrival_event = segrellion_black_unicorn.0040

				on_arrival_destinations = first

				on_travel_planner_cancel_event = segrellion_black_unicorn.0090
			}
		}


		ai_chance = {
			base = 100
		}
	}


	# =====================================================
	# NOT YET
	#
	# Refund the decision cost.
	# =====================================================

	option = {

		name = segrellion_black_unicorn.0030.c

		custom_tooltip = segrellion_black_unicorn.0030.c.tt


		add_gold = 150


		ai_chance = {
			base = 0
		}
	}
}

# =========================================================
# 0042 - RETURN FROM BITTER DRYNN
# =========================================================

segrellion_black_unicorn.0042 = {

	type = character_event
	hidden = yes


	immediate = {


		var:segrellion_black_unicorn_origin ?= {

			save_scope_as = black_unicorn_origin
		}


		set_location = title:b_bitter_dryn.title_province


		if = {

			limit = {
				exists = scope:black_unicorn_origin
			}


			start_travel_plan = {

				destination = scope:black_unicorn_origin

				return_trip = no

				on_arrival_event = segrellion_black_unicorn.0043

				on_arrival_destinations = first

				on_travel_planner_cancel_event = segrellion_black_unicorn.0090
			}
		}


		else = {

			return_home = yes
		}
	}
}

# =========================================================
# 0045 - BITTER DRYNN -> COLD PEAKS
#
# A completely NEW travel plan.
# =========================================================

segrellion_black_unicorn.0045 = {

	type = character_event
	hidden = yes


	trigger = {

		this = character:Segrellion_60

		has_character_flag = segrellion_black_unicorn_mountain_route

		NOT = {
			exists = current_travel_plan
		}
	}


	immediate = {


		# Critical for landless characters:
		# begin the next journey FROM Bitter Drynn,
		# not from Daemon's camp/home position.

		set_location = title:b_bitter_dryn.title_province


		start_travel_plan = {

			destination = title:b_cold_peaks.title_province

			return_trip = no

			on_arrival_event = segrellion_black_unicorn.0050

			on_arrival_destinations = first

			on_travel_planner_cancel_event = segrellion_black_unicorn.0090
		}
	}
}

# 0080 - DUEL RESULT

# STORY CURRENTLY STOPS HERE.

segrellion_black_unicorn.0080 = {

	type = character_event
	window = duel_event

	title = segrellion_black_unicorn.0080.t

	desc = {

		first_valid = {

			triggered_desc = {

				trigger = {
					scope:actor = scope:sc_victor
				}

				desc = segrellion_black_unicorn.0080.desc_daemon_wins
			}

			desc = segrellion_black_unicorn.0080.desc_woman_wins
		}
	}

	theme = war


	left_portrait = {

		character = scope:actor
		animation = personality_bold
	}


	right_portrait = {

		character = scope:recipient
		animation = personality_bold
	}


	immediate = {


		scope:actor ?= {

			segrellion_black_unicorn_restore_duel_injuries_effect = yes
		}


		scope:recipient ?= {

			segrellion_black_unicorn_restore_duel_injuries_effect = yes
		}


		if = {

			limit = {
				scope:actor = scope:sc_victor
			}


			scope:actor = {

				add_prestige = 250

				add_character_flag = segrellion_black_unicorn_daemon_won_duel
			}


			scope:recipient = {

				add_prestige = 75

				add_character_flag = segrellion_black_unicorn_woman_lost_duel
			}
		}


		else = {


			scope:recipient = {

				add_prestige = 250

				add_character_flag = segrellion_black_unicorn_woman_won_duel
			}


			scope:actor = {

				add_prestige = 75

				add_character_flag = segrellion_black_unicorn_daemon_lost_duel
			}
		}


		character:Segrellion_60 ?= {

			add_character_flag = segrellion_black_unicorn_duel_resolved
		}
	}


	option = {

		name = segrellion_black_unicorn.0080.a

		trigger = {
			scope:actor = scope:sc_victor
		}
	}


	option = {

		name = segrellion_black_unicorn.0080.b

		trigger = {
			scope:recipient = scope:sc_victor
		}
	}
}
