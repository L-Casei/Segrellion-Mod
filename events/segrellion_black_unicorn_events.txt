# =========================================================
# 0000 - RED UNICORN CAMP LOCK
#
# She-Unicorn must remain hidden at the Cold Peaks until
# the storyline explicitly releases her.
# =========================================================

segrellion_black_unicorn.0000 = {

	type = character_event

	hidden = yes


	trigger = {

		this = character:Black_Unicorn_1

		is_alive = yes

		has_character_flag = segrellion_black_unicorn_lock_camp_skagos
	}


	immediate = {


		current_travel_plan ?= {

			abort_travel_plan = yes
		}


		domicile ?= {

			if = {

				limit = {

					domicile_location != title:b_cold_peaks.title_province
				}


				move_domicile = title:b_cold_peaks.title_province
			}
		}


		if = {

			limit = {

				location != title:b_cold_peaks.title_province
			}


			set_location = title:b_cold_peaks.title_province
		}


		trigger_event = {

			id = segrellion_black_unicorn.0000

			days = 7
		}
	}
}

# =========================================================
# 0310 - SHIPMASTER RESULT
#
# 5% chance to return.
# =========================================================

segrellion_black_unicorn.0310 = {

	type = character_event
	hidden = yes


	immediate = {

		add_character_flag = segrellion_black_unicorn_shipmaster_resolved


		var:segrellion_black_unicorn_shipmaster ?= {

			save_scope_as = black_unicorn_resolving_scout
		}


		if = {

			limit = {

				exists = scope:black_unicorn_resolving_scout

				scope:black_unicorn_resolving_scout = {
					is_alive = yes
				}
			}


			random_list = {

				# =========================================
				# SURVIVES
				# =========================================

				5 = {

					scope:black_unicorn_resolving_scout = {

						add_trait = maimed
						add_trait = one_eyed

						add_character_flag = segrellion_black_unicorn_scout_survivor
						add_character_flag = segrellion_black_unicorn_scout_horrified

						set_location = root.location
					}


					add_courtier = scope:black_unicorn_resolving_scout


					if = {

						limit = {

							NOT = {
								has_character_flag = segrellion_black_unicorn_survivor_pending
							}

							NOT = {
								has_character_flag = segrellion_black_unicorn_letter_received
							}
						}


						set_variable = {

							name = segrellion_black_unicorn_returning_knight

							value = scope:black_unicorn_resolving_scout
						}


						add_character_flag = segrellion_black_unicorn_survivor_pending


						trigger_event = {

							id = segrellion_black_unicorn.0320

							days = 1
						}
					}
				}


				# =========================================
				# NEVER RETURNS
				# =========================================

				95 = {

					scope:black_unicorn_resolving_scout = {

						add_character_flag = segrellion_black_unicorn_lost_on_skagos

						death = {
							death_reason = death_vanished
						}
					}
				}
			}
		}


		segrellion_black_unicorn_check_three_scouts_effect = yes
	}
}

segrellion_black_unicorn.0080 = {

	type = character_event

	window = duel_event

	title = segrellion_black_unicorn.0080.t


	desc = {

		first_valid = {


			triggered_desc = {

				trigger = {

					root = scope:sc_victor
				}

				desc = segrellion_black_unicorn.0080.desc_daemon_wins
			}


			desc = segrellion_black_unicorn.0080.desc_woman_wins
		}
	}


	theme = war


	left_portrait = {

		character = root

		animation = personality_bold
	}


	right_portrait = {

		character = character:Black_Unicorn_1

		animation = personality_bold
	}


	immediate = {


		character:Black_Unicorn_1 = {

			save_scope_as = black_unicorn_woman
		}


		# =================================================
		# DEFENSIVE CLEANUP
		#
		# This duel must never leave permanent wounds.
		# FATALITY = practice already prevents them, but this
		# also cleans old-test residue.
		# =================================================

		character:Black_Unicorn_1 = {

			remove_trait = wounded_1
			remove_trait = wounded_2
			remove_trait = wounded_3
		}


		# =================================================
		# REMOVE DUEL GRUDGE
		# =================================================

		if = {

			limit = {

				has_relation_grudge = character:Black_Unicorn_1
			}

			remove_relation_grudge = character:Black_Unicorn_1
		}


		character:Black_Unicorn_1 = {

			if = {

				limit = {

					has_relation_grudge = root
				}

				remove_relation_grudge = root
			}
		}


		# =================================================
		# CLEAN OLD RESULT FLAGS
		# =================================================

		remove_character_flag = segrellion_black_unicorn_daemon_won_duel
		remove_character_flag = segrellion_black_unicorn_daemon_lost_duel


		character:Black_Unicorn_1 = {

			remove_character_flag = segrellion_black_unicorn_woman_won_duel
			remove_character_flag = segrellion_black_unicorn_woman_lost_duel
		}


		# =================================================
		# DAEMON WON
		# =================================================

		if = {

			limit = {

				root = scope:sc_victor
			}


			add_prestige = 250

			add_character_flag = segrellion_black_unicorn_daemon_won_duel


			character:Black_Unicorn_1 = {

				add_prestige = 75

				add_character_flag = segrellion_black_unicorn_woman_lost_duel
			}
		}


		# =================================================
		# VISENYA WON
		# =================================================

		else = {


			add_prestige = 75

			add_character_flag = segrellion_black_unicorn_daemon_lost_duel


			character:Black_Unicorn_1 = {

				add_prestige = 250

				add_character_flag = segrellion_black_unicorn_woman_won_duel
			}
		}


		add_character_flag = segrellion_black_unicorn_duel_resolved
	}


	option = {

		name = segrellion_black_unicorn.0080.a

		trigger = {

			has_character_flag = segrellion_black_unicorn_daemon_won_duel
		}
	}


	option = {

		name = segrellion_black_unicorn.0080.b

		trigger = {

			has_character_flag = segrellion_black_unicorn_daemon_lost_duel
		}
	}


	# =====================================================
	# CONTINUE AFTER CLOSING DUEL WINDOW
	#
	# Much more reliable than firing from the duel option.
	# =====================================================

	after = {


		if = {

			limit = {

				has_character_flag = segrellion_black_unicorn_daemon_won_duel
			}


			trigger_event = {

				id = segrellion_black_unicorn.0100

				days = 1
			}
		}


		else_if = {

			limit = {

				has_character_flag = segrellion_black_unicorn_daemon_lost_duel
			}


			trigger_event = {

				id = segrellion_black_unicorn.0200

				days = 1
			}
		}
	}
}

# 0160 - SHE CAME
# =========================================================

segrellion_black_unicorn.0160 = {

	type = character_event

	title = segrellion_black_unicorn.0160.t
	desc = segrellion_black_unicorn.0160.desc

	theme = dynasty


	override_background = {
		reference = segrellion_westerosi_northern_tavern
	}


	left_portrait = {

		character = root
		animation = admiration
	}


	right_portrait = {

		character = character:Black_Unicorn_1
		animation = personality_bold
	}


	immediate = {

		character:Black_Unicorn_1 = {

			save_scope_as = black_unicorn_woman
		}


		remove_character_flag = segrellion_black_unicorn_waiting_for_visenya

		add_character_flag = segrellion_black_unicorn_visenya_accepted


		character:Black_Unicorn_1 = {

			add_character_flag = segrellion_black_unicorn_visenya_chose_daemon

			remove_character_flag = segrellion_black_unicorn_lock_camp_skagos


			primary_title ?= {

				save_scope_as = red_unicorn_camp_title
			}
		}


		# She abandons the Red Unicorn camp.

		if = {

			limit = {
				exists = scope:red_unicorn_camp_title
			}

			destroy_title = scope:red_unicorn_camp_title
		}


		character:Black_Unicorn_1 = {

			set_location = root.location

			set_employer = root
		}


		# =================================================
		# LOVERS
		#
		# Adult-only.
		# =================================================

		if = {

			limit = {

				character:Black_Unicorn_1 = {
					age >= 18
				}

				NOT = {
					has_relation_lover = character:Black_Unicorn_1
				}
			}


			set_relation_lover = {
				target = character:Black_Unicorn_1
			}


			add_character_flag = segrellion_daemon_visenya_lovers


			character:Black_Unicorn_1 = {
				add_character_flag = segrellion_visenya_daemon_lovers
			}
		}


		else_if = {

			limit = {

				character:Black_Unicorn_1 = {
					age < 18
				}
			}

			add_character_flag = segrellion_black_unicorn_romance_waiting_for_adulthood
		}
	}


	option = {

		name = segrellion_black_unicorn.0160.a

		custom_tooltip = segrellion_black_unicorn.0160.a.tt


		add_character_flag = segrellion_black_unicorn_return_with_visenya


		set_location = title:b_bitter_dryn.title_province


		segrellion_black_unicorn_return_to_daemon_camp_effect = yes
	}
}

# =========================================================
# 0180 - TWO HALVES OF THE UNICORN
#
# WEDDING
# =========================================================

segrellion_black_unicorn.0180 = {

	type = character_event

	title = segrellion_black_unicorn.0180.t
	desc = segrellion_black_unicorn.0180.desc

	theme = marriage


	trigger = {

		is_alive = yes

		character:Black_Unicorn_1 = {

			is_alive = yes
			age >= 18
		}
	}


	# No manual background override.
	# The marriage theme supplies the wedding presentation.


	left_portrait = {

		character = root

		animation = admiration

		outfit_tags = {
			western_high_nobility
		}
	}


	right_portrait = {

		character = character:Black_Unicorn_1

		animation = wedding_happy_cry

		outfit_tags = {
			western_high_nobility
		}
	}


	immediate = {

		character:Black_Unicorn_1 = {

			save_scope_as = black_unicorn_woman
		}


		# =================================================
		# CLEAR PREVIOUS UNIONS
		# =================================================

		if = {

			limit = {

				OR = {

					is_married = yes

					exists = betrothed
				}
			}


			add_character_flag = segrellion_daemon_broke_previous_union_for_visenya
		}


		if = {

			limit = {
				exists = betrothed
			}

			break_betrothal = betrothed
		}


		every_spouse = {
			divorce = root
		}


		character:Black_Unicorn_1 = {

			if = {

				limit = {
					exists = betrothed
				}

				break_betrothal = betrothed
			}


			every_spouse = {
				divorce = character:Black_Unicorn_1
			}
		}


		# =================================================
		# ENSURE LOVERS
		# =================================================

		if = {

			limit = {

				NOT = {
					has_relation_lover = character:Black_Unicorn_1
				}
			}

			set_relation_lover = {
				target = character:Black_Unicorn_1
			}
		}


		# =================================================
		# MARRIAGE
		# =================================================

		marry = character:Black_Unicorn_1


		add_character_flag = segrellion_black_unicorn_daemon_married_visenya


		character:Black_Unicorn_1 = {

			add_character_flag = segrellion_black_unicorn_visenya_married_daemon

			add_character_flag = segrellion_visenya_canon_aegon_pregnancy_pending
		}


		remove_character_flag = segrellion_black_unicorn_wedding_planned
	}


	option = {

		name = segrellion_black_unicorn.0180.a

		custom_tooltip = segrellion_black_unicorn.0180.a.tt


		trigger_event = {

			id = segrellion_black_unicorn.0181

			days = 1
		}
	}
}

segrellion_black_unicorn.0181 = {

	type = character_event

	title = segrellion_black_unicorn.0181.t
	desc = segrellion_black_unicorn.0181.desc

	theme = romance


	trigger = {

		age >= 18

		is_married = yes

		character:Black_Unicorn_1 = {

			is_alive = yes
			age >= 18
		}
	}


	override_background = {
		reference = army_camp
	}


	left_portrait = {

		character = root
		animation = flirtation
	}


	right_portrait = {

		character = character:Black_Unicorn_1
		animation = flirtation_left
	}


	immediate = {

		character:Black_Unicorn_1 = {

			save_scope_as = black_unicorn_woman

			add_character_flag = is_naked
		}


		add_character_flag = is_naked
	}


	option = {

		name = segrellion_black_unicorn.0181.a

		custom_tooltip = segrellion_black_unicorn.0181.a.tt


		# Records the intimate encounter without interfering
		# with the future scripted canonical pregnancy.

		had_sex_with_effect = {

			CHARACTER = character:Black_Unicorn_1

			PREGNANCY_CHANCE = 0
		}


		if = {

			limit = {

				NOT = {
					has_relation_soulmate = character:Black_Unicorn_1
				}
			}


			set_relation_soulmate = {

				target = character:Black_Unicorn_1

				reason = soulmate_laamp_adventure
			}
		}


		add_character_flag = segrellion_daemon_visenya_soulmates


		character:Black_Unicorn_1 = {

			add_character_flag = segrellion_visenya_daemon_soulmates
		}
	}


	after = {

		remove_character_flag = is_naked


		character:Black_Unicorn_1 = {

			remove_character_flag = is_naked
		}
	}
}

# 0190 - RETURN TO DAEMON'S CAMP COMPLETE
# =========================================================

segrellion_black_unicorn.0190 = {

	type = character_event

	hidden = yes


	immediate = {


		if = {

			limit = {

				exists = current_travel_plan
			}


			current_travel_plan = {

				complete_travel_plan = yes
			}
		}


		var:segrellion_black_unicorn_origin ?= {

			save_scope_as = black_unicorn_return_destination
		}


		if = {

			limit = {

				exists = scope:black_unicorn_return_destination
			}


			set_location = scope:black_unicorn_return_destination
		}


		add_character_flag = segrellion_black_unicorn_skagos_visit_complete


		# =================================================
		# VISENYA CAME WITH DAEMON
		# =================================================

		if = {

			limit = {

				has_character_flag = segrellion_black_unicorn_return_with_visenya
			}


			remove_character_flag = segrellion_black_unicorn_return_with_visenya


			character:Black_Unicorn_1 = {

				set_location = root.location
			}


			trigger_event = {

				id = segrellion_black_unicorn.0170

				days = 1
			}
		}


		# =================================================
		# VISENYA REMAINED ON SKAGOS
		# =================================================

		else_if = {

			limit = {

				has_character_flag = segrellion_black_unicorn_return_alone
			}


			remove_character_flag = segrellion_black_unicorn_return_alone


			trigger_event = {

				id = segrellion_black_unicorn.0171

				days = 1
			}
		}


		remove_variable = segrellion_black_unicorn_return_start
	}
}
