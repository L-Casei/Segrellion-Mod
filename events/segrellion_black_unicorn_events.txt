# =========================================================
# 0000 - RED UNICORN CAMP LOCK
#
# She-Unicorn must remain hidden at the Cold Peaks until
# the storyline explicitly releases her.
# =========================================================

segrellion_black_unicorn.0000 = {

	type = character_event

	hidden = yes


	trigger = {

		this = character:Black_Unicorn_1

		is_alive = yes

		has_character_flag = segrellion_black_unicorn_lock_camp_skagos
	}


	immediate = {


		current_travel_plan ?= {

			abort_travel_plan = yes
		}


		domicile ?= {

			if = {

				limit = {

					domicile_location != title:b_cold_peaks.title_province
				}


				move_domicile = title:b_cold_peaks.title_province
			}
		}


		if = {

			limit = {

				location != title:b_cold_peaks.title_province
			}


			set_location = title:b_cold_peaks.title_province
		}


		trigger_event = {

			id = segrellion_black_unicorn.0000

			days = 7
		}
	}
}

segrellion_black_unicorn.0080 = {

	type = character_event

	window = duel_event

	title = segrellion_black_unicorn.0080.t


	desc = {

		first_valid = {


			triggered_desc = {

				trigger = {

					root = scope:sc_victor
				}

				desc = segrellion_black_unicorn.0080.desc_daemon_wins
			}


			desc = segrellion_black_unicorn.0080.desc_woman_wins
		}
	}


	theme = war


	left_portrait = {

		character = root

		animation = personality_bold
	}


	right_portrait = {

		character = character:Black_Unicorn_1

		animation = personality_bold
	}


	immediate = {


		character:Black_Unicorn_1 = {

			save_scope_as = black_unicorn_woman
		}


		# =================================================
		# DEFENSIVE CLEANUP
		#
		# This duel must never leave permanent wounds.
		# FATALITY = practice already prevents them, but this
		# also cleans old-test residue.
		# =================================================

		character:Black_Unicorn_1 = {

			remove_trait = wounded_1
			remove_trait = wounded_2
			remove_trait = wounded_3
		}


		# =================================================
		# REMOVE DUEL GRUDGE
		# =================================================

		if = {

			limit = {

				has_relation_grudge = character:Black_Unicorn_1
			}

			remove_relation_grudge = character:Black_Unicorn_1
		}


		character:Black_Unicorn_1 = {

			if = {

				limit = {

					has_relation_grudge = root
				}

				remove_relation_grudge = root
			}
		}


		# =================================================
		# CLEAN OLD RESULT FLAGS
		# =================================================

		remove_character_flag = segrellion_black_unicorn_daemon_won_duel
		remove_character_flag = segrellion_black_unicorn_daemon_lost_duel


		character:Black_Unicorn_1 = {

			remove_character_flag = segrellion_black_unicorn_woman_won_duel
			remove_character_flag = segrellion_black_unicorn_woman_lost_duel
		}


		# =================================================
		# DAEMON WON
		# =================================================

		if = {

			limit = {

				root = scope:sc_victor
			}


			add_prestige = 250

			add_character_flag = segrellion_black_unicorn_daemon_won_duel


			character:Black_Unicorn_1 = {

				add_prestige = 75

				add_character_flag = segrellion_black_unicorn_woman_lost_duel
			}
		}


		# =================================================
		# VISENYA WON
		# =================================================

		else = {


			add_prestige = 75

			add_character_flag = segrellion_black_unicorn_daemon_lost_duel


			character:Black_Unicorn_1 = {

				add_prestige = 250

				add_character_flag = segrellion_black_unicorn_woman_won_duel
			}
		}


		add_character_flag = segrellion_black_unicorn_duel_resolved
	}


	option = {

		name = segrellion_black_unicorn.0080.a
		custom_tooltip = segrellion_black_unicorn.ui.0080.a.tt

		trigger = {

			has_character_flag = segrellion_black_unicorn_daemon_won_duel
		}
	}


	option = {

		name = segrellion_black_unicorn.0080.b
		custom_tooltip = segrellion_black_unicorn.ui.0080.b.tt

		trigger = {

			has_character_flag = segrellion_black_unicorn_daemon_lost_duel
		}
	}


	# =====================================================
	# CONTINUE AFTER CLOSING DUEL WINDOW
	#
	# Much more reliable than firing from the duel option.
	# =====================================================

	after = {


		if = {

			limit = {

				has_character_flag = segrellion_black_unicorn_daemon_won_duel
			}


			trigger_event = {

				id = segrellion_black_unicorn.0100

				days = 1
			}
		}


		else_if = {

			limit = {

				has_character_flag = segrellion_black_unicorn_daemon_lost_duel
			}


			trigger_event = {

				id = segrellion_black_unicorn.0200

				days = 1
			}
		}
	}
}
