namespace = segrellion_jon

# =========================================================
# 1299 - START CHAIN
# =========================================================

segrellion_jon.1299 = {
	type = character_event
	hidden = yes

	immediate = {

		if = {
			limit = {
				NOT = { has_character_flag = segrellion_jon_post_resurrection_chain_started }
			}

			add_character_flag = segrellion_jon_post_resurrection_chain_started
			add_character_flag = segrellion_jon_post_resurrection_chain_running

			trigger_event = {
				id = segrellion_jon.1300
				days = 10
			}
		}
	}
}

# =========================================================
# 1300 - FOR THE WATCH
# =========================================================

segrellion_jon.1300 = {
	type = character_event

	title = segrellion_jon.1300.t
	desc = segrellion_jon.1300.desc

	theme = death

	trigger = {
		is_alive = yes
		has_character_flag = segrellion_jon_post_resurrection_chain_started
		NOT = { has_character_flag = segrellion_jon_mutiny_closed }
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	option = {
		name = segrellion_jon.1300.a
		custom_tooltip = segrellion_jon.1300.a.tt

		segrellion_jon_execute_surviving_mutineers_effect = yes

		add_prestige = 250
		add_dread = 20

		segrellion_jon_close_mutiny_storyline_effect = yes

		trigger_event = {
			id = segrellion_jon.1310
			days = { 10 14 }
		}
	}
}

# =========================================================
# 1350 - THE DRAGON IN THE WOLF
# =========================================================

segrellion_jon.1350 = {
	type = character_event

	title = segrellion_jon.1350.t
	desc = segrellion_jon.1350.desc

	theme = dynasty

	trigger = {
		is_alive = yes
		has_character_flag = segrellion_jon_targaryen_revealed
		NOT = { has_character_flag = segrellion_jon_dragon_in_the_wolf_done }
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	option = {
		name = segrellion_jon.1350.a
		custom_tooltip = segrellion_jon.1350.a.tt

		add_character_flag = segrellion_jon_dragon_in_the_wolf_done
		add_character_flag = segrellion_jon_targaryen_claim_storyline_started

		if = {
			limit = {
				has_trait = just
			}
			remove_trait = just
		}

		if = {
			limit = {
				NOT = { has_trait = ambitious }
			}
			add_trait = ambitious
		}

		add_prestige = 500
		add_legitimacy = 0.15
	}
}
