# =========================================================
# 0090 - STARTUP ROUTER
#
# One day after Roose's murder. This is the stable startup
# anchor: it prepares all scopes and then launches 0091.
# =========================================================

segrellion_stannis.0090 = {

	hidden = yes


	trigger = {

		this = character:Baratheon_3

		is_alive = yes

		has_character_flag = segrellion_stannis_north_story_initialized

		NOT = {
			has_character_flag = segrellion_stannis_opening_seen
		}

		NOT = {
			has_character_flag = segrellion_stannis_opening_scheduled
		}

		NOT = {
			has_character_flag = segrellion_stannis_northern_campaign_started
		}
	}


	immediate = {

		add_character_flag = {
			flag = segrellion_stannis_opening_scheduled
			days = 45
		}

		set_location = title:c_castle_black.title_province

		# Apply Stannis' 300 AC narrative reputation package.
		# The visible modifier contains only flavor; selective
		# opinion corrections are handled inside the effect.
		segrellion_apply_one_true_king_reputation_effect = yes


		title:e_the_north.holder ?= {

			add_character_flag = segrellion_stannis_northern_enemy

			save_scope_as = stannis_bolton_lord
		}


		trigger_event = {

			id = segrellion_stannis.0091

			days = { 5 10 }
		}
	}
}

# =========================================================
# 0091 - FURY BURNS
#
# Root = Stannis Baratheon
#
# Opening event for Stannis in the 300 AC bookmark.
# A retrospective of the Blackwater, Dragonstone,
# the Wall and the war against the wildlings.
#
# This is the first visible event of his storyline.
# =========================================================

segrellion_stannis.0091 = {

	type = character_event

	title = segrellion_stannis.0091.t
	desc = segrellion_stannis.0091.desc

	theme = crown


	override_background = {
		reference = segrellion_castle_black_great_hall
	}


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	trigger = {
		this = character:Baratheon_3
		is_alive = yes
		has_character_flag = segrellion_stannis_north_story_initialized
		NOT = { has_character_flag = segrellion_stannis_opening_seen }
		NOT = { has_character_flag = segrellion_stannis_northern_campaign_started }
	}


	immediate = {
		add_character_flag = segrellion_stannis_opening_seen
		remove_character_flag = segrellion_stannis_opening_scheduled
	}


	left_portrait = {

		character = root

		animation = personality_rational
	}


	option = {

		name = segrellion_stannis.0091.a

		custom_tooltip = segrellion_stannis.0091.a.tt

		if = {

			limit = {
				NOT = {
					has_character_modifier = segrellion_stannis_fury_burns_modifier
				}
			}

			add_character_modifier = {
				modifier = segrellion_stannis_fury_burns_modifier
			}
		}


		# Continue with the original first campaign event after a
		# deliberate narrative pause.
		trigger_event = {

			id = segrellion_stannis.0001

			days = { 5 10 }
		}

		ai_chance = {
			base = 100
		}
	}
}

	desc = segrellion_stannis.0015.desc

		reference = army_camp
	}

			is_alive = yes
		}
	}


	immediate = {

		title:d_the_drench.holder ?= {
			save_scope_as = stannis_davos
		}
	}


	left_portrait = {

		character = root

		animation = personality_rational
	}


	right_portrait = {

		character = scope:stannis_davos

		animation = thinking
	}


	option = {

		name = segrellion_stannis.0015.a

		custom_tooltip = segrellion_stannis.0015.a.tt

		add_character_flag = segrellion_stannis_only_forward

		if = {

			limit = {
				NOT = {
					has_character_modifier = segrellion_stannis_only_forward_modifier
				}
			}

			add_character_modifier = {
				modifier = segrellion_stannis_only_forward_modifier
			}
		}

		# The second interlude happens while the host is already
		# struggling south through the snow.
		trigger_event = {

			id = segrellion_stannis.0017

			days = { 5 10 }
		}


		# Late campaign-start failsafe. Normally 0017 schedules 0018;
		# this copy only matters if Davos dies before the second
		# interlude can fire. Never skip 0018 because it establishes
		# the Deepwood state required by 0020.
		trigger_event = {

			id = segrellion_stannis.0018

			days = { 15 20 }
		}


		ai_chance = {
			base = 100
		}
	}
}

# =========================================================
# 1012 - JON NORTH TRANSFER
#
# Root = Jon
# Hidden mechanical transfer.
# =========================================================

segrellion_stannis.1012 = {

	hidden = yes


	trigger = {

		segrellion_is_current_jon_trigger = yes

		has_character_flag = segrellion_jon_accepted_stannis_north_offer

		NOT = {
			has_character_flag = segrellion_jon_north_handover_completed
		}


		character:Baratheon_3 ?= {
			is_alive = yes
		}
	}


	immediate = {

		segrellion_jon_accepts_north_effect = yes


		if = {

			limit = {
				has_character_flag = segrellion_jon_north_transfer_failed
			}


			trigger_event = {
				id = segrellion_stannis.1012
				days = 10
			}
		}


		else = {

			remove_character_flag = segrellion_jon_accepted_stannis_north_offer

			add_character_flag = segrellion_jon_north_handover_completed


			# Northern lords approve of Stannis restoring a Stark.
			every_vassal = {
