# =========================================================
# 0090 - STARTUP ROUTER
#
# One day after Roose's murder. This is the stable startup
# anchor: it prepares all scopes and then launches 0091.
# =========================================================

segrellion_stannis.0090 = {

	hidden = yes


	trigger = {

		this = character:Baratheon_3

		is_alive = yes

		has_character_flag = segrellion_stannis_north_story_initialized

		NOT = {
			has_character_flag = segrellion_stannis_opening_seen
		}

		NOT = {
			has_character_flag = segrellion_stannis_opening_scheduled
		}

		NOT = {
			has_character_flag = segrellion_stannis_northern_campaign_started
		}
	}


	immediate = {

		immediate = {

		add_character_flag = {
			flag = segrellion_stannis_opening_scheduled
			days = 45
		}

		set_location = title:c_castle_black.title_province

		# Apply Stannis' 300 AC narrative reputation package.
		# The visible modifier contains only flavor; selective
		# opinion corrections are handled inside the effect.
		segrellion_apply_one_true_king_reputation_effect = yes


		title:e_the_north.holder ?= {

			add_character_flag = segrellion_stannis_northern_enemy

			save_scope_as = stannis_bolton_lord
		}


		trigger_event = {

			id = segrellion_stannis.0091

			days = { 5 10 }
		}
	}
}

# =========================================================
# 0091 - FURY BURNS
#
# Root = Stannis Baratheon
#
# Opening event for Stannis in the 300 AC bookmark.
# A retrospective of the Blackwater, Dragonstone,
# the Wall and the war against the wildlings.
#
# This is the first visible event of his storyline.
# =========================================================

segrellion_stannis.0091 = {

	type = character_event

	title = segrellion_stannis.0091.t
	desc = segrellion_stannis.0091.desc

	theme = crown


	override_background = {
		reference = segrellion_castle_black_great_hall
	}


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	immediate = {
		add_character_flag = segrellion_stannis_opening_seen
		remove_character_flag = segrellion_stannis_opening_scheduled
	}


	left_portrait = {

		character = root

		animation = personality_rational
	}


	option = {

		name = segrellion_stannis.0091.a

		custom_tooltip = segrellion_stannis.0091.a.tt

		if = {

			limit = {
				NOT = {
					has_character_modifier = segrellion_stannis_fury_burns_modifier
				}
			}

			add_character_modifier = {
				modifier = segrellion_stannis_fury_burns_modifier
			}
		}


		# Continue with the original first campaign event after a
		# deliberate narrative pause.
		trigger_event = {

			id = segrellion_stannis.0001

			days = { 5 10 }
		}

		ai_chance = {
			base = 100
		}
	}
}


# =========================================================
# 0001 - THE WAR AFTER THE WAR
#
# Root = Stannis
#
# The wildling threat has been checked.
# Now Stannis decides what the North requires.
# =========================================================

segrellion_stannis.0001 = {

	type = character_event

	title = segrellion_stannis.0001.t
	desc = {

		first_valid = {

			triggered_desc = {

				trigger = {

					exists = scope:stannis_jon
				}

				desc = segrellion_stannis.0001.desc
			}


			desc = segrellion_stannis.0001.desc_jon_dead
		}
	}

	theme = war


	override_background = {
		reference = segrellion_castle_black_great_hall
	}


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		this = character:Baratheon_3

		is_alive = yes

		has_character_flag = segrellion_stannis_preparing_northern_campaign

		title:e_the_north = {
			exists = holder
		}
	}


	immediate = {

		if = {

			limit = {
				has_global_variable = segrellion_resurrected_jon
			}

			global_var:segrellion_resurrected_jon ?= {

				if = {

					limit = {
						is_alive = yes
					}

					save_scope_as = stannis_jon
				}
			}
		}


		else_if = {

			limit = {
				character:Stark_7 ?= {
					is_alive = yes
				}
			}

			character:Stark_7 = {
				save_scope_as = stannis_jon
			}
		}

		title:k_the_wall.holder ?= {

			save_scope_as = stannis_wall_commander
		}


		title:e_the_north.holder = {

			save_scope_as = stannis_bolton_lord
		}
	}


	left_portrait = {

		character = root

		animation = personality_rational
	}


	right_portrait = {

		character = scope:stannis_wall_commander

		animation = personality_honorable
	}


	# =====================================================
	# MARCH SOUTH
	# =====================================================

	option = {

		name = segrellion_stannis.0001.a

		custom_tooltip = segrellion_stannis.0001.a.tt


		add_character_flag = segrellion_stannis_northern_campaign_committed


		trigger_event = {

			id = segrellion_stannis.0010

			days = { 5 10 }
		}


		ai_chance = {
			base = 100
		}
	}


	# =====================================================
	# PLAYER DELAY
	# =====================================================

	option = {

		name = segrellion_stannis.0001.b

		trigger = {
			is_ai = no
		}

		custom_tooltip = segrellion_stannis.0001.b.tt


		trigger_event = {

			id = segrellion_stannis.0001

			days = { 30 60 }
		}


		ai_chance = {
			base = 0
		}
	}
}

#
# Root = Jon
#
# Stannis offers Jon release from his vows,
# legitimacy, Winterfell and the North.
#
# AI:
# 90% - Accept
# 10% - Refuse
# =========================================================

segrellion_stannis.1010 = {

	type = character_event

	title = segrellion_stannis.1010.t
	desc = segrellion_stannis.1010.desc

	theme = crown


	override_background = {
		reference = segrellion_winterfell_great_hall
	}


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		OR = {

			this = character:Stark_7

			has_character_flag = segrellion_canonical_current_jon
		}

		is_alive = yes

		character:Baratheon_3 ?= {

			is_alive = yes

			has_character_flag = segrellion_stannis_northern_campaign_victory
		}
	}


	immediate = {

		character:Baratheon_3 = {
			save_scope_as = jon_stannis
		}

		title:e_the_north.holder ?= {
			save_scope_as = jon_bolton_lord
		}
	}


	left_portrait = {

		character = root

		animation = thinking
	}


	right_portrait = {

		character = character:Baratheon_3

		animation = personality_rational
	}


	# =====================================================
	# ACCEPT
	#
	# Jon leaves the Watch, is legitimized and receives
	# Winterfell and the North.
	#
	# AI chance: 90%
	# =====================================================

	option = {

		name = segrellion_stannis.1010.a

		custom_tooltip = segrellion_stannis.1010.a.tt


		segrellion_jon_accepts_north_effect = yes


		# Only continue if the Night's Watch transfer
		# succeeded and Jon genuinely received the North.

		if = {

			limit = {

				has_character_flag = segrellion_jon_restored_to_winterfell

				has_title = title:e_the_north

				has_title = title:c_winterfell
			}


			character:Baratheon_3 = {

				trigger_event = {

					id = segrellion_stannis.1020

					days = { 5 10 }
				}
			}
		}


		ai_chance = {

			base = 90
		}
	}

	# REFUSE
	#
	# Jon remains Lord Commander of the Night's Watch.
	# Stannis keeps temporary custody of the North.
	#
	# AI chance: 10%
	# =====================================================

		name = segrellion_stannis.1010.b

		custom_tooltip = segrellion_stannis.1010.b.tt


		add_character_flag = segrellion_jon_refused_north


		# Jon remains in the Watch.
		# Nothing is transferred and his vows remain intact.

		give_nickname = nick_segrellion_lord_snow


		character:Baratheon_3 = {

			add_character_flag = segrellion_stannis_jon_refused_north

			add_character_flag = segrellion_stannis_searching_northern_heir


			trigger_event = {

				id = segrellion_stannis.1021

				days = { 5 10 }
			}
		}


		ai_chance = {

			base = 10
		}
	}
}
