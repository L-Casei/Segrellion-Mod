# =========================================================
# 0090 - STARTUP ROUTER
#
# One day after Roose's murder.
# Gives CK3 enough time to resolve inheritance.
# =========================================================

segrellion_stannis.0090 = {

	hidden = yes


	trigger = {

		this = character:Baratheon_3

		is_alive = yes

		has_character_flag = segrellion_stannis_north_story_initialized

		NOT = {
			has_character_flag = segrellion_stannis_northern_campaign_started
		}
	}


	immediate = {

		set_location = title:c_castle_black.title_province


		title:e_the_north.holder ?= {

			add_character_flag = segrellion_stannis_northern_enemy

			save_scope_as = stannis_bolton_lord
		}


		trigger_event = {

			id = segrellion_stannis.0001

			days = { 2 4 }
		}
	}
}


# =========================================================
# 0001 - THE WAR AFTER THE WAR
#
# Root = Stannis
#
# The wildling threat has been checked.
# Now Stannis decides what the North requires.
# =========================================================

segrellion_stannis.0001 = {

	type = character_event

	title = segrellion_stannis.0001.t
	desc = segrellion_stannis.0001.desc

	theme = war


	override_background = {
		reference = castle
	}


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		this = character:Baratheon_3

		is_alive = yes

		has_character_flag = segrellion_stannis_preparing_northern_campaign

		character:Stark_7 ?= {
			is_alive = yes
		}

		title:e_the_north = {
			exists = holder
		}
	}


	immediate = {

		character:Stark_7 = {
			save_scope_as = stannis_jon
		}

		title:e_the_north.holder = {
			save_scope_as = stannis_bolton_lord
		}
	}


	left_portrait = {

		character = root

		animation = personality_rational
	}


	right_portrait = {

		character = character:Stark_7

		animation = personality_honorable
	}


	# =====================================================
	# MARCH SOUTH
	# =====================================================

	option = {

		name = segrellion_stannis.0001.a

		custom_tooltip = segrellion_stannis.0001.a.tt


		add_character_flag = segrellion_stannis_northern_campaign_committed


		trigger_event = {

			id = segrellion_stannis.0010

			days = { 2 5 }
		}


		ai_chance = {
			base = 100
		}
	}


	# =====================================================
	# PLAYER DELAY
	# =====================================================

	option = {

		name = segrellion_stannis.0001.b

		trigger = {
			is_ai = no
		}

		custom_tooltip = segrellion_stannis.0001.b.tt


		trigger_event = {

			id = segrellion_stannis.0001

			days = { 30 60 }
		}


		ai_chance = {
			base = 0
		}
	}
}

# =========================================================
# 1010 - THE NAME OF STARK
#
# Root = Jon
#
# Stannis offers Jon release from his vows,
# legitimacy, Winterfell and the North.
#
# AI:
# 90% - Accept
# 10% - Refuse
# =========================================================

segrellion_stannis.1010 = {

	type = character_event

	title = segrellion_stannis.1010.t
	desc = segrellion_stannis.1010.desc

	theme = crown


	override_background = {
		reference = segrellion_winterfell_great_hall
	}


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		this = character:Stark_7

		is_alive = yes

		character:Baratheon_3 ?= {

			is_alive = yes

			has_character_flag = segrellion_stannis_northern_campaign_victory
		}
	}


	immediate = {

		character:Baratheon_3 = {
			save_scope_as = jon_stannis
		}

		title:e_the_north.holder ?= {
			save_scope_as = jon_bolton_lord
		}
	}


	left_portrait = {

		character = root

		animation = thinking
	}


	right_portrait = {

		character = character:Baratheon_3

		animation = personality_rational
	}


	# =====================================================
	# ACCEPT
	#
	# Jon leaves the Watch, is legitimized and receives
	# Winterfell and the North.
	#
	# AI chance: 90%
	# =====================================================

	option = {

		name = segrellion_stannis.1010.a

		custom_tooltip = segrellion_stannis.1010.a.tt


		segrellion_jon_accepts_north_effect = yes


		# Only continue if the Night's Watch transfer
		# succeeded and Jon genuinely received the North.

		if = {

			limit = {

				has_character_flag = segrellion_jon_restored_to_winterfell

				has_title = title:e_the_north

				has_title = title:c_winterfell
			}


			character:Baratheon_3 = {

				trigger_event = {

					id = segrellion_stannis.1020

					days = { 2 4 }
				}
			}
		}


		ai_chance = {

			base = 90
		}
	}


	# =====================================================
	# REFUSE
	#
	# Jon remains Lord Commander of the Night's Watch.
	# Stannis keeps temporary custody of the North.
	#
	# AI chance: 10%
	# =====================================================

	option = {

		name = segrellion_stannis.1010.b

		custom_tooltip = segrellion_stannis.1010.b.tt


		add_character_flag = segrellion_jon_refused_north


		# Jon remains in the Watch.
		# Nothing is transferred and his vows remain intact.

		give_nickname = nick_segrellion_lord_snow


		character:Baratheon_3 = {

			add_character_flag = segrellion_stannis_jon_refused_north

			add_character_flag = segrellion_stannis_searching_northern_heir


			trigger_event = {

				id = segrellion_stannis.1021

				days = { 2 4 }
			}
		}


		ai_chance = {

			base = 10
		}
	}
}

# =========================================================
# 1030 - THE PINK LETTER'S PROMISE
#
# Root = Jon
# =========================================================

segrellion_stannis.1030 = {

	type = character_event

	title = segrellion_stannis.1030.t
	desc = segrellion_stannis.1030.desc

	theme = vengeance


	override_background = {
		reference = segrellion_winterfell_great_hall
	}


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		this = character:Stark_7

		is_alive = yes

		has_title = title:e_the_north


		any_prisoner = {

			is_alive = yes

			has_character_flag = segrellion_ramsay_northern_takeover
		}
	}


	immediate = {

		random_prisoner = {

			limit = {

				is_alive = yes

				has_character_flag = segrellion_ramsay_northern_takeover
			}


			save_scope_as = jon_ramsay
		}
	}


	left_portrait = {

		character = root

		animation = personality_rational
	}


	right_portrait = {

		character = scope:jon_ramsay

		animation = fear
	}


	# =====================================================
	# THE HEART
	# =====================================================

	option = {

		name = segrellion_stannis.1030.a

		custom_tooltip = segrellion_stannis.1030.a.tt


		scope:jon_ramsay = {

			death = {

				death_reason = death_execution

				killer = root
			}
		}


		if = {

			limit = {

				NOT = {
					has_trait = lunatic
				}
			}


			add_trait = lunatic_1
		}


		add_character_flag = segrellion_jon_ramsay_heart_execution
	}


	# =====================================================
	# HIS OWN HOUNDS
	# =====================================================

	option = {

		name = segrellion_stannis.1030.b

		custom_tooltip = segrellion_stannis.1030.b.tt


		scope:jon_ramsay = {

			death = {

				death_reason = death_execution

				killer = root
			}
		}


		add_character_flag = segrellion_jon_ramsay_hounds_execution
	}


	# =====================================================
	# KEEP HIM
	# =====================================================

	option = {

		name = segrellion_stannis.1030.c

		custom_tooltip = segrellion_stannis.1030.c.tt


		scope:jon_ramsay = {

			change_prison_type = dungeon
		}


		add_character_flag = segrellion_jon_kept_ramsay_alive
	}
}

# =========================================================
# 1100 - THE BLOOD OF WINTERFELL
#
# Jon Snow has refused Stannis' offer.
#
# Stannis therefore remains temporary custodian of
# Winterfell and the North while a legitimate Northern
# ruler is sought.
#
# Lord Manderly brings two pieces of information:
#
# - Alayne Baelish may actually be Sansa Stark.
# - Rickon Stark may still be alive on Skagos.
#
# Stannis may also simply retain the North temporarily
# and demand that the Northern lords help him take
# the Iron Throne first.
# =========================================================

segrellion_stannis.1100 = {

	type = character_event

	title = segrellion_stannis.1100.t
	desc = segrellion_stannis.1100.desc

	theme = crown


	override_background = {
		reference = segrellion_winterfell_great_hall
	}


	trigger = {

		this = character:Baratheon_3

		is_alive = yes

		has_character_flag = segrellion_stannis_searching_northern_heir

		has_title = title:e_the_north

		has_title = title:c_winterfell
	}


	immediate = {

		title:c_white_harbor.holder ?= {

			save_scope_as = stannis_manderly
		}
	}


	left_portrait = {

		character = root

		animation = thinking
	}


	right_portrait = {

		character = scope:stannis_manderly

		animation = personality_rational
	}


	# =====================================================
	# SANSA STARK / ALAYNE BAELISH
	#
	# AI: 80%
	# =====================================================

	option = {

		name = segrellion_stannis.1100.a

		custom_tooltip = segrellion_stannis.1100.a.tt


		add_character_flag = segrellion_stannis_north_successor_sansa_path


		trigger_event = {

			id = segrellion_stannis.1110

			days = { 3 7 }
		}


		ai_chance = {

			base = 80
		}
	}


	# =====================================================
	# RICKON STARK / SKAGOS
	#
	# AI: 10%
	#
	# We only flag this route for now.
	# The actual expedition to Skagos comes later.
	# =====================================================

	option = {

		name = segrellion_stannis.1100.b

		custom_tooltip = segrellion_stannis.1100.b.tt


		add_character_flag = segrellion_stannis_north_successor_rickon_path

		add_character_flag = segrellion_rickon_search_pending


		ai_chance = {

			base = 10
		}
	}


	# =====================================================
	# KEEP THE NORTH
	#
	# AI: 10%
	#
	# Stannis refuses to waste more time choosing a
	# Northern ruler. He keeps temporary control and
	# demands Northern support for his royal campaign.
	#
	# The large Northern opinion penalty and special
	# negative modifier will be developed later.
	# =====================================================

	option = {

		name = segrellion_stannis.1100.c

		custom_tooltip = segrellion_stannis.1100.c.tt


		add_character_flag = segrellion_stannis_keeps_north_until_iron_throne

		remove_character_flag = segrellion_stannis_searching_northern_heir


		ai_chance = {

			base = 10
		}
	}
}
