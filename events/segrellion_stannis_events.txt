# =========================================================
# 0090 - STARTUP ROUTER
#
# One day after Roose's murder.
# Gives CK3 enough time to resolve inheritance.
# =========================================================

segrellion_stannis.0090 = {

	hidden = yes


	trigger = {

		this = character:Baratheon_3

		is_alive = yes

		has_character_flag = segrellion_stannis_north_story_initialized

		NOT = {
			has_character_flag = segrellion_stannis_northern_campaign_started
		}
	}


	immediate = {

		set_location = title:c_castle_black.title_province


		title:e_the_north.holder ?= {

			add_character_flag = segrellion_stannis_northern_enemy

			save_scope_as = stannis_bolton_lord
		}


		trigger_event = {

			id = segrellion_stannis.0001

			days = { 2 4 }
		}
	}
}


# =========================================================
# 0001 - THE WAR AFTER THE WAR
#
# Root = Stannis
#
# The wildling threat has been checked.
# Now Stannis decides what the North requires.
# =========================================================

segrellion_stannis.0001 = {

	type = character_event

	title = segrellion_stannis.0001.t
	desc = segrellion_stannis.0001.desc

	theme = war


	override_background = {
		reference = castle
	}


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		this = character:Baratheon_3

		is_alive = yes

		has_character_flag = segrellion_stannis_preparing_northern_campaign

		character:Stark_7 ?= {
			is_alive = yes
		}

		title:e_the_north = {
			exists = holder
		}
	}


	immediate = {

		character:Stark_7 = {
			save_scope_as = stannis_jon
		}

		title:e_the_north.holder = {
			save_scope_as = stannis_bolton_lord
		}
	}


	left_portrait = {

		character = root

		animation = personality_rational
	}


	right_portrait = {

		character = character:Stark_7

		animation = personality_honorable
	}


	# =====================================================
	# MARCH SOUTH
	# =====================================================

	option = {

		name = segrellion_stannis.0001.a

		custom_tooltip = segrellion_stannis.0001.a.tt


		add_character_flag = segrellion_stannis_northern_campaign_committed


		trigger_event = {

			id = segrellion_stannis.0010

			days = { 2 5 }
		}


		ai_chance = {
			base = 100
		}
	}


	# =====================================================
	# PLAYER DELAY
	# =====================================================

	option = {

		name = segrellion_stannis.0001.b

		trigger = {
			is_ai = no
		}

		custom_tooltip = segrellion_stannis.0001.b.tt


		trigger_event = {

			id = segrellion_stannis.0001

			days = { 30 60 }
		}


		ai_chance = {
			base = 0
		}
	}
}

# =========================================================
# 1010 - THE NAME OF STARK
#
# Root = Jon
#
# Stannis offers Jon release from his vows,
# legitimacy, Winterfell and the North.
#
# AI:
# 90% - Accept
# 10% - Refuse
# =========================================================

segrellion_stannis.1010 = {

	type = character_event

	title = segrellion_stannis.1010.t
	desc = segrellion_stannis.1010.desc

	theme = crown


	override_background = {
		reference = throne_room
	}


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}


	trigger = {

		this = character:Stark_7

		is_alive = yes

		character:Baratheon_3 ?= {

			is_alive = yes

			has_character_flag = segrellion_stannis_northern_campaign_victory
		}
	}


	immediate = {

		character:Baratheon_3 = {
			save_scope_as = jon_stannis
		}

		title:e_the_north.holder ?= {
			save_scope_as = jon_bolton_lord
		}
	}


	left_portrait = {

		character = root

		animation = thinking
	}


	right_portrait = {

		character = character:Baratheon_3

		animation = personality_rational
	}


	# =====================================================
	# ACCEPT
	#
	# Jon leaves the Watch, is legitimized and receives
	# Winterfell and the North.
	#
	# AI chance: 90%
	# =====================================================

	option = {

		name = segrellion_stannis.1010.a

		custom_tooltip = segrellion_stannis.1010.a.tt


		segrellion_jon_accepts_north_effect = yes


		# Only continue if the Night's Watch transfer
		# succeeded and Jon genuinely received the North.

		if = {

			limit = {

				has_character_flag = segrellion_jon_restored_to_winterfell

				has_title = title:e_the_north

				has_title = title:c_winterfell
			}


			character:Baratheon_3 = {

				trigger_event = {

					id = segrellion_stannis.1020

					days = { 2 4 }
				}
			}
		}


		ai_chance = {

			base = 90
		}
	}


	# =====================================================
	# REFUSE
	#
	# Jon remains Lord Commander of the Night's Watch.
	# Stannis keeps temporary custody of the North.
	#
	# AI chance: 10%
	# =====================================================

	option = {

		name = segrellion_stannis.1010.b

		custom_tooltip = segrellion_stannis.1010.b.tt


		add_character_flag = segrellion_jon_refused_north


		# Jon remains in the Watch.
		# Nothing is transferred and his vows remain intact.

		give_nickname = nick_segrellion_lord_snow


		character:Baratheon_3 = {

			add_character_flag = segrellion_stannis_jon_refused_north

			add_character_flag = segrellion_stannis_searching_northern_heir


			trigger_event = {

				id = segrellion_stannis.1021

				days = { 2 4 }
			}
		}


		ai_chance = {

			base = 10
		}
	}
}
