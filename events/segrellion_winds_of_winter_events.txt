# =========================================================
# THE UNICORN RETURNS
# DAEMON SEGRELLION
# =========================================================

segrellion_winds.0001 = {

	type = character_event

	title = segrellion_winds.0001.t
	desc = segrellion_winds.0001.desc

	theme = war

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	left_portrait = root


	immediate = {

		title:e_flamelands.holder ?= {
			save_scope_as = flamelands_defender
		}
	}


	option = {

		name = segrellion_winds.0001.a

		remove_character_flag = segrellion_unicorn_returns_event_pending

		add_character_flag = segrellion_unicorn_returns_cb_authorized

		custom_tooltip = segrellion_unicorn_returns_war_begins_tt

		start_war = {
			cb = segrellion_unicorn_returns_cb
			target = scope:flamelands_defender
			target_title = title:e_flamelands
		}
	}
}

# 0101 - THE PLATE RETURNS

# =========================================================

segrellion_winds.0101 = {

	type = character_event

	title = segrellion_winds.0101.t
	desc = segrellion_winds.0101.desc

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	override_background = {
		reference = segrellion_vhalerion_throne_room
	}

	left_portrait = root

	trigger = {
		this = character:Segrellion_43

		NOT = {
			any_character_artifact = {
				has_variable = segrellion_plate_of_segrell_artifact
			}
		}
	}


	option = {

		name = segrellion_winds.0101.a

		segrellion_create_artifact_plate_of_segrell_effect = {
			OWNER = root
		}

		add_character_flag = segrellion_plate_returned_by_sneakys_kin

		custom_tooltip = segrellion_plate_returns_to_hugh_tt
	}
}


# =========================================================
# 0102 - DAEMON DEMANDS THE PLATE
# Hugh chooses his answer.
# =========================================================

segrellion_winds.0102 = {

	type = character_event

	title = segrellion_winds.0102.t
	desc = segrellion_winds.0102.desc

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	override_background = {
		reference = segrellion_vhalerion_throne_room
	}

	left_portrait = root

	trigger = {

		this = character:Segrellion_43

		trigger_if = {
			limit = {
				NOT = {
					debug_only = yes
				}
			}

			any_character_artifact = {
				has_variable = segrellion_plate_of_segrell_artifact
			}

			character:Segrellion_60 ?= {
				is_alive = yes
				has_character_flag = segrellion_plate_demand_pending
			}
		}
	}


	# =====================================================
	# 20% - GIVE THE PLATE AND JOIN DAEMON
	# =====================================================

	option = {

		name = segrellion_winds.0102.a

		ai_chance = {
			base = 20
		}

		segrellion_transfer_plate_to_daemon_effect = yes

		add_character_flag = segrellion_hugh_will_support_daemon

		character:Segrellion_60 = {

			remove_character_flag = segrellion_plate_demand_pending
			add_character_flag = segrellion_plate_demand_resolved

			trigger_event = {
				id = segrellion_winds.0103
				days = 1
			}
		}
	}


	# =====================================================
	# 70% - CANON
	# GIVE THE PLATE, BUT DO NOT JOIN THE WAR
	# =====================================================

	option = {

		name = segrellion_winds.0102.b

		ai_chance = {
			base = 70
		}

		segrellion_transfer_plate_to_daemon_effect = yes

		character:Segrellion_60 = {

			remove_character_flag = segrellion_plate_demand_pending
			add_character_flag = segrellion_plate_demand_resolved

			trigger_event = {
				id = segrellion_winds.0104
				days = 1
			}
		}
	}


	# =====================================================
	# 10% - REFUSE
	# =====================================================

	option = {

		name = segrellion_winds.0102.c

		ai_chance = {
			base = 10
		}

		add_character_flag = segrellion_hugh_refused_daemon_plate

		character:Segrellion_60 = {

			remove_character_flag = segrellion_plate_demand_pending
			add_character_flag = segrellion_plate_demand_resolved

			trigger_event = {
				id = segrellion_winds.0105
				days = 1
			}
		}
	}
}
