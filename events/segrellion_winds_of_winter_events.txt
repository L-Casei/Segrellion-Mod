segrellion_winds.0001 = {

	type = character_event

	title = segrellion_winds.0001.t
	desc = segrellion_winds.0001.desc

	theme = war

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	left_portrait = root


	immediate = {

		title:e_flamelands.holder ?= {
			save_scope_as = flamelands_defender
		}
	}


	option = {

		name = segrellion_winds.0001.a

		remove_character_flag = segrellion_unicorn_returns_event_pending

			cb = segrellion_unicorn_returns_cb
			target = scope:flamelands_defender
			target_title = title:e_flamelands
		}
	}
}


# =========================================================
# 0101 - SOMETHING SURVIVED
#
# Hugh discovers documentary evidence that the Plate
# survived the fall of House Segrellion.
# =========================================================

segrellion_winds.0101 = {

	type = character_event

	title = segrellion_winds.0101.t
	desc = segrellion_winds.0101.desc

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	override_background = {
		reference = segrellion_vhalerion_throne_room
	}

	left_portrait = root


	trigger = {

		this = character:Segrellion_43

		is_alive = yes

		has_character_flag = segrellion_plate_story_stage_1_pending

		NOT = {
			has_character_flag = segrellion_plate_story_stage_1_complete
		}

		NOT = {
			segrellion_plate_recovery_blocked_trigger = yes
		}
	}


	option = {

		name = segrellion_winds.0101.a


		remove_character_flag = segrellion_plate_story_stage_1_pending

		add_character_flag = segrellion_plate_story_stage_1_complete
		add_character_flag = segrellion_plate_story_stage_2_pending


		# =====================================================
		# SEGRELLION LORE
		# =====================================================

		custom_tooltip = segrellion_lore_minor_gain_tt

		hidden_effect = {

			if = {

				limit = {
					NOT = {
						exists = var:segrellion_lore_plate_stage_1
					}
				}

				segrellion_add_lore_effect = {
					AMOUNT = 1
					VISITED_VARIABLE = segrellion_lore_plate_stage_1
				}
			}
		}


		trigger_event = {
			id = segrellion_winds.0110
			days = { 45 90 }
		}
	}
}

# =========================================================
# 0102 - DAEMON DEMANDS THE PLATE
# Hugh chooses his answer.
# =========================================================

segrellion_winds.0102 = {

	type = character_event

	title = segrellion_winds.0102.t
	desc = segrellion_winds.0102.desc

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	override_background = {
		reference = segrellion_vhalerion_throne_room
	}

	right_portrait = character:Segrellion_60

	trigger = {

		this = character:Segrellion_43

		trigger_if = {
			limit = {
				NOT = {
					debug_only = yes
				}
			}

			any_character_artifact = {
				has_variable = segrellion_plate_of_segrell_artifact
			}

			character:Segrellion_60 ?= {
				is_alive = yes
				has_character_flag = segrellion_plate_demand_pending
			}
		}
	}


	# =====================================================
	# 20% - GIVE THE PLATE AND JOIN DAEMON
	# =====================================================

	option = {

		name = segrellion_winds.0102.a

		ai_chance = {
			base = 20
		}

		segrellion_transfer_plate_to_daemon_effect = yes

		add_character_flag = segrellion_hugh_will_support_daemon

		character:Segrellion_60 = {

			remove_character_flag = segrellion_plate_demand_pending
			add_character_flag = segrellion_plate_demand_resolved

			trigger_event = {
				id = segrellion_winds.0103
				days = 1
			}
		}
	}


	# =====================================================
	# 70% - CANON
	# GIVE THE PLATE, BUT DO NOT JOIN THE WAR
	# =====================================================

	option = {

		name = segrellion_winds.0102.b

		ai_chance = {
			base = 70
		}

		segrellion_transfer_plate_to_daemon_effect = yes

		character:Segrellion_60 = {

			remove_character_flag = segrellion_plate_demand_pending
			add_character_flag = segrellion_plate_demand_resolved

			trigger_event = {
				id = segrellion_winds.0104
				days = 1
			}
		}
	}


	# =====================================================
	# 10% - REFUSE
	# =====================================================

	option = {

		name = segrellion_winds.0102.c

		ai_chance = {
			base = 10
		}

		add_character_flag = segrellion_hugh_refused_daemon_plate

		character:Segrellion_60 = {

			remove_character_flag = segrellion_plate_demand_pending
			add_character_flag = segrellion_plate_demand_resolved

			trigger_event = {
				id = segrellion_winds.0105
				days = 1
			}
		}
	}
}

			base = 20
		}

		add_character_flag = segrellion_young_griff_refused_arianne

# =========================================================
# 0402 - LEO'S ANSWER
# =========================================================

segrellion_winds.0402 = {

	type = character_event

	title = segrellion_winds.0402.t
	desc = segrellion_winds.0402.desc

	override_background = {
		reference = throne_room
	}

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	left_portrait = root
	right_portrait = character:Segrellion_60


	# Surrender.
	option = {

		name = segrellion_winds.0402.a

		ai_chance = {

			base = 30

			modifier = {
				add = 70

				character:Segrellion_60 = {
					has_character_flag = segrellion_killed_clifford_event_done
				}
			}
		}

		segrellion_capture_stormflame_court_effect = yes

		character:Segrellion_60 = {

			add_character_flag = segrellion_leo_surrendered_beacon

			trigger_event = {
				id = segrellion_winds.0403
				days = 1
			}

			var:segrellion_unicorn_returns_war ?= {
				end_war = attacker
			}
		}
	}


	# Fight.
	option = {

		name = segrellion_winds.0402.b

		ai_chance = {

			base = 70

			modifier = {
				factor = 0

				character:Segrellion_60 = {
					has_character_flag = segrellion_killed_clifford_event_done
				}
			}
		}

		character:Segrellion_60 = {

			add_character_flag = segrellion_leo_refused_beacon_surrender
		}
	}
}
