# =========================================================
# 0121 - AI HUGH RECOVERS THE PLATE
#
# Safety net for the Winds of Winter storyline.
#
# The normal decision remains available and the AI can
# take it first. This event only fires if Hugh still has
# not acted ten days after locating the Plate.
# =========================================================

segrellion_winds.0121 = {

	type = character_event
	hidden = yes

	trigger = {

		this = character:Segrellion_43

		is_ai = yes

		has_character_flag = segrellion_plate_recovery_ready

		NOT = {
			has_character_flag = segrellion_plate_recovery_in_progress
		}

		NOT = {
			has_character_flag = segrellion_plate_recovery_complete
		}
	}

	immediate = {

		if = {

			limit = {
				NOT = {
					segrellion_plate_recovery_blocked_trigger = yes
				}
			}

			# Hugh pays the full recovery cost even if
			# doing so forces Vhal temporarily into debt.

			add_gold = -200

			remove_character_flag = segrellion_plate_recovery_ready

			add_character_flag = segrellion_plate_recovery_in_progress

			trigger_event = {
				id = segrellion_winds.0130
				days = { 20 60 }
			}
		}
	}
}

# =========================================================
# 0102 - DAEMON DEMANDS THE PLATE
# Hugh chooses his answer.
# =========================================================

segrellion_winds.0102 = {

	type = character_event

	title = segrellion_winds.0102.t
	desc = segrellion_winds.0102.desc

	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	override_background = {
		reference = segrellion_vhalerion_throne_room
	}

	left_portrait = {
		character = root
		animation = thinking
	}

	right_portrait = {
		character = character:Segrellion_60
		animation = personality_bold
	}

	trigger = {

		this = character:Segrellion_43

		trigger_if = {
			limit = {
				NOT = {
					debug_only = yes
				}
			}

			any_character_artifact = {
				has_variable = segrellion_plate_of_segrell_artifact
			}

			character:Segrellion_60 ?= {
				is_alive = yes
				has_character_flag = segrellion_plate_demand_pending
			}
		}
	}


	# =====================================================
	# 20% - GIVE THE PLATE AND JOIN DAEMON
	# =====================================================

	option = {

		name = segrellion_winds.0102.a

		ai_chance = {
			base = 15
		}

		segrellion_transfer_plate_to_daemon_effect = yes

		add_character_flag = segrellion_hugh_will_support_daemon

		character:Segrellion_60 = {

			remove_character_flag = segrellion_plate_demand_pending
			add_character_flag = segrellion_plate_demand_resolved

			trigger_event = {
				id = segrellion_winds.0103
				days = 1
			}
		}
	}


	# =====================================================
	# 70% - CANON
	# GIVE THE PLATE, BUT DO NOT JOIN THE WAR
	# =====================================================

	option = {

		name = segrellion_winds.0102.b

		ai_chance = {
			base = 20
		}

		segrellion_transfer_plate_to_daemon_effect = yes

		character:Segrellion_60 = {

			remove_character_flag = segrellion_plate_demand_pending
			add_character_flag = segrellion_plate_demand_resolved

			trigger_event = {
				id = segrellion_winds.0104
				days = 1
			}
		}
	}

	# =====================================================
	# DEMAND REIMBURSEMENT
	# =====================================================

	option = {

		name = segrellion_winds.0102.d

		trigger = {

			character:Segrellion_60 = {
				gold >= 200
			}
		}

		ai_chance = {
			base = 55
		}

		add_character_flag = segrellion_hugh_requested_payment_for_plate

		character:Segrellion_60 = {

			trigger_event = {
				id = segrellion_winds.0106
				days = 2
			}
		}
	}


	# =====================================================
	# 10% - REFUSE
	# =====================================================

	option = {

		name = segrellion_winds.0102.c

		ai_chance = {
			base = 10
		}

		add_character_flag = segrellion_hugh_refused_daemon_plate

		character:Segrellion_60 = {

			remove_character_flag = segrellion_plate_demand_pending
			add_character_flag = segrellion_plate_demand_resolved

			trigger_event = {
				id = segrellion_winds.0105
				days = 1
			}
		}
	}

	
}

	# =====================================================

	# Leo offers his own life in exchange for his family.
	# =====================================================

		name = segrellion_winds.0402.a


		ai_chance = {

			base = 30

			modifier = {

				add = 70

				character:Segrellion_60 = {
					has_character_flag = segrellion_killed_clifford_event_done
				}
			}
		}


		# Daemon's terms included mercy for Leo's family.
		character:Segrellion_60 = {

			add_character_flag = segrellion_leo_surrendered_beacon

			add_character_flag = segrellion_promised_stormflame_family_mercy

			remove_character_modifier = segrellion_beacon_awaits_modifier
		}


		# The gates open.
		# Every surviving Stormflame passes into Daemon's custody.
		segrellion_capture_stormflame_court_effect = yes


		# Daemon now decides whether his word means anything.
		character:Segrellion_60 = {

			trigger_event = {
				id = segrellion_winds.0403
			}
		}
	}


	# Fight.
	option = {

		name = segrellion_winds.0402.b

		ai_chance = {

			base = 70

			modifier = {
				factor = 0

				character:Segrellion_60 = {
					has_character_flag = segrellion_killed_clifford_event_done
				}
			}
		}

		character:Segrellion_60 = {

			add_character_flag = segrellion_leo_refused_beacon_surrender
		}
	}
}
