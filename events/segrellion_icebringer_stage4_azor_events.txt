# =========================================================
# 7130 - WHITE WALKER ATTEMPT DISPATCHER
# =========================================================

segrellion_icebringer.7130 = {
	type = character_event
	hidden = yes

	immediate = {

		# =================================================
		# AFTER THE FIRST VICTORY
		#
		# Returning again is a hidden persistence path.
		# =================================================

		if = {
			limit = {
				has_character_flag = segrellion_azor_white_walker_rematch_two
			}

			trigger_event = segrellion_icebringer.7139
		}

		else_if = {
			limit = {
				has_character_flag = segrellion_azor_white_walker_rematch_one
			}

			trigger_event = segrellion_icebringer.7138
		}

		else_if = {
			limit = {
				has_character_flag = segrellion_azor_white_walker_defeated
			}

			trigger_event = segrellion_icebringer.7137
		}


		# =================================================
		# ORIGINAL LEARNING LOOP
		# =================================================

		else_if = {
			limit = {
				has_character_flag = segrellion_azor_white_walker_attempt_four
			}

			trigger_event = segrellion_icebringer.7135
		}

		else_if = {
			limit = {
				has_character_flag = segrellion_azor_white_walker_attempt_three
			}

			trigger_event = segrellion_icebringer.7134
		}

		else_if = {
			limit = {
				has_character_flag = segrellion_azor_white_walker_attempt_two
			}

			trigger_event = segrellion_icebringer.7133
		}

		else_if = {
			limit = {
				has_character_flag = segrellion_azor_white_walker_attempt_one
			}

			trigger_event = segrellion_icebringer.7132
		}

		else = {
			trigger_event = segrellion_icebringer.7131
		}
	}
}
