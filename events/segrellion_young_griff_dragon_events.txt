# =========================================================
# 0001 - A SCHOLAR'S TASK
#
# Young Griff orders a capable courtier to investigate
# possible surviving dragon eggs.
# =========================================================

segrellion_young_griff_dragon.0001 = {

	type = character_event

	title = segrellion_young_griff_dragon.0001.t

	desc = segrellion_young_griff_dragon.0001.desc

	theme = intrigue


	trigger = {

		this = character:IAF_Young_Griff_1

		is_alive = yes

		has_character_flag = segrellion_young_griff_dragon_path_egg

		NOT = {
			has_character_flag = segrellion_young_griff_dragon_route_resolved
		}
	}


	immediate = {

		random_courtier = {

			limit = {

				is_alive = yes

				is_adult = yes

				is_imprisoned = no

				learning >= 8

				stewardship >= 8
			}


			save_scope_as = segrellion_young_griff_dragon_researcher
		}


		if = {

			limit = {
				exists = scope:segrellion_young_griff_dragon_researcher
			}


			set_variable = {

				name = segrellion_young_griff_dragon_researcher

				value = scope:segrellion_young_griff_dragon_researcher
			}
		}
	}


	left_portrait = {

		character = root

		animation = thinking
	}


	right_portrait = {

		character = var:segrellion_young_griff_dragon_researcher

		animation = personality_rational
	}


	option = {

		name = segrellion_young_griff_dragon.0001.a


		trigger = {
			exists = var:segrellion_young_griff_dragon_researcher
		}


		add_character_flag = segrellion_young_griff_dragon_research_underway


		trigger_event = {

			id = segrellion_young_griff_dragon.0010

			days = { 45 75 }
		}
	}


	# Failsafe. This should never normally appear.

	option = {

		name = segrellion_young_griff_dragon.0001.b


		trigger = {

			NOT = {
				exists = var:segrellion_young_griff_dragon_researcher
			}
		}


		remove_character_flag = segrellion_young_griff_dragon_path_chosen

		remove_character_flag = segrellion_young_griff_dragon_path_egg
	}
}

# =========================================================
# 0032 - WHAT THE FIRE SPARED
#
# EXACT CHANCES:
#
# 10% egg
# 90% nothing
# =========================================================

segrellion_young_griff_dragon.0032 = {

	type = character_event

	title = segrellion_young_griff_dragon.0032.t


	desc = {

		first_valid = {


			triggered_desc = {

				trigger = {
					has_character_flag = segrellion_young_griff_found_summerhall_egg
				}

				desc = segrellion_young_griff_dragon.0032.desc_found
			}


			desc = segrellion_young_griff_dragon.0032.desc_failed
		}
	}


	theme = intrigue


	trigger = {

		this = character:IAF_Young_Griff_1

		is_alive = yes

		has_character_flag = segrellion_young_griff_dragon_search_summerhall

		NOT = {
			has_character_flag = segrellion_young_griff_dragon_route_resolved
		}
	}


	immediate = {

		random_list = {


			# =================================================
			# 10% - THE EGG SURVIVED
			# =================================================

			10 = {

				add_character_flag = segrellion_young_griff_found_dragon_egg

				add_character_flag = segrellion_young_griff_found_summerhall_egg


				agot_create_weighted_dragon_egg_artifact = {

					OWNER = root
				}


				# Preserve the fact that this specific egg
				# came from Summerhall.

				scope:newly_created_artifact ?= {

					set_variable = {

						name = segrellion_summerhall_dragon_egg

						value = yes
					}


					set_artifact_name = segrellion_summerhall_dragon_egg_name

					set_artifact_description = segrellion_summerhall_dragon_egg_description
				}
			}


			# =================================================
			# 90% - NOTHING
			# =================================================

			90 = {

				add_character_flag = segrellion_young_griff_dragon_egg_search_failed

				add_character_flag = segrellion_young_griff_summerhall_search_failed
			}
		}


		add_character_flag = segrellion_young_griff_dragon_route_resolved

		remove_character_flag = segrellion_young_griff_dragon_search_summerhall

		remove_variable = segrellion_young_griff_dragon_researcher
	}


	left_portrait = {

		character = root

		animation = personality_bold
	}


	option = {

		name = segrellion_young_griff_dragon.0032.a
	}
}

# =========================================================
# 0100 - A DRAGON WITHOUT A RIDER
#
# Only Rhaegal or Viserion may be targeted.
# =========================================================

segrellion_young_griff_dragon.0100 = {

	type = character_event

	title = segrellion_young_griff_dragon.0100.t

	desc = segrellion_young_griff_dragon.0100.desc

	theme = danger


	trigger = {

		this = character:IAF_Young_Griff_1

		is_alive = yes

		has_character_flag = segrellion_young_griff_dragon_path_daenerys

		NOT = {
			has_character_flag = segrellion_young_griff_dragon_route_resolved
		}
	}


	immediate = {

		# =================================================
		# BOTH ARE AVAILABLE
		# =================================================

		if = {

			limit = {

				character:dragon_rhaegal ?= {

					is_alive = yes

					NOT = {
						exists = var:current_rider
					}
				}


				character:dragon_viserion ?= {

					is_alive = yes

					NOT = {
						exists = var:current_rider
					}
				}
			}


			random_list = {

				50 = {

					character:dragon_rhaegal = {
						save_scope_as = segrellion_young_griff_target_dragon
					}
				}

				50 = {

					character:dragon_viserion = {
						save_scope_as = segrellion_young_griff_target_dragon
					}
				}
			}
		}


		# =================================================
		# ONLY RHAEGAL
		# =================================================

		else_if = {

			limit = {

				character:dragon_rhaegal ?= {

					is_alive = yes

					NOT = {
						exists = var:current_rider
					}
				}
			}


			character:dragon_rhaegal = {
				save_scope_as = segrellion_young_griff_target_dragon
			}
		}


		# =================================================
		# ONLY VISERION
		# =================================================

		else_if = {

			limit = {

				character:dragon_viserion ?= {

					is_alive = yes

					NOT = {
						exists = var:current_rider
					}
				}
			}


			character:dragon_viserion = {
				save_scope_as = segrellion_young_griff_target_dragon
			}
		}


		if = {

			limit = {
				exists = scope:segrellion_young_griff_target_dragon
			}


			set_variable = {

				name = segrellion_young_griff_target_dragon

				value = scope:segrellion_young_griff_target_dragon
			}
		}
	}


	left_portrait = {

		character = root

		animation = personality_bold
	}


	right_portrait = {

		character = var:segrellion_young_griff_target_dragon
	}


	# =====================================================
	# GO TO THE DRAGON
	# =====================================================

	option = {

		name = segrellion_young_griff_dragon.0100.a


		trigger = {
			exists = var:segrellion_young_griff_target_dragon
		}


		custom_tooltip = segrellion_young_griff_dragon.0100.a.tt


		hidden_effect = {

			start_travel_plan = {

				destination = var:segrellion_young_griff_target_dragon.location

				return_trip = yes

				on_arrival_event = segrellion_young_griff_dragon.0110

				on_travel_planner_cancel_event = segrellion_young_griff_dragon.0190
			}
		}
	}


	# =====================================================
	# FAILSAFE - BOTH DRAGONS UNAVAILABLE
	# =====================================================

	option = {

		name = segrellion_young_griff_dragon.0100.b


		trigger = {

			NOT = {
				exists = var:segrellion_young_griff_target_dragon
			}
		}


		remove_character_flag = segrellion_young_griff_dragon_path_chosen

		remove_character_flag = segrellion_young_griff_dragon_path_daenerys
	}
}


# =========================================================
# 0110 - BLOOD OF THE DRAGON
#
# EXACT CHANCES:
#
# 40% death
# 50% burned
# 10% tame
# =========================================================

segrellion_young_griff_dragon.0110 = {

	type = character_event

	title = segrellion_young_griff_dragon.0110.t

	desc = segrellion_young_griff_dragon.0110.desc

	theme = danger


	trigger = {

		this = character:IAF_Young_Griff_1

		is_alive = yes

		has_character_flag = segrellion_young_griff_dragon_path_daenerys

		NOT = {
			has_character_flag = segrellion_young_griff_dragon_route_resolved
		}
	}


	left_portrait = {

		character = root

		animation = personality_bold
	}


	right_portrait = {

		character = var:segrellion_young_griff_target_dragon
	}


	# =====================================================
	# APPROACH
	# =====================================================

	option = {

		name = segrellion_young_griff_dragon.0110.a


		trigger = {

			var:segrellion_young_griff_target_dragon ?= {

				is_alive = yes

				has_trait = dragon

				NOT = {
					exists = var:current_rider
				}
			}
		}


		custom_tooltip = segrellion_young_griff_dragon.0110.a.tt


		hidden_effect = {

			random_list = {


				# =============================================
				# 40% - DEATH
				# =============================================

				40 = {

					death = {

						death_reason = death_dragon_ate_failed_tamed

						killer = var:segrellion_young_griff_target_dragon
					}
				}


				# =============================================
				# 50% - BURNED
				# =============================================

				50 = {

					if = {

						limit = {

							NOT = {
								has_trait = burned
							}
						}

						add_trait = burned
					}


					add_trait_xp = {

						trait = burned

						track = burned

						value = 100
					}


					add_character_flag = segrellion_young_griff_burned_by_dragon

					add_character_flag = segrellion_young_griff_dragon_route_resolved


					character:Targaryen_9 ?= {

						add_character_flag = segrellion_daenerys_caught_young_griff_dragon_theft


						add_opinion = {

							target = root

							modifier = segrellion_daenerys_caught_dragon_thief_opinion
						}
					}


					trigger_event = {

						id = segrellion_young_griff_dragon.0120

						days = 1
					}
				}


				# =============================================
				# 10% - TAME
				# =============================================

				10 = {

					agot_tame_dragon = {

						TAMER = root

						DRAGON = var:segrellion_young_griff_target_dragon
					}


					add_character_flag = segrellion_young_griff_tamed_daenerys_dragon

					add_character_flag = segrellion_young_griff_stole_daenerys_dragon

					add_character_flag = segrellion_young_griff_dragon_route_resolved


					character:Targaryen_9 ?= {

						add_character_flag = segrellion_daenerys_dragon_stolen_by_young_griff


						add_opinion = {

							target = root

							modifier = segrellion_daenerys_dragon_stolen_opinion
						}
					}


					trigger_event = {

						id = segrellion_young_griff_dragon.0120

						days = 1
					}
				}
			}
		}
	}


	# =====================================================
	# TARGET NO LONGER AVAILABLE
	# =====================================================

	option = {

		name = segrellion_young_griff_dragon.0110.b


		trigger = {

			NOT = {

				var:segrellion_young_griff_target_dragon ?= {

					is_alive = yes

					has_trait = dragon

					NOT = {
						exists = var:current_rider
					}
				}
			}
		}


		hidden_effect = {

			add_character_flag = segrellion_young_griff_dragon_target_lost

			add_character_flag = segrellion_young_griff_dragon_route_resolved


			trigger_event = {

				id = segrellion_young_griff_dragon.0120

				days = 1
			}
		}
	}
}
