	type = character_event

	title = segrellion_daenerys_dragonstone.0002.t

	override_background = {
		reference = segrellion_dragonstone_throne_room
	}

		reference = "gfx/interface/window_event/event_header_blue.dds"
	}

	left_portrait = {

		animation = personality_rational
	}

	right_portrait = {
		character = character:Baratheon_13
		animation = worry
	}


	# =====================================================
	# DAVOS + JON
	# =====================================================

	option = {

		name = segrellion_daenerys_dragonstone.0002.a

		add_character_flag = segrellion_daenerys_invited_davos_and_jon


		character:Seaworth_1 ?= {

			add_character_flag = segrellion_davos_invited_to_dragonstone
			add_character_flag = segrellion_davos_invited_to_daenerys_parley
		}


		segrellion_prepare_davos_for_dragonstone_story_effect = yes


		trigger_event = {

			id = segrellion_daenerys_dragonstone.0010
			days = 1
		}


		ai_chance = {

			base = 1000
		}
	}


	# =====================================================
	# EXILE
	# =====================================================

	option = {

		name = segrellion_daenerys_dragonstone.0002.b


		character:Baratheon_13 = {

			release_from_prison = yes

			set_location = title:c_pentos.title_province

			add_character_flag = segrellion_exiled_to_essos
		}


		character:Baratheon_13 = {

			mother ?= {

				if = {

					limit = {
						is_alive = yes
					}

					release_from_prison = yes

					set_location = title:c_pentos.title_province

					add_character_flag = segrellion_exiled_to_essos
				}
			}
		}


		ai_chance = {

			base = 40
		}
	}


	# =====================================================
	# EXECUTE
	# =====================================================

	option = {

		name = segrellion_daenerys_dragonstone.0002.c


		character:Baratheon_13 = {

			death = {
				death_reason = death_execution
				killer = character:Targaryen_9
			}
		}


		character:Baratheon_13 = {

			mother ?= {

				if = {

					limit = {
						is_alive = yes
					}

					death = {
						death_reason = death_execution
						killer = character:Targaryen_9
					}
				}
			}
		}


		ai_chance = {
			base = 0
		}
	}
}

# =========================================================
# 0090 - TAKE DRAGONSTONE ARRIVAL
#
# Hidden arrival router for the Take Dragonstone decision.
#
# Daenerys has ALREADY landed in Westeros.
# Therefore we seize Dragonstone and redirect into the
# existing Dragonstone arrival storyline without repeating
# the generic Westeros landing sequence.
# =========================================================

segrellion_daenerys_dragonstone.0090 = {

	hidden = yes


	trigger = {

		this = character:Targaryen_9

		is_alive = yes

		has_character_flag = segrellion_daenerys_taking_dragonstone
	}


	immediate = {

		remove_character_flag = segrellion_daenerys_taking_dragonstone


		if = {

			limit = {

				has_character_flag = segrellion_daenerys_landed_in_westeros


				segrellion_daenerys_dragonstone_landing_available_trigger = yes


				NOT = {
					has_character_flag = segrellion_daenerys_seized_dragonstone
				}
			}


			add_character_flag = segrellion_daenerys_took_dragonstone_after_landing


			segrellion_daenerys_seize_dragonstone_effect = yes


			# =================================================
			# EXISTING DRAGONSTONE ARRIVAL EVENT
			# =================================================

			if = {

				limit = {

					NOT = {
						has_character_flag = segrellion_daenerys_dragonstone_reflection_seen
					}
				}


				trigger_event = {

					id = segrellion_daenerys_invasion.0019

					days = 1
				}
			}


			else = {

				remove_character_flag = segrellion_daenerys_took_dragonstone_after_landing
			}
		}
	}
}
