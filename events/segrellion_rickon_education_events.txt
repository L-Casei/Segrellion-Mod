namespace = segrellion_rickon_education

segrellion_rickon_education.0000 = {
	type = character_event
	hidden = yes
	trigger = {
		this = character:Stark_Dummy_11
		is_alive = yes
		NOT = { has_character_flag = segrellion_rickon_education_chain_ended }
	}
	immediate = {
		if = {
			limit = { is_adult = yes }
			segrellion_rickon_end_education_chain_effect = yes
		}
		else_if = {
			limit = { has_character_flag = segrellion_rickon_education_chain_active }
			segrellion_rickon_clean_unapproved_personality_traits_effect = yes

			if = {
				limit = {
					has_character_flag = segrellion_rickon_tutor_daemon
					NOT = { has_relation_guardian = character:Segrellion_60 }
					character:Segrellion_60 = { is_alive = no }
					character:Black_Unicorn_1 = { is_alive = yes is_adult = yes is_imprisoned = no is_incapable = no }
				}
				remove_character_flag = segrellion_rickon_tutor_daemon
				add_character_flag = segrellion_rickon_tutor_visenya
				set_relation_guardian = character:Black_Unicorn_1
				character:Black_Unicorn_1 = { trigger_event = { id = segrellion_rickon_education.0002 days = 1 } }
			}
			else_if = {
				limit = { NOT = { segrellion_rickon_story_guardian_valid_trigger = yes } }
				segrellion_rickon_end_education_chain_effect = yes
			}

			if = {
				limit = { has_character_flag = segrellion_rickon_education_chain_active }
				trigger_event = { id = segrellion_rickon_education.0000 days = 45 }
			}
		}
		else = {
			if = {
				limit = {
					has_character_flag = segrellion_rickon_under_daemon_protection
					NOT = { has_character_flag = segrellion_rickon_education_start_queued }
					character:Segrellion_60 = { is_alive = yes is_adult = yes is_imprisoned = no is_incapable = no }
				}
				add_character_flag = segrellion_rickon_education_start_queued
				character:Segrellion_60 = { trigger_event = { id = segrellion_rickon_education.0001 days = 1 } }
			}
			else_if = {
				limit = {
					has_character_flag = segrellion_rickon_under_visenya_protection
					NOT = { has_character_flag = segrellion_rickon_education_start_queued }
					OR = { NOT = { exists = character:Segrellion_60 } character:Segrellion_60 = { is_alive = no } }
					character:Black_Unicorn_1 = { is_alive = yes is_adult = yes is_imprisoned = no is_incapable = no }
				}
				add_character_flag = segrellion_rickon_education_start_queued
				character:Black_Unicorn_1 = { trigger_event = { id = segrellion_rickon_education.0001 days = 1 } }
			}
			if = {
				limit = {
					is_adult = no
					NOT = { has_character_flag = segrellion_rickon_education_chain_active }
					NOT = { has_character_flag = segrellion_rickon_education_chain_ended }
				}
				trigger_event = { id = segrellion_rickon_education.0000 days = 60 }
			}
		}
	}
}

segrellion_rickon_education.0001 = {
	type = character_event
	title = segrellion_rickon_education.0001.t
	desc = {
		first_valid = {
			triggered_desc = { trigger = { this = character:Segrellion_60 } desc = segrellion_rickon_education.0001.desc_daemon }
			desc = segrellion_rickon_education.0001.desc_visenya
		}
	}
	theme = education
	trigger = {
		OR = { this = character:Segrellion_60 this = character:Black_Unicorn_1 }
		is_alive = yes
		is_adult = yes
		is_imprisoned = no
		is_incapable = no
		character:Stark_Dummy_11 = {

		}
	}
	on_trigger_fail = { character:Stark_Dummy_11 ?= { remove_character_flag = segrellion_rickon_education_start_queued } }
	left_portrait = { character = root animation = personality_rational }
	right_portrait = { character = character:Stark_Dummy_11 animation = child_personality_bold }
	option = {
		name = segrellion_rickon_education.0001.a
		character:Stark_Dummy_11 = {
			remove_character_flag = segrellion_rickon_education_start_queued
			add_character_flag = segrellion_rickon_education_chain_active
			if = {
				limit = { root = character:Segrellion_60 }
				add_character_flag = segrellion_rickon_tutor_daemon
				remove_character_flag = segrellion_rickon_tutor_visenya
			}
			else = {
				add_character_flag = segrellion_rickon_tutor_visenya
				remove_character_flag = segrellion_rickon_tutor_daemon
			}
			set_relation_guardian = root
			segrellion_rickon_seed_existing_story_traits_effect = yes
			segrellion_rickon_clean_unapproved_personality_traits_effect = yes
			segrellion_rickon_queue_first_lesson_effect = yes
			trigger_event = { id = segrellion_rickon_education.0000 days = 45 }
		}
	}
}
