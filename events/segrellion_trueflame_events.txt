		name = segrellion_trueflame.0100.b

		add_character_flag = segrellion_trueflame_agents_expedition

		trigger_event = {
			id = segrellion_trueflame.0120
			days = { 90 150 }
		}
	}
}

# =========================================================
# 0400 - RECLAIM TRUEFLAME
# =========================================================

segrellion_trueflame.0400 = {

	type = character_event

	title = segrellion_trueflame.0400.t
	desc = segrellion_trueflame.0400.desc

	theme = diplomacy

	immediate = {

		global_var:segrellion_trueflame ?= {

			artifact_owner ?= {
				save_scope_as = trueflame_current_holder
			}
		}
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:trueflame_current_holder
		animation = personality_bold
	}

	# Buy it back.
	option = {
		name = segrellion_trueflame.0400.a

		trigger = {
			gold >= 500
		}

		add_gold = -500

		scope:trueflame_current_holder ?= {
			add_gold = 500
		}

		segrellion_transfer_trueflame_effect = {
			NEW_OWNER = root
		}

		add_prestige = 250
	}

	# Assert the ancestral right of the House.
	option = {
		name = segrellion_trueflame.0400.b

		# The decision already charges 500 prestige and checks fame.

		hidden_effect = {

			if = {
				limit = {
					OR = {
						diplomacy >= 14
						intrigue >= 14
						dread >= 50
					}
				}

				segrellion_transfer_trueflame_effect = {
					NEW_OWNER = root
				}

				trigger_event = {
					id = segrellion_trueflame.0410
				}
			}

			else = {
				trigger_event = {
					id = segrellion_trueflame.0420
				}
			}
		}
	}

	option = {
		name = segrellion_trueflame.0400.c
	}
}
