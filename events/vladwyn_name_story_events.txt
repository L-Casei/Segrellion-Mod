# =========================================================
# 4000
# THE LORD OF STORM'S END
#
# ROOT = TRISTAN
#
# scope:vladwyn_storm_lord = CURRENT HOLDER OF STORM'S END
# =========================================================

vladwyn_name.4000 = {

	type = letter_event

	title = vladwyn_name.4000.t

	opening = {
		desc = vladwyn_name.4000.opening
	}

	desc = {

		first_valid = {


			# =================================================
			# ACCEPT
			# OPINION >= 50
			# =================================================

			triggered_desc = {

				trigger = {

					scope:vladwyn_storm_lord = {

						opinion = {
							target = root
							value >= 50
						}
					}
				}

				desc = vladwyn_name.4000.desc_accept
			}


			# =================================================
			# DEATH THREAT
			# OPINION <= -75
			# =================================================

			triggered_desc = {

				trigger = {

					scope:vladwyn_storm_lord = {

						opinion = {
							target = root
							value <= -75
						}
					}
				}

				desc = vladwyn_name.4000.desc_threat
			}


			# =================================================
			# THIRD FAILED REQUEST
			# =================================================

			triggered_desc = {

				trigger = {

					scope:vladwyn_storm_lord = {

						exists = var:vladwyn_tristan_storms_end_failed_requests

						var:vladwyn_tristan_storms_end_failed_requests >= 2
					}
				}

				desc = vladwyn_name.4000.desc_final_rejection
			}


			# =================================================
			# NORMAL REJECTION
			# =================================================

			desc = vladwyn_name.4000.desc_reject
		}
	}


	sender = scope:vladwyn_storm_lord


	# =====================================================
	# ACCEPTED
	# =====================================================

	option = {

		name = vladwyn_name.4000.a

		trigger = {

			scope:vladwyn_storm_lord = {

				opinion = {
					target = root
					value >= 50
				}
			}
		}


		custom_tooltip = vladwyn_name.4000.a.tt


		hidden_effect = {

			scope:vladwyn_storm_lord = {

				add_character_flag = vladwyn_tristan_storms_end_permission

				remove_character_flag = vladwyn_tristan_storms_end_request_cooldown
			}
		}
	}


	# =====================================================
	# DEATH THREAT
	#
	# OPINION <= -75
	#
	# PERMANENT FOR THIS LORD.
	# A NEW LORD GETS A CLEAN STATE.
	# =====================================================

	option = {

		name = vladwyn_name.4000.b

		trigger = {

			scope:vladwyn_storm_lord = {

				opinion = {
					target = root
					value <= -75
				}
			}
		}


		custom_tooltip = vladwyn_name.4000.b.tt


		hidden_effect = {

			scope:vladwyn_storm_lord = {

				add_character_flag = vladwyn_tristan_storms_end_barred

				add_character_flag = vladwyn_tristan_storms_end_death_threat

				remove_character_flag = vladwyn_tristan_storms_end_request_cooldown
			}
		}
	}


	# =====================================================
	# NORMAL REJECTION
	#
	# -74 TO +49
	# =====================================================

	option = {

		name = vladwyn_name.4000.c

		trigger = {

			scope:vladwyn_storm_lord = {

				opinion = {
					target = root
					value < 50
				}

				opinion = {
					target = root
					value > -75
				}
			}
		}


		custom_tooltip = vladwyn_name.4000.c.tt


		hidden_effect = {

			scope:vladwyn_storm_lord = {


				# =============================================
				# FIRST FAILED REQUEST
				# =============================================

				if = {

					limit = {

						NOT = {
							exists = var:vladwyn_tristan_storms_end_failed_requests
						}
					}

					set_variable = {
						name = vladwyn_tristan_storms_end_failed_requests
						value = 1
					}
				}


				# =============================================
				# THIRD FAILED REQUEST
				#
				# BEFORE THIS REQUEST:
				# VARIABLE = 2
				# =============================================

				else_if = {

					limit = {

						var:vladwyn_tristan_storms_end_failed_requests >= 2
					}

					change_variable = {
						name = vladwyn_tristan_storms_end_failed_requests
						add = 1
					}

					add_character_flag = vladwyn_tristan_storms_end_barred
				}


				# =============================================
				# SECOND FAILED REQUEST
				# =============================================

				else = {

					change_variable = {
						name = vladwyn_tristan_storms_end_failed_requests
						add = 1
					}
				}


				# =============================================
				# TWO YEAR COOLDOWN
				# =============================================

				if = {

					limit = {

						NOT = {
							has_character_flag = vladwyn_tristan_storms_end_barred
						}
					}

					add_character_flag = {
						flag = vladwyn_tristan_storms_end_request_cooldown
						years = 2
					}
				}
			}
		}
	}
}

# =========================================================
# 4900
# YOU WERE WARNED
#
# ROOT = TRISTAN
#
# Fires if Tristan enters Storm's End while the SAME ruler
# who threatened him still controls the castle.
# =========================================================

vladwyn_name.4900 = {

	type = character_event

	title = vladwyn_name.4900.t
	desc = vladwyn_name.4900.desc


	trigger = {

		has_character_flag = is_Vladwyn_1

		is_alive = yes

		exists = current_travel_plan

		current_travel_plan.travel_plan_owner = root

		location = title:b_storms_end.title_province

		exists = title:b_storms_end.holder

		title:b_storms_end.holder = {

			has_character_flag = vladwyn_tristan_storms_end_death_threat
		}
	}


	immediate = {

		title:b_storms_end.holder = {
			save_scope_as = vladwyn_storm_lord
		}
	}


	option = {

		name = vladwyn_name.4900.a


		hidden_effect = {

			scope:vladwyn_storm_lord = {

				imprison = {
					target = root
					type = dungeon
				}
			}

			root = {

				death = {
					death_reason = death_execution
					killer = scope:vladwyn_storm_lord
				}
			}
		}
	}
}
