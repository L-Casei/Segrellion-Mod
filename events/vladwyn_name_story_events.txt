# =========================================================
# 4000
# THE LORD OF STORM'S END
# =========================================================

vladwyn_name.4000 = {

	type = character_event

	title = vladwyn_name.4000.t

	theme = diplomacy
	override_header_background = {
		reference = "gfx/interface/window_event/event_header_blue.dds"
	}


	override_icon = {
		reference = "gfx/interface/icons/event_types/segrellion_ice_and_fire_event_icon.dds"
	}

	override_background = {
		reference = segrellion_insidious_castle
	}


	left_portrait = root

	right_portrait = scope:vladwyn_storm_lord


	desc = {

		first_valid = {

			triggered_desc = {

				trigger = {

					character:Baratheon_34 = {
						is_alive = yes
					}
				}

				desc = vladwyn_name.4000.desc_boremund_alive
			}


			# =================================================
			# ACCEPT
			# =================================================

			triggered_desc = {

				trigger = {

					scope:vladwyn_storm_lord = {

						opinion = {
							target = root
							value >= 50
						}
					}
				}

				desc = vladwyn_name.4000.desc_accept
			}


			# =================================================
			# DEATH THREAT
			# =================================================

			triggered_desc = {

				trigger = {

					scope:vladwyn_storm_lord = {

						opinion = {
							target = root
							value <= -75
						}
					}
				}

				desc = vladwyn_name.4000.desc_threat
			}


			# =================================================
			# THIRD FAILED REQUEST
			# =================================================

			triggered_desc = {

				trigger = {

					scope:vladwyn_storm_lord = {

						exists = var:vladwyn_tristan_storms_end_failed_requests

						var:vladwyn_tristan_storms_end_failed_requests >= 2
					}
				}

				desc = vladwyn_name.4000.desc_final_rejection
			}


			# =================================================
			# NORMAL REJECTION
			# =================================================

			desc = vladwyn_name.4000.desc_reject
		}
	}


	# =====================================================
	# ACCEPTED
	# =====================================================

	option = {

		name = vladwyn_name.4000.a


		trigger = {

			character:Baratheon_34 = {
				is_alive = no
			}

			scope:vladwyn_storm_lord = {

				opinion = {
					target = root
					value >= 50
				}
			}
		}


		custom_tooltip = vladwyn_name.4000.a.tt


		hidden_effect = {

			scope:vladwyn_storm_lord = {

				add_character_flag = vladwyn_tristan_storms_end_permission

				remove_character_flag = vladwyn_tristan_storms_end_request_cooldown
			}
		}
	}


	# =====================================================
	# DEATH THREAT
	# =====================================================

	option = {

		name = vladwyn_name.4000.b


		trigger = {

			character:Baratheon_34 = {
				is_alive = no
			}

			scope:vladwyn_storm_lord = {

				opinion = {
					target = root
					value <= -75
				}
			}
		}


		custom_tooltip = vladwyn_name.4000.b.tt


		hidden_effect = {

			scope:vladwyn_storm_lord = {

				add_character_flag = vladwyn_tristan_storms_end_barred

				add_character_flag = vladwyn_tristan_storms_end_death_threat

				remove_character_flag = vladwyn_tristan_storms_end_request_cooldown
			}
		}
	}


	# =====================================================
	# NORMAL REJECTION
	# =====================================================

	option = {

		name = vladwyn_name.4000.c


		trigger = {

			character:Baratheon_34 = {
				is_alive = no
			}

			scope:vladwyn_storm_lord = {

				opinion = {
					target = root
					value < 50
				}

				opinion = {
					target = root
					value > -75
				}
			}
		}


		custom_tooltip = vladwyn_name.4000.c.tt


		hidden_effect = {

			scope:vladwyn_storm_lord = {


				# =============================================
				# FIRST FAILED REQUEST
				# =============================================

				if = {

					limit = {

						NOT = {
							exists = var:vladwyn_tristan_storms_end_failed_requests
						}
					}

					set_variable = {
						name = vladwyn_tristan_storms_end_failed_requests
						value = 1
					}
				}


				# =============================================
				# THIRD FAILED REQUEST
				# =============================================

				else_if = {

					limit = {

						var:vladwyn_tristan_storms_end_failed_requests >= 2
					}

					change_variable = {
						name = vladwyn_tristan_storms_end_failed_requests
						add = 1
					}

					add_character_flag = vladwyn_tristan_storms_end_barred
				}


				# =============================================
				# SECOND FAILED REQUEST
				# =============================================

				else = {

					change_variable = {
						name = vladwyn_tristan_storms_end_failed_requests
						add = 1
					}
				}


				# =============================================
				# TWO YEAR COOLDOWN
				# =============================================

				if = {

					limit = {

						NOT = {
							has_character_flag = vladwyn_tristan_storms_end_barred
						}
					}

					add_character_flag = {
						flag = vladwyn_tristan_storms_end_request_cooldown
						years = 2
					}
				}
			}
		}
	}

	option = {

		name = vladwyn_name.4000.d


		trigger = {

			character:Baratheon_34 = {
				is_alive = yes
			}
		}


		custom_tooltip = vladwyn_name.4000.d.tt


		hidden_effect = {

			scope:vladwyn_storm_lord = {

				add_character_flag = {
					flag = vladwyn_tristan_storms_end_request_cooldown
					years = 2
				}
			}
		}
	}
}

	type = character_event

	theme = martial

	override_header_background = {
		reference = "gfx/interface/window_event/event_header_blue.dds"
	}

	}

	override_background = {
		reference = segrellion_winterfell_great_hall
	}

	left_portrait = root


	trigger = {

		has_character_flag = is_Vladwyn_1

		is_alive = yes

		exists = current_travel_plan

		current_travel_plan.travel_plan_owner = root

		location.county = {
			this = title:c_winterfell
		}

		exists = title:c_winterfell.holder


		title:c_winterfell.holder = {

			has_character_flag = vladwyn_tristan_winterfell_death_threat
		}
	}


	immediate = {

		title:c_winterfell.holder = {
			save_scope_as = vladwyn_winterfell_lord
		}
	}


	option = {

		name = vladwyn_name.4990.a


		hidden_effect = {

			scope:vladwyn_winterfell_lord = {

				imprison = {
					target = root
					type = dungeon
				}
			}


			root = {

				death = {
					death_reason = death_execution
					killer = scope:vladwyn_winterfell_lord
				}
			}
		}
	}
}
