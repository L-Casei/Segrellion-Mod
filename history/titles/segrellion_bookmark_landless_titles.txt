############################################
# TRISTAN VLADWYN
############################################

d_laamp_tristan_vladwyn = {

	# Tristan turns sixteen and establishes his own camp.
	# The camp therefore already exists when the 106 AC bookmark
	# is loaded on the following day.
	8106.4.18 = {
		liege = 0
		holder = Vladwyn_1
		government = landless_adventurer_government

		succession_laws = {
			landless_adventurer_succession_law
		}

		effect = {
			create_landless_adventurer_title_history_effect = yes

			set_variable = {
				name = adventurer_creation_reason
				value = flag:historical
			}
		}
	}

	# Only remove the camp if Roads to Power is unavailable.
	# This is intentionally placed on the actual bookmark date.
	8106.4.19 = {
		effect = {
			destroy_landless_title_no_dlc_effect = {
				DATE = 8106.4.19
			}
		}
	}

	# Tristan enters the Kingsguard.
	8131.1.28 = {
		holder = 0
	}
}

# JOFFREY VLADWYN - THE TRUE SENTINELS
############################################

d_laamp_true_sentinels = {

	8295.12.31 = {
		liege = 0
		holder = Vladwyn_5
		government = landless_adventurer_government

		succession_laws = {
			landless_adventurer_succession_law
		}

		effect = {

			holder ?= {

				if = {
					limit = {
						NOT = {
							has_realm_law = camp_purpose_mercenaries
						}
					}

					add_realm_law = camp_purpose_mercenaries
				}
			}

			create_landless_adventurer_title_history_effect = yes

			set_variable = {
				name = adventurer_creation_reason
				value = flag:historical
			}
		}
	}

	8300.1.1 = {
		effect = {
			destroy_landless_title_no_dlc_effect = {
				DATE = 8300.1.1
			}
		}
	}
}

# DAENERYS TARGARYEN - THE DRAGON QUEEN
# 300 AC
# =========================================================

d_laamp_daenerys_targaryen = {

    8300.8.1 = {

        liege = 0
        holder = Targaryen_9

        government = landless_adventurer_government

        succession_laws = {
            landless_adventurer_succession_law
        }

        effect = {

            holder ?= {

                if = {
                    limit = {
                        NOT = {
                            has_realm_law = camp_purpose_legitimists
                        }
                    }

                    add_realm_law_skip_effects = camp_purpose_legitimists
                }
            }

            create_landless_adventurer_title_history_effect = yes

            set_variable = {
                name = adventurer_creation_reason
                value = flag:historical
            }
        }
    }

    8300.8.19 = {

        effect = {

            # Tyrion Lannister
            character:Lannister_8 ?= {
                agot_safe_set_location_history_effect = {
                    CASTLE = title:d_laamp_daenerys_targaryen
                }
            }

            # Barristan Selmy
            character:Selmy_3 ?= {
                agot_safe_set_location_history_effect = {
                    CASTLE = title:d_laamp_daenerys_targaryen
                }
            }
