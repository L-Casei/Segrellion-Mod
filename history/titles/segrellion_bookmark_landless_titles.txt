d_laamp_tristan_vladwyn = {

	# =====================================================
	# 106 AC
	#
	# Tristan already walks his own road by the beginning
	# of the White Shadow bookmark.
	# =====================================================

	8106.4.18 = {

		liege = 0

		holder = Vladwyn_1

		government = landless_adventurer_government


		succession_laws = {
			landless_adventurer_succession_law
		}


		effect = {


			# =================================================
			# HONORABLE ADVENTURER SETUP
			# =================================================

			holder ?= {


				# ---------------------------------------------
				# MERCENARY CAMP PURPOSE
				# ---------------------------------------------

				if = {

					limit = {

						NOT = {
							has_realm_law = camp_purpose_mercenaries
						}
					}

					add_realm_law_skip_effects = camp_purpose_mercenaries
				}


				# ---------------------------------------------
				# TRISTAN IS NOT GALLOWSBAIT
				# ---------------------------------------------

				if = {

					limit = {
						has_trait = gallowsbait
					}

					remove_trait = gallowsbait
				}


				# ---------------------------------------------
				# CANON HONORABLE IDENTITY
				# ---------------------------------------------

				add_character_flag = vladwyn_honorable_adventurer


				# ---------------------------------------------
				# BLOCK GENERIC LAAMP OPENING
				# ---------------------------------------------

				add_character_flag = block_laamp_starting_event
			}


			# =================================================
			# CREATE THE HISTORICAL ADVENTURER CAMP
			# =================================================

			create_landless_adventurer_title_history_effect = yes


			set_variable = {
				name = adventurer_creation_reason
				value = flag:historical
			}
		}
	}


	# =====================================================
	# ROAD TO POWER SAFETY
	# =====================================================

	8106.4.18 = {

		effect = {

			destroy_landless_title_no_dlc_effect = {
				DATE = 8106.4.18
			}
		}
	}


	# =====================================================
	# KINGSGUARD
	# =====================================================

	8131.1.28 = {
		holder = 0
	}
}

d_laamp_daemon_segrellion = {

	8295.12.31 = {
		liege = 0
		holder = Segrellion_60
		government = landless_adventurer_government

		succession_laws = {
			landless_adventurer_succession_law
		}

		effect = {
			holder ?= {
				# Daemon has a fully scripted opening and historical followers.
				# Never run AGOT's generic LAAMP opening on top of it.
				add_character_flag = block_laamp_starting_event
			}

			create_landless_adventurer_title_history_effect = yes

			set_variable = {
				name = adventurer_creation_reason
				value = flag:historical
			}

			# Do not leave a historical adventurer without an explicit domicile.
			# Without this, the engine can resolve the camp from stale/random
			# character location data on some setups.
			agot_safe_move_laamp_history_effect = {
				CASTLE = title:b_storms_end
			}
		}
	}

	8300.1.1 = {
		effect = {
			destroy_landless_title_no_dlc_effect = {
				DATE = 8300.1.1
			}
		}
	}
}

d_laamp_daenerys_targaryen = {

    8300.8.1 = {

        liege = 0
        holder = Targaryen_9

        government = landless_adventurer_government

        succession_laws = {
            landless_adventurer_succession_law
        }

        effect = {

            holder ?= {

                # Daenerys has her own scripted 300 AC opening.
                add_character_flag = block_laamp_starting_event

                if = {
                    limit = {
                        NOT = {
                            has_realm_law = camp_purpose_legitimists
                        }
                    }

                    add_realm_law_skip_effects = camp_purpose_legitimists
                }
            }

            create_landless_adventurer_title_history_effect = yes

            set_variable = {
                name = adventurer_creation_reason
                value = flag:historical
            }

            # Anchor the historical host at Ghoyan Drohe instead of
            # allowing an unresolved character location to place it.
            agot_safe_move_laamp_history_effect = {
                CASTLE = title:b_ghoyan_drohe
            }
        }
    }

    8300.8.19 = {

        effect = {

            # Tyrion Lannister
            character:Lannister_8 ?= {
                agot_safe_set_location_history_effect = {
                    CASTLE = title:d_laamp_daenerys_targaryen
                }
            }

            # Barristan Selmy
            character:Selmy_3 ?= {
                agot_safe_set_location_history_effect = {
                    CASTLE = title:d_laamp_daenerys_targaryen
                }
            }

            # Jorah Mormont
            character:Mormont_3 ?= {
                agot_safe_set_location_history_effect = {
                    CASTLE = title:d_laamp_daenerys_targaryen
                }
            }

            # Missandei
            character:IAF_Missandei_1 ?= {
                agot_safe_set_location_history_effect = {
                    CASTLE = title:d_laamp_daenerys_targaryen
                }
            }

            # Torgo Nudho
            character:IAF_Torgo_Nudho_1 ?= {
                agot_safe_set_location_history_effect = {
                    CASTLE = title:d_laamp_daenerys_targaryen
                }
            }
        }
    }
}
