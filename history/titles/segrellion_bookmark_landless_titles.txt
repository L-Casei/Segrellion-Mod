############################################
# TRISTAN VLADWYN
############################################

d_laamp_tristan_vladwyn = {

	# Tristan turns sixteen and establishes his own camp.
	# The camp therefore already exists when the 106 AC bookmark
	# is loaded on the following day.
	8106.4.18 = {
		liege = 0
		holder = Vladwyn_1
		government = landless_adventurer_government

		succession_laws = {
			landless_adventurer_succession_law
		}

		effect = {
			create_landless_adventurer_title_history_effect = yes

			set_variable = {
				name = adventurer_creation_reason
				value = flag:historical
			}
		}
	}

	# Only remove the camp if Roads to Power is unavailable.
	# This is intentionally placed on the actual bookmark date.
	8106.4.19 = {
		effect = {
			destroy_landless_title_no_dlc_effect = {
				DATE = 8106.4.19
			}
		}
	}

	# Tristan enters the Kingsguard.
	8131.1.28 = {
		holder = 0
	}
}
